242 XSpreadsheet xSheet,
243 String scoreSchluessel,
244 int spalte, int zeile) {
245 try {
246 XNamedRanges namedRanges = namedRangesAusDoc(xDoc);
247 if (namedRanges == null) return;
248 String sheetName = Lo.qi(XNamed.class, xSheet).getName();
249 int sheetIdx = sheetIndex(xDoc, xSheet);
250 String escaped = sheetName.replace("'", "''");
251 String inhalt = "$'" + escaped + "'.$" + spaltenNummer2Buchstabe(spalte) + "$" + (zeile + 1);
252 CellAddress refAddr = new CellAddress();

254 if (namedRanges.hasByName(scoreSchluessel)) {

256 if (existing != null) {
257 existing.setContent(inhalt);
258 return;
259 }
260 }
261 namedRanges.addNewByName(scoreSchluessel, inhalt, refAddr, 0);
262 } catch (Exception e) {
263 LogUtil.error(logger, "Score-Zellen-Metadaten schreiben fehlgeschlagen für '" + scoreSchluessel + "'", e);
264 } catch (Error e) {

277 public static String leseScoreText(XSpreadsheetDocument xDoc, String scoreSchluessel) {

287 } catch (Exception e) {

300 private static Optional<CellRangeAddress> aufloeseNamedRangeAddresse(XSpreadsheetDocument xDoc,
301 String scoreSchluessel) throws Exception {
302 XNamedRanges namedRanges = namedRangesAusDoc(xDoc);
303 if (namedRanges == null || !namedRanges.hasByName(scoreSchluessel)) return Optional.empty();
304 XNamedRange range = Lo.qi(XNamedRange.class, namedRanges.getByName(scoreSchluessel));
305 if (range == null) return Optional.empty();
306 if (istKaputteReferenz(range.getContent())) return Optional.empty();
307 XCellRangeReferrer referrer = Lo.qi(XCellRangeReferrer.class, range);
308 if (referrer == null) return Optional.empty();
309 XCellRangeAddressable addrAble = Lo.qi(XCellRangeAddressable.class, referrer.getReferredCells());
310 if (addrAble == null) return Optional.empty();
311 return Optional.of(addrAble.getRangeAddress());

327 private static String spaltenNummer2Buchstabe(int spalte) {
328 var sb = new StringBuilder();
329 int n = spalte + 1;
330 while (n > 0) {
331 n--;
332 sb.insert(0, (char) ('A' + (n % 26)));
333 n /= 26;
334 }
335 return sb.toString();
336 }

436 private static void entferneFremdeIdentitaetsSchluessel(XNamedRanges namedRanges,
437 int sheetIdx, String behalteSchluessel) {
438 if (namedRanges == null || sheetIdx < 0) {
439 return;
440 }
441 String[] namen = namedRanges.getElementNames();
442 if (namen == null) {
443 return;
444 }
445 List<String> zuEntfernen = new ArrayList<>();
446 for (String name : namen) {
447 if (!name.startsWith("__PTM_") || name.startsWith("__PTM_SCORE_") || name.equals(behalteSchluessel)) {
448 continue;
449 }
450 try {
451 int idx = sheetIndexAusNamedRangeObj(namedRanges.getByName(name));
452 if (idx >= 0 && idx == sheetIdx) {
453 zuEntfernen.add(name);
454 }
455 } catch (Exception e) {
456 LogUtil.warn(logger, "Fremdschlüssel-Prüfung fehlgeschlagen für '" + name + "'", e);
457 } catch (Error e) {
458 throw e;
459 }
460 }
461 for (String name : zuEntfernen) {
462 try {
463 namedRanges.removeByName(name);
464 logger.debug("Fremden Identitäts-Schlüssel '{}' von Blatt-Index {} entfernt (Blatt-Eindeutigkeit).",
465 name, sheetIdx);
466 } catch (Exception e) {
467 LogUtil.warn(logger, "Entfernen des Fremdschlüssels '" + name + "' fehlgeschlagen", e);
468 } catch (Error e) {
469 throw e;
470 }
471 }

545 public static XSpreadsheet findeSheetUndHeile(XSpreadsheetDocument xDoc,
546 String schluessel, String fallbackName) {
547 // 1. Metadaten-Lookup (überlebt Umbenennungen)
548 Optional<XSpreadsheet> found = findeSheet(xDoc, schluessel);
549 if (found.isPresent()) return found.get();
550
551 // 2. Name-Fallback für alte Dateien ohne Metadaten
552 try {
553 XSpreadsheets sheets = xDoc.getSheets();
554 if (!sheets.hasByName(fallbackName)) return null;
555 XSpreadsheet sheet = Lo.qi(XSpreadsheet.class, sheets.getByName(fallbackName));
556 if (sheet == null) return null;
557
558 // 3. Datei heilen: Metadaten nachschreiben damit zukünftige Lookups direkt greifen
559 schreibeSheetMetadaten(xDoc, sheet, schluessel);
560 logger.debug("Sheet '{}' geheilt: Metadaten-Schlüssel '{}' nachgetragen", fallbackName, schluessel);
561 return sheet;
562 } catch (Exception e) {
563 LogUtil.warn(logger, "Fallback-Lookup für Sheet '" + fallbackName + "' fehlgeschlagen", e);
564 return null;
565 } catch (Error e) {
566 throw e;
567 }
