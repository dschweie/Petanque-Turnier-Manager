109 @Override
110 protected void doRun() throws GenerateException {
111 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.MAASTRICHTER)
112 .prefix(getLogPrefix()).validate()) {
113 return;
114 }
115
116 // 1. Meldeliste erstellen (löscht alle vorhandenen Sheets)
117 meldelisteTestDaten.erstelleTestdaten();
118
119 // Konfiguration setzen (überschreibt ggf. die Defaults der Testdaten-Klasse)
120 MaastrichterKonfigurationSheet konfigSheet = new MaastrichterKonfigurationSheet(getWorkingSpreadsheet());
121 konfigSheet.setSpielrundeSpielbahn(SpielrundeSpielbahn.R);
122 konfigSheet.setAnzVorrunden(anzVorrunden);
123 konfigSheet.setGruppenGroesse(gruppenGroesse);
124 konfigSheet.setMaastrichterGruppenModus(MaastrichterGruppenModus.NACH_GROESSE);
125
126 // 2. Vorrunden erstellen und mit Zufallsergebnissen füllen
127 for (int runde = 1; runde <= anzVorrunden; runde++) {
128 SheetRunner.testDoCancelTask();
129 processBoxinfo("processbox.erstelle.vorrunde", runde, anzVorrunden);
130 naechsteVorrunde.erstelleNaechsteVorrunde();
131
132 String legacyName = runde + ". " + SheetNamen.LEGACY_MAASTRICHTER_VORRUNDE_PRAEFIX;
133 XSpreadsheet sheet = SheetMetadataHelper.findeSheetUndHeile(
134 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
135 SheetMetadataHelper.schluesselMaastrichterVorrunde(runde), legacyName);
136 if (sheet != null) {
137 ergebnisseEinfuegen(sheet);
138 }
139 }
140
141 // 3. Vorrunden-Rangliste erstellen
142 SheetRunner.testDoCancelTask();
143 processBoxinfo("processbox.erstelle.rangliste");
144 ranglisteSheet.doRun();
145
146 // 4. Finalrunden erstellen
147 SheetRunner.testDoCancelTask();
148 processBoxinfo("processbox.erstelle.finalrunde");
149 finalrundeSheet.doRun();
150
151 // 5. Teilnehmerliste erstellen
152 SheetRunner.testDoCancelTask();
153 processBoxinfo("processbox.erstelle.teilnehmerliste");
154 teilnehmerSheet.generate();
155
156 // 6. Kopfzeile und Werbefußzeile setzen
157 konfigSheet.setKopfZeileMitte(getTurnierSystem().getBezeichnung());
158 konfigSheet.seitenstileAktualisieren();
159 }

164 private void ergebnisseEinfuegen(XSpreadsheet sheet) throws GenerateException {
165 RangePosition readRange = RangePosition.from(
166 SchweizerAbstractSpielrundeSheet.TEAM_A_SPALTE,
167 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE,
168 SchweizerAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE,
169 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE + 100);
170
171 RangeData data = RangeHelper
172 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
173 .getDataFromRange();
174
175 for (int i = 0; i < data.size(); i++) {
176 RowData row = data.get(i);
177 if (row.size() < 2) break;
178
179 int nrA = row.get(0).getIntVal(0);
180 if (nrA <= 0) break;
181 int nrB = row.get(1).getIntVal(0);
182 if (nrB <= 0) continue; // Freilos – kein Ergebnis nötig
183
184 int zeile = SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE + i;
185 int winner = RandomSource.nextInt(2);
186 int loserPts = RandomSource.nextInt(0, 13);
187 int ergA = (winner == 0) ? 13 : loserPts;
188 int ergB = (winner == 0) ? loserPts : 13;
189
190 getSheetHelper().setNumberValueInCell(NumberCellValue
191 .from(sheet, Position.from(SchweizerAbstractSpielrundeSheet.ERG_TEAM_A_SPALTE, zeile))
192 .setValue(ergA));
193 getSheetHelper().setNumberValueInCell(NumberCellValue
194 .from(sheet, Position.from(SchweizerAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE, zeile))
195 .setValue(ergB));
196 }
197 }
