122 static Path pdfZieldatei(String sheetName, Path zielVerzeichnis, String location) throws GenerateException {
123 String fallback = sheetName + ".pdf";
124 if (StringUtils.isBlank(location)) {
125 return zielVerzeichnis.resolve(fallback);
126 }
127 try {
128 Path dateiname = Path.of(URI.create(location).toURL().toURI()).getFileName();
129 if (dateiname == null) {
130 return zielVerzeichnis.resolve(fallback);
131 }
132 String basisName = FilenameUtils.removeExtension(dateiname.toString());
133 return zielVerzeichnis.resolve(sheetName + "_" + basisName + ".pdf");
134 } catch (IllegalArgumentException | MalformedURLException | URISyntaxException e) {
135 throw new GenerateException(e.getMessage());
136 }

169 static TurnierlogoExport bereiteTurnierlogoVor(Path zielVerzeichnis, String logoUrl)
170 throws IOException, GenerateException {
171 String quelle = StringUtils.stripToEmpty(logoUrl);
172 if (StringUtils.isBlank(quelle) || istUnveraenderteLogoQuelle(quelle)) {
173 return new TurnierlogoExport(quelle, Optional.empty());
174 }
175
176 Path logoDatei = lokalerLogoPfad(quelle);
177 if (!Files.isRegularFile(logoDatei) || !Files.isReadable(logoDatei)) {
178 throw new GenerateException(I18n.get("export.fehler.turnierlogo.nicht.lesbar", logoDatei));
179 }
180
181 Path zielDatei = zielVerzeichnis.resolve("turnier-logo" + logoEndung(logoDatei));
182 Files.copy(logoDatei, zielDatei, StandardCopyOption.REPLACE_EXISTING);
183 return new TurnierlogoExport(zielDatei.getFileName().toString(), Optional.of(zielDatei));
184 }

186 private static boolean istUnveraenderteLogoQuelle(String quelle) {
187 String lower = quelle.toLowerCase(Locale.ROOT);
188 return lower.startsWith("http://") || lower.startsWith("https://") || lower.startsWith("data:");
189 }

191 private static Path lokalerLogoPfad(String quelle) throws GenerateException {
192 try {
193 URI uri = URI.create(quelle);
194 if ("file".equalsIgnoreCase(uri.getScheme())) {
195 return Path.of(uri);
196 }
197 } catch (IllegalArgumentException e) {
198 // Kein URI, sondern normaler lokaler Pfad.
199 }
200 return Path.of(quelle);
201 }

256 protected HtmlExportErgebnis exportiereHtmlMitMeldelisteDruckbereich(
257 boolean meldelisteExportieren, String meldelisteSheetName,
258 Path zielVerzeichnis, String fallbackDateiname, String titel, String logoUrl,
259 List<ExportHtmlSeite.Section> sections) throws GenerateException {
260
261 if (!meldelisteExportieren || StringUtils.isBlank(meldelisteSheetName)) {
262 return exportiereHtml(zielVerzeichnis, fallbackDateiname, titel, logoUrl, sections);
263 }
264
265 var sheet = getSheetHelper().findByName(meldelisteSheetName);
266 if (sheet == null) {
267 return exportiereHtml(zielVerzeichnis, fallbackDateiname, titel, logoUrl, sections);
268 }
269 var printAreas = Lo.qi(XPrintAreas.class, sheet);
270 if (printAreas == null) {
271 return exportiereHtml(zielVerzeichnis, fallbackDateiname, titel, logoUrl, sections);
272 }
273 var originalBereiche = printAreas.getPrintAreas();
274 if (originalBereiche == null || originalBereiche.length == 0) {
275 return exportiereHtml(zielVerzeichnis, fallbackDateiname, titel, logoUrl, sections);
276 }
277
278 var original = originalBereiche[0];
279 var neuerBereich = new CellRangeAddress();
280 neuerBereich.Sheet = original.Sheet;
281 neuerBereich.StartRow = MeldeListeKonstanten.ZWEITE_HEADER_ZEILE;
282 neuerBereich.StartColumn = original.StartColumn;
283 neuerBereich.EndRow = letzteZeileMitDaten(sheet, original.EndRow);
284 neuerBereich.EndColumn = original.EndColumn;
285 printAreas.setPrintAreas(new CellRangeAddress[] { neuerBereich });
286 try {
287 return exportiereHtml(zielVerzeichnis, fallbackDateiname, titel, logoUrl, sections);
288 } finally {
289 printAreas.setPrintAreas(originalBereiche);
290 }
291 }

293 private int letzteZeileMitDaten(XSpreadsheet sheet, int maxZeile) {
294 for (int zeile = maxZeile; zeile >= MeldeListeKonstanten.ERSTE_DATEN_ZEILE; zeile--) {
295 try {
296 var zelle = sheet.getCellByPosition(MeldeListeKonstanten.SPIELER_NR_SPALTE, zeile);
297 if (zelle.getType() == CellContentType.VALUE && zelle.getValue() > 0) {
298 return zeile;
299 }
300 } catch (Exception e) {
301 logger.debug("Fehler beim Prüfen von Zeile {}", zeile, e);
302 }
303 }
304 return MeldeListeKonstanten.ZWEITE_HEADER_ZEILE;
305 }
