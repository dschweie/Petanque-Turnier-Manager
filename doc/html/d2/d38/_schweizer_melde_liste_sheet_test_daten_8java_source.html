118 @Override
119 public void doRun() throws GenerateException {
120 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.SCHWEIZER)
121 .prefix(getLogPrefix()).validate()) {
122 return;
123 }
124
125 getSheetHelper().removeAllSheetsExclude();
126 setAktiveSpielRunde(SpielRundeNr.from(1));
127
128 // Meldeliste mit Standardparametern anlegen (kein Dialog)
129 meldeListe.createMeldelisteWithParams(TEST_FORMATION, true, true);
130
131 testNamenEinfuegen();
132
133 // Teamnummern vergeben und Meldeliste aktualisieren
134 new SchweizerMeldeListeSheetUpdate(getWorkingSpreadsheet()).doRun();
135 }

137 private void testNamenEinfuegen() throws GenerateException {
138 int anzSpielerProTeam = getKonfigurationSheet().getMeldeListeFormation().getAnzSpieler();
139 boolean teamnameAktiv = getKonfigurationSheet().isMeldeListeTeamnameAnzeigen();
140 boolean vereinsnameAktiv = getKonfigurationSheet().isMeldeListeVereinsnameAnzeigen();
141 var spieler = testnamenLoader.listeMitSpielerTestNamen(anzTeams * anzSpielerProTeam);
142
143 var data = new RangeData();
144 for (int team = 0; team < anzTeams; team++) {
145 testDoCancelTask();
146 var zeile = data.addNewRow();
147 if (teamnameAktiv) {
148 zeile.newString("Team " + (team + 1));
149 }
150 for (int s = 0; s < anzSpielerProTeam; s++) {
151 var stn = spieler.get(team * anzSpielerProTeam + s);
152 zeile.newString(stn.vorname());
153 zeile.newString(stn.nachname());
154 if (vereinsnameAktiv) {
155 zeile.newString("Verein " + ((team % 5) + 1));
156 }
157 }
158 zeile.newEmpty(); // Setzposition
159 zeile.newInt(SchweizerListeDelegate.AKTIV_WERT_NIMMT_TEIL);
160 }
161
162 var xSheet = meldeListe.getXSpreadSheet();
163 var startPos = Position.from(1, ERSTE_DATEN_ZEILE);
164 RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
165 data.getRangePosition(startPos)).setDataInRange(data);
166
167 meldeListe.upDateSheet();
168 }
