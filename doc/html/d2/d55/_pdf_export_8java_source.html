117 public URI doExport() throws GenerateException {
118 URI pdfFile = null;
119
120 if (!istGespeichert() && zielVerzeichnis == null) {
121 String errMsg = I18n.get("error.dokument.nicht.gespeichert");
122 logger.warn(errMsg);
123 throw new GenerateException(errMsg);
124 }
125
126 try {
127 String newFileName = istGespeichert()
128 ? newFileName(null)
129 : newFileName(null, ungespeicherterBasisDateiname());
130 newFileName = FilenameUtils.removeExtension(newFileName);
131 pdfFile = zielVerzeichnis != null
132 ? zielVerzeichnis.resolve(newFileName + ".pdf").toUri()
133 : newLocationInSameDir(newFileName + ".pdf");
134
135 logger.info("Erstelle PDF :" + pdfFile);
136 Map<String, Object> filterData = new HashMap<>();
137
138 Map<String, Object> saveprops = new HashMap<>();
139 saveprops.put(SAVE_PROP_OVERWRITE, Boolean.TRUE);
140 saveprops.put(SAVE_PROP_FILTER_NAME, CALC_PDF_EXPORT_FILTER);
141
142 filterData.put(SAVE_PROP_FILTER_EXPORTBOOKMARKS, Boolean.FALSE);
143 filterData.put(SAVE_PROP_FILTER_EXPORTNOTES, Boolean.FALSE);
144
145 if (rangePosition != null) {
146 filterData.put(SAVE_PROP_FILTER_SELECTION, selectRangetoExport());
147 } else if (sheetName != null) {
148 var xSheet = this.getSheetHelper().findByName(sheetName);
149 if (xSheet != null) {
150 filterData.put(SAVE_PROP_FILTER_SELECTION, sheetSelektion(xSheet));
151 } else {
152 throw new GenerateException(I18n.get("error.tabelle.nicht.vorhanden", sheetName));
153 }
154 }
155
156 if (!filterData.isEmpty()) {
157 saveprops.put(SAVE_PROP_FILTERDATA, PropertyValueHelper.map2Proplist(filterData));
158 }
159
160 // stores the object's persistent data to a URL and continues to be a representation of the old URL
161 getxStorable().storeToURL(pdfFile.toString(), PropertyValueHelper.map2Proplist(saveprops));
162
163 } catch (MalformedURLException | URISyntaxException | IOException e) {
164 logger.error(e.getMessage(), e);
165 }
166
167 return pdfFile;
168 }
