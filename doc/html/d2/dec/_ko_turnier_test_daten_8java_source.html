70 @Override
71 protected void doRun() throws GenerateException {
72 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.KO)
73 .prefix(getLogPrefix()).validate()) {
74 return;
75 }
76
77 // 1. Konfiguration
78 var konfig = getKonfigurationSheet();
79 konfig.setSpielbaumSpielbahn(SpielrundeSpielbahn.N);
80 // Bei 16 Teams: 2 Gruppen à 8 erstellen (demonstriert Gruppen-Feature)
81 if (anzTeams == 16) {
82 konfig.setGruppenGroesse(8);
83 }
84
85 // 2. Meldeliste mit Testdaten füllen
86 meldeListeTestDaten.erstelleMeldelisteWithTestdaten();
87
88 // 3. Turnierbaum ohne Dialog erstellen
89 turnierbaumSheet.erstelleTurnierbaumOhneDialog();
90
91 // 4. Runde-1 + Cadrage-Ergebnisse in jeden Gruppen-Turnierbaum eintragen.
92 // Folgerunden lösen sich danach automatisch via WENN-Sieger-Formeln auf.
93 ergebnisseInAlleTurnierbaeumeEintragen();
94
95 // 5. Kopfzeile und Werbefußzeile setzen
96 konfig.setKopfZeileMitte(getTurnierSystem().getBezeichnung());
97 konfig.seitenstileAktualisieren();
98 }
99
105 private void ergebnisseInAlleTurnierbaeumeEintragen() throws GenerateException {
106 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
107 String praefix = SheetNamen.koTurnierbaumEinzel();
108 for (String sheetName : xDoc.getSheets().getElementNames()) {
109 if (sheetName.startsWith(praefix)) {
110 turnierbaumSheet.schreibeRunde1TestErgebnisse(sheetName);
111 }
112 }
113 XCalculatable xCal = Lo.qi(XCalculatable.class, xDoc);
114 if (xCal != null) {
115 xCal.calculateAll();
116 }
117 // Nach Propagation der Runde-1-Sieger in die Folgerunden zusätzlich Scores für
118 // Halbfinale, Finale etc. eintragen, damit der Bracket vollständig "gespielt" wird.
119 for (String sheetName : xDoc.getSheets().getElementNames()) {
120 if (sheetName.startsWith(praefix)) {
121 turnierbaumSheet.schreibeFolgerundenTestErgebnisse(sheetName);
122 }
123 }
124 if (xCal != null) {
125 xCal.calculateAll();
126 }
127 }
