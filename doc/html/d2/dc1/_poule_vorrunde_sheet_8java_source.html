50 @Override
51 public void doRun() throws GenerateException {
52 processBoxinfo("processbox.poule.vorrunde.erstellen");
53 meldeliste.upDateSheet();
54
55 var meldungen = meldeliste.getAktiveMeldungen();
56 int anzTeams = meldungen.size();
57
58 if (anzTeams < 3) {
59 MessageBox.from(getWorkingSpreadsheet(), MessageBoxTypeEnum.ERROR_OK)
60 .caption(I18n.get("msg.caption.fehler"))
61 .message(I18n.get("poule.vorrunde.fehler.zu.wenige.meldungen", anzTeams)).show();
62 return;
63 }
64
65 var gruppenGroessen = PouleGruppenRechner.berechneGruppenGroessen(anzTeams);
66 var poules = PouleSeedingService.verteileTeams(meldungen, gruppenGroessen);
67
68 NewSheet.from(this, SheetNamen.pouleVorrunde(), SheetMetadataHelper.SCHLUESSEL_POULE_VORRUNDE)
69 .tabColor(getKonfigurationSheet().getPouleVorrundeTabFarbe()).pos(DefaultSheetPos.POULE_WORK)
70 .forceCreate().hideGrid().create();
71
72 var xSheet = getXSpreadSheet();
73
74 headerSchreiben(xSheet);
75 spaltenBreitenSetzen(xSheet);
76
77 int aktuelleZeile = ERSTE_DATEN_ZEILE;
78 var trennzeilenNachBlock = new ArrayList<Integer>();
79
80 for (int i = 0; i < poules.size(); i++) {
81 SheetRunner.testDoCancelTask();
82 var poule = poules.get(i);
83 if (poule.teams().size() == 4) {
84 schreibeViererPoule(xSheet, poule, aktuelleZeile, aktuelleZeile);
85 aktuelleZeile += VIERER_POULE_ZEILEN;
86 } else {
87 schreibeDreierPoule(xSheet, poule, aktuelleZeile, aktuelleZeile);
88 aktuelleZeile += DREIER_POULE_ZEILEN;
89 }
90 if (i < poules.size() - 1) {
91 trennzeilenNachBlock.add(aktuelleZeile - 1);
92 }
93 }
94
95 int letzteDatenZeile = aktuelleZeile - 1;
96 formatierungDurchfuehren(xSheet, letzteDatenZeile);
97
98 for (var trennzeile : trennzeilenNachBlock) {
99 schreibePouleBlockTrenner(xSheet, trennzeile);
100 }
101
102 printBereichSetzen(xSheet, letzteDatenZeile);
103
104 if (SheetRunner.isRunning()) {
105 SheetFreeze.from(xSheet, getWorkingSpreadsheet()).anzZeilen(ERSTE_DATEN_ZEILE).doFreeze();
106 }
107 }
