131 public void generate(TeamMeldungen meldungen) throws GenerateException {
132 if (!meldungen.isValid()) {
133 processBoxinfo("processbox.abbruch");
134 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
135 .caption(I18n.get("msg.caption.liga.spielplan"))
136 .message(I18n.get("msg.text.ungueltige.anzahl.meldungen")).show();
137 return;
138 }
139
140 if (!NewSheet.from(this, sheetName(), METADATA_SCHLUESSEL)
141 .pos(DefaultSheetPos.LIGA_WORK).setActiv().hideGrid()
142 .tabColor(konfigurationSheet.getSpielrundeTabFarbe()).create().isDidCreate()) {
143 ProcessBox.from().info("Abbruch vom Benutzer, Liga SpielPlan wurde nicht erstellt");
144 return;
145 }
146
147 LigaSpielPlan ligaSpielPlan = new LigaSpielPlan(meldungen);
148 // nur einmal schufflePlan ! damit beim wechsel von hr nach rr keine 2 x hintereinander
149 List<List<TeamPaarung>> spielPlanHRunde = ligaSpielPlan.schufflePlan().getSpielPlanClone();
150 List<List<TeamPaarung>> spielPlanRRunde = ligaSpielPlan.flipTeams().getSpielPlanClone();
151
152 getxCalculatable().enableAutomaticCalculation(false);
153 try {
154 insertDatenHeaderUndSpalteBreite();
155 insertArbeitsspalten(spielPlanHRunde, spielPlanRRunde);
156 insertSpieltageDaten(spielPlanHRunde);
157 insertFormulaTeamNamen();
158 insertFormulaPunkte();
159 insertDatumFormulaUndFormatierungen();
160 formatieren(spielPlanHRunde);
161 printBereichDefinieren();
162 SheetFreeze.from(getTurnierSheet()).anzZeilen(2).doFreeze();
163 } finally {
164 getxCalculatable().enableAutomaticCalculation(true);
165 }
166 }

358 private void insertArbeitsspalten(List<List<TeamPaarung>> spielPlanHRunde, List<List<TeamPaarung>> spielPlanRRunde)
359 throws GenerateException {
360 RangeData rangeData = new RangeData();
361
362 List<List<TeamPaarung>> alleSpieltage = new ArrayList<>();
363 alleSpieltage.addAll(spielPlanHRunde);
364 alleSpieltage.addAll(spielPlanRRunde);
365
366 // hr und rr runde
367 for (List<TeamPaarung> spielTag : alleSpieltage) {
368 for (TeamPaarung teamPaarung : spielTag) {
369 SheetRunner.testDoCancelTask();
370 RowData teamPaarungData = rangeData.addNewRow();
371 teamPaarungData.newInt(teamPaarung.getA().getNr());
372 teamPaarungData.newInt(teamPaarung.getOptionalB().isPresent() ? teamPaarung.getB().getNr() : 0);
373 }
374 }
375 Position startPos = Position.from(TEAM_A_NR_SPALTE, ERSTE_SPIELTAG_DATEN_ZEILE);
376 RangeHelper.from(this, rangeData.getRangePosition(startPos)).setDataInRange(rangeData).setRangeProperties(
377 RangeProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().toBorder()));
378
379 ColumnProperties spalteBreite = ColumnProperties.from().setWidth(MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH)
380 .isVisible(false);
381 XSpreadsheet sheet = getXSpreadSheet();
382 getSheetHelper().setColumnProperties(sheet, TEAM_A_NR_SPALTE, spalteBreite);
383 getSheetHelper().setColumnProperties(sheet, TEAM_B_NR_SPALTE, spalteBreite);
384
385 // SPIELPNKT-Zellen für Freispiel-Reihen vorbelegen
386 int freispielPlus = getKonfigurationSheet().getFreispielPunktePlus();
387 int freispielMinus = getKonfigurationSheet().getFreispielPunkteMinus();
388 int zeile = ERSTE_SPIELTAG_DATEN_ZEILE;
389 for (List<TeamPaarung> spielTag : alleSpieltage) {
390 for (TeamPaarung teamPaarung : spielTag) {
391 if (!teamPaarung.getOptionalB().isPresent()) {
392 getSheetHelper().setNumberValueInCell(
393 NumberCellValue.from(sheet, SPIELPNKT_A_SPALTE, zeile, freispielPlus));
394 getSheetHelper().setNumberValueInCell(
395 NumberCellValue.from(sheet, SPIELPNKT_B_SPALTE, zeile, freispielMinus));
396 }
397 zeile++;
398 }
399 }
