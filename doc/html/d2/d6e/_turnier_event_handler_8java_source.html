57 public void trigger(TurnierEventType type, ITurnierEvent eventObj) {

67 private void onPropertiesChanged(ITurnierEvent eventObj) {
68 if (SheetRunner.isRunning()) {
69 pending.set(eventObj);
70 coalescedSeitFlush.incrementAndGet();
71 return;
72 }
73 dispatch(eventObj, "trigger");
74 }

81 public void flushPending() {
82 ITurnierEvent ev = pending.getAndSet(null);
83 int coalesced = coalescedSeitFlush.getAndSet(0);
84 if (ev != null) {
85 PerfLog.log(logger, "[WORKER-TIMING] TurnierEventHandler.flushPending koalesziert={}",
86 coalesced);
87 dispatch(ev, "flushPending");
88 }
89 }

91 private void dispatch(ITurnierEvent eventObj, String quelle) {
92 XRequestCallback d = getDispatcher();
93 if (d == null) {
94 broadcastSync(eventObj, quelle, true);
95 return;
96 }
97 try {
98 d.addCallback((XCallback) data -> broadcastSync(eventObj, quelle, false), null);
99 } catch (RuntimeException e) {
100 logger.debug("AsyncCallback-Post fehlgeschlagen – synchroner Fallback", e);
101 broadcastSync(eventObj, quelle, true);
102 }
103 }
104
105 private void broadcastSync(ITurnierEvent eventObj, String quelle, boolean synchronerPfad) {
106 long startNs = System.nanoTime();
107 int anzahl;
108 String eventDoc = eventObj == null ? "null"
109 : de.petanqueturniermanager.comp.ProtocolHandler.beschreibeDokument(
110 eventObj.getWorkingSpreadsheetDocument());
111 synchronized (listeners) {
112 anzahl = listeners.size();
113 logger.trace("[FOKUS-TRACE] TurnierEvent broadcast START quelle={} pfad={} eventDoc={} listeners={} thread={}",
114 quelle, synchronerPfad ? "sync" : "main", eventDoc, anzahl,
115 Thread.currentThread().getName());
116 for (ITurnierEventListener listner : listeners) {
117 try {
118 listner.onPropertiesChanged(eventObj);
119 } catch (RuntimeException e) {
120 logger.warn("Listener-Aufruf fehlgeschlagen", e);
121 }
122 }
123 }
124 long dauerMs = (System.nanoTime() - startNs) / 1_000_000L;
125 logger.trace("[FOKUS-TRACE] TurnierEvent broadcast ENDE quelle={} dauerMs={} listener={}",
126 quelle, dauerMs, anzahl);
127 PerfLog.log(logger, "[WORKER-TIMING] TurnierEventHandler broadcast PropertiesChanged quelle={} pfad={} : {} ms, listener={}, thread={}",
128 quelle, synchronerPfad ? "sync" : "main", dauerMs, anzahl,
129 Thread.currentThread().getName());
130 }
131
132 private XRequestCallback getDispatcher() {
133 XRequestCallback d = dispatcher;
134 if (d != null) {
135 return d;
136 }
137 XComponentContext ctx = PetanqueTurnierMngrSingleton.getContext();
138 if (ctx == null) {
139 return null;
140 }
141 synchronized (this) {
142 if (dispatcher != null) {
143 return dispatcher;
144 }
145 try {
146 Object asyncCallback = ctx.getServiceManager()
147 .createInstanceWithContext("com.sun.star.awt.AsyncCallback", ctx);
148 dispatcher = Lo.qi(XRequestCallback.class, asyncCallback);
149 } catch (com.sun.star.uno.Exception e) {
150 logger.warn("AsyncCallback-Service nicht verfügbar – Fallback synchron", e);
151 } catch (DisposedException e) {
152 // LO-Bridge tot (z.B. nach Office-Shutdown zwischen UI-Test-Klassen).
153 // Synchroner Fallback. sharedContext bleibt stale; das ist Aufruferproblem.
154 logger.debug("LO-Bridge disposed – Fallback synchron");
155 }
156 return dispatcher;
157 }
158 }
