183 @Override
184 protected void doRun() throws GenerateException {
185 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.JGJ)
186 .prefix(getLogPrefix()).validate()) {
187 return;
188 }
189
190 // Dialog zuerst – bei Abbruch keine Änderungen am Dokument
191 Optional<JGJStartDialog.StartParameter> param;
192 try {
193 param = JGJStartDialog.from(getWorkingSpreadsheet()).show();
194 } catch (com.sun.star.uno.Exception e) {
195 String errMsg = I18n.get("error.dialog.startdialog", e.getMessage());
196 logger.error(errMsg, e);
197 throw new GenerateException(errMsg);
198 }
199
200 if (param.isEmpty()) {
201 return; // Benutzer hat abgebrochen – keine Dokument-Änderungen
202 }
203
204 // Erst nach Bestätigung: TurnierSystem + Page Styles setzen
205 getKonfigurationSheet().update();
206
207 getSheetHelper().removeAllSheetsExclude();
208 createMeldelisteWithParams(param.get().formation(),
209 param.get().teamnameAnzeigen(), param.get().vereinsnameAnzeigen(),
210 param.get().spielplanTeamAnzeige(), param.get().gruppengroesse(), param.get().mitRueckrunde());
