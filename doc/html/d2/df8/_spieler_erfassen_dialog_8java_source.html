157 private short[] indizesFuer(List<Integer> labelNrs) {
158 Set<Integer> gesucht = new HashSet<>(labelNrs);
159 List<Short> indizes = new ArrayList<>();
160 for (int i = 0; i < alleLabels.size(); i++) {
161 Integer nr = alleLabels.get(i).nr();
162 if (nr != null && gesucht.contains(nr)) {
163 indizes.add((short) i);
164 }
165 }
166 short[] out = new short[indizes.size()];
167 for (int i = 0; i < out.length; i++) {
168 out[i] = indizes.get(i);
169 }
170 return out;
171 }

173 private List<Integer> labelNrsAusAuswahl(short[] indizes) {
174 List<Integer> nrs = new ArrayList<>(indizes.length);
175 for (short idx : indizes) {
176 if (idx >= 0 && idx < alleLabels.size()) {
177 Integer nr = alleLabels.get(idx).nr();
178 if (nr != null) {
179 nrs.add(nr);
180 }
181 }
182 }
183 return nrs;

186 private void beimOkKlick() {
187 UnoControlsHelper c = this.controls;
188 XDialog dlg = this.xDialog;
189 if (c == null || dlg == null) {
190 return;
191 }
192 String vorname = c.leseText("txtVorname").strip();
193 String nachname = c.leseText("txtNachname").strip();
194 String vereinEing = c.leseText("cbxVerein").strip();
195 List<Integer> labelNrs = labelNrsAusAuswahl(c.ausgewaehlteIndizes("lstLabels"));
196 String lizenz = c.leseText("txtLizenznr").strip();
197
198 if (vorname.isEmpty() || nachname.isEmpty()) {
199 zeigeFehler(I18n.get("spielerdb.fehler.pflichtfeld"));
200 return;
201 }
202
203 try {
204 Integer vereinNr = aufloeseOderAnlegenVerein(vereinEing).orElse(null);
205 if (vereinNr == null && !vereinEing.isEmpty()) {
206 // User hat Anlegen-Rückfrage abgelehnt — Eingabe bleibt, Dialog offen.
207 return;
208 }
209
210 if (istNamensduplikat(vorname, nachname, vereinNr)) {
211 return;
212 }
213
214 SpielerDatensatz neu = new SpielerDatensatz(
215 bearbeiten == null ? null : bearbeiten.nr(),
216 vorname, nachname, vereinNr, labelNrs,
217 lizenz.isEmpty() ? null : lizenz);
218
219 if (bearbeiten == null) {
220 ergebnis = spielerRepo.insert(neu);
221 } else {
222 spielerRepo.update(neu);
223 ergebnis = neu;
224 }
225 dlg.endExecute();
226 } catch (NameDuplikatException e) {
227 zeigeFehler(I18n.get("spielerdb.fehler.spieler_db_doppelt",
228 vorname, nachname, vereinEing));
229 } catch (LizenzDuplikatException e) {
230 zeigeFehler(I18n.get("spielerdb.fehler.lizenz_doppelt", lizenz));
231 } catch (SpielerDbException e) {
232 logger.error("Spieler speichern fehlgeschlagen", e);
233 zeigeFehler(I18n.get("spielerdb.fehler.dbinit", e.getMessage()));
234 }
235 }

244 private boolean istNamensduplikat(String vorname, String nachname, @Nullable Integer vereinNr)
245 throws SpielerDbException {
246 Integer ausserNr = bearbeiten == null ? null : bearbeiten.nr();
247 Optional<SpielerMitVerein> dup = spielerRepo.findeDuplikat(vorname, nachname, vereinNr, ausserNr);
248 if (dup.isEmpty()) {
249 return false;
250 }
251 SpielerMitVerein s = dup.get();
252 String vereinAnzeige = s.vereinName() == null ? "" : s.vereinName();
253 zeigeFehler(I18n.get("spielerdb.fehler.spieler_db_doppelt",
254 s.vorname(), s.nachname(), vereinAnzeige));
255 return true;
256 }

258 private Optional<Integer> aufloeseOderAnlegenVerein(String eingabe) throws SpielerDbException {
259 if (eingabe.isEmpty()) {
260 return Optional.empty();
261 }
262 Optional<VereinDatensatz> bestehend = vereinRepo.findByName(eingabe);
263 if (bestehend.isPresent()) {
264 return bestehend.get().nr() == null ? Optional.empty() : Optional.of(bestehend.get().nr());
265 }
266 // Rückfrage anlegen?
267 MessageBoxResult res = MessageBox.from(xContext, MessageBoxTypeEnum.QUESTION_YES_NO)
268 .caption(I18n.get("spielerdb.frage.titel"))
269 .message(I18n.get("spielerdb.frage.verein_anlegen", eingabe))
270 .show();
271 if (res != MessageBoxResult.YES) {
272 return Optional.empty();
273 }
274 try {
275 VereinDatensatz angelegt = vereinRepo.insert(eingabe);
276 return Optional.ofNullable(angelegt.nr());
277 } catch (DuplikatException e) {
278 // Race: zweiter Lookup, dann sicher ohne Rückfrage akzeptieren.
279 return vereinRepo.findByName(eingabe).map(VereinDatensatz::nr);
280 }
