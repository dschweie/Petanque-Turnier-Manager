123 private void registriereListener(Object source) {
124 try {
125 XModel xModel = Lo.qi(XModel.class, source);
126 if (xModel == null) return;
127 XSpreadsheetDocument xDoc = Lo.qi(XSpreadsheetDocument.class, xModel);
128 if (xDoc == null) return;
129
130 synchronized (registriert) {
131 if (registriert.contains(xDoc)) return;
132 registriert.add(xDoc);
133 }
134
135 registriereSelectionChangeListener(xModel, xDoc);

137 } catch (RuntimeException e) {
138 logger.error("Fehler beim Registrieren des Formatierer-Listeners", e);

142 private void registriereSelectionChangeListener(XModel xModel, XSpreadsheetDocument xDoc) {
143 var controller = xModel.getCurrentController();
144 if (controller == null) return;
145 if (Lo.qi(XSpreadsheetView.class, controller) == null) return;
146 XSelectionSupplier selSupplier = Lo.qi(XSelectionSupplier.class, controller);
147 if (selSupplier == null) return;
148
149 String initialesSheetToken = aktivesSheetToken(xModel, xDoc);
150 selSupplier.addSelectionChangeListener(new XSelectionChangeListener() {
151 private String letztesSheetToken = initialesSheetToken;
152
153 @Override
154 public void selectionChanged(EventObject e) {
155 try {
156 XSpreadsheetView view = Lo.qi(XSpreadsheetView.class, e.Source);
157 if (view == null) return;
158
159 XSpreadsheet aktuellesSheet = view.getActiveSheet();
160 if (aktuellesSheet == null) return;
161
162 String aktuellesSheetToken = fokusToken(xDoc, aktuellesSheet);
163 if (aktuellesSheetToken.equals(letztesSheetToken)) {
164 return;
165 }
166 letztesSheetToken = aktuellesSheetToken;
167
168 if (!zielSheetMatch.test(xDoc, aktuellesSheet)) return;
169 if (SheetRunner.isRunning()) return;
170
171 planeFormatierer(xDoc, aktuellesSheet);
172
173 } catch (RuntimeException ex) {
174 logger.error("Fehler im Formatierer-SelectionChangeListener", ex);
175 }
176 }

179 public void disposing(EventObject e) {

182 });

186 XSpreadsheetView view = Lo.qi(XSpreadsheetView.class, xModel.getCurrentController());
187 if (view == null) return null;

236 boolean merkeNeuenFokus(XSpreadsheetDocument xDoc, XSpreadsheet xSheet) {
237 String neuesToken = fokusToken(xDoc, xSheet);
238 String altesToken = letztesFokusToken;
239 if (neuesToken != null && neuesToken.equals(altesToken)) {
240 return false;
241 }
242 letztesFokusToken = neuesToken;
243 return true;
244 }

246 static String fokusToken(XSpreadsheetDocument xDoc, XSpreadsheet xSheet) {
247 if (xDoc == null || xSheet == null) return null;
248 String sheetName = "<unbekannt>";
249 try {
250 XNamed xNamed = Lo.qi(XNamed.class, xSheet);
251 if (xNamed != null && xNamed.getName() != null) {
252 sheetName = xNamed.getName();
253 }
254 } catch (RuntimeException e) {
255 logger.trace("Sheet-Name für Fokus-Token nicht ermittelbar", e);
256 }
257 return System.identityHashCode(xDoc) + ":" + sheetName;
258 }

260 private void planeFormatierer(XSpreadsheetDocument xDoc, XSpreadsheet xSheet) {
261 SheetSyncDebouncer.get().schedule(xDoc, debounceSchluessel, () -> {
262 try {
263 if (SheetRunner.isRunning()) return;
264 if (!istDokumentLebendig(xDoc)) return;
265 var ws = new WorkingSpreadsheet(xContext, xDoc);
266 formatiererFactory.apply(ws, xSheet).startSilent();
267 } catch (DisposedException e) {
268 logger.debug("Formatierer übersprungen: Dokument bereits geschlossen", e);
269 } catch (RuntimeException e) {
270 logger.error("Formatierer konnte nicht gestartet werden", e);
271 }
272 });
273 }

275 private static boolean istDokumentLebendig(XSpreadsheetDocument xDoc) {

277 try {
278 xDoc.getSheets();
279 return true;

281 logger.debug("Formatierer übersprungen: Dokument bereits geschlossen", e);
282 return false;
283 } catch (RuntimeException e) {
284 logger.debug("Formatierer übersprungen: Dokument nicht nutzbar", e);
285 return false;
