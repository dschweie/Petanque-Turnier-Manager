179 public static String[] sheetTypenFuer(TurnierSystem aktiv) {
180 if (aktiv == null || aktiv == KEIN) {
181 return SHEET_TYPEN.clone();
182 }
183 return RESOLVER_MAP.entrySet().stream()
184 .filter(e -> e.getValue().giltFuer(aktiv))
185 .map(Map.Entry::getKey)
186 .toArray(String[]::new);
187 }

195 public static SheetResolver erstellen(String configWert) {
196 var key = configWert.toUpperCase();
197 var aufgeloesterKey = LEGACY_ALIASE.getOrDefault(key, key);
198 var definition = RESOLVER_MAP.get(aufgeloesterKey);
199 if (definition != null) {
200 return definition.factory().get();
201 }
202 // Default: Exakter Sheet-Name
203 return new StaticSheetResolver(configWert);
204 }
