57 private int[] zaehleVorrundeSpiele(XSpreadsheet sheet) {
58 int gesamt = 0;
59 int gespielt = 0;
60 try {
61 for (int zeile = AbstractPouleVorrundeSheet.ERSTE_DATEN_ZEILE; zeile < MAX_ZEILEN; zeile++) {
62 var beschrZelle = sheet.getCellByPosition(AbstractPouleVorrundeSheet.SPALTE_BESCHR, zeile);
63 if (beschrZelle.getType() == CellContentType.EMPTY) {
64 break;
65 }
66 gesamt++;
67 var ergZelle = sheet.getCellByPosition(AbstractPouleVorrundeSheet.SPALTE_ERG_A, zeile);
68 if (ergZelle.getType() == CellContentType.VALUE) {
69 gespielt++;
70 }
71 }
72 } catch (IndexOutOfBoundsException e) {
73 logger.error("Fehler beim Lesen der Poule-Vorrunde-Spiele", e);
74 }
75 return new int[] { gespielt, gesamt };
76 }

82 private boolean pruefeKoBeendet(
83 com.sun.star.sheet.XSpreadsheetDocument xDoc, XSpreadsheet koSheet) {
84 var scoreKey = SheetMetadataHelper.scoreSchluessel(SheetMetadataHelper.schluesselPouleKo("A"));
85 var scoreText = SheetMetadataHelper.leseScoreText(xDoc, scoreKey);
86 if (scoreText == null || !scoreText.startsWith(SCORE_PRAEFIX)) {
87 return false;
88 }
89 var positionenStr = scoreText.substring(SCORE_PRAEFIX.length());
90 var positionen = positionenStr.split(POSITIONS_TRENNZEICHEN);
91 try {
92 for (var pos : positionen) {
93 if (pos.isBlank()) {
94 continue;
95 }
96 var teile = pos.split(KOORDINATEN_TRENNZEICHEN);
97 if (teile.length < 2) {
98 continue;
99 }
100 int spalte = Integer.parseInt(teile[0].trim());
101 int zeile = Integer.parseInt(teile[1].trim());
102 var zelle = koSheet.getCellByPosition(spalte, zeile);
103 if (zelle.getType() != CellContentType.VALUE) {
104 return false;
105 }
106 }
107 } catch (Exception e) {
108 logger.error("Fehler beim Prüfen der KO-Scores für Beendet-Erkennung", e);
109 return false;
110 }
111 return positionen.length > 0;
112 }
