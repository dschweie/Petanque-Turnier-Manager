38 Path datei = request.source();
39 if (!Files.isRegularFile(datei)) {
40 throw new SpielerDbException("JSON-Datei fehlt: " + datei);

43
44 JsonObject root = leseRoot(datei);
45
46 List<RohSpieler> spieler = entities.contains(ExportEntity.SPIELER)

48 : List.of();
49 List<RohVerein> vereine = entities.contains(ExportEntity.VEREINE)

51 : List.of();
52 List<RohLabel> labels = entities.contains(ExportEntity.LABELS)

54 : List.of();
55 List<RohSpielerLabel> junction =
56 entities.contains(ExportEntity.SPIELER) && entities.contains(ExportEntity.LABELS)
57 ? leseJunction(datei, holeArrayPflicht(datei, root, "spielerLabels"))
58 : List.of();
59
60 return new ImportRohdaten(spieler, vereine, labels, junction);

63 private static JsonObject leseRoot(Path datei) throws SpielerDbException {
64 try (Reader r = Files.newBufferedReader(datei, StandardCharsets.UTF_8)) {
65 JsonElement root = JsonParser.parseReader(r);
66 if (!root.isJsonObject()) {
67 throw new SpielerDbException("JSON-Wurzel in " + datei + " ist kein Objekt");
68 }
69 return root.getAsJsonObject();
70 } catch (IOException | JsonSyntaxException e) {
71 throw new SpielerDbException("JSON-Lesen fehlgeschlagen: " + datei, e);
72 }
73 }

75 private static JsonArray holeArrayPflicht(Path datei, JsonObject root, String key)
