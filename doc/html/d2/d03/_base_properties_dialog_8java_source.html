106 @Override
107 protected final void erstelleFelder(
108 XMultiComponentFactory mcf, XMultiServiceFactory xMSF,
109 XNameContainer cont, XToolkit xToolkit, XWindowPeer peer,
110 XPropertySet dlgProps, XDialog xDialog
111 ) throws com.sun.star.uno.Exception {
112
113 // Window-Adapter für Resize (xDialog ist dasselbe UNO-Objekt wie xControl)
114 XWindow xWindow = de.petanqueturniermanager.helper.Lo.qi(XWindow.class, xDialog);
115 if (xWindow != null) {
116 xWindow.addWindowListener(windowAdapter);
117 }
118
119 int margin = 2;
120 layout = new VerticalLayout(0, margin);
121
122 // Felder hinzufügen
123 GuiFactoryCreateParam guiFactoryCreateParam = new GuiFactoryCreateParam(
124 mcf, xContext, xToolkit, peer);
125 List<ConfigProperty<?>> konfigProperties =
126 KonfigurationSingleton.getKonfigProperties(currentSpreadsheet);
127
128 AddConfigElementsToWindow addConfigElementsToWindow =
129 new AddConfigElementsToWindow(guiFactoryCreateParam, currentSpreadsheet, layout);
130 AtomicInteger anzElementen = new AtomicInteger(0);
131 if (konfigProperties != null) {
132 var stream = konfigProperties.stream().filter(getKonfigFieldFilter());
133 var comparator = getKonfigFieldComparator();
134 if (comparator != null) {
135 stream = stream.sorted(comparator);
136 }
137 stream.forEach(konfigprop -> {
138 addConfigElementsToWindow.addPropToPanel(konfigprop);
139 anzElementen.addAndGet(1);
140 });
141 }
142
143 erstelleNachHauptFelder(guiFactoryCreateParam, addConfigElementsToWindow);
144
145 // Höhe anpassen
146 int dialogHeight = Math.min(Math.max(layout.getHeight() / 2, DIALOG_MIN_HEIGHT), DIALOG_MAX_HEIGHT);
147 try {
148 dlgProps.setPropertyValue("Height", Integer.valueOf(dialogHeight));
149 } catch (Exception e) {
150 logger.error(e.getMessage(), e);
151 }
152 }
