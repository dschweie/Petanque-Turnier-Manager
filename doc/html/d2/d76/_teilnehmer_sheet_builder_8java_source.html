107 public TeilnehmerSheetBuilder aufbauen() throws GenerateException {
108 int spaltenProBlock = teamnameAktiv ? 4 : 3; // Nr · [Teamname ·] Spieler · Leer
109 int datenSpaltenProBlock = teamnameAktiv ? 3 : 2;
110 int anzahlEintraege = daten.size();
111 int zeilenProBlock = Math.min(maxProBlock, Math.max(anzahlEintraege, 1));
112 int anzBloecke = Math.max(1, (anzahlEintraege + maxProBlock - 1) / maxProBlock);
113 // letzte Daten-Spalte: keine Leer-Trennspalte hinter dem letzten Block
114 int totalSpalten = (anzBloecke - 1) * spaltenProBlock + datenSpaltenProBlock - 1;
115
116 spaltenbreitenSetzen(anzBloecke, spaltenProBlock, datenSpaltenProBlock);
117 headerSchreiben(anzBloecke, spaltenProBlock);
118
119 if (anzahlEintraege == 0) {
120 this.letzteDatenZeile = HEADER_ZEILE;
121 this.letzteDatenSpalte = totalSpalten;
122 this.aufgebaut = true;
123 return this;
124 }
125
126 datenSchreiben(anzBloecke, spaltenProBlock, zeilenProBlock, totalSpalten);
127 optimaleBreiteSetzen(anzBloecke, spaltenProBlock);
128 datenBorderSetzen(anzBloecke, spaltenProBlock, zeilenProBlock);
129 zebraFarbenSetzen(anzBloecke, spaltenProBlock, zeilenProBlock);
130
131 this.letzteDatenZeile = ERSTE_DATEN_ZEILE + zeilenProBlock - 1;
132 this.letzteDatenSpalte = totalSpalten;
133 this.aufgebaut = true;
134 return this;
135 }

177 private void spaltenbreitenSetzen(int anzBloecke, int spaltenProBlock, int datenSpaltenProBlock)
178 throws GenerateException {
179 SheetHelper sheetHelper = sheet.getSheetHelper();
180 var xSheet = sheet.getXSpreadSheet();
181 ColumnProperties propNr = ColumnProperties.from().setHoriJustify(CellHoriJustify.CENTER).setWidth(NR_SPALTE_WIDTH);
182 ColumnProperties propTeamname = ColumnProperties.from().setHoriJustify(CellHoriJustify.LEFT);
183 ColumnProperties propSpieler = ColumnProperties.from().setHoriJustify(CellHoriJustify.LEFT);
184 ColumnProperties propLeer = ColumnProperties.from().setWidth(NR_SPALTE_WIDTH);
185
186 for (int b = 0; b < anzBloecke; b++) {
187 int base = b * spaltenProBlock;
188 sheetHelper.setColumnProperties(xSheet, base, propNr);
189 if (teamnameAktiv) {
190 sheetHelper.setColumnProperties(xSheet, base + 1, propTeamname);
191 sheetHelper.setColumnProperties(xSheet, base + 2, propSpieler);
192 } else {
193 sheetHelper.setColumnProperties(xSheet, base + 1, propSpieler);
194 }
195 if (b < anzBloecke - 1) {
196 sheetHelper.setColumnProperties(xSheet, base + datenSpaltenProBlock, propLeer);
197 }
198 }

264 row.newString(t.spielerNamen() != null ? t.spielerNamen() : "");
265 } else {
266 row.newEmpty();

268 row.newEmpty();
