56 @Override
57 protected void doRun() throws GenerateException {
58 ExtensionsHelper extensionsHelper = ExtensionsHelper.from(getxContext());
59 processBoxinfo("processbox.download.start");
60 processBoxinfo("processbox.installierte.version", extensionsHelper.getVersionNummer());
61
62 ReleaseInfo release;
63 try {
64 release = ReleaseUpdateService.get().getAktuellesRelease().orElse(null);
65 } catch (IllegalStateException e) {
66 release = null;
67 }
68 if (release == null) {
69 processBox().fehler("Keine Version verfügbar.");
70 return;
71 }
72
73 var oxtAsset = release.findeAsset(EXTENSION_FILE_PREFIX, EXTENSION_FILE_SUFFIX).orElse(null);
74 if (oxtAsset == null) {
75 processBox().fehler("keine " + EXTENSION_FILE_SUFFIX + " Datei zum Download vorhanden.");
76 return;
77 }
78 URL downloadURL;
79 try {
80 downloadURL = URI.create(oxtAsset.downloadUrl()).toURL();
81 } catch (IllegalArgumentException | java.net.MalformedURLException e) {
82 processBox().fehler("Ungültige Download-URL: " + e.getMessage());
83 return;
84 }
85
86 processBoxinfo("processbox.github.version", release.name());
87 processBox().info(downloadURL.toString());
88
89 try {
90 File selectedPath = zielVerzeichnis;
91 File targetFile = new File(selectedPath, oxtAsset.name());
92 if (targetFile.exists()) {
93 processBoxinfo("processbox.datei.bereits.vorhanden", targetFile);
94 MessageBoxResult answer = MessageBox
95 .from(getxContext(), MessageBoxTypeEnum.QUESTION_YES_NO)
96 .caption(I18n.get("msg.caption.datei.vorhanden"))
97 .message(I18n.get("msg.text.datei.vorhanden.ueberschreiben", targetFile))
98 .show();
99 if (answer == MessageBoxResult.NO) {
100 processBoxinfo("processbox.abbruch");
101 return;
102 }
103 processBoxinfo("processbox.ueberschreiben");
104 }
105
106 processBoxinfo("processbox.speichern.in", selectedPath.getPath());
107 if (selectedPath.canWrite()) {
108 FileUtils.copyURLToFile(downloadURL, targetFile, 10000, 10000);
109 } else {
110 processBox().fehler("keine Schreibrechte");
111 }
112 processBox().infoText("");
113 } catch (IOException e) {
114 logger.error(e);
115 processBox().fehler(e.getMessage());
116 }
117 }
