62 @Override
63 public void handle(HttpExchange exchange) throws IOException {
64 try (exchange) {
65 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
66 exchange.sendResponseHeaders(405, -1);
67 return;
68 }
69 String path = exchange.getRequestURI().getPath();
70 try {
71 if ("/api/stats".equals(path)) {
72 sendeJson(exchange, 200, baueStats());
73 } else if ("/api/vereine".equals(path)) {
74 sendeJson(exchange, 200, vereinRepo.findAll());
75 } else if ("/api/labels".equals(path)) {
76 sendeJson(exchange, 200, labelRepo.findAll());
77 } else if ("/api/spieler".equals(path)) {
78 handleSpielerListe(exchange);
79 } else if (path.startsWith("/api/spieler/")) {
80 handleSpielerEinzeln(exchange, path.substring("/api/spieler/".length()));
81 } else {
82 exchange.sendResponseHeaders(404, -1);
83 }
84 } catch (SpielerDbException e) {
85 logger.error("Spieler-DB-API-Fehler bei {}", path, e);
86 sendeFehler(exchange, 500, e.getMessage());
87 } catch (RuntimeException e) {
88 logger.error("Unerwarteter Fehler bei {}", path, e);
89 sendeFehler(exchange, 500, e.getClass().getSimpleName() + ": " + e.getMessage());
90 }
91 }

94 private void handleSpielerListe(HttpExchange exchange) throws SpielerDbException, IOException {
95 Map<String, String> params = parseQuery(exchange.getRequestURI());
96 String q = params.getOrDefault("q", "").strip();
97 int limit = leseInt(params.get("limit"), DEFAULT_LIMIT, 1, MAX_LIMIT);
98 int offset = leseInt(params.get("offset"), 0, 0, Integer.MAX_VALUE);
99
100 List<SpielerMitVerein> alle = q.isEmpty()
101 ? spielerRepo.findAll()
102 : spielerRepo.findeMitWildcard(q, true, MAX_LIMIT);
103
104 int total = alle.size();
105 int von = Math.min(offset, total);
106 int bis = Math.min(von + limit, total);
107 List<SpielerMitVerein> seite = alle.subList(von, bis);
108
109 Map<String, Object> body = new LinkedHashMap<>();
110 body.put("items", seite);
111 body.put("total", total);
112 body.put("limit", limit);
113 body.put("offset", offset);
114 sendeJson(exchange, 200, body);
115 }

158 private static int leseInt(@Nullable String roh, int defaultWert, int min, int max) {
159 if (roh == null || roh.isEmpty()) {
160 return defaultWert;
161 }
162 try {
163 return Math.max(min, Math.min(max, Integer.parseInt(roh)));
164 } catch (NumberFormatException e) {
165 return defaultWert;
166 }
167 }

169 private static Map<String, String> parseQuery(URI uri) {
170 String roh = uri.getRawQuery();
171 if (roh == null || roh.isEmpty()) {
172 return Map.of();
173 }
174 Map<String, String> ergebnis = new HashMap<>();
175 for (String paar : roh.split("&")) {
176 int eq = paar.indexOf('=');
177 String name = eq < 0 ? paar : paar.substring(0, eq);
178 String wert = eq < 0 ? "" : paar.substring(eq + 1);
179 ergebnis.put(URLDecoder.decode(name, StandardCharsets.UTF_8),
180 URLDecoder.decode(wert, StandardCharsets.UTF_8));
181 }
182 return ergebnis;
183 }
