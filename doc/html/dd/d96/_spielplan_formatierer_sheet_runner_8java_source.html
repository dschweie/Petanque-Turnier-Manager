90 formatiereZebra(konfig);

110 private boolean cfFehlt(SpielplanFormatiererKonfig konfig) {
111 List<RangePosition> editierbar = konfig.editierbareRanges();
112 if (editierbar.isEmpty()) return false;
113 RangePosition erste = editierbar.get(0);
114 try {
115 var cell = xSheet.getCellByPosition(erste.getStartSpalte(), erste.getStartZeile());
116 XSheetConditionalEntries cf = Lo.qi(XSheetConditionalEntries.class,
117 Lo.qi(XPropertySet.class, cell).getPropertyValue("ConditionalFormat"));
118 return cf == null || cf.getCount() == 0;
119 } catch (Exception e) {
120 logger.warn("CF-Check fehlgeschlagen – überspringe CF-Repair", e);
121 return false;
122 }
123 }
