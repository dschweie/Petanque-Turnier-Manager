166 private Formation leseFormation(XControlContainer xcc) {
167 return switch (leseListBoxAuswahl(xcc, "lstFormation")) {

182 private short leseListBoxAuswahl(XControlContainer xcc, String name) {
183 XControl ctrl = xcc.getControl(name);

189 }

191 private boolean leseCheckBoxZustand(XControlContainer xcc, String name) {

194 return false;
195 }
196 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
197 return cb != null && cb.getState() == 1;
198 }

200 private int leseSpinnerWert(XControlContainer xcc, String name, int fallback) {
201 XControl ctrl = xcc.getControl(name);
202 if (ctrl == null) {
203 return fallback;
204 }
205 XSpinField sf = Lo.qi(XSpinField.class, ctrl);
206 if (sf == null) {
207 return fallback;
208 }
209 try {
210 XPropertySet props = Lo.qi(XPropertySet.class, ctrl.getModel());
211 Object val = props.getPropertyValue("Value");
212 return val instanceof Number n ? n.intValue() : fallback;
213 } catch (Exception e) {
214 return fallback;
215 }
216 }
