66 public SpieltagErgebnisse lese(int anzRunden) throws GenerateException {
67 SpieltagErgebnisse ergebnisse = new SpieltagErgebnisse();
68 var xDoc = workingSpreadsheet.getWorkingSpreadsheetDocument();
69
70 for (int rundeNr = 1; rundeNr <= anzRunden; rundeNr++) {
71 String schluessel = SheetMetadataHelper.schluesselSupermeleeSpielrunde(spielTag.getNr(), rundeNr);
72 String legacyName = SheetNamen.supermeleeSpielrunde(spielTag.getNr(), rundeNr);
73 XSpreadsheet sheet = SheetMetadataHelper.findeSheetUndHeile(xDoc, schluessel, legacyName);
74 if (sheet == null) {
75 logger.warn("Spielrunde-Sheet {}.{} nicht gefunden", spielTag.getNr(), rundeNr);
76 ergebnisse.nichtGespielteRunden.add(rundeNr);
77 continue;
78 }
79 // Vertikale Ergebnis-Spalten T/U sind formel-getrieben aus den horizontalen
80 // Ergebnissen. Bei leerer Runde liefern sie 0/0 (Formelresultat für leeres H);
81 // das ist mathematisch konsistent mit der alten VLOOKUP-Auswertung.
82 Map<Integer, RundenErgebnis> proSpieler = leseRunde(sheet);
83 ergebnisse.proRunde.put(rundeNr, proSpieler);
84 if (!hatHorizontaleErgebnisse(sheet)) {
85 // Runde gilt für die NG-Markierung als "nicht gespielt" (keine Einträge
86 // in der horizontalen H-Spalte) – unabhängig von den 0-Werten der Vertikalen.
87 ergebnisse.nichtGespielteRunden.add(rundeNr);
88 }
89 }
90 return ergebnisse;
91 }

93 private Map<Integer, RundenErgebnis> leseRunde(XSpreadsheet sheet) throws GenerateException {
94 Map<Integer, RundenErgebnis> proSpieler = new HashMap<>();
95
96 int ersteSpalte = SpielrundeSheetKonstanten.ERSTE_SPALTE_VERTIKALE_ERGEBNISSE;
97 int letzteSpalte = SpielrundeSheetKonstanten.SPALTE_VERTIKALE_ERGEBNISSE_MINUS;
98 int ersteZeile = SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE;
99 int letzteZeile = ersteZeile + MAX_DATENZEILEN - 1;
100
101 RangePosition range = RangePosition.from(ersteSpalte, ersteZeile, letzteSpalte, letzteZeile);
102 RangeData daten = RangeHelper
103 .from(sheet, workingSpreadsheet.getWorkingSpreadsheetDocument(), range)
104 .getDataFromRange();
105
106 for (RowData row : daten) {
107 if (row.isEmpty()) {
108 continue;
109 }
110 int spielerNr = row.get(0).getIntVal(-1);
111 if (spielerNr <= 0) {
112 continue;
113 }
114 int plus = row.size() > 1 ? row.get(1).getIntVal(-1) : -1;
115 int minus = row.size() > 2 ? row.get(2).getIntVal(-1) : -1;
116 if (plus < 0 || minus < 0) {
117 // Spieler war zugeteilt, aber kein Ergebnis erfasst → für Spieler-Sicht
118 // "nicht gespielt"; nicht in Map aufnehmen.
119 continue;
120 }
121 proSpieler.put(spielerNr, new RundenErgebnis(plus, minus));
122 }
123 return proSpieler;
124 }

126 private boolean hatHorizontaleErgebnisse(XSpreadsheet sheet) throws GenerateException {
127 int spalte = SpielrundeSheetKonstanten.ERSTE_SPALTE_ERGEBNISSE;
128 int ersteZeile = SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE;
129 int letzteZeile = ersteZeile + MAX_DATENZEILEN - 1;
130 RangeData spalteH = RangeHelper
131 .from(sheet, workingSpreadsheet.getWorkingSpreadsheetDocument(),
132 RangePosition.from(spalte, ersteZeile, spalte, letzteZeile))
133 .getDataFromRange();
134 for (RowData row : spalteH) {
135 if (row.isEmpty()) {
136 continue;
137 }
138 Object data = row.get(0).getData();
139 if (data instanceof Number) {
140 return true;
141 }
142 if (data instanceof String s && !s.isBlank()) {
143 return true;
144 }
145 }
146 return false;
147 }
