145 protected void doRun() throws GenerateException {
146 XSpreadsheet xSheet = getXSpreadSheet();
147 if (xSheet == null) {
148 logger.warn("Poule Meldeliste nicht gefunden");
149 return;
150 }

158 private void stringsBesinigen(XSpreadsheet xSheet) throws GenerateException {
159 Formation formation = getKonfigurationSheet().getMeldeListeFormation();
160 boolean teamnameAktiv = getKonfigurationSheet().isMeldeListeTeamnameAnzeigen();
161 boolean vereinsnameAktiv = getKonfigurationSheet().isMeldeListeVereinsnameAnzeigen();
162 int letzteZeile = letzteZeileMitDaten(xSheet) + MIN_ANZAHL_MELDUNGEN_ZEILEN;
163 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
164 if (teamnameAktiv) {
165 bereinigeSpalte(xSheet, getTeamnameSpalte(), zeile);
166 }
167 for (int s = 0; s < formation.getAnzSpieler(); s++) {
168 bereinigeSpalte(xSheet, getVornameSpalte(s), zeile);
169 bereinigeSpalte(xSheet, getNachnameSpalte(s), zeile);
170 if (vereinsnameAktiv) {
171 bereinigeSpalte(xSheet, getVereinsnameSpalte(s), zeile);
172 }
173 }
174 }
175 }

177 private void bereinigeSpalte(XSpreadsheet xSheet, int spalte, int zeile) throws GenerateException {
178 if (spalte < 0) {
179 return;
180 }
181 String original = getSheetHelper().getTextFromCell(xSheet, Position.from(spalte, zeile));
182 if (original == null || original.isEmpty()) {
183 return;
184 }
185 String bereinigt = original.replaceAll("[\\p{Cntrl}]", "").strip();
186 if (!bereinigt.equals(original)) {
187 getSheetHelper().setStringValueInCell(StringCellValue.from(xSheet, Position.from(spalte, zeile), bereinigt));
188 }
189 }

191 private void teamnummernVergeben(XSpreadsheet xSheet) throws GenerateException {
192 int letzteZeile = letzteZeileMitDaten(xSheet);
193 if (letzteZeile < ERSTE_DATEN_ZEILE) {
194 return;
195 }
196 int vornameSpalte = getVornameSpalte(0);
197 RangePosition range = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
198 getSetzPositionSpalte(), letzteZeile);
199 SortHelper.from(this, range).spalteToSort(getTeamNrSpalte()).abSteigendSortieren().doSort();
200
201 int letztNr = Math.max(0,
202 getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE)));
203
204 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
205 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
206 if (vorname == null || vorname.isEmpty()) {
207 continue;
208 }
209 int nr = getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
210 if (nr <= 0) {
211 letztNr++;
212 getSheetHelper().setNumberValueInCell(
213 NumberCellValue.from(xSheet, Position.from(getTeamNrSpalte(), zeile)).setValue(letztNr));
214 }
215 }
216 }

225 SortHelper.from(this, range).spalteToSort(getTeamNrSpalte()).aufSteigendSortieren(true).doSort();
226 }
