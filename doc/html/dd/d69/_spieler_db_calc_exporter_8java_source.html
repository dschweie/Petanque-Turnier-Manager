84 private void schreibeAlleSheets(XSpreadsheetDocument doc, SpielerDbExportData data,
85 ExportRequest request) throws SpielerDbException {
86 XSpreadsheets sheets = doc.getSheets();
87 List<String> geschriebeneSheetNamen = new ArrayList<>();
88
89 if (request.entities().contains(ExportEntity.SPIELER)) {
90 geschriebeneSheetNamen.add(schreibeSpielerSheet(doc, sheets, data));
91 }
92 if (request.entities().contains(ExportEntity.VEREINE)) {
93 geschriebeneSheetNamen.add(schreibeVereineSheet(doc, sheets, data));
94 }
95 if (request.entities().contains(ExportEntity.LABELS)) {
96 geschriebeneSheetNamen.add(schreibeLabelsSheet(doc, sheets, data));
97 }
98 if (request.entities().contains(ExportEntity.SPIELER)
99 && request.entities().contains(ExportEntity.LABELS)) {
100 geschriebeneSheetNamen.add(schreibeJunctionSheet(doc, sheets, data));
101 }
102
103 if (geschriebeneSheetNamen.isEmpty()) {
104 throw new SpielerDbException("Keine Entities im Scope — Export würde leeres Dokument erzeugen");
105 }
106 entferneStandardSheet(sheets, geschriebeneSheetNamen);
107 }

109 private String schreibeSpielerSheet(XSpreadsheetDocument doc, XSpreadsheets sheets,
110 SpielerDbExportData data) throws SpielerDbException {
111 String name = "Spieler";
112 XSpreadsheet sheet = sheetAnlegen(sheets, name);
113 String[] header = { "nr", "vorname", "nachname", "vereinNr", "vereinName", "lizenznr" };
114 RangeData rd = new RangeData();
115 rd.add(headerZeile(header));
116 for (SpielerMitVerein s : data.spieler()) {
117 RowData row = new RowData();
118 row.newInt(s.nr());
119 row.newString(s.vorname());
120 row.newString(s.nachname());
121 if (s.vereinNr() == null) {
122 row.newEmpty();
123 } else {
124 row.newInt(s.vereinNr());
125 }
126 row.newString(s.vereinName() == null ? "" : s.vereinName());
127 row.newString(s.lizenznr() == null ? "" : s.lizenznr());
128 rd.add(row);
129 }
130 schreibeBlock(doc, sheet, rd, header.length);
131 return name;
132 }

243 private static void entferneStandardSheet(XSpreadsheets sheets,
244 List<String> geschriebene) {
245 try {
246 XIndexAccess idx = Lo.qi(XIndexAccess.class, sheets);
247 int anz = idx.getCount();
248 for (int i = 0; i < anz; i++) {
249 XSpreadsheet sh = Lo.qi(XSpreadsheet.class, idx.getByIndex(i));
250 if (sh == null) {
251 continue;
252 }
253 String name = Lo.qi(com.sun.star.container.XNamed.class, sh).getName();
254 if (!geschriebene.contains(name)) {
255 sheets.removeByName(name);
256 return;
257 }
258 }
259 } catch (com.sun.star.uno.Exception e) {
260 // Default-Sheet bleibt drin — Export funktioniert trotzdem.
261 logger.warn("Default-Sheet konnte nicht entfernt werden: {}", e.getMessage());
262 }

265 private static void speichereAls(XSpreadsheetDocument doc, String absoluterPfad)
266 throws SpielerDbException {
267 XStorable storable = Lo.qi(XStorable.class, doc);
268 if (storable == null) {
269 throw new SpielerDbException("XStorable nicht verfügbar");

271 PropertyValue[] props = {
272 propertyValue("FilterName", FILTER_ODS),
273 propertyValue("Overwrite", Boolean.TRUE)
274 };
275 try {
276 String url = pfadAlsUrl(absoluterPfad);
277 storable.storeToURL(url, props);
278 } catch (IOException | MalformedURLException e) {
279 throw new SpielerDbException("Calc-Export-Speichern fehlgeschlagen: " + absoluterPfad, e);
280 }
281 }
