52 public CompositeViewInstanz(CompositeViewKonfiguration konfiguration) throws IOException {
53 this.konfiguration = konfiguration;
54 httpServer = HttpServer.create(new InetSocketAddress(konfiguration.port()), 10);
55 httpServer.setExecutor(Executors.newCachedThreadPool());
56 httpServer.createContext("/events", this::handleEvents);
57 httpServer.createContext("/debug/sse", this::handleDebugSse);
58 httpServer.createContext("/", this::handleStatischOderRoot);
59 keepAliveExecutor = Executors.newSingleThreadScheduledExecutor(r -> {
60 var t = new Thread(r, "PTM-CompositeView-KeepAlive-" + konfiguration.port());
61 t.setDaemon(true);
62 return t;
63 });
64 }

67 public void starten() {
68 httpServer.start();
69 laeuft = true;
70 keepAliveExecutor.scheduleAtFixedRate(
71 () -> sseVerbindungen.forEach(SseVerbindung::keepAlive),
72 KEEPALIVE_INTERVALL_SEKUNDEN,
73 KEEPALIVE_INTERVALL_SEKUNDEN,
74 TimeUnit.SECONDS);
75 logger.info("CompositeView-Server gestartet auf Port {}", konfiguration.port());
76 }

163 private void handleStatischOderRoot(HttpExchange exchange) throws IOException {
164 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
165 exchange.sendResponseHeaders(405, -1);
166 return;
167 }
168 String path = exchange.getRequestURI().getPath();
169 if ("/".equals(path) || path.isEmpty()) {
170 serviereRessource(exchange, "/index.html", CONTENT_TYPE_HTML);
171 } else if (path.startsWith("/assets/")) {
172 String dateiname = path.substring("/assets/".length());
173 serviereRessource(exchange, "/assets/" + dateiname, ermittleContentType(dateiname));
174 } else if (path.startsWith("/images/")) {
175 String dateiname = path.substring("/images/".length());
176 serviereRessource(exchange, "/images/" + dateiname, ermittleContentType(dateiname));
177 } else {
178 exchange.sendResponseHeaders(404, -1);
179 }
180 }

182 private void handleEvents(HttpExchange exchange) throws IOException {
183 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
184 exchange.sendResponseHeaders(405, -1);
185 return;
186 }
187 exchange.getResponseHeaders().set("Content-Type", CONTENT_TYPE_SSE);
188 exchange.getResponseHeaders().set("Cache-Control", "no-cache");
189 exchange.getResponseHeaders().set("Connection", "keep-alive");
190 exchange.getResponseHeaders().set("X-Accel-Buffering", "no");
191 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
192 exchange.sendResponseHeaders(200, 0);
193
194 OutputStream os = exchange.getResponseBody();
195 var verbindung = new SseVerbindung(os, this);
196 sseVerbindungen.add(verbindung);
197 logger.debug("Neue SSE-Verbindung auf CompositeView-Port {}, aktiv: {}",
198 konfiguration.port(), sseVerbindungen.size());
199
200 try {
201 os.write("retry: 3000\n\n".getBytes(StandardCharsets.UTF_8));
202 os.flush();
203 } catch (IOException e) {
204 verbindungEntfernen(verbindung);
205 return;
206 }
207 verbindung.sendeInitNachricht();
208 }

210 private void handleDebugSse(HttpExchange exchange) throws IOException {
211 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
212 exchange.sendResponseHeaders(405, -1);
213 return;
214 }
215 String json = cachedInitJson != null ? cachedInitJson : "{\"info\":\"kein cachedInitJson vorhanden\"}";
216 byte[] body = json.getBytes(StandardCharsets.UTF_8);
217 exchange.getResponseHeaders().set("Content-Type", "application/json; charset=UTF-8");
218 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
219 exchange.sendResponseHeaders(200, body.length);
220 try (OutputStream os = exchange.getResponseBody()) {
221 os.write(body);
222 }

225 private void serviereRessource(HttpExchange exchange, String relativerPfad, String contentType)
226 throws IOException {
227 String ressourcePfad = (STATIC_RESOURCE_PREFIX + relativerPfad).replaceFirst("^/", "");
228 InputStream gefunden = getClass().getClassLoader().getResourceAsStream(ressourcePfad);
229 if (gefunden == null) {
230 gefunden = getClass().getResourceAsStream("/" + ressourcePfad);
231 }
232 if (gefunden == null) {
233 logger.warn("Classpath-Ressource nicht gefunden: {}", ressourcePfad);
234 exchange.sendResponseHeaders(404, -1);
235 return;
236 }
237 try (InputStream in = gefunden) {
238 byte[] body = in.readAllBytes();
239 var headers = exchange.getResponseHeaders();
240 headers.set("Content-Type", contentType);
241 headers.set("Cache-Control", "no-cache");
242 headers.set("Access-Control-Allow-Origin", "*");
243 exchange.sendResponseHeaders(200, body.length);
244 try (OutputStream os = exchange.getResponseBody()) {
245 os.write(body);
246 }
247 }

250 private static String ermittleContentType(String dateiname) {
251 if (dateiname.endsWith(".js")) return "text/javascript; charset=UTF-8";
252 if (dateiname.endsWith(".css")) return "text/css; charset=UTF-8";
253 if (dateiname.endsWith(".html")) return CONTENT_TYPE_HTML;
254 if (dateiname.endsWith(".svg")) return "image/svg+xml";
255 if (dateiname.endsWith(".png")) return "image/png";
256 if (dateiname.endsWith(".ico")) return "image/x-icon";
257 return "application/octet-stream";
