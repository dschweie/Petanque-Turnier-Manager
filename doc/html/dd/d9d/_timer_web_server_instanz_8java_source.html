65 public TimerWebServerInstanz(int port) throws IOException {
66 this.port = port;
67 httpServer = HttpServer.create(new InetSocketAddress(port), 10);
68 httpServer.setExecutor(Executors.newCachedThreadPool());
69 httpServer.createContext("/events", this::handleEvents);
70 httpServer.createContext("/gong.wav", this::handleGong);
71 httpServer.createContext("/snooze", this::handleSnooze);
72 httpServer.createContext("/", this::handleRoot);
73 keepAliveExecutor = Executors.newSingleThreadScheduledExecutor(r -> {
74 var t = new Thread(r, "PTM-Timer-WebServer-KeepAlive-" + port);
75 t.setDaemon(true);
76 return t;
77 });
78 cachedInitJson = zuStateJson(TimerState.inaktiv());
79 }

82 public void starten() {
83 httpServer.start();
84 laeuft = true;
85 keepAliveExecutor.scheduleAtFixedRate(
86 () -> sseVerbindungen.forEach(SseVerbindung::keepAlive),
87 KEEPALIVE_INTERVALL_SEKUNDEN,
88 KEEPALIVE_INTERVALL_SEKUNDEN,
89 TimeUnit.SECONDS);
90 logger.info("Timer-Webserver gestartet auf Port {}", port);
91 }

94 public void stoppen() {

100 logger.info("Timer-Webserver gestoppt auf Port {}", port);

139 private void handleRoot(HttpExchange exchange) throws IOException {
140 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
141 exchange.sendResponseHeaders(405, -1);
142 return;
143 }
144 InputStream ressource = getClass().getClassLoader().getResourceAsStream(TIMER_HTML_RESSOURCE);
145 if (ressource == null) {
146 logger.warn("timer.html nicht gefunden im Classpath: {}", TIMER_HTML_RESSOURCE);
147 exchange.sendResponseHeaders(404, -1);
148 return;
149 }
150 try (InputStream in = ressource) {
151 byte[] body = in.readAllBytes();
152 exchange.getResponseHeaders().set("Content-Type", CONTENT_TYPE_HTML);
153 exchange.getResponseHeaders().set("Cache-Control", "no-cache");
154 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
155 exchange.sendResponseHeaders(200, body.length);
156 try (OutputStream os = exchange.getResponseBody()) {
157 os.write(body);
158 }
159 }
160 }

185 private void handleSnooze(HttpExchange exchange) throws IOException {
186 if (!"POST".equalsIgnoreCase(exchange.getRequestMethod())) {
187 exchange.sendResponseHeaders(405, -1);
188 return;
189 }
190 try {
191 TimerManager.get().snooze();
192 } catch (IllegalStateException e) {
193 logger.warn("Snooze ignoriert – TimerManager nicht initialisiert");
194 }
195 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
196 exchange.sendResponseHeaders(204, -1);
197 exchange.close();
198 }

200 private void handleEvents(HttpExchange exchange) throws IOException {
201 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
202 exchange.sendResponseHeaders(405, -1);
203 return;
204 }
205 exchange.getResponseHeaders().set("Content-Type", CONTENT_TYPE_SSE);
206 exchange.getResponseHeaders().set("Cache-Control", "no-cache");
207 exchange.getResponseHeaders().set("Connection", "keep-alive");
208 exchange.getResponseHeaders().set("X-Accel-Buffering", "no");
209 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
210 exchange.sendResponseHeaders(200, 0);
211
212 OutputStream os = exchange.getResponseBody();
213 var verbindung = new SseVerbindung(os, this);
214 sseVerbindungen.add(verbindung);
215 try {
216 os.write("retry: 30000\n\n".getBytes(StandardCharsets.UTF_8));
217 os.flush();
218 } catch (IOException e) {
219 verbindungEntfernen(verbindung);
220 return;
221 }
222 verbindung.sendeInitNachricht();
223 }
