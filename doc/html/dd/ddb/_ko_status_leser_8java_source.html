68 private boolean pruefeBaumBeendet(com.sun.star.sheet.XSpreadsheetDocument xDoc, String baumSchluessel) {
69 var scoreKey = SheetMetadataHelper.scoreSchluessel(baumSchluessel);
70 var scoreText = SheetMetadataHelper.leseScoreText(xDoc, scoreKey);
71 if (scoreText == null || !scoreText.startsWith(SCORE_PRAEFIX)) {
72 return false;
73 }
74 var positionenStr = scoreText.substring(SCORE_PRAEFIX.length());
75 if (positionenStr.isBlank()) {
76 return false;
77 }
78 var positionen = positionenStr.split("\\|");
79 var sheet = SheetMetadataHelper.findeSheet(xDoc, baumSchluessel);
80 if (sheet.isEmpty()) {
81 return false;
82 }
83 try {
84 for (var pos : positionen) {
85 if (pos.isBlank()) {
86 continue;
87 }
88 var teile = pos.split(",");
89 if (teile.length < 2) {
90 continue;
91 }
92 int spalte = Integer.parseInt(teile[0].trim());
93 int zeile = Integer.parseInt(teile[1].trim());
94 var zelle = sheet.get().getCellByPosition(spalte, zeile);
95 if (zelle.getType() == com.sun.star.table.CellContentType.EMPTY) {
96 return false;
97 }
98 }
99 } catch (Exception e) {
100 logger.error("Fehler beim Prüfen der KO-Turnierbaum-Scores für '{}'", baumSchluessel, e);
101 return false;
102 }
103 return positionen.length > 0;
104 }
