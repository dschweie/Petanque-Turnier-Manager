184 protected void allesFelderEntfernenUndNeuFenster() {
185 String panel = this.getClass().getSimpleName();
186 long startNs = System.nanoTime();
187 long t = startNs;
188 try {
189 vorFensterDispose();
190 } catch (Exception e) {
191 logger.error("Fehler beim Freigeben vor Window-Dispose", e);
192 }
193 long tVorFensterDispose = System.nanoTime();
194 if (window != null) {
195 window.dispose();
196 }
197 long tWindowDisposed = System.nanoTime();
198 uiGeneration.incrementAndGet(); // erst nach dispose – alte UI ist jetzt tot, neue Generation beginnt
199 window = null;
200 if (guiFactoryCreateParam != null) {
201 guiFactoryCreateParam.clear();
202 guiFactoryCreateParam = null;
203 }
204 ausstehendAktualisierung = null; // Rebuild ist vollständige Aktualisierung – pending Event obsolet
205 neuesBasisFenster();
206 long tNeuesBasisFenster = System.nanoTime();
207 felderHinzufuegen();
208 long tFelderHinzu = System.nanoTime();
209 requestLayout();
210 long tFertig = System.nanoTime();
211 PerfLog.log(logger,
212 "[SIDEBAR-TIMING] {} allesFelderEntfernenUndNeuFenster: vorDispose={} ms, dispose={} ms, neuesFenster={} ms, felderHinzufuegen={} ms, requestLayout={} ms, GESAMT={} ms, thread={}",
213 panel,
214 (tVorFensterDispose - t) / 1_000_000L,
215 (tWindowDisposed - tVorFensterDispose) / 1_000_000L,
216 (tNeuesBasisFenster - tWindowDisposed) / 1_000_000L,
217 (tFelderHinzu - tNeuesBasisFenster) / 1_000_000L,
218 (tFertig - tFelderHinzu) / 1_000_000L,
219 (tFertig - startNs) / 1_000_000L,
220 Thread.currentThread().getName());

253 void bereinigen() {
254 if (istBereinigt) {
255 return;
256 }
257 istBereinigt = true;
258 uiReadyGeneration = -1;
259 ausstehendAktualisierung = null;
260 logger.debug("BaseSidebarContent.bereinigen");
261
262 try {
263 SidebarPanelDelegator.get().entfernen(this);
264 } catch (RuntimeException e) {
265 logger.error("Fehler beim Entfernen des GlobalEventListeners", e);
266 }
267 try {
268 PetanqueTurnierMngrSingleton.removeTurnierEventListener(this);
269 } catch (RuntimeException e) {
270 logger.error("Fehler beim Entfernen des TurnierEventListeners", e);
271 }
272 if (parentWindow != null) {
273 try {
274 parentWindow.removeWindowListener(fensterAdapter);
275 } catch (RuntimeException e) {
276 logger.error("Fehler beim Entfernen des FensterListeners", e);
277 }
278 parentWindow = null;
279 }
280 layout = null;
281 try {
282 onDisposing(null);
283 } catch (RuntimeException e) {
284 logger.error("Fehler in onDisposing", e);
285 }
286 currentSpreadsheet = null;
287 if (guiFactoryCreateParam != null) {
288 guiFactoryCreateParam.clear();
289 guiFactoryCreateParam = null;
290 }
291 xSidebar = null;
292 if (window != null && !ausstehendInit) {
293 try {
294 window.dispose();
295 } catch (RuntimeException e) {
296 logger.error("Fehler beim Dispose des Fensters", e);
297 }
298 }
299 window = null;
300 ausstehendInit = false;

416 private void aktualisiereSpreadsheetUndFelder(Object source) {
417 if (istBereinigt || ausstehendInit) {
418 return;
419 }
420 XModel xModel = Lo.qi(XModel.class, source);
421 if (xModel == null) {
422 return;
423 }
424 XSpreadsheetDocument xSpreadsheetDocument = Lo.qi(XSpreadsheetDocument.class, xModel);
425 XSpreadsheetView xSpreadsheetView = Lo.qi(XSpreadsheetView.class, xModel.getCurrentController());
426 if (xSpreadsheetDocument != null && xSpreadsheetView != null) {
427 // Globale OnNew/OnLoad-Events werden an alle Sidebars aller Calc-Fenster verteilt.
428 // Nur reagieren, wenn das Event zu UNSEREM Dokument gehört.
429 if (eigenesDokument != null && !eigenesDokument.equals(xSpreadsheetDocument)) {
430 return;
431 }
432 eigenesDokument = xSpreadsheetDocument;
433 currentSpreadsheet = new WorkingSpreadsheet(currentSpreadsheet.getxContext(), xSpreadsheetDocument);
434 onSpreadsheetGewechselt(currentSpreadsheet);
435 var event = new OnProperiesChangedEvent(currentSpreadsheet.getWorkingSpreadsheetDocument());
436 if (isUiReady()) {
437 felderAktualisieren(event);
438 } else {
439 ausstehendAktualisierung = event;
440 }
441 }

474 @Override
475 public void onPropertiesChanged(ITurnierEvent eventObj) {
476 if (istBereinigt || ausstehendInit) {
477 return;
478 }
479 var doc = getCurrentSpreadsheet().getWorkingSpreadsheetDocument();
480 if (doc == null || !doc.equals(eventObj.getWorkingSpreadsheetDocument())) {
481 return;
482 }
483 if (!isUiReady()) {
484 ausstehendAktualisierung = eventObj;
485 return;
486 }
487 String panel = this.getClass().getSimpleName();
488 long startNs = System.nanoTime();
489 try {
490 felderAktualisieren(eventObj);
491 } finally {
492 long dauerMs = (System.nanoTime() - startNs) / 1_000_000L;
493 PerfLog.log(logger, "[SIDEBAR-TIMING] {} onPropertiesChanged.felderAktualisieren: {} ms, thread={}",
494 panel, dauerMs, Thread.currentThread().getName());
495 }
496 }

498 protected void doLayout() {
499 if (istBereinigt || ausstehendInit) {
500 return;
501 }
502 try {
503 Rectangle posSizeParent = new Rectangle(0, 0, getParentWindow().getPosSize().Width,
504 getParentWindow().getPosSize().Height);
505 if (posSizeParent.Width <= 0) {
506 return;
507 }
508 if (window != null) {
509 window.setPosSize(0, 0, posSizeParent.Width, posSizeParent.Height, PosSize.POSSIZE);
510 }
511 if (getLayout() != null) {
512 getLayout().layout(posSizeParent);
513 }
514 int genJetzt = uiGeneration.get();
515 if (uiReadyGeneration != genJetzt) {
516 uiReadyGeneration = genJetzt;
517 var pending = ausstehendAktualisierung;
518 if (pending != null) {
519 ausstehendAktualisierung = null;
520 felderAktualisieren(pending);
521 }
522 }
523 } catch (Exception e) {
524 logger.error(e.getMessage(), e);
525 }
