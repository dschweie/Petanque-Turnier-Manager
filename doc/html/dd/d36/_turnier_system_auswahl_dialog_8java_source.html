139 @Override
140 protected void beiOkGeklickt() throws Exception {
141 if (listBox == null) {
142 return;
143 }
144 short ausgewaehltIndex = listBox.getSelectedItemPos();
145 if (ausgewaehltIndex < 0 || ausgewaehltIndex >= AUSWAHL_SYSTEME.length) {
146 return;
147 }
148 TurnierSystem gewaehltesTurnierSystem = AUSWAHL_SYSTEME[ausgewaehltIndex];
149 logger.info("Turnier-Start: {}", gewaehltesTurnierSystem.getBezeichnung());
150 starteNeueTurnierInDokument(ws, gewaehltesTurnierSystem);
151 }
152
153 protected static void starteNeueTurnierInDokument(WorkingSpreadsheet zielWs, TurnierSystem system)
154 throws Exception {
155 switch (system) {
156 case SUPERMELEE -> new MeldeListeSheet_New(zielWs).startSilent();
157 case SCHWEIZER -> new SchweizerMeldeListeSheetNew(zielWs).startSilent();
158 case MAASTRICHTER -> new MaastrichterMeldeListeSheetNew(zielWs).startSilent();
159 case POULE -> new PouleMeldeListeSheetNew(zielWs).startSilent();
160 case LIGA -> new LigaMeldeListeSheetNew(zielWs).startSilent();
161 case JGJ -> new JGJMeldeListeSheet_New(zielWs).startSilent();
162 case KO -> new KoMeldeListeSheetNew(zielWs).startSilent();
163 case KASKADE -> new KaskadeMeldeListeSheetNew(zielWs).startSilent();
164 case FORMULEX -> new FormuleXMeldeListeSheetNew(zielWs).startSilent();
165 case TRIPTETE -> new TripTeteMeldeListeSheetNew(zielWs).startSilent();
166 default -> logger.warn("Unbekanntes Turniersystem für Start-Aktion: {}", system);
167 }
168 }
