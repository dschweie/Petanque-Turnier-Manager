75 private Map<SheetGruppe, List<BlattKnoten>> schluesselNachGruppenSortieren(
76 XSpreadsheetDocument xDoc) {
77 var gruppenMap = new EnumMap<SheetGruppe, List<BlattKnoten>>(SheetGruppe.class);
78 var allKeys = SheetMetadataHelper.getSchluesselMitPrefix(xDoc, "__PTM_");
79 // Deterministische Reihenfolge, damit bei (in Alt-Dokumenten möglichen) Mehrfach-Schlüsseln
80 // auf demselben Blatt stets derselbe Eintrag gewinnt.
81 Arrays.sort(allKeys);
82 // Blatt-Eindeutigkeit in der Anzeige: ein Blatt, das (z.B. nach Systemwechsel) mehrere
83 // Identitäts-Schlüssel trägt, darf nur EINEN Eintrag erzeugen. Nicht-destruktiv – heilt
84 // bestehende Dokumente in der Ansicht, ohne sie zu verändern (der schreib-seitige Purge
85 // in SheetMetadataHelper bereinigt sie dauerhaft bei der nächsten Metadaten-Schreibung).
86 var belegteBlaetter = new HashSet<String>();
87 for (var schluessel : allKeys) {
88 if (schluessel.startsWith("__PTM_SCORE_")) {
89 continue;
90 }
91 var gruppeOpt = SheetGruppe.fuerSchluessel(schluessel);
92 var gruppe = gruppeOpt.orElse(SheetGruppe.ALLGEMEIN);
93 SheetMetadataHelper.findeSheet(xDoc, schluessel).ifPresent(sheet -> {
94 var named = Lo.qi(XNamed.class, sheet);
95 if (named != null && !belegteBlaetter.add(named.getName())) {
96 logger.debug("Blatt '{}' bereits durch anderen Schlüssel belegt – Schlüssel '{}' übersprungen",
97 named.getName(), schluessel);
98 return;
99 }
100 // Diese Gruppen erscheinen auf oberster Ebene (keine Einrückung)
101 var einrueckung = (gruppe == SheetGruppe.SUPERMELEE || gruppe == SheetGruppe.LIGA
102 || gruppe == SheetGruppe.SCHWEIZER || gruppe == SheetGruppe.KO
103 || gruppe == SheetGruppe.FORMULEX || gruppe == SheetGruppe.JGJ) ? "" : " ";
104 var knoten = knoten(sheet, schluessel, einrueckung);
105 if (knoten != null) {
106 gruppenMap.computeIfAbsent(gruppe, g -> new ArrayList<>()).add(knoten);
107 }
108 });
109 }
110 gruppenMap.values().forEach(liste -> liste.sort(this::knotenVergleichen));
111 return gruppenMap;
112 }

114 private BlattKnoten knoten(XSpreadsheet sheet, String schluessel, String einrueckung) {
115 var named = Lo.qi(XNamed.class, sheet);
116 if (named == null) {
117 logger.warn("BlattKnoten: XNamed nicht verfügbar für Schlüssel '{}'", schluessel);
118 return null;
119 }
120 return new BlattKnoten(sheet, einrueckung + named.getName(), schluessel);

123 private List<BlattBaumEintrag> eintraegeMitKopfAufbauen(
124 Map<SheetGruppe, List<BlattKnoten>> gruppenMap,
125 Set<SheetGruppe> kollabiert,
126 Set<Integer> kollabierteSpielTage,
127 Set<String> kollabierteUnterGruppen) {
128 var ergebnis = new ArrayList<BlattBaumEintrag>();
129 // Gruppen, deren ALLGEMEIN-Teilnehmer bereits integriert wurden (Schweizer, Poule)
130 var verbrauchteGruppen = new HashSet<SheetGruppe>();
131 for (var gruppe : SheetGruppe.values()) {
132 if (verbrauchteGruppen.contains(gruppe)) {
133 continue;
134 }
135 var knoten = gruppenMap.get(gruppe);
136 if (knoten == null || knoten.isEmpty()) {
137 continue;
138 }
139 if (gruppe == SheetGruppe.SUPERMELEE) {
140 ergebnis.addAll(supermeleeEintraege(knoten, kollabierteSpielTage));
141 } else if (gruppe == SheetGruppe.LIGA) {
142 ergebnis.addAll(ligaEintraege(knoten));
143 } else if (gruppe == SheetGruppe.SCHWEIZER) {
144 var allgemeinKnoten = gruppenMap.getOrDefault(SheetGruppe.ALLGEMEIN, List.of());
145 ergebnis.addAll(schweizerEintraege(knoten, allgemeinKnoten));
146 verbrauchteGruppen.add(SheetGruppe.ALLGEMEIN);
147 } else if (gruppe == SheetGruppe.POULE) {
148 var allgemeinKnoten = gruppenMap.getOrDefault(SheetGruppe.ALLGEMEIN, List.of());
149 ergebnis.addAll(pouleEintraege(knoten, allgemeinKnoten, kollabierteUnterGruppen));
150 verbrauchteGruppen.add(SheetGruppe.ALLGEMEIN);
151 } else if (gruppe == SheetGruppe.MAASTRICHTER) {
152 var allgemeinKnoten = gruppenMap.getOrDefault(SheetGruppe.ALLGEMEIN, List.of());
153 ergebnis.addAll(maastrichterEintraege(knoten, allgemeinKnoten, kollabierteUnterGruppen));
154 verbrauchteGruppen.add(SheetGruppe.ALLGEMEIN);
155 } else if (gruppe == SheetGruppe.KO) {
156 var allgemeinKnoten = gruppenMap.getOrDefault(SheetGruppe.ALLGEMEIN, List.of());
157 ergebnis.addAll(koEintraege(knoten, allgemeinKnoten));
158 verbrauchteGruppen.add(SheetGruppe.ALLGEMEIN);
159 } else if (gruppe == SheetGruppe.KASKADE) {
160 ergebnis.addAll(kaskadeEintraege(knoten, kollabierteUnterGruppen));
161 } else if (gruppe == SheetGruppe.FORMULEX) {
162 var allgemeinKnoten = gruppenMap.getOrDefault(SheetGruppe.ALLGEMEIN, List.of());
163 ergebnis.addAll(formulexEintraege(knoten, allgemeinKnoten));
164 verbrauchteGruppen.add(SheetGruppe.ALLGEMEIN);
165 } else if (gruppe == SheetGruppe.JGJ) {
166 var allgemeinKnoten = gruppenMap.getOrDefault(SheetGruppe.ALLGEMEIN, List.of());
167 ergebnis.addAll(jgjEintraege(knoten, allgemeinKnoten));
168 verbrauchteGruppen.add(SheetGruppe.ALLGEMEIN);
169 } else {
170 var expandiert = !kollabiert.contains(gruppe);
171 ergebnis.add(new GruppenKopf(gruppe, expandiert));
172 if (expandiert) {
173 ergebnis.addAll(knoten);
174 }
175 }
176 }
177 return ergebnis;
178 }

184 private List<BlattBaumEintrag> supermeleeEintraege(List<BlattKnoten> knoten,
185 Set<Integer> kollabierteSpielTage) {
186 var ergebnis = new ArrayList<BlattBaumEintrag>();
187
188 Optional<BlattKnoten> meldeliste = knoten.stream()
189 .filter(k -> SUPERMELEE_MELDELISTE_SCHLUESSEL.equals(k.metadatenSchluessel()))
190 .findFirst();
191 Optional<BlattKnoten> endrangliste = knoten.stream()
192 .filter(k -> SUPERMELEE_ENDRANGLISTE_SCHLUESSEL.equals(k.metadatenSchluessel()))
193 .findFirst();
194
195 // Spieltage nach Nummer gruppieren; Blätter ohne Spieltagnummer (Teams) → Spieltag 1
196 var spieltageMap = new TreeMap<Integer, List<BlattKnoten>>();
197 for (var k : knoten) {
198 if (SUPERMELEE_MELDELISTE_SCHLUESSEL.equals(k.metadatenSchluessel())
199 || SUPERMELEE_ENDRANGLISTE_SCHLUESSEL.equals(k.metadatenSchluessel())) {
200 continue;
201 }
202 int spieltagNr = Optional.ofNullable(spieltagNrAusSchluessel(k.metadatenSchluessel()))
203 .orElse(1);
204 var mitEinrueckung = new BlattKnoten(k.sheet(), " " + blattName(k), k.metadatenSchluessel());
205 spieltageMap.computeIfAbsent(spieltagNr, n -> new ArrayList<>()).add(mitEinrueckung);
206 }
207 spieltageMap.values().forEach(liste -> liste.sort(this::knotenVergleichen));
208
209 meldeliste.map(k -> new BlattKnoten(k.sheet(), blattName(k), k.metadatenSchluessel()))
210 .ifPresent(ergebnis::add);
211
212 for (var eintrag : spieltageMap.entrySet()) {
213 int nr = eintrag.getKey();
214 var expandiert = !kollabierteSpielTage.contains(nr);
215 ergebnis.add(new SpieltagKopf(nr, expandiert));
216 if (expandiert) {
217 ergebnis.addAll(eintrag.getValue());
218 }
219 }
220
221 endrangliste.map(k -> new BlattKnoten(k.sheet(), blattName(k), k.metadatenSchluessel()))
222 .ifPresent(ergebnis::add);
223
224 return ergebnis;
225 }

381 private List<BlattBaumEintrag> pouleEintraege(
382 List<BlattKnoten> pouleKnoten,
383 List<BlattKnoten> allgemeinKnoten,
384 Set<String> kollabierteUnterGruppen) {
385 var ergebnis = new ArrayList<BlattBaumEintrag>();
386
387 // Meldeliste an erster Stelle ohne Einrückung
388 pouleKnoten.stream()
389 .filter(k -> SheetMetadataHelper.SCHLUESSEL_POULE_MELDELISTE.equals(k.metadatenSchluessel()))
390 .map(k -> new BlattKnoten(k.sheet(), blattName(k), k.metadatenSchluessel()))
391 .forEach(ergebnis::add);
392
393 // Checkin-Liste direkt nach der Meldeliste
394 pouleKnoten.stream()
395 .filter(k -> SheetMetadataHelper.SCHLUESSEL_POULE_CHECKIN_LISTE.equals(k.metadatenSchluessel()))
396 .map(k -> new BlattKnoten(k.sheet(), blattName(k), k.metadatenSchluessel()))
397 .forEach(ergebnis::add);
398
399 // Teilnehmer aus ALLGEMEIN direkt danach
400 allgemeinKnoten.stream()
401 .map(k -> new BlattKnoten(k.sheet(), blattName(k), k.metadatenSchluessel()))
402 .forEach(ergebnis::add);
403
404 // Vorrunde-Untergruppe
405 var vorrundeKnoten = pouleKnoten.stream()
406 .filter(k -> !SheetMetadataHelper.SCHLUESSEL_POULE_MELDELISTE.equals(k.metadatenSchluessel())
407 && !SheetMetadataHelper.SCHLUESSEL_POULE_CHECKIN_LISTE.equals(k.metadatenSchluessel())
408 && !k.metadatenSchluessel().startsWith(SheetMetadataHelper.SCHLUESSEL_POULE_KO_PREFIX))
409 .map(k -> new BlattKnoten(k.sheet(), " " + blattName(k), k.metadatenSchluessel()))
410 .toList();
411
412 if (!vorrundeKnoten.isEmpty()) {
413 var vorrundeExpandiert = !kollabierteUnterGruppen.contains(POULE_VORRUNDE_GRUPPE_ID);
414 ergebnis.add(new UnterGruppenKopf(
415 POULE_VORRUNDE_GRUPPE_ID,
416 I18n.get("sidebar.sheets.poule.vorrunde.gruppe"),
417 vorrundeExpandiert));
418 if (vorrundeExpandiert) {
419 ergebnis.addAll(vorrundeKnoten);
420 }
421 }
422
423 // KO-Runde-Untergruppe
424 var koKnoten = pouleKnoten.stream()
425 .filter(k -> k.metadatenSchluessel().startsWith(SheetMetadataHelper.SCHLUESSEL_POULE_KO_PREFIX))
426 .map(k -> new BlattKnoten(k.sheet(), " " + blattName(k), k.metadatenSchluessel()))
427 .toList();
428
429 if (!koKnoten.isEmpty()) {
430 var koExpandiert = !kollabierteUnterGruppen.contains(POULE_KO_GRUPPE_ID);
431 ergebnis.add(new UnterGruppenKopf(
432 POULE_KO_GRUPPE_ID,
433 I18n.get("sidebar.poule.ko"),
434 koExpandiert));
435 if (koExpandiert) {
436 ergebnis.addAll(koKnoten);
437 }
438 }
439
440 return ergebnis;
441 }

589 private Integer ersteZahl(String text) {
590 int start = 0;
591 while (start < text.length() && !Character.isDigit(text.charAt(start))) {
592 start++;
593 }
594 int end = start;
595 while (end < text.length() && Character.isDigit(text.charAt(end))) {
596 end++;
597 }
598 if (start == end) {
599 return null;
600 }
601 try {
602 return Integer.parseInt(text.substring(start, end));

611 int rangB = gruppeA.reihenfolgeDesSchluessels(b.metadatenSchluessel());
