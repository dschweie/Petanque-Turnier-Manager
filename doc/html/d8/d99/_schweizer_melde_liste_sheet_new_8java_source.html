134 @Override
135 protected void doRun() throws GenerateException {
136 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.SCHWEIZER)
137 .prefix(getLogPrefix()).validate()) {
138 return;
139 }
140
141 // Dialog zuerst – bei Abbruch keine Änderungen am Dokument
142 Optional<SchweizerTurnierParameterDialog.TurnierParameter> param;
143 try {
144 param = SchweizerTurnierParameterDialog.from(getWorkingSpreadsheet()).show(Formation.DOUBLETTE, false, false,
145 SpielplanTeamAnzeige.NR, getKonfigurationSheet().getRankingModus());
146 } catch (Exception e) {
147 String errMsg = I18n.get("error.dialog.parameterdialog", e.getMessage());
148 logger.error(errMsg, e);
149 throw new GenerateException(errMsg);
150 }
151
152 if (param.isEmpty()) {
153 return; // Benutzer hat abgebrochen – keine Dokument-Änderungen
154 }
155
156 // Erst nach Bestätigung: TurnierSystem + Page Styles setzen
157 getKonfigurationSheet().update();
158
159 getSheetHelper().removeAllSheetsExclude();
160 getKonfigurationSheet().setRankingModus(param.get().rankingModus);
161 createMeldelisteWithParams(param.get().formation, param.get().teamnameAnzeigen, param.get().vereinsnameAnzeigen,
162 param.get().spielplanTeamAnzeige);
163 }
164
169 public void createMeldelisteWithParams(Formation formation, boolean teamnameAnzeigen, boolean vereinsnameAnzeigen)
170 throws GenerateException {
171 createMeldelisteWithParams(formation, teamnameAnzeigen, vereinsnameAnzeigen, SpielplanTeamAnzeige.NR);
172 }
173
174 public void createMeldelisteWithParams(Formation formation, boolean teamnameAnzeigen, boolean vereinsnameAnzeigen,
175 SpielplanTeamAnzeige spielplanTeamAnzeige) throws GenerateException {
176 var neuesSheet = NewSheet.from(this, SheetNamen.meldeliste(), METADATA_SCHLUESSEL)
177 .pos(DefaultSheetPos.MELDELISTE).hideGrid().tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).setDocVersionWhenNew().create();
178 if (neuesSheet.isDidCreate()) {
179 getKonfigurationSheet().setMeldeListeFormation(formation);
180 getKonfigurationSheet().setMeldeListeTeamnameAnzeigen(teamnameAnzeigen);
181 getKonfigurationSheet().setMeldeListeVereinsnameAnzeigen(vereinsnameAnzeigen);
182 getKonfigurationSheet().setSpielplanTeamAnzeige(spielplanTeamAnzeige);
183 upDateSheet();
184 }
