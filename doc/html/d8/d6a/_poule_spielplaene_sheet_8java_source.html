67 @Override
68 public void doRun() throws GenerateException {
69 processBoxinfo("processbox.poule.spielplaene.erstellen");
70 meldeliste.upDateSheet();
71
72 var meldungen = meldeliste.getAktiveMeldungen();
73 int anzTeams = meldungen.size();
74
75 if (anzTeams < 3) {
76 MessageBox.from(getWorkingSpreadsheet(), MessageBoxTypeEnum.ERROR_OK)
77 .caption(I18n.get("msg.caption.fehler"))
78 .message(I18n.get("poule.vorrunde.fehler.zu.wenige.meldungen", anzTeams)).show();
79 return;
80 }
81
82 var gruppenGroessen = PouleGruppenRechner.berechneGruppenGroessen(anzTeams);
83 var poules = PouleSeedingService.verteileTeams(meldungen, gruppenGroessen);
84
85 int vorrundeZeile = ERSTE_DATEN_ZEILE;
86 for (var poule : poules) {
87 SheetRunner.testDoCancelTask();
88 erstellePouleSpielplanSheet(poule, vorrundeZeile);
89 vorrundeZeile += poule.teams().size() == 4 ? VIERER_POULE_ZEILEN : DREIER_POULE_ZEILEN;
90 }
91 }

93 private void erstellePouleSpielplanSheet(PouleSeedingService.Poule poule, int vorrundeStartZeile)
94 throws GenerateException {
95 var sheetName = SheetNamen.pouleSpielplan(poule.pouleNr());
96 var metaKey = SheetMetadataHelper.schluesselPouleSpielplan(poule.pouleNr());
97
98 NewSheet.from(this, sheetName, metaKey)
99 .tabColor(getKonfigurationSheet().getSpielrundeTabFarbe())
100 .pos(DefaultSheetPos.POULE_WORK)
101 .forceCreate().hideGrid().create();
102
103 aktuellesPouleSheet = SheetMetadataHelper.findeSheetUndHeile(
104 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
105 metaKey, sheetName);
106
107 try {
108 headerSchreiben(aktuellesPouleSheet);
109 spaltenBreitenSetzen(aktuellesPouleSheet);
110
111 int letzteDatenZeile;
112 if (poule.teams().size() == 4) {
113 schreibeViererPoule(aktuellesPouleSheet, poule, ERSTE_DATEN_ZEILE, vorrundeStartZeile);
114 letzteDatenZeile = ERSTE_DATEN_ZEILE + VIERER_POULE_DATEN_ZEILEN - 1;
115 } else {
116 schreibeDreierPoule(aktuellesPouleSheet, poule, ERSTE_DATEN_ZEILE, vorrundeStartZeile);
117 letzteDatenZeile = ERSTE_DATEN_ZEILE + DREIER_POULE_DATEN_ZEILEN - 1;
118 }
119
120 formatierungDurchfuehren(aktuellesPouleSheet, letzteDatenZeile);
121 printBereichSetzen(aktuellesPouleSheet, letzteDatenZeile);
122
123 if (SheetRunner.isRunning()) {
124 SheetFreeze.from(aktuellesPouleSheet, getWorkingSpreadsheet()).anzZeilen(ERSTE_DATEN_ZEILE).doFreeze();
125 }
126 } finally {
127 aktuellesPouleSheet = null;
128 }
129 }
