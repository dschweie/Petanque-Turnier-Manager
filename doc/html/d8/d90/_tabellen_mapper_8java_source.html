91 private TabelleModel mapBereich(XSpreadsheet sheet, XSpreadsheetDocument doc, CellRangeAddress bereich) {
92 int numZeilen = bereich.EndRow - bereich.StartRow + 1;
93 int numSpalten = bereich.EndColumn - bereich.StartColumn + 1;
94
95 // Gitter: null = Merge-Slave (kein <td>), "" = noch nicht verarbeitet
96 var gitterRaw = new String[numZeilen][numSpalten];
97 for (var zeile : gitterRaw) {
98 Arrays.fill(zeile, "");
99 }
100
101 Map<String, ZelleModel> zellenMap = new LinkedHashMap<>();
102 Map<Long, CellRangeAddress> mergeCache = new HashMap<>();
103 ZahlFormatierer zahlenFormatierer = ZahlFormatierer.erstelleAus(doc);
104
105 for (int r = 0; r < numZeilen; r++) {
106 for (int c = 0; c < numSpalten; c++) {
107 if (gitterRaw[r][c] == null) {
108 continue; // Merge-Slave – von einer Master-Zelle bereits markiert
109 }
110
111 int absRow = bereich.StartRow + r;
112 int absCol = bereich.StartColumn + c;
113 long mergeKey = toMergeKey(absRow, absCol);
114
115 try {
116 XCell cell = sheet.getCellByPosition(absCol, absRow);
117 XPropertySet props = Lo.qi(XPropertySet.class, cell);
118
119 int colspan = 1;
120 int rowspan = 1;
121
122 // Merge-Erkennung via Cursor: zuverlässiger als IsMerged-Property
123 // Ergebnis wird gecacht, um wiederholte UNO-Aufrufe zu vermeiden
124 CellRangeAddress mergeAdresse = mergeCache.get(mergeKey);
125 if (mergeAdresse == null) {
126 var einzelZelle = Lo.qi(XSheetCellRange.class,
127 sheet.getCellRangeByPosition(absCol, absRow, absCol, absRow));
128 var mergeKursor = sheet.createCursorByRange(einzelZelle);
129 mergeKursor.collapseToMergedArea();
130 mergeAdresse = Lo.qi(XCellRangeAddressable.class, mergeKursor).getRangeAddress();
131 for (int mr = mergeAdresse.StartRow; mr <= mergeAdresse.EndRow; mr++) {
132 for (int mc = mergeAdresse.StartColumn; mc <= mergeAdresse.EndColumn; mc++) {
133 mergeCache.put(toMergeKey(mr, mc), mergeAdresse);
134 }
135 }
136 }
137
138 boolean istMaster = mergeAdresse.StartRow == absRow
139 && mergeAdresse.StartColumn == absCol
140 && (mergeAdresse.EndRow > absRow || mergeAdresse.EndColumn > absCol);
141 if (istMaster) {
142 colspan = mergeAdresse.EndColumn - absCol + 1;
143 rowspan = mergeAdresse.EndRow - absRow + 1;
144 markiereSlaves(gitterRaw, r, c, rowspan, colspan, numZeilen, numSpalten);
145 }
146
147 String id = TabelleModel.zelleId(r, c);
148 gitterRaw[r][c] = id;
149
150 String wert = extrahiereZellwert(cell, props, zahlenFormatierer);
151
152 zellenMap.put(id, new ZelleModel(id, wert, extrahiereStil(props, colspan, rowspan)));
153
154 } catch (Exception e) {
155 logger.debug("Fehler beim Mappen von Zelle [{},{}]", r, c, e);
156 gitterRaw[r][c] = TabelleModel.zelleId(r, c);
157 }
158 }
159 }
160
161 var kopfFuss = ermittleKopfUndFusszeile(sheet, doc);
162 return new TabelleModel(
163 numZeilen, numSpalten,
164 rohGitterZuListe(gitterRaw, numZeilen),
165 zellenMap,
166 ermittleSpaltenBreiten(sheet, bereich, numSpalten),
167 ermittleZeilenHoehen(sheet, bereich, numZeilen),
168 bereich.StartRow,
169 bereich.StartColumn,
170 ermittleKopfZeilenAnzahl(sheet, bereich),
171 kopfFuss[0], kopfFuss[1], kopfFuss[2],
172 kopfFuss[3], kopfFuss[4], kopfFuss[5]);
173 }

187 try {
188 XPrintAreas printAreas = Lo.qi(XPrintAreas.class, sheet);
189 if (printAreas == null || !printAreas.getPrintTitleRows()) {
190 return 0;
191 }
192 CellRangeAddress titelZeilen = printAreas.getTitleRows();

201 return Math.max(0, Math.min(anzahl, bereich.EndRow - bereich.StartRow + 1));
202 } catch (Exception e) {

204 return 0;

330 private String extrahiereZellwert(XCell cell, XPropertySet props, ZahlFormatierer formatierer) {
331 try {
332 CellContentType type = cell.getType();
333
334 if (CellContentType.TEXT.equals(type)) {
335 XText txt = Lo.qi(XText.class, cell);
336 String text = (txt != null) ? txt.getString() : "";
337 if (text.startsWith(INTERN_METADATEN_PRAEFIX)) {
338 return null;
339 }
340 return text;
341 }
342
343 if (CellContentType.VALUE.equals(type)) {
344 // getString() liefert den formatierten Anzeigewert für alle Formate:
345 // Datum, Uhrzeit, Prozent, Währung, benutzerdefiniert.
346 XText txt = Lo.qi(XText.class, cell);
347 String anzeige = (txt != null) ? txt.getString() : "";
348 if (!anzeige.isEmpty()) {
349 return anzeige;
350 }
351 double wert = cell.getValue();
352 if (Double.isNaN(wert) || Double.isInfinite(wert)) {
353 return null;
354 }
355 if (formatierer != null) {
356 String formatiert = formatierer.formatiere(props, wert);
357 if (formatiert != null) {
358 return formatiert;
359 }
360 }
361 if (wert == Math.rint(wert)) {
362 return String.valueOf((long) wert);
363 }
364 return String.valueOf(wert);
365 }
366
367 if (CellContentType.FORMULA.equals(type)) {
368 // getString() liefert den formatierten Anzeigewert universell:
369 // "" bei leerem Text-Ergebnis, "0" bei numerischer Null,
370 // formatierten Text für Datum/Uhrzeit und sonstige Zahlenformate,
371 // den eigentlichen Text für String-Ergebnisse (z.B. Spielernamen).
372 XText txt = Lo.qi(XText.class, cell);
373 String anzeige = (txt != null) ? txt.getString() : "";
374 if (!anzeige.isEmpty()) {
375 return anzeige;
376 }
377 // getString() = "": leeres Text-Formelergebnis oder numerisch 0 ohne Anzeige
378 double wert = cell.getValue();
379 if (Double.isNaN(wert) || Double.isInfinite(wert)) {
380 return null;
381 }
382 if (wert == 0.0) {
383 return null;
384 }
385 if (formatierer != null) {
386 String formatiert = formatierer.formatiere(props, wert);
387 if (formatiert != null) {
388 return formatiert;
389 }
390 }
391 if (wert == Math.rint(wert)) {
392 return String.valueOf((long) wert);
393 }
394 return String.valueOf(wert);
395 }
396
397 return null;
398
399 } catch (Exception e) {
400 logger.debug("Zellwert Fehler", e);
401 return null;
402 }
403 }

420 private StyleModel extrahiereStil(XPropertySet props, int colspan, int rowspan) {
421 boolean fett = false;
422 boolean kursiv = false;
423 String hintergrundfarbe = null;
424 String schriftfarbe = null;
425 String ausrichtung = "left";
426 String vertikaleAusrichtung = "top";
427 int rotationGrad = 0;
428 boolean zeilenumbruch = false;
429 String schriftart = null;
430 float schriftgroesse = 0f;
431 String linienOben = null;
432 String linienUnten = null;
433 String linienLinks = null;
434 String linienRechts = null;
435
436 // Jede Eigenschaftsgruppe wird separat abgefangen, damit ein Fehler in einer Gruppe
437 // die anderen nicht abbricht und der Browser maximal viele Stilinfos erhält.
438
439 try {
440 // Schrifteigenschaften
441 Object fontWeightObj = props.getPropertyValue("CharWeight");
442 fett = fontWeightObj instanceof Float fw && fw >= FontWeight.BOLD;
443
444 Object fontSlantObj = props.getPropertyValue("CharPosture");
445 kursiv = fontSlantObj instanceof FontSlant slant && slant == FontSlant.ITALIC;
446
447 Object fontNameObj = props.getPropertyValue("CharFontName");
448 if (fontNameObj instanceof String fn && !fn.isBlank()) {
449 schriftart = fn;
450 }
451
452 Object heightObj = props.getPropertyValue("CharHeight");
453 if (heightObj instanceof Number n) {
454 schriftgroesse = n.floatValue();
455 }
456 } catch (Exception e) {
457 logger.debug("Schrifteigenschaften nicht ermittelbar", e);
458 }
459
460 try {
461 // Farben (-1 = automatisch/transparent)
462 Object bgObj = props.getPropertyValue("CellBackColor");
463 if (bgObj instanceof Integer bg && bg != -1) {
464 hintergrundfarbe = String.format("#%06X", 0xFFFFFF & bg);
465 }
466
467 Object ccObj = props.getPropertyValue("CharColor");
468 if (ccObj instanceof Integer cc && cc != -1) {
469 schriftfarbe = String.format("#%06X", 0xFFFFFF & cc);
470 }
471 } catch (Exception e) {
472 logger.debug("Farb-Eigenschaften nicht ermittelbar", e);
473 }
474
475 try {
476 // Ausrichtung und Layout
477 // CellHoriJustify ist kein Java-Enum → equals-Vergleich
478 Object hJustObj = props.getPropertyValue("HoriJustify");
479 if (hJustObj instanceof CellHoriJustify hJust) {
480 if (CellHoriJustify.CENTER.equals(hJust)) {
481 ausrichtung = "center";
482 } else if (CellHoriJustify.RIGHT.equals(hJust)) {
483 ausrichtung = "right";
484 } else if (CellHoriJustify.BLOCK.equals(hJust)) {
485 ausrichtung = "justify";
486 } else {
487 ausrichtung = "left";
488 }
489 }
490
491 // CellVertJustify2-Konstante als int
492 Object vJustObj = props.getPropertyValue("VertJustify");
493 if (vJustObj instanceof Integer vJust) {
494 vertikaleAusrichtung = switch (vJust) {
495 case 2 -> "middle"; // CENTER
496 case 3, 4 -> "bottom"; // BOTTOM, BLOCK
497 default -> "top";
498 };
499 }
500
501 // Textrotation (1/100 Grad → Grad)
502 Object rotObj = props.getPropertyValue("RotateAngle");
503 if (rotObj instanceof Integer winkel) {
504 rotationGrad = winkel / 100;
505 }
506
507 Object wrapObj = props.getPropertyValue(ICommonProperties.IS_TEXT_WRAPPED);
508 zeilenumbruch = Boolean.TRUE.equals(wrapObj);
509 } catch (Exception e) {
510 logger.debug("Ausrichtungs-Eigenschaften nicht ermittelbar", e);
511 }
512
513 try {
514 // Rahmenlinien (TableBorder2)
515 Object borderObj = props.getPropertyValue("TableBorder2");
516 if (borderObj instanceof com.sun.star.table.TableBorder2 tb) {
517 linienOben = linienZuCss(tb.TopLine, tb.IsTopLineValid);
518 linienUnten = linienZuCss(tb.BottomLine, tb.IsBottomLineValid);
519 linienLinks = linienZuCss(tb.LeftLine, tb.IsLeftLineValid);
520 linienRechts = linienZuCss(tb.RightLine, tb.IsRightLineValid);
521 }
522 } catch (Exception e) {
523 logger.debug("Rahmen-Eigenschaften nicht ermittelbar", e);
524 }
525
526 return new StyleModel(fett, kursiv, hintergrundfarbe, schriftfarbe,
527 ausrichtung, vertikaleAusrichtung, colspan, rowspan,
528 rotationGrad, zeilenumbruch, schriftart, schriftgroesse,
529 linienOben, linienUnten, linienLinks, linienRechts);
530 }

639 private CellRangeAddress begrenzungsrahmen(CellRangeAddress[] bereiche) {
640 var box = new CellRangeAddress();
641 box.Sheet = bereiche[0].Sheet;
642 box.StartColumn = bereiche[0].StartColumn;
643 box.StartRow = bereiche[0].StartRow;
644 box.EndColumn = bereiche[0].EndColumn;
645 box.EndRow = bereiche[0].EndRow;
646 for (int i = 1; i < bereiche.length; i++) {
647 box.StartColumn = Math.min(box.StartColumn, bereiche[i].StartColumn);
648 box.StartRow = Math.min(box.StartRow, bereiche[i].StartRow);
649 box.EndColumn = Math.max(box.EndColumn, bereiche[i].EndColumn);
650 box.EndRow = Math.max(box.EndRow, bereiche[i].EndRow);
