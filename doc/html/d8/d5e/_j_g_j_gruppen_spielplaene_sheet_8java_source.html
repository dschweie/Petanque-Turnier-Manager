125 public void erstelle(String buchstabe, int zentraleStartZeile,
126 List<List<TeamPaarung>> hRunde, List<List<TeamPaarung>> rRunde,
127 boolean zeigeNr, Map<Integer, String> teamNamen, String freispielText) throws GenerateException {
128
129 String sheetName = SheetNamen.jgjGruppeSpielplan(buchstabe);
130 String metaKey = SheetMetadataHelper.schluesselJgjGruppeSpielplan(buchstabe);
131
132 NewSheet.from(this, sheetName, metaKey)
133 .tabColor(konfigurationSheet.getSpielrundeTabFarbe())
134 .pos(DefaultSheetPos.JGJ_WORK)
135 .forceCreate().hideGrid().create();
136
137 aktuellesSheet = SheetMetadataHelper.findeSheetUndHeile(
138 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), metaKey, sheetName);
139 try {
140 schreibeSpaltenBreiten(zeigeNr);
141 schreibeTitelZeile(buchstabe);
142 schreibeSpaltenHeader();
143
144 int letzteDatenZeile = schreibeSpieleDaten(hRunde, rRunde, zeigeNr, teamNamen, freispielText);
145 schreibeErgebnisFormeln(letzteDatenZeile, zentraleStartZeile);
146
147 formatiereSheet(letzteDatenZeile, hRunde, rRunde);
148 definierePrintBereich(letzteDatenZeile);
149
150 if (SheetRunner.isRunning()) {
151 SheetFreeze.from(aktuellesSheet, getWorkingSpreadsheet())
152 .anzZeilen(ERSTE_DATEN_ZEILE).doFreeze();
153 }
154 } finally {
155 aktuellesSheet = null;
156 }

217 private int schreibeSpieleDaten(List<List<TeamPaarung>> hRunde, List<List<TeamPaarung>> rRunde,
218 boolean zeigeNr, Map<Integer, String> teamNamen, String freispielText) throws GenerateException {
219
220 List<List<TeamPaarung>> alleSpieltage = new ArrayList<>();
221 alleSpieltage.addAll(hRunde);
222 alleSpieltage.addAll(rRunde);
223
224 RangeData daten = new RangeData();
225 int hIndex = 1;
226 int rIndex = 1;
227 boolean inHinrunde = true;
228 int hAnzahl = anzPaarungen(hRunde);
229
230 for (List<TeamPaarung> spieltag : alleSpieltage) {
231 for (TeamPaarung paarung : spieltag) {
232 SheetRunner.testDoCancelTask();
233 RowData row = daten.addNewRow();
234 if (inHinrunde) {
235 row.newString(JGJSpielPlanSheet.NR_HINRUNDE_PREFIX + hIndex++);
236 } else {
237 row.newString(JGJSpielPlanSheet.NR_RUECKRUNDE_PREFIX + rIndex++);
238 }
239 int nrA = paarung.getA().getNr();
240 boolean hatB = paarung.getOptionalB().isPresent();
241 int nrB = hatB ? paarung.getB().getNr() : 0;
242
243 row.newString(zeigeNr ? String.valueOf(nrA) : teamNamen.getOrDefault(nrA, ""));
244 row.newString(hatB
245 ? (zeigeNr ? String.valueOf(nrB) : teamNamen.getOrDefault(nrB, ""))
246 : freispielText);
247 }
248 if (inHinrunde && hIndex > hAnzahl) {
249 inHinrunde = false;
250 }
251 }
252
253 Position startPos = Position.from(SPALTE_NR, ERSTE_DATEN_ZEILE);
254 int letzteDatenZeile = ERSTE_DATEN_ZEILE + daten.size() - 1;
255
256 RangeHelper.from(this, daten.getRangePosition(startPos)).setDataInRange(daten);
257
258 // Border/Formatierung über die komplette Datenzeile (inkl. der per Formel
259 // befüllten Ergebnis-Spalten) anwenden.
260 RangeHelper.from(this,
261 RangePosition.from(SPALTE_NR, ERSTE_DATEN_ZEILE, LETZTE_SPALTE, letzteDatenZeile))
262 .setRangeProperties(RangeProperties.from()
263 .setBorder(BorderFactory.from().allThin().toBorder())
264 .centerJustify().setShrinkToFit(true)
265 .topMargin(110).bottomMargin(110).setCharHeight(12));
266
267 return letzteDatenZeile;
268 }

302 private void formatiereSheet(int letzteDatenZeile, List<List<TeamPaarung>> hRunde,
303 List<List<TeamPaarung>> rRunde) throws GenerateException {
304 RangePosition zebraRange = RangePosition.from(SPALTE_NR, ERSTE_DATEN_ZEILE,
305 LETZTE_SPALTE, letzteDatenZeile);
306 RanglisteGeradeUngeradeFormatHelper.from(this, zebraRange)
307 .geradeFarbe(konfigurationSheet.getSpielPlanHintergrundFarbeGerade())
308 .ungeradeFarbe(konfigurationSheet.getSpielPlanHintergrundFarbeUnGerade())
309 .apply();
310
311 RangeProperties horTrennerDouble = RangeProperties.from()
312 .setBorder(BorderFactory.from().doubleLn().forBottom().toBorder());
313 RangeProperties horTrennerBoldBottom = RangeProperties.from()
314 .setBorder(BorderFactory.from().boldLn().forBottom().toBorder());
315
316 int zeile = ERSTE_DATEN_ZEILE;
317 int anzRunden = hRunde.size();
318 int anzPaarungenProRunde = hRunde.isEmpty() ? 0 : hRunde.get(0).size();
319
320 for (int i = 1; i < anzRunden; i++) {
321 zeile += anzPaarungenProRunde - 1;
322 RangeHelper.from(this,
323 RangePosition.from(SPALTE_NR, zeile, LETZTE_SPALTE, zeile))
324 .setRangeProperties(horTrennerDouble);
325 zeile++;
326 }
327 zeile += anzPaarungenProRunde - 1;
328 RangeHelper.from(this,
329 RangePosition.from(SPALTE_NR, zeile, LETZTE_SPALTE, zeile))
330 .setRangeProperties(horTrennerBoldBottom);
331 zeile++;
332
333 if (!rRunde.isEmpty()) {
334 for (int i = 1; i < anzRunden; i++) {
335 zeile += anzPaarungenProRunde - 1;
336 RangeHelper.from(this,
337 RangePosition.from(SPALTE_NR, zeile, LETZTE_SPALTE, zeile))
338 .setRangeProperties(horTrennerDouble);
339 zeile++;
340 }
341 }
342
343 RangeProperties vertTrennerBoldLeft = RangeProperties.from()
344 .setBorder(BorderFactory.from().boldLn().forLeft().toBorder());
345 RangeProperties vertTrennerDoubleLeft = RangeProperties.from()
346 .setBorder(BorderFactory.from().doubleLn().forLeft().toBorder());
347
348 RangePosition vertikal = RangePosition.from(SPALTE_NR, HEADER_SPALTEN_ZEILE,
349 SPALTE_NR, letzteDatenZeile);
350 RangeHelper.from(this, vertikal.spalte(SPALTE_NR)).setRangeProperties(vertTrennerBoldLeft);
351 RangeHelper.from(this, vertikal.spalte(SPALTE_TEAM_B)).setRangeProperties(vertTrennerDoubleLeft);
352 RangeHelper.from(this, vertikal.spalte(SPALTE_ERGEBNIS_A)).setRangeProperties(vertTrennerBoldLeft);
353 RangeHelper.from(this, vertikal.spalte(LETZTE_SPALTE + 1)).setRangeProperties(vertTrennerBoldLeft);
354 }
