107 private void ergebnisseEinfuegen(XSpreadsheet sheet) throws GenerateException {
108 RangePosition readRange = RangePosition.from(
109 TEAM_A_SPALTE, ERSTE_DATEN_ZEILE, ERG_TEAM_B_SPALTE, ERSTE_DATEN_ZEILE + 100);
110 RangeData data = RangeHelper
111 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
112 .getDataFromRange();
113
114 for (int i = 0; i < data.size(); i++) {
115 RowData row = data.get(i);
116 if (row.size() < 2) break;
117
118 // Funktioniert sowohl für SpielplanTeamAnzeige.NR (int) als auch .NAME (String)
119 int nrA = row.get(0).getIntVal(0);
120 if (nrA <= 0) {
121 String valA = row.get(0).getStringVal();
122 if (valA == null || valA.isBlank()) break; // Ende der Daten
123 }
124 int nrB = row.get(1).getIntVal(0);
125 if (nrB <= 0) {
126 String valB = row.get(1).getStringVal();
127 if (valB == null || valB.isBlank()) continue; // Freilos – kein Ergebnis nötig
128 }
129
130 int zeile = ERSTE_DATEN_ZEILE + i;
131 int winner = RandomSource.nextInt(2); // 0 oder 1
132 int loserPts = RandomSource.nextInt(0, 13); // 0–12
133 int ergA = (winner == 0) ? 13 : loserPts;
134 int ergB = (winner == 0) ? loserPts : 13;
135
136 getSheetHelper().setNumberValueInCell(
137 NumberCellValue.from(sheet, Position.from(ERG_TEAM_A_SPALTE, zeile)).setValue(ergA));
138 getSheetHelper().setNumberValueInCell(
139 NumberCellValue.from(sheet, Position.from(ERG_TEAM_B_SPALTE, zeile)).setValue(ergB));
140 }
141 }
