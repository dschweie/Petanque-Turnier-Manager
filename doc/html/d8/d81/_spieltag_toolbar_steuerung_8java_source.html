51 public static void aktualisiereInAllenFrames(XComponentContext xContext) {
52 try {
53 var xDesktop = DocumentHelper.getCurrentDesktop(xContext);
54 if (xDesktop == null) {
55 logger.debug("aktualisiereInAllenFrames: kein Desktop gefunden");
56 return;
57 }
58 var xFramesSupplier = Lo.qi(XFramesSupplier.class, xDesktop);
59 if (xFramesSupplier == null) {
60 logger.debug("aktualisiereInAllenFrames: XFramesSupplier nicht verfügbar");
61 return;
62 }
63 var xFrames = xFramesSupplier.getFrames();
64 if (xFrames == null) {
65 return;
66 }
67 int anzahl = xFrames.getCount();
68 for (int i = 0; i < anzahl; i++) {
69 try {
70 var xFrame = Lo.qi(XFrame.class, xFrames.getByIndex(i));
71 aktualisiereInFrame(xFrame, xContext);
72 } catch (Exception e) {
73 logger.error("Fehler beim Aktualisieren der Spieltag-Toolbar in Frame {}", i, e);
74 }
75 }
76 } catch (Exception e) {
77 logger.error("Fehler beim Durchsuchen aller Frames für Spieltag-Toolbar", e);
78 }
79 }

83 private static void aktualisiereInFrame(XFrame xFrame, XComponentContext xContext) {
84 if (xFrame == null) {
85 return;
86 }
87 try {
88 // Druckvorschau-Frames überspringen: ScPreviewController implementiert XSpreadsheetView nicht
89 if (Lo.qi(XSpreadsheetView.class, xFrame.getController()) == null) {
90 logger.debug("aktualisiereInFrame '{}': Druckvorschau-Frame erkannt – übersprungen", xFrame.getName());
91 return;
92 }
93 var xFrameProps = Lo.qi(XPropertySet.class, xFrame);
94 if (xFrameProps == null) {
95 return;
96 }
97 var xLayoutManager = Lo.qi(XLayoutManager.class, xFrameProps.getPropertyValue("LayoutManager"));
98 if (xLayoutManager == null) {
99 return;
100 }
101 boolean sichtbar = ermittleSichtbarkeit(xFrame, xContext);
102 if (sichtbar) {
103 xLayoutManager.requestElement(SPIELTAG_TOOLBAR_URL);
104 xLayoutManager.showElement(SPIELTAG_TOOLBAR_URL);
105 } else {
106 xLayoutManager.hideElement(SPIELTAG_TOOLBAR_URL);
107 }
108 String frameTitel = "<unknown>";
109 try {
110 var xTitle = Lo.qi(com.sun.star.frame.XTitle.class, xFrame);
111 if (xTitle != null) frameTitel = xTitle.getTitle();
112 } catch (Exception e) {
113 logger.debug("Frame-Titel konnte nicht ermittelt werden: {}", e.getMessage());
114 }
115 logger.trace("[FOKUS-TRACE] SpieltagToolbar.aktualisiereInFrame frameTitle='{}' frameHash={} sichtbar={}",
116 frameTitel, System.identityHashCode(xFrame), sichtbar);
117 } catch (Exception e) {
118 logger.error("Fehler beim Aktualisieren der Spieltag-Toolbar in Frame '{}'",
119 xFrame.getName(), e);
120 }
121 }

123 private static boolean ermittleSichtbarkeit(XFrame xFrame, XComponentContext xContext) {
124 try {
125 // Dokument aus dem eigenen Frame ermitteln – nicht das global aktive Dokument.
126 // Bei mehreren offenen Dokumenten könnte sonst der falsche TurnierSystem-Wert
127 // für diesen Frame verwendet werden.
128 var controller = xFrame.getController();
129 if (controller == null) {
130 return false;
131 }
132 var doc = Lo.qi(XSpreadsheetDocument.class, controller.getModel());
133 if (doc == null) {
134 return false;
135 }
136 var ws = new WorkingSpreadsheet(xContext, doc);
137 TurnierSystem ts = new DocumentPropertiesHelper(ws).getTurnierSystemAusDocument();
138 return ts != null && ts.hatMehrereSpielTage();
139 } catch (Exception e) {
140 logger.debug("Sichtbarkeit konnte nicht ermittelt werden: {}", e.getMessage());
141 return false;
142 }
143 }
