197 private void readProperties() {
198 fileLock.lock();
199 try {
200 var path = Path.of(System.getProperty("user.home"), FILENAME);
201
202 try {
203 Files.createFile(path);
204 } catch (java.nio.file.FileAlreadyExistsException ignored) {
205 }
206
207 Properties loaded = new Properties();
208
209 try (var in = Files.newInputStream(path)) {
210 loaded.load(in);
211 }
212
213 propMap.clear();
214 for (String key : loaded.stringPropertyNames()) {
215 propMap.put(key, loaded.getProperty(key));
216 }
217
218 } catch (Exception e) {
219 logger.error("Fehler beim Laden", e);
220 } finally {
221 fileLock.unlock();
222 }
223 }
224
225 private static void speichernDatei() {
226
227 fileLock.lock();
228
229 try {
230 var path = Path.of(System.getProperty("user.home"), FILENAME);
231 var tmp = path.resolveSibling(FILENAME + ".tmp");
232
233 Properties props = new Properties();
234 for (var e : propMap.entrySet()) {
235 props.setProperty(e.getKey(), e.getValue());
236 }
237
238 try (var out = Files.newOutputStream(tmp)) {
239 props.store(out, "Petanque Turnier Manager Konfiguration");
240 }
241
242 Files.move(tmp, path,
243 java.nio.file.StandardCopyOption.REPLACE_EXISTING,
244 java.nio.file.StandardCopyOption.ATOMIC_MOVE);
245
246 } catch (Exception e) {
247 logger.error("Fehler beim Speichern", e);
248 } finally {
249 fileLock.unlock();
250 }

342 public List<CompositeViewEintragRoh> getCompositeViewEintraege() {
343 List<CompositeViewEintragRoh> eintraege = new ArrayList<>();
344 try {
345 var portsStr = propMap.getOrDefault(WEBSERVER_COMPOSITE_PORTS_PROP, "").trim();
346 if (portsStr.isEmpty()) return eintraege;
347
348 for (var portStr : portsStr.split(",")) {
349 portStr = portStr.trim();
350 if (portStr.isEmpty()) continue;
351 try {
352 int port = Integer.parseInt(portStr);
353 boolean aktiv = getBoolean(WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_AKTIV_SUFFIX);
354 String name = propMap.getOrDefault(WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_NAME_SUFFIX, "").trim();
355 int zoom = parseZoom(propMap.get(WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_ZOOM_SUFFIX));
356 boolean mitHeaderFooter = getBooleanMitDefault(
357 WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_MIT_HEADER_FOOTER_SUFFIX, true);
358 String layoutJson = propMap.getOrDefault(WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_LAYOUT_SUFFIX, "").trim();
359 if (layoutJson.isEmpty()) continue;
360
361 int panelCount = 0;
362 try {
363 panelCount = Integer.parseInt(propMap.getOrDefault(
364 WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_PANEL_COUNT_SUFFIX, "0").trim());
365 } catch (NumberFormatException ignored) {
366 }
367
368 List<PanelEintragRoh> panels = new ArrayList<>();
369 for (int i = 0; i < panelCount; i++) {
370 String panelPrefix = WEBSERVER_COMPOSITE_PREFIX + port + WEBSERVER_COMPOSITE_PANEL_INFIX + i;
371 PanelTyp panelTyp = parsePanelTyp(propMap.get(panelPrefix + WEBSERVER_COMPOSITE_PANEL_TYP_SUFFIX));
372 String sheetConfig = propMap.getOrDefault(panelPrefix + WEBSERVER_COMPOSITE_PANEL_SHEET_SUFFIX, "").trim();
373 String externeUrl = propMap.getOrDefault(panelPrefix + WEBSERVER_COMPOSITE_PANEL_URL_SUFFIX, "").trim();
374 if (panelTyp == PanelTyp.BLATT && sheetConfig.isEmpty()) continue;
375 if (panelTyp == PanelTyp.URL && externeUrl.isEmpty()) continue;
376 int panelZoom = parseZoom(propMap.get(panelPrefix + WEBSERVER_COMPOSITE_PANEL_ZOOM_SUFFIX));
377 String panelHAlign = propMap.get(panelPrefix + WEBSERVER_COMPOSITE_PANEL_HALIGN_SUFFIX);
378 String panelVAlign = propMap.get(panelPrefix + WEBSERVER_COMPOSITE_PANEL_VALIGN_SUFFIX);
379 if (panelHAlign == null && panelVAlign == null
380 && getBoolean(panelPrefix + LEGACY_WEBSERVER_COMPOSITE_PANEL_ZENTRIERT_SUFFIX)) {
381 // Migration: alte boolean-Property "_zentriert" → Horizontal=Mitte
382 panelHAlign = PanelAusrichtung.H_MITTE;
383 }
384 boolean panelBlattnameAnzeigen = getBoolean(panelPrefix + WEBSERVER_COMPOSITE_PANEL_BLATTNAME_SUFFIX);
385 panels.add(new PanelEintragRoh(panelTyp, sheetConfig, panelZoom, panelHAlign, panelVAlign, panelBlattnameAnzeigen, externeUrl));
386 }
387 if (!panels.isEmpty()) {
388 eintraege.add(new CompositeViewEintragRoh(port, name, aktiv, zoom, mitHeaderFooter, layoutJson, panels));
389 }
390 } catch (Exception e) {
391 logger.warn("Ungültiger Composite-Port-Eintrag '{}'", portStr, e);
392 }
393 }
394 } catch (Exception e) {
395 logger.warn("Fehler beim Lesen der Composite-View-Einträge", e);
396 }
397 return eintraege;
398 }

403 public List<CompositeViewKonfiguration> getCompositeViewKonfigurationen() {
404 List<CompositeViewKonfiguration> konfigs = new ArrayList<>();
405 var gson = new GsonBuilder()
406 .registerTypeAdapter(SplitKnoten.class, new SplitKnotenAdapter())
407 .create();
408
409 for (var eintrag : getCompositeViewEintraege()) {
410 if (!eintrag.aktiv()) continue;
411 try {
412 SplitKnoten wurzel = gson.fromJson(eintrag.layoutJson(), SplitKnoten.class);
413 if (wurzel == null) {
414 logger.warn("Ungültiger Layout-JSON für Port {}", eintrag.port());
415 continue;
416 }
417 List<PanelKonfiguration> panels = new ArrayList<>();
418 for (var p : eintrag.panels()) {
419 if (p.typ() == PanelTyp.TIMER) {
420 panels.add(new PanelKonfiguration(PanelTyp.TIMER, "", null, p.zoom(),
421 p.horizontalAusrichtung(), p.vertikalAusrichtung(), false, ""));
422 continue;
423 }
424 if (p.typ() == PanelTyp.URL) {
425 panels.add(new PanelKonfiguration(PanelTyp.URL, "", null, p.zoom(),
426 p.horizontalAusrichtung(), p.vertikalAusrichtung(), p.blattnameAnzeigen(), p.externeUrl()));
427 continue;
428 }
429 var resolver = SheetResolverFactory.erstellen(p.sheetConfig());
430 if (resolver == null) {
431 logger.warn("Resolver null für Panel-Config '{}'", p.sheetConfig());
432 continue;
433 }
434 panels.add(new PanelKonfiguration(PanelTyp.BLATT, p.sheetConfig(), resolver, p.zoom(),
435 p.horizontalAusrichtung(), p.vertikalAusrichtung(), p.blattnameAnzeigen(), ""));
436 }
437 if (!panels.isEmpty()) {
438 konfigs.add(new CompositeViewKonfiguration(eintrag.port(), eintrag.name(), eintrag.zoom(), wurzel, panels, eintrag.mitHeaderFooter()));
439 }
440 } catch (Exception e) {
441 logger.error("Fehler bei Composite-View-Konfiguration {}", eintrag, e);
442 }
443 }
444 return konfigs;
445 }

451 public void speichernCompositeViews(boolean aktiv, List<CompositeViewEintragRoh> eintraege) {
452 try {
453 setBooleanProp(WEBSERVER_AKTIV_PROP, aktiv);
454 // Alte Einträge löschen
455 for (var alt : getCompositeViewEintraege()) {
456 String prefix = WEBSERVER_COMPOSITE_PREFIX + alt.port();
457 propMap.remove(prefix + WEBSERVER_COMPOSITE_AKTIV_SUFFIX);
458 propMap.remove(prefix + WEBSERVER_COMPOSITE_NAME_SUFFIX);
459 propMap.remove(prefix + WEBSERVER_COMPOSITE_ZOOM_SUFFIX);
460 propMap.remove(prefix + WEBSERVER_COMPOSITE_MIT_HEADER_FOOTER_SUFFIX);
461 propMap.remove(prefix + WEBSERVER_COMPOSITE_LAYOUT_SUFFIX);
462 propMap.remove(prefix + WEBSERVER_COMPOSITE_PANEL_COUNT_SUFFIX);
463 for (int i = 0; i < alt.panels().size(); i++) {
464 String panelPrefix = prefix + WEBSERVER_COMPOSITE_PANEL_INFIX + i;
465 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_SHEET_SUFFIX);
466 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_ZOOM_SUFFIX);
467 propMap.remove(panelPrefix + LEGACY_WEBSERVER_COMPOSITE_PANEL_ZENTRIERT_SUFFIX);
468 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_HALIGN_SUFFIX);
469 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_VALIGN_SUFFIX);
470 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_BLATTNAME_SUFFIX);
471 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_TYP_SUFFIX);
472 propMap.remove(panelPrefix + WEBSERVER_COMPOSITE_PANEL_URL_SUFFIX);
473 }
474 }
475 propMap.remove(WEBSERVER_COMPOSITE_PORTS_PROP);
476
477 if (!eintraege.isEmpty()) {
478 var ports = new StringBuilder();
479 for (var eintrag : eintraege) {
480 if (!ports.isEmpty()) ports.append(",");
481 ports.append(eintrag.port());
482 String prefix = WEBSERVER_COMPOSITE_PREFIX + eintrag.port();
483 if (eintrag.aktiv())
484 propMap.put(prefix + WEBSERVER_COMPOSITE_AKTIV_SUFFIX, "true");
485 if (!eintrag.name().isBlank())
486 propMap.put(prefix + WEBSERVER_COMPOSITE_NAME_SUFFIX, eintrag.name().trim());
487 if (eintrag.zoom() != DEFAULT_ZOOM)
488 propMap.put(prefix + WEBSERVER_COMPOSITE_ZOOM_SUFFIX, String.valueOf(eintrag.zoom()));
489 // Default = true → nur explizit "false" persistieren (migrationssicher).
490 if (!eintrag.mitHeaderFooter())
491 propMap.put(prefix + WEBSERVER_COMPOSITE_MIT_HEADER_FOOTER_SUFFIX, "false");
492 propMap.put(prefix + WEBSERVER_COMPOSITE_LAYOUT_SUFFIX, eintrag.layoutJson());
493 propMap.put(prefix + WEBSERVER_COMPOSITE_PANEL_COUNT_SUFFIX, String.valueOf(eintrag.panels().size()));
494 for (int i = 0; i < eintrag.panels().size(); i++) {
495 var panel = eintrag.panels().get(i);
496 String panelPrefix = prefix + WEBSERVER_COMPOSITE_PANEL_INFIX + i;
497 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_TYP_SUFFIX, panel.typ().name());
498 if (panel.typ() == PanelTyp.URL) {
499 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_URL_SUFFIX, panel.externeUrl());
500 } else {
501 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_SHEET_SUFFIX, panel.sheetConfig());
502 }
503 if (panel.zoom() != DEFAULT_ZOOM)
504 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_ZOOM_SUFFIX, String.valueOf(panel.zoom()));
505 if (!PanelAusrichtung.KEIN.equals(panel.horizontalAusrichtung()))
506 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_HALIGN_SUFFIX, panel.horizontalAusrichtung());
507 if (!PanelAusrichtung.KEIN.equals(panel.vertikalAusrichtung()))
508 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_VALIGN_SUFFIX, panel.vertikalAusrichtung());
509 if (panel.blattnameAnzeigen())
510 propMap.put(panelPrefix + WEBSERVER_COMPOSITE_PANEL_BLATTNAME_SUFFIX, "true");
511 }
512 }
513 propMap.put(WEBSERVER_COMPOSITE_PORTS_PROP, ports.toString());
514 }
515 speichernDatei();
516 } catch (Exception e) {
517 logger.error("Fehler beim Speichern der Composite Views", e);
518 }

592 private static int parseZoom(String value) {
593 if (value == null || value.isBlank()) return DEFAULT_ZOOM;
594 try {
595 int zoom = Integer.parseInt(value.trim());
596 if (zoom >= 10 && zoom <= 500) return zoom;
597 logger.warn("Zoom-Wert außerhalb des erlaubten Bereichs (10-500): {}, verwende Standard {}", zoom, DEFAULT_ZOOM);
598 return DEFAULT_ZOOM;

605 private void safeSetLogLevel() {
606 try {
607 var logLevel = getLogLevel();
608 Level level = switch (logLevel) {
609 case "trace" -> Level.TRACE;
610 case "debug" -> Level.DEBUG;
611 case "info" -> Level.INFO;
612 case "warn" -> Level.WARN;
613 case "error" -> Level.ERROR;
614 case "" -> Level.INFO;
615 default -> {
616 logger.warn("Unbekanntes LogLevel: {}", logLevel);
617 yield null;
618 }
619 };
620 if (level != null) {
621 // setLevel(String, Level) ruft intern ctx.updateLoggers() auf und propagiert
622 // die Änderung an alle bereits erzeugten Logger-Instanzen.
623 Configurator.setLevel(Log4J.LOGGERNAME, level);
624 }
625 } catch (Exception e) {
626 logger.error("Fehler beim Setzen des LogLevels", e);
627 }
