184 void insertHeaderInSheet(int headerColor) throws GenerateException {
185 sheet.processBoxinfo("processbox.schweizer.meldeliste.einlesen");
186
187 // Turniersystem oben links (Zeile 0 = ERSTE_HEADER_ZEILE) – inkl. Ranking-Modus
188 SchweizerRankingModus rankingModus = konfigurationSheet.getRankingModus();
189 String rankingInfo = rankingModus == SchweizerRankingModus.OHNE_BUCHHOLZ
190 ? I18n.get("schweizer.meldeliste.ranking.ohne.buchholz")
191 : I18n.get("schweizer.meldeliste.ranking.mit.buchholz");
192 sheet.getSheetHelper().setStringValueInCell(StringCellValue
193 .from(sheet.getXSpreadSheet(), Position.from(0, ERSTE_HEADER_ZEILE),
194 I18n.get("schweizer.meldeliste.header.turniersystem",
195 turnierSystem.getBezeichnung(), rankingInfo))
196 .setEndPosMergeSpaltePlus(2).setCharWeight(FontWeight.BOLD)
197 .setHoriJustify(CellHoriJustify.LEFT).setVertJustify(CellVertJustify2.TOP)
198 .setShrinkToFit(true).setCharColor("00599d"));
199
200 Formation formation = konfigurationSheet.getMeldeListeFormation();
201 int anzSpieler = formation.getAnzSpieler();
202 int spaltenProSpieler = getSpaltenProSpieler();
203 boolean teamnameAktiv = konfigurationSheet.isMeldeListeTeamnameAnzeigen();
204 boolean vereinsnameAktiv = konfigurationSheet.isMeldeListeVereinsnameAnzeigen();
205
206 ColumnProperties colPropNr = ColumnProperties.from().setWidth(NR_SPALTE_WIDTH)
207 .setHoriJustify(CellHoriJustify.CENTER).setVertJustify(CellVertJustify2.CENTER)
208 .margin(MeldeListeKonstanten.CELL_MARGIN);
209 ColumnProperties colPropName = ColumnProperties.from().setWidth(NAME_SPALTE_WIDTH)
210 .setHoriJustify(CellHoriJustify.LEFT).setVertJustify(CellVertJustify2.CENTER)
211 .margin(MeldeListeKonstanten.CELL_MARGIN);

214 StringCellValue nrHeader = StringCellValue
215 .from(sheet.getXSpreadSheet(), Position.from(getTeamNrSpalte(), ZWEITE_HEADER_ZEILE),
216 I18n.get("column.header.nr"))
217 .addColumnProperties(colPropNr)
218 .setCellBackColor(headerColor)
219 .setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder())
220 .setVertJustify(CellVertJustify2.CENTER)
221 .setEndPosMergeZeilePlus(1);

224 // Teamname-Spalte (optional): merged über Zeile 1+2 (ZWEITE + DRITTE Header-Zeile)

226 StringCellValue teamnameHeader = StringCellValue
227 .from(sheet.getXSpreadSheet(), Position.from(1, ZWEITE_HEADER_ZEILE),
228 I18n.get("column.header.teamname"))
229 .addColumnProperties(colPropName.setWidth(TEAMNAME_SPALTE_WIDTH))
230 .setCellBackColor(headerColor)
231 .setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder())
232 .setVertJustify(CellVertJustify2.CENTER)
233 .setEndPosMergeZeilePlus(1);
234 sheet.getSheetHelper().setStringValueInCell(teamnameHeader);
235 }
236
237 // Setzposition-Spalte (SP) – merged über beide Header-Zeilen
238 ColumnProperties colPropSP = ColumnProperties.from().setWidth(NR_SPALTE_WIDTH)
239 .setHoriJustify(CellHoriJustify.CENTER).setVertJustify(CellVertJustify2.CENTER)
240 .margin(MeldeListeKonstanten.CELL_MARGIN);

319 void formatDatenSpalten() throws GenerateException {
320 Formation formation = konfigurationSheet.getMeldeListeFormation();
321 int anzSpieler = formation.getAnzSpieler();
322 int letzteDatenZeile = getLetzteDatenZeileUseMin();
323
324 // Team-Nr Spalte
325 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
326 getTeamNrSpalte(), letzteDatenZeile);
327 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), nrRange,
328 CellProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
329
330 // Bedingte Formatierung Team-Nr
331 String kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
332 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
333 ConditionalFormatHelper.from(sheet, nrRange).clear()
334 .formulaIsText().styleIsFehler().applyAndDoReset()
335 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
336 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
337 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset();
338
339 // Teamname-Spalte (optional)
340 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
341 RangePosition teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, 1, letzteDatenZeile);
342 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), teamnameRange,
343 CellProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()).setShrinkToFit(true));
344 }
345
346 // Spieler-Spalten (Vorname + Nachname [+ Vereinsname])
347 for (int s = 0; s < anzSpieler; s++) {
348 int ersteSpielSpalte = getVornameSpalte(s);
349 int letzteSpielSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
350 ? getVereinsnameSpalte(s)
351 : getNachnameSpalte(s);
352 RangePosition spielerRange = RangePosition.from(ersteSpielSpalte, ERSTE_DATEN_ZEILE,
353 letzteSpielSpalte, letzteDatenZeile);
354 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), spielerRange,
355 CellProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()).setShrinkToFit(true));
356 }
357
358 // Setzposition-Spalte
359 RangePosition spRange = RangePosition.from(getSetzPositionSpalte(), ERSTE_DATEN_ZEILE,
360 getSetzPositionSpalte(), letzteDatenZeile);
361 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), spRange,
362 CellProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
363
364 // Aktiv-Spalte
365 RangePosition aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE,
366 getAktivSpalte(), letzteDatenZeile);
367 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), aktivRange,
368 CellProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
369
370 // Editierbare Felder hervorheben: Spalten 1..Aktiv (ohne Nr-Spalte 0)
371 int aktivSpalte = getAktivSpalte();
372 RangePosition editierbareRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, aktivSpalte, letzteDatenZeile);
373 EditierbaresZelleFormatHelper.anwenden(sheet, editierbareRange);
374 }

376 void formatZeilenfarben() throws GenerateException {
377 Integer geradeColor = konfigurationSheet.getMeldeListeHintergrundFarbeGerade();
378 Integer ungeradeColor = konfigurationSheet.getMeldeListeHintergrundFarbeUnGerade();
379
380 int letzteDatenZeile = getLetzteDatenZeileUseMin();
381 int letzteSpalte = getAktivSpalte();
382
383 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteDatenZeile; zeile++) {
384 RangePosition zeileRange = RangePosition.from(getTeamNrSpalte(), zeile, letzteSpalte, zeile);
385 Integer color = ((zeile - ERSTE_DATEN_ZEILE) % 2 == 0) ? geradeColor : ungeradeColor;
386 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), zeileRange,
387 CellProperties.from().setCellBackColor(color));
388 }
389 }

402 TeamMeldungen getAktiveMeldungen() throws GenerateException {
403 XSpreadsheet xSheet = sheet.getXSpreadSheet();
404 int letzteZeile = letzteZeileMitDaten(xSheet);
405 TeamMeldungen meldungen = new TeamMeldungen();
406 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
407 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(getVornameSpalte(0), zeile));
408 if (vorname == null || vorname.isEmpty()) {
409 continue;
410 }
411 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
412 if (nr <= 0) {
413 continue;
414 }
415 int aktivWert = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getAktivSpalte(), zeile));
416 if (aktivWert == AKTIV_WERT_NIMMT_TEIL) {
417 int setzPos = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getSetzPositionSpalte(), zeile));
418 meldungen.addTeamWennNichtVorhanden(Team.from(nr).setSetzPos(setzPos));
419 }
420 }
421 return meldungen;
422 }

509 int letzteZeileMitDaten(XSpreadsheet xSheet) throws GenerateException {
510 int vornameSpalte = getVornameSpalte(0);
511 int letzte = ERSTE_DATEN_ZEILE - 1;
512 int maxZeile = ERSTE_DATEN_ZEILE + 500;
513 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= maxZeile; zeile++) {
514 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
515 if (vorname != null && !vorname.isEmpty()) {
516 letzte = zeile;
517 }
518 }
519 return letzte;
520 }

527 int letzteZeile = letzteZeileMitDaten(xSheet);
528 if (letzteZeile < ERSTE_DATEN_ZEILE) {
529 return;
530 }
531 int vornameSpalte = getVornameSpalte(0);
532 Map<Integer, List<Integer>> alleNrn = new LinkedHashMap<>();
533 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
534 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
535 if (vorname == null || vorname.isEmpty()) {
536 continue;
537 }
538 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
539 if (nr <= 0) {
540 continue;
541 }
542 alleNrn.computeIfAbsent(nr, k -> new ArrayList<>()).add(zeile);
543 }
544 Map<Integer, List<Integer>> duplikate = new LinkedHashMap<>();
545 for (Map.Entry<Integer, List<Integer>> entry : alleNrn.entrySet()) {
546 if (entry.getValue().size() > 1) {
547 duplikate.put(entry.getKey(), entry.getValue());
548 }
549 }
550 if (duplikate.isEmpty()) {
551 return;
552 }
553 StringBuilder sb = new StringBuilder(I18n.get("error.meldeliste.doppelte.startnummern"));
554 for (Map.Entry<Integer, List<Integer>> entry : duplikate.entrySet()) {
555 sb.append("\nNr. ").append(entry.getKey()).append(": Zeilen ");
556 List<Integer> zeilen = entry.getValue();
557 for (int i = 0; i < zeilen.size(); i++) {
558 if (i > 0) {
559 sb.append(", ");
560 }
561 sb.append(zeilen.get(i) + 1); // 0-basiert → 1-basiert für Benutzer
562 }
563 }
564 throw new GenerateException(sb.toString());
565 }
