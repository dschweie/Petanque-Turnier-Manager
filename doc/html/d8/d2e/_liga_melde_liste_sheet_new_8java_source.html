149 public void createMeldelisteWithParams(String gruppenname) throws GenerateException {
150 var neuesSheet = NewSheet.from(this, SheetNamen.meldeliste(), METADATA_SCHLUESSEL)
151 .pos(DefaultSheetPos.MELDELISTE).hideGrid().tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).setDocVersionWhenNew().create();
152 if (neuesSheet.isDidCreate()) {
153 getKonfigurationSheet().setGruppenname(gruppenname);
154 delegate.upDateSheet();
155 }
156 }

158 @Override

160 return false;
161 }

163 @Override
164 protected void doRun() throws GenerateException {
165 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.LIGA)
166 .prefix(getLogPrefix()).validate()) {
167 return;
168 }
169
170 // Dialog zuerst – bei Abbruch keine Änderungen am Dokument
171 Optional<LigaStartDialog.StartParameter> param;
172 try {
173 param = LigaStartDialog.from(getWorkingSpreadsheet()).show();
174 } catch (com.sun.star.uno.Exception e) {
175 String errMsg = I18n.get("error.dialog.startdialog", e.getMessage());
176 logger.error(errMsg, e);
177 throw new GenerateException(errMsg);
178 }
179
180 if (param.isEmpty()) {
181 return; // Benutzer hat abgebrochen – keine Dokument-Änderungen
182 }
183
184 // Erst nach Bestätigung: TurnierSystem + Page Styles setzen
185 getKonfigurationSheet().update();
186
187 getSheetHelper().removeAllSheetsExclude();
188 createMeldelisteWithParams(param.get().gruppenname());
189 }
