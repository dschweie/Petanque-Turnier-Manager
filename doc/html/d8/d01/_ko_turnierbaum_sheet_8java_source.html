307 static int berechneBracketGroesse(int teamCount) {
308 int klein = Integer.highestOneBit(teamCount);
309 return (teamCount - klein) > (klein / 2) ? klein * 2 : klein;
310 }
311
317 int[] berechneSetzliste(int n) {
318 if (n == 1) {
319 return new int[] { 1 };
320 }
321 int[] prev = berechneSetzliste(n / 2);
322 int[] result = new int[n];
323 for (int i = 0; i < n / 2; i++) {
324 result[2 * i] = prev[i];
325 result[2 * i + 1] = n + 1 - prev[i];
326 }
327 return result;
328 }

608 private void erstelleAlleGruppenBaeume(TeamMeldungen alleMeldungen, int gruppenGroesse,
609 int minLetzteGruppe) throws GenerateException {
610 List<Integer> gruppenGroessen = GruppenAufteilungRechner.berechne(
611 alleMeldungen.size(), gruppenGroesse, minLetzteGruppe);
612 int anzGruppen = gruppenGroessen.size();
613
614 alleGruppenSheetNamenLoeschen();
615
616 var alleTeams = alleMeldungen.teams();
617 int startIndex = 0;
618
619 for (int g = 0; g < anzGruppen; g++) {
620 int groesse = gruppenGroessen.get(g);
621 var gruppenMeldungen = new TeamMeldungen();
622 for (int i = startIndex; i < startIndex + groesse; i++) {
623 gruppenMeldungen.addTeamWennNichtVorhanden(alleTeams.get(i));
624 }
625 startIndex += groesse;
626
627 int bracketGroesse = berechneBracketGroesse(gruppenMeldungen.size());
628 int numRunden = Integer.numberOfTrailingZeros(bracketGroesse);
629 String sheetName = sheetNameFuerGruppe(g, anzGruppen);
630
631 // aktuellerGruppenSheetName setzen BEVOR NewSheet.create() aufgerufen wird,
632 // damit PageStyleHelper.applytoSheet() via getXSpreadSheet() das richtige Sheet trifft
633 this.aktuellerGruppenSheetName = sheetName;
634 try {
635 NewSheet.from(this, sheetName, schluesselFuerGruppe(g, anzGruppen))
636 .pos((short) (DefaultSheetPos.KO_TURNIERBAUM + g))
637 .hideGrid()
638 .tabColor(getKonfigurationSheet().getKoTurnierbaumTabFarbe())
639 .setActiv()
640 .create();
641
642 XSpreadsheet xSheet = getSheetHelper().findByName(sheetName);
643 TurnierSheet.from(xSheet, getWorkingSpreadsheet()).setActiv();
644 String gruppenLabel = (anzGruppen > 1) ? String.valueOf((char) ('A' + g)) : null;
645 erstelleTurnierbaum(xSheet, gruppenMeldungen, numRunden, bracketGroesse,
646 getKonfigurationSheet(), schluesselFuerGruppe(g, anzGruppen), gruppenLabel);
647 } finally {
648 this.aktuellerGruppenSheetName = null;
649 }
650 }

922 public static List<RangePosition> decodeScoreBereiche(String encoded) {
923 if (encoded == null || encoded.isBlank()) {
924 return List.of();
925 }
926 String data = encoded.startsWith(SCORE_DATA_PREFIX)
927 ? encoded.substring(SCORE_DATA_PREFIX.length())
928 : encoded; // Rückwärtskompatibilität: alte Daten ohne Präfix
929 if (data.isBlank()) {
930 return List.of();
931 }
932 var spalteZuZeilen = new TreeMap<Integer, TreeSet<Integer>>();
933 for (var token : data.split("\\|")) {
934 var parts = token.split(",");
935 if (parts.length != 2) continue;
936 try {
937 int spalte = Integer.parseInt(parts[0].trim());
938 int zeile = Integer.parseInt(parts[1].trim());
939 spalteZuZeilen.computeIfAbsent(spalte, _ -> new TreeSet<>()).add(zeile);
940 } catch (NumberFormatException ignoriert) {
941 // Fehlerhafte Tokens überspringen
942 }
943 }
944 var bereiche = new ArrayList<RangePosition>();
945 for (var eintrag : spalteZuZeilen.entrySet()) {
946 int spalte = eintrag.getKey();
947 int startZeile = -1;
948 int letzteZeile = -1;
949 for (int zeile : eintrag.getValue()) {
950 if (startZeile == -1) {
951 startZeile = zeile;
952 letzteZeile = zeile;
953 } else if (zeile == letzteZeile + 1) {
954 letzteZeile = zeile;
955 } else {
956 bereiche.add(RangePosition.from(spalte, startZeile, spalte, letzteZeile));
957 startZeile = zeile;
958 letzteZeile = zeile;
959 }
960 }
961 if (startZeile != -1) {
962 bereiche.add(RangePosition.from(spalte, startZeile, spalte, letzteZeile));
963 }
964 }
965 return bereiche;
966 }

977 private int[] berechneBahnNummern(int anzMatches) {
978 int[] bahnen = new int[anzMatches];
979 if (spielbahn == SpielrundeSpielbahn.N) {
980 for (int i = 0; i < anzMatches; i++) {
981 bahnen[i] = i + 1;
982 }
983 } else if (spielbahn == SpielrundeSpielbahn.R) {
984 for (int i = 0; i < anzMatches; i++) {
985 bahnen[i] = i + 1;
986 }
987 Random rng = RandomSource.asJavaRandom();
988 for (int i = anzMatches - 1; i > 0; i--) {
989 int j = rng.nextInt(i + 1);
990 int tmp = bahnen[i];
991 bahnen[i] = bahnen[j];
992 bahnen[j] = tmp;
993 }
994 }
995 // L und X → bahnen bleibt alle 0
996 return bahnen;
997 }
