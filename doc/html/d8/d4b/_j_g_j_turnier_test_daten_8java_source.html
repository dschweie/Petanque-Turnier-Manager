57 protected void doRun() throws GenerateException {

59 .prefix(getLogPrefix()).validate()) {
60 return;
61 }
62 getSheetHelper().removeAllSheetsExclude(new String[] {});
63
64 // 1. Meldeliste erstellen und mit Testnamen befüllen
65 var meldeListeTestDaten = new JGJMeldeListeSheetTestDaten(
66 getWorkingSpreadsheet(), getFormation(), getAnzTeams(), getGruppengroesse());
67 meldeListeTestDaten.erstellenUndBefuellen();
68
69 // 2. Spielplan generieren
70 TeamMeldungen aktiveMeldungen = meldeListeTestDaten.getAktiveMeldungen();
71 generate(aktiveMeldungen);
72
73 // 3. Zufallsergebnisse einfügen
74 processBoxinfo("processbox.jgj.testdaten.ergebnisse");
75 ergebnisseEinfuegen();
76
77 // 4. Rangliste erstellen
78 processBoxinfo("processbox.erstelle.rangliste");
79 new JGJRanglisteSheet(getWorkingSpreadsheet()).upDateSheet();
80
81 // 5. Kopfzeile und Werbefußzeile setzen

83 getKonfigurationSheet().seitenstileAktualisieren();
84

86 var ranglisteSheet = new JGJRanglisteSheet(getWorkingSpreadsheet());
87 TurnierSheet.from(ranglisteSheet.getXSpreadSheet(), getWorkingSpreadsheet()).setActiv();
88 }

90 private void ergebnisseEinfuegen() throws GenerateException {
91 XSpreadsheet sheet = getXSpreadSheet();
92 RangePosition leseBereich = RangePosition.from(
93 TEAM_A_NR_SPALTE, ERSTE_SPIELTAG_DATEN_ZEILE,
94 TEAM_B_NR_SPALTE, ERSTE_SPIELTAG_DATEN_ZEILE + 500);
95 RangeData data = RangeHelper
96 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), leseBereich)
97 .getDataFromRange();
98
99 for (int i = 0; i < data.size(); i++) {
100 RowData zeile = data.get(i);
101 if (zeile.size() < 2) break;
102
103 int nrA = zeile.get(0).getIntVal(0);
104 if (nrA <= 0) continue; // Gruppenheader-Zeile oder leere Zeile überspringen
105
106 int nrB = zeile.get(1).getIntVal(0);
107 if (nrB <= 0) continue; // Freispiel – kein Ergebnis nötig
108
109 int spielZeile = ERSTE_SPIELTAG_DATEN_ZEILE + i;
110 int gewinner = RandomSource.nextInt(2);
111 int verliererPunkte = RandomSource.nextInt(0, 13);
112 int ergA = (gewinner == 0) ? 13 : verliererPunkte;
113 int ergB = (gewinner == 0) ? verliererPunkte : 13;
114
115 getSheetHelper().setNumberValueInCell(
116 NumberCellValue.from(sheet, Position.from(SPIELPNKT_A_SPALTE, spielZeile)).setValue(ergA));
117 getSheetHelper().setNumberValueInCell(
118 NumberCellValue.from(sheet, Position.from(SPIELPNKT_B_SPALTE, spielZeile)).setValue(ergB));
119 }
120 }
