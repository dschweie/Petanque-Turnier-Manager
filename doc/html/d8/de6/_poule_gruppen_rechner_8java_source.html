101 private static GruppenAnzahl berechneGruppenAnzahl(int anzTeams) {
102 validiereAnzTeams(anzTeams);
103 int q = anzTeams / 4;
104 int r = anzTeams % 4;
105 int vierer;
106 int dreier;
107 switch (r) {
108 case 0 -> { vierer = q; dreier = 0; }
109 case 1 -> { vierer = q - 2; dreier = 3; }
110 case 2 -> { vierer = q - 1; dreier = 2; }
111 case 3 -> { vierer = q; dreier = 1; }
112 default -> throw new IllegalStateException("Unerreichbar: r=" + r);
113 }
114 return new GruppenAnzahl(vierer, dreier);
115 }

117 private static void validiereAnzTeams(int anzTeams) {
118 checkArgument(anzTeams >= 3,
119 "anzTeams muss mindestens 3 sein, war: %s", anzTeams);
120 int r = anzTeams % 4;
121 if (r == 1) {
122 checkArgument(anzTeams >= 9,
123 "Bei Rest 1 (anzTeams %% 4 == 1) werden 3 Dreiergruppen benoetigt "
124 + "(q-2 Vierergruppen), daher mindestens 9 Teams erforderlich; war: %s",

126 }
