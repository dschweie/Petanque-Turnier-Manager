79 public void insertFormulaFuerDoppelteNamenGeradeUngradeFarbe(int erstNameSpalte, int letzteNamespalte,
80 int letzteDatenZeile, ISheet sheet,
81 MeldungenHintergrundFarbeGeradeStyle meldungenHintergrundFarbeGeradeStyle,
82 MeldungenHintergrundFarbeUnGeradeStyle meldungenHintergrundFarbeUnGeradeStyle) throws GenerateException {
83 // -----------------------------------------------
84 // Spieler Namen prüfen auf doppelte namen
85 // -----------------------------------------------
86 RangePosition nameSetPosRange = RangePosition.from(erstNameSpalte, ERSTE_DATEN_ZEILE, letzteNamespalte,
87 letzteDatenZeile);
88 String conditionfindDoppeltNamen;
89 if (erstNameSpalte == letzteNamespalte) {
90 conditionfindDoppeltNamen = "COUNTIF(" + Position.from(erstNameSpalte, 0).getSpalteAddressWith$() + ";"
91 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
92 } else {
93 // Mehrere Namens-Spalten (z.B. Vorname+Nachname): Kombination per COUNTIFS prüfen.
94 // Spalten absolut, Zeile relativ ($B3 / $C3) damit die Formel pro Zeile mitläuft.
95 final int ankerZeile = ERSTE_DATEN_ZEILE + 1; // 1-basiert für Zell-Adresse
96 final int endZeile = MeldungenSpalte.MAX_ANZ_MELDUNGEN + 1;
97 StringBuilder sb = new StringBuilder("COUNTIFS(");
98 for (int sp = erstNameSpalte; sp <= letzteNamespalte; sp++) {
99 if (sp > erstNameSpalte) {
100 sb.append(';');
101 }
102 String spalteAbs = Position.from(sp, 0).getSpalteAddressWith$();
103 sb.append(spalteAbs).append(ankerZeile).append(":").append(spalteAbs).append(endZeile).append(';')
104 .append(spalteAbs).append(ankerZeile);
105 }
106 sb.append(")>1");
107 conditionfindDoppeltNamen = sb.toString();
108 }
109 ConditionalFormatHelper.from(sheet, nameSetPosRange).clear()
110 .formula1(conditionfindDoppeltNamen).operator(ConditionOperator.FORMULA).styleIsFehler()
111 .applyAndDoReset();
112 SheetHelper.faerbeZeilenAbwechselnd(sheet, nameSetPosRange,
113 meldungenHintergrundFarbeGeradeStyle.getFarbe(),
114 meldungenHintergrundFarbeUnGeradeStyle.getFarbe());
115 // -----------------------------------------------
116 }

302 public static String teamNameFormel(String nrAdresse, boolean teamnameAnzeigen,
303 Formation formation, boolean vereinsnameAnzeigen) {
304 String range = "$'" + SheetNamen.meldeliste() + "'.$A$1:$Z$999";
305 if (teamnameAnzeigen) {
306 return "VLOOKUP(" + nrAdresse + ";" + range + ";2;0)";
307 }
308 int spaltenProSpieler = vereinsnameAnzeigen ? 3 : 2;
309 // Erste Spieler-Spalte ohne Teamname: Spalte B (1-basiert für VLOOKUP = 2)
310 int ersterSpielerVlookupIdx = 2;
311 int anzSpieler = formation.getAnzSpieler();
312 StringBuilder sb = new StringBuilder();
313 for (int s = 0; s < anzSpieler; s++) {
314 int vornameIdx = ersterSpielerVlookupIdx + s * spaltenProSpieler;
315 int nachnameIdx = vornameIdx + 1;
316 if (s > 0) {
317 sb.append(" & \" / \" & ");
318 }
319 sb.append("VLOOKUP(").append(nrAdresse).append(";").append(range)
320 .append(";").append(vornameIdx).append(";0)")
321 .append(" & \" \" & ")
322 .append("VLOOKUP(").append(nrAdresse).append(";").append(range)
323 .append(";").append(nachnameIdx).append(";0)");
324 }
325 return sb.toString();
326 }

328 public String formulaSverweisSpielernamen(String spielrNrAdresse) {
329 int ersteName = meldeListe.getMeldungenSpalte().getErsteMeldungNameSpalte();
330 int letzteName = meldeListe.getMeldungenSpalte().getLetzteMeldungNameSpalte();
331 String ersteZelleAddress = Position.from(SPIELER_NR_SPALTE, ERSTE_DATEN_ZEILE).getAddressWith$();
332 String letzteZelleAddress = Position.from(letzteName, 999).getAddressWith$();
333 String range = "$'" + SheetNamen.meldeliste() + "'." + ersteZelleAddress + ":" + letzteZelleAddress;
334
335 if (ersteName == letzteName) {
336 return "VLOOKUP(" + spielrNrAdresse + ";" + range + ";2;0)";
337 }
338 // 2 Namens-Spalten (Supermelee): "Nachname, Vorname" zusammensetzen.
339 int vornameIdx = ersteName - SPIELER_NR_SPALTE + 1;
340 int nachnameIdx = letzteName - SPIELER_NR_SPALTE + 1;
341 return "VLOOKUP(" + spielrNrAdresse + ";" + range + ";" + nachnameIdx + ";0)"
342 + " & \", \" & "
343 + "VLOOKUP(" + spielrNrAdresse + ";" + range + ";" + vornameIdx + ";0)";
344 }

354 public IMeldungen<MLD_LIST_TYPE, MLDTYPE> getMeldungen(final SpielTagNr spieltag,
355 final List<SpielrundeGespielt> spielrundeGespielt, IMeldungen<MLD_LIST_TYPE, MLDTYPE> meldungen)
356 throws GenerateException {
357 checkNotNull(spieltag, "spieltag == null");
358 checkNotNull(meldungen, "meldungen == null");
359
360 int letzteZeile = meldeListe.getMeldungenSpalte().getLetzteMitDatenZeileInSpielerNrSpalte();
361
362 if (letzteZeile >= ERSTE_DATEN_ZEILE) {
363 // daten vorhanden
364 int spieltagSpalte = spieltagSpalte(spieltag);
365
366 // Use getDataArray to get an Array off all Spieler bis SpieltagSpalte
367 RangePosition rangebisSpieltagSpalte = RangePosition.from(SPIELER_NR_SPALTE, ERSTE_DATEN_ZEILE,
368 spieltagSpalte, letzteZeile);
369 RangeData meldungenDaten = RangeHelper.from(meldeListe, rangebisSpieltagSpalte).getDataFromRange();
370
371 // 0 = nr
372 // 1-3 = namen je nach Modus, Tete,Doubl,Tripl
373 // 4 = setzpos
374 // letzte ! spalte = aktuelle Spieltag status
375 for (RowData meldungZeile : meldungenDaten) {
376 int isAktivStatus = meldungZeile.getLast().getIntVal(-1);
377 SpielrundeGespielt status = SpielrundeGespielt.findById(isAktivStatus);
378 if (spielrundeGespielt == null || spielrundeGespielt.contains(status)) {
379 meldungen.addNewWennNichtVorhanden(meldungZeile);
380 }
381 }
382 }
383 return meldungen;
384 }

403 meldeListe.processBoxinfo("processbox.meldeliste.nummern.aktualisieren");

405 int letzteSpielZeile = meldeListe.getMeldungenSpalte().letzteZeileMitSpielerName();

420 int ersteZeileOhneNummer = meldeListe.naechsteFreieDatenZeileInSpielerNrSpalte(); // letzte Zeile ohne Spieler Nr

422 // lücken füllen
423 NumberCellValue celVal = NumberCellValue.from(xSheet, Position.from(SPIELER_NR_SPALTE, ERSTE_DATEN_ZEILE));

427 if (spielrNr == -1) {
428 meldeListe.getSheetHelper()
429 .setNumberValueInCell(celVal.setValue((double) ++letzteSpielerNr).zeile(spielerZeilecntr));

431 }
432 // spieler nach Alphabet sortieren
