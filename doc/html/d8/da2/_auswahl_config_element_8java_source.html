32 public AuswahlConfigElement(GuiFactoryCreateParam guiFactoryCreateParam,
33 AuswahlConfigProperty configProperty, WorkingSpreadsheet workingSpreadsheet) {
34 this.configProperty = checkNotNull(configProperty);
35 this.workingSpreadsheet = checkNotNull(workingSpreadsheet);
36 var labelText = configProperty.getDescription() != null ? configProperty.getDescription() : configProperty.getKey();
37 labelPlusCombobox = LabelPlusCombobox.from(guiFactoryCreateParam).labelText(labelText)
38 .helpText(labelText).addAuswahlItems(configProperty.getAuswahl())
39 .addListener(this).select(getComboboxItemValue());
40 }

47 return new HorizontalLayout();

50 private void setPropertyValue(String keyVal) {
51 if (StringTools.equalsIgnoreCase(getPropertyValue(), keyVal)) {
52 // nichts zu tun
53 return;
54 }
55 DocumentPropertiesHelper docPropHelper = new DocumentPropertiesHelper(workingSpreadsheet);
56 docPropHelper.setStringProperty(configProperty.getKey(), keyVal);
57 }
58
59 private String getComboboxItemValue() {
60 // find text
61 String key = getPropertyValue();
62 ComboBoxItem itemFromVal = configProperty.getAuswahl().stream()
63 .filter(cmbItem -> key.equalsIgnoreCase(cmbItem.getKey())).findAny()
64 .orElse(configProperty.getAuswahl().get(0));
65 return itemFromVal.getText();
66 }
67
71 private String getPropertyValue() {
72 DocumentPropertiesHelper docPropHelper = new DocumentPropertiesHelper(workingSpreadsheet);
73 return docPropHelper.getStringProperty(configProperty.getKey(), configProperty.getDefaultVal());
74 }
75
76 @Override
77 public void disposing(EventObject arg0) {
78 workingSpreadsheet = null;
79 configProperty = null;
80 labelPlusCombobox = null;
81 }

88 }
