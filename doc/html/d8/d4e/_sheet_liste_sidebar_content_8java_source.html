358 return;
359 }

361 if (idx < 0 || idx >= baumEintraege.size()) {

368 // (innerhalb von pBox->Select()) verursachen VCL-Re-Entranz und SIGSEGV, weil

373 if (!isUiAlive(gen) || sheetListBox == null) {
374 return;
375 }
376 logger.debug("itemDispatcher: verarbeiteItemAuswahl({})", idx);
377 verarbeiteItemAuswahl(idx);
378 }, null);
379 } else {
380 verarbeiteItemAuswahl(idx);
