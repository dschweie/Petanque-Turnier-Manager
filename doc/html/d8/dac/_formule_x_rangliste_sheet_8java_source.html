209 private record RanglisteAccumulation(List<FormuleXErgebnis> ergebnisse,
210 Map<Integer, Integer> siegeMap, Map<Integer, Integer> wertungMap) {}

221 private static final int STATS_EIGENE = 0;
222 private static final int STATS_KASSIERT = 1;
223 private static final int STATS_SIEGE = 2;
224 private static final int STATS_WERTUNG = 3;
225 private static final int STATS_LEN = 4;

227 private RanglisteAccumulation leseAlleRunden(TeamMeldungen aktiveMeldungen, int bisSpielrunde)
228 throws GenerateException {
229 Map<Integer, int[]> statsMap = new HashMap<>();
230 Map<Integer, Boolean> freilosMap = new HashMap<>();
231
232 for (Team team : aktiveMeldungen.teams()) {
233 statsMap.put(team.getNr(), new int[STATS_LEN]);
234 freilosMap.put(team.getNr(), false);
235 }
236
237 int siegaufschlag = new FormuleX().getSiegaufschlag(bisSpielrunde);
238 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
239 for (int runde = 1; runde <= bisSpielrunde; runde++) {
240 SheetRunner.testDoCancelTask();
241 XSpreadsheet rundeSheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
242 SheetMetadataHelper.SCHLUESSEL_FORMULEX_SPIELRUNDE_PREFIX + runde,
243 SpielRundeNr.from(runde).getNr() + ". Spielrunde");
244 if (rundeSheet == null) {
245 LOGGER.debug("leseAlleRunden: Runde {} – Sheet nicht gefunden, übersprungen", runde);
246 continue;
247 }
248 leseRundeEin(rundeSheet, aktiveMeldungen, statsMap, freilosMap, siegaufschlag);
249 }
250
251 List<FormuleXErgebnis> ergebnisse = new ArrayList<>();
252 Map<Integer, Integer> siegeMap = new HashMap<>();
253 Map<Integer, Integer> wertungMap = new HashMap<>();
254 for (Team team : aktiveMeldungen.teams()) {
255 int[] stats = statsMap.getOrDefault(team.getNr(), new int[STATS_LEN]);
256 boolean hatteFreilos = freilosMap.getOrDefault(team.getNr(), false);
257 ergebnisse.add(new FormuleXErgebnis(team.getNr(),
258 stats[STATS_EIGENE], stats[STATS_KASSIERT], List.of(), hatteFreilos));
259 siegeMap.put(team.getNr(), stats[STATS_SIEGE]);
260 wertungMap.put(team.getNr(), stats[STATS_WERTUNG]);
261 }
262 return new RanglisteAccumulation(ergebnisse, siegeMap, wertungMap);
263 }

265 private void leseRundeEin(XSpreadsheet rundeSheet, TeamMeldungen aktiveMeldungen,
266 Map<Integer, int[]> statsMap, Map<Integer, Boolean> freilosMap, int siegaufschlag)
267 throws GenerateException {
268 var readRange = RangePosition.from(
269 FormuleXAbstractSpielrundeSheet.TEAM_A_SPALTE,
270 FormuleXAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE,
271 FormuleXAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE,
272 FormuleXAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE + 999);
273 var rowsData = RangeHelper
274 .from(rundeSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
275 .getDataFromRange();
276
277 for (var row : rowsData) {
278 if (row.size() < 2) {
279 break;
280 }
281 int nrA = row.get(0).getIntVal(0);
282 if (nrA <= 0) {
283 break;
284 }
285 if (aktiveMeldungen.getTeam(nrA) == null) {
286 continue;
287 }
288
289 int nrB = row.get(1).getIntVal(0);
290 if (nrB <= 0) {
291 freilosMap.put(nrA, true);
292 int[] s = statsMap.computeIfAbsent(nrA, k -> new int[STATS_LEN]);
293 s[STATS_SIEGE]++; // Freilos zählt als Sieg (Doku 4.2)
294 s[STATS_WERTUNG] += 126;
295 continue;
296 }
297 if (aktiveMeldungen.getTeam(nrB) == null) {
298 continue;
299 }
300
301 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0;
302 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0;
303
304 if (ergA > 0 || ergB > 0) {
305 int[] sA = statsMap.computeIfAbsent(nrA, k -> new int[STATS_LEN]);
306 int[] sB = statsMap.computeIfAbsent(nrB, k -> new int[STATS_LEN]);
307 sA[STATS_EIGENE] += ergA;
308 sA[STATS_KASSIERT] += ergB;
309 sB[STATS_EIGENE] += ergB;
310 sB[STATS_KASSIERT] += ergA;
311 if (ergA > ergB) {
312 sA[STATS_SIEGE]++;
313 sA[STATS_WERTUNG] += siegaufschlag + ergA + (ergA - ergB);
314 sB[STATS_WERTUNG] += ergB;
315 } else if (ergB > ergA) {
316 sB[STATS_SIEGE]++;
317 sB[STATS_WERTUNG] += siegaufschlag + ergB + (ergB - ergA);
318 sA[STATS_WERTUNG] += ergA;
319 } else {
320 // Unentschieden: kein Sieg, beide bekommen Verlierer-Formel (= eigene Punkte)
321 sA[STATS_WERTUNG] += ergA;
322 sB[STATS_WERTUNG] += ergB;
323 }
324 }
325 }

328 private Map<Integer, String> leseTeamnamen(FormuleXMeldeListeSheetUpdate meldeliste) throws GenerateException {
329 Map<Integer, String> result = new HashMap<>();
330 XSpreadsheet mlSheet = meldeliste.getXSpreadSheet();
331 if (mlSheet == null) {
332 return result;
333 }
334
335 XSpreadsheetDocument doc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
336 int nrSpalte = meldeliste.getNrSpalte();
337 int ersteZeile = meldeliste.getErsteDatenZeile();
338
339 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
340 int nameSpalte = meldeliste.getTeamnameSpalte();
341 int maxSpalte = Math.max(nrSpalte, nameSpalte);
342 var data = RangeHelper
343 .from(mlSheet, doc, RangePosition.from(0, ersteZeile, maxSpalte, ersteZeile + 999))
344 .getDataFromRange();
345 for (RowData row : data) {
346 if (row.size() <= nrSpalte) {
347 break;
348 }
349 int nr = row.get(nrSpalte).getIntVal(0);
350 if (nr <= 0) {
351 break;
352 }
353 String name = nameSpalte < row.size() ? row.get(nameSpalte).getStringVal() : null;
354 result.put(nr, name != null ? name.trim() : "");
355 }
356 } else {
357 int anzSpieler = konfigurationSheet.getMeldeListeFormation().getAnzSpieler();
358 int[] vorSpalten = new int[anzSpieler];
359 int[] nachSpalten = new int[anzSpieler];
360 int maxSpalte = nrSpalte;
361 for (int s = 0; s < anzSpieler; s++) {
362 vorSpalten[s] = meldeliste.getVornameSpalte(s);
363 nachSpalten[s] = meldeliste.getNachnameSpalte(s);
364 maxSpalte = Math.max(maxSpalte, Math.max(vorSpalten[s], nachSpalten[s]));
365 }
366 var data = RangeHelper
367 .from(mlSheet, doc, RangePosition.from(0, ersteZeile, maxSpalte, ersteZeile + 999))
368 .getDataFromRange();
369 for (RowData row : data) {
370 if (row.size() <= nrSpalte) {
371 break;
372 }
373 int nr = row.get(nrSpalte).getIntVal(0);
374 if (nr <= 0) {
375 break;
376 }
377 var sb = new StringBuilder();
378 for (int s = 0; s < anzSpieler; s++) {
379 String vn = vorSpalten[s] < row.size() ? row.get(vorSpalten[s]).getStringVal() : null;
380 String nn = nachSpalten[s] < row.size() ? row.get(nachSpalten[s]).getStringVal() : null;
381 String spielerName = erstelleSpielerName(vn, nn);
382 if (!spielerName.isEmpty()) {
383 if (sb.length() > 0) {
384 sb.append(" / ");
385 }
386 sb.append(spielerName);
387 }
388 }
389 result.put(nr, sb.toString());
390 }
391 }
392 return result;
393 }

395 private static String erstelleSpielerName(String vorname, String nachname) {
396 String vn = vorname != null ? vorname.trim() : "";
397 String nn = nachname != null ? nachname.trim() : "";
398 if (vn.isEmpty() && nn.isEmpty()) {
399 return "";
400 }
401 if (vn.isEmpty()) {
402 return nn;
403 }
404 if (nn.isEmpty()) {
405 return vn;
406 }
407 return vn + " " + nn;
408 }
