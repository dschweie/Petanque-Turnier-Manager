182 private void beimExport() {
183 UnoControlsHelper c = this.controls;
184 if (c == null) {
185 return;
186 }
187 SpielerDbDateiFormat format = aktuellesFormat();
188 String pfadText = c.leseText("edZiel").strip();
189 if (pfadText.isEmpty()) {
190 zeigeFehler(I18n.get("spielerdb.export.fehler.kein_pfad"));
191 return;
192 }
193 EnumSet<ExportEntity> scope = leseScope(format, c);
194 if (format != SpielerDbDateiFormat.SQLITE_BACKUP && scope.isEmpty()) {
195 zeigeFehler(I18n.get("spielerdb.export.fehler.kein_scope"));
196 return;
197 }
198
199 Path target = Path.of(pfadText);
200 ExportRequest request = new ExportRequest(format,
201 format == SpielerDbDateiFormat.SQLITE_BACKUP ? EnumSet.allOf(ExportEntity.class) : scope,
202 new AllExportFilter(), target);
203
204 try {
205 SpielerDbExportData data = ladeDaten(format, scope);
206 SpielerDbExporter exporter = exporterFuer(format);
207 exporter.export(data, request);
208 settings.merkeLetztenPfad(format, pfadText);
209 zeigeErfolg(format, scope, target, data);
210 } catch (SpielerDbException e) {
211 logger.error("Export fehlgeschlagen", e);
212 zeigeFehler(I18n.get("spielerdb.export.fehler.allgemein",
213 e.getMessage() == null ? "" : e.getMessage()));
214 }

225 private SpielerDbExporter exporterFuer(SpielerDbDateiFormat format) {
226 return switch (format) {
227 case CSV -> new SpielerDbCsvExporter();
228 case JSON -> new SpielerDbJsonExporter();
229 case CALC -> new SpielerDbCalcExporter(xContext);
230 case SQLITE_BACKUP -> new SpielerDbSqliteBackupExporter(dbConnection);
231 };
232 }

234 private static EnumSet<ExportEntity> leseScope(SpielerDbDateiFormat format, UnoControlsHelper c) {
235 if (format == SpielerDbDateiFormat.SQLITE_BACKUP) {
236 return EnumSet.noneOf(ExportEntity.class);
237 }
238 if (format == SpielerDbDateiFormat.CSV) {
239 // Flache CSV deckt nur Spieler+Vereinsname ab — Scope ist fix.
240 return EnumSet.of(ExportEntity.SPIELER);
241 }
242 EnumSet<ExportEntity> scope = EnumSet.noneOf(ExportEntity.class);
243 if (c.istAngekreuzt("cbSpieler")) {
244 scope.add(ExportEntity.SPIELER);
245 }
246 if (c.istAngekreuzt("cbVereine")) {
247 scope.add(ExportEntity.VEREINE);
248 }
249 if (c.istAngekreuzt("cbLabels")) {
250 scope.add(ExportEntity.LABELS);
251 }
252 return scope;
253 }

255 private void zeigeErfolg(SpielerDbDateiFormat format, EnumSet<ExportEntity> scope, Path target,
256 SpielerDbExportData data) {
257 List<String> zeilen = new ArrayList<>();
258 zeilen.add(I18n.get("spielerdb.export.fertig.kopf", target.toString()));
259 zeilen.add("");
260 if (format == SpielerDbDateiFormat.SQLITE_BACKUP) {
261 zeilen.add(I18n.get("spielerdb.export.fertig.sqlite"));
262 } else {
263 if (scope.contains(ExportEntity.SPIELER)) {
264 zeilen.add(I18n.get("spielerdb.export.fertig.spieler", data.spieler().size()));
265 }
266 if (scope.contains(ExportEntity.VEREINE)) {
267 zeilen.add(I18n.get("spielerdb.export.fertig.vereine", data.vereine().size()));
268 }
269 if (scope.contains(ExportEntity.LABELS)) {
270 zeilen.add(I18n.get("spielerdb.export.fertig.labels", data.labels().size()));
271 }
272 if (scope.contains(ExportEntity.SPIELER) && scope.contains(ExportEntity.LABELS)) {
273 zeilen.add(I18n.get("spielerdb.export.fertig.junction", data.spielerLabels().size()));
274 }
275 }
276 MessageBox.from(xContext, MessageBoxTypeEnum.INFO_OK)
277 .caption(I18n.get("spielerdb.export.dialog.titel"))
278 .message(String.join("\n", zeilen))
279 .show();
280 }
