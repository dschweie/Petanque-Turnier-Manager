212 private void beimImport() {
213 UnoControlsHelper c = this.controls;
214 if (c == null) {
215 return;
216 }
217 SpielerDbDateiFormat format = aktuellesFormat();
218 String pfadText = c.leseText("edQuelle").strip();
219 if (pfadText.isEmpty()) {
220 zeigeFehler(I18n.get("spielerdb.import.fehler.kein_pfad"));
221 return;
222 }
223 Path quelle = Path.of(pfadText);
224
225 if (format == SpielerDbDateiFormat.SQLITE_BACKUP) {
226 fuehreSqliteRestoreAus(quelle);
227 settings.merkeLetztenPfad(format, pfadText);
228 return;
229 }
230
231 EnumSet<ExportEntity> scope = scopeFuer(format, c);
232 if (scope.isEmpty()) {
233 zeigeFehler(I18n.get("spielerdb.import.fehler.kein_scope"));
234 return;
235 }
236 ImportModus modus = aktuellerModus();
237 ImportRequest request = new ImportRequest(format, scope, quelle, modus, false);
238
239 try {
240 SpielerDbImportReader reader = readerFuer(format);
241 ImportRohdaten roh = reader.read(request);
242 ValidierteDaten vd = new SpielerDbImportValidator().validiere(roh);
243 ImportErgebnis erg = new SpielerDbImporter(dbConnection).importiere(vd, request);
244 settings.merkeLetztenPfad(format, pfadText);
245 zeigeErfolg(erg);
246 } catch (SpielerDbValidationException e) {
247 logger.error("Import-Validierung fehlgeschlagen", e);
248 zeigeFehler(I18n.get("spielerdb.import.fehler.validierung",
249 String.join("\n• ", e.fehler())));
250 } catch (SpielerDbException e) {
251 logger.error("Import fehlgeschlagen", e);
252 zeigeFehler(I18n.get("spielerdb.import.fehler.allgemein",
253 e.getMessage() == null ? "" : e.getMessage()));
254 }
255 }

257 private void fuehreSqliteRestoreAus(Path quelle) {
258 MessageBoxResult antwort = MessageBox.from(xContext, MessageBoxTypeEnum.WARN_YES_NO)
259 .caption(I18n.get("spielerdb.import.dialog.titel"))
260 .message(I18n.get("spielerdb.import.restore.bestaetigung", quelle.toString()))
261 .show();
262 if (antwort != MessageBoxResult.YES) {
263 return;
264 }
265 try {
266 new SpielerDbSqliteRestoreImporter().restore(quelle);
267 MessageBox.from(xContext, MessageBoxTypeEnum.INFO_OK)
268 .caption(I18n.get("spielerdb.import.dialog.titel"))
269 .message(I18n.get("spielerdb.import.restore.fertig"))
270 .show();
271 } catch (SpielerDbException e) {
272 logger.error("SQLite-Restore fehlgeschlagen", e);
273 zeigeFehler(I18n.get("spielerdb.import.fehler.allgemein",
274 e.getMessage() == null ? "" : e.getMessage()));
275 }

278 private SpielerDbImportReader readerFuer(SpielerDbDateiFormat format)
279 throws SpielerDbException {
280 return switch (format) {
281 case CSV -> new SpielerDbCsvImportReader();
282 case JSON -> new SpielerDbJsonImportReader();
283 case CALC -> new SpielerDbCalcImportReader(xContext);
284 case SQLITE_BACKUP ->
285 throw new SpielerDbException("SQLite-Restore läuft über separaten Pfad");
286 };

289 private static EnumSet<ExportEntity> scopeFuer(SpielerDbDateiFormat format, UnoControlsHelper c) {
290 if (format == SpielerDbDateiFormat.CSV) {
291 // Flache CSV deckt nur Spieler+Vereinsname ab — Scope ist fix.
292 return EnumSet.of(ExportEntity.SPIELER);
293 }
294 EnumSet<ExportEntity> scope = EnumSet.noneOf(ExportEntity.class);
295 if (c.istAngekreuzt("cbSpieler")) {
296 scope.add(ExportEntity.SPIELER);
297 }
298 if (c.istAngekreuzt("cbVereine")) {
299 scope.add(ExportEntity.VEREINE);
300 }
301 if (c.istAngekreuzt("cbLabels")) {
302 scope.add(ExportEntity.LABELS);
303 }
304 return scope;
305 }

316 erg.labelsEingefuegt(), erg.labelsAktualisiert(), erg.labelsUebersprungen()));
317 zeilen.add(I18n.get("spielerdb.import.fertig.junction", erg.junctionEingefuegt()));
318 if (!erg.warnungen().isEmpty()) {
319 zeilen.add("");
320 zeilen.add(I18n.get("spielerdb.import.fertig.warnungen", erg.warnungen().size()));

322 zeilen.add("• " + w.text());

325 MessageBox.from(xContext, MessageBoxTypeEnum.INFO_OK)
326 .caption(I18n.get("spielerdb.import.dialog.titel"))
327 .message(String.join("\n", zeilen))
