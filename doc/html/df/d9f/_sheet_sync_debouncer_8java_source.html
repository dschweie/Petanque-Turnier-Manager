80 private void scheduleMitDelay(XSpreadsheetDocument xDoc, String schluessel, Runnable check,
81 long delayMs) {
82 checkNotNull(xDoc);
83 checkNotNull(schluessel);
84 checkNotNull(check);
85 String key = mapKey(xDoc, schluessel);
86 ScheduledFuture<?> neu = executor.schedule(() -> ausfuehren(key, check), delayMs,
87 TimeUnit.MILLISECONDS);
88 ScheduledFuture<?> alt = ausstehend.put(key, neu);
89 if (alt != null) {
