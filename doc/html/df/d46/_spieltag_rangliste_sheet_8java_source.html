247 protected void berechnungUndSchreiben(XSpreadsheet sheet, int anzSpielRunden) throws GenerateException {
248 processBoxinfo("processbox.spieltage.ergebnisse.einfuegen");
249
250 if (anzSpielRunden < 1) {
251 return;
252 }
253
254 List<Integer> spielerNrList = leseSpielerNrInSheetOrdnung();
255 if (spielerNrList.isEmpty()) {
256 return;
257 }
258
259 // Die vertikalen Ergebnisspalten der Spielrunde-Sheets sind formel-gestützte
260 // Views auf die horizontalen Ergebnisse. Bei deaktiviertem AutoCalc können
261 // sie veraltete Werte zeigen, wenn unmittelbar zuvor Ergebnisse geändert
262 // wurden. Vor dem Block-Read einmal recalculaten.
263 getxCalculatable().calculate();
264
265 SpieltagErgebnisse ergebnisse = new SpielrundeErgebnisLeser(getWorkingSpreadsheet(), getSpieltagNr())
266 .lese(anzSpielRunden);
267
268 int nichtgespieltPlus = getKonfigurationSheet().getNichtGespielteRundePlus();
269 int nichtgespieltMinus = getKonfigurationSheet().getNichtGespielteRundeMinus();
270 int ersteSummeSpalte = getErsteSummeSpalte();
271 int letzteSummenSpalte = ersteSummeSpalte + SuperMeleeSummenSpalten.PUNKTE_DIV_OFFS;
272 int ngSpalte = nichtGespieltSpalteNr();
273 int ersteDatenzeile = ERSTE_DATEN_ZEILE;
274 int letzteDatenzeile = ersteDatenzeile + spielerNrList.size() - 1;
275
276 RangeData runden = new RangeData();
277 RangeData nichtGespielt = new RangeData();
278
279 for (Integer spielerNr : spielerNrList) {
280 SheetRunner.testDoCancelTask();
281 RowData rundenRow = runden.addNewRow();
282 int spielePlus = 0;
283 int spieleMinus = 0;
284 int punktePlus = 0;
285 int punkteMinus = 0;
286 boolean hatLuecke = false;
287
288 for (int rundeNr = 1; rundeNr <= anzSpielRunden; rundeNr++) {
289 Optional<RundenErgebnis> erg = ergebnisse.ergebnis(rundeNr, spielerNr);
290 int plus;
291 int minus;
292 if (erg.isPresent()) {
293 plus = erg.get().plus();
294 minus = erg.get().minus();
295 } else {
296 plus = nichtgespieltPlus;
297 minus = nichtgespieltMinus;
298 hatLuecke = true;
299 }
300 if (ergebnisse.istRundeNichtGespielt(rundeNr)) {
301 // Runde komplett leer – die Vertikalen liefern zwar 0/0, semantisch
302 // hat dieser Spieler aber nicht gespielt.
303 hatLuecke = true;
304 }
305 rundenRow.newInt(plus);
306 rundenRow.newInt(minus);
307 punktePlus += plus;
308 punkteMinus += minus;
309 if (plus > minus) {
310 spielePlus++;
311 } else if (minus > plus) {
312 spieleMinus++;
313 }
314 }
315 rundenRow.newInt(spielePlus);
316 rundenRow.newInt(spieleMinus);
317 rundenRow.newInt(spielePlus - spieleMinus);
318 rundenRow.newInt(punktePlus);
319 rundenRow.newInt(punkteMinus);
320 rundenRow.newInt(punktePlus - punkteMinus);
321
322 nichtGespielt.addNewRow().newString(hatLuecke ? "x" : "");
323 }
324
325 RangeHelper.from(this,
326 RangePosition.from(ERSTE_SPIELRUNDE_SPALTE, ersteDatenzeile, letzteSummenSpalte, letzteDatenzeile))
327 .setDataInRange(runden);
328 RangeHelper.from(this,
329 RangePosition.from(ngSpalte, ersteDatenzeile, ngSpalte, letzteDatenzeile))
330 .setDataInRange(nichtGespielt);
331 }

365 private void formatNichtGespieltRunden(int anzSpielRunden, int nichtGespieltGeradeFarbe,
366 int nichtGespieltUnGeradeFarbe) throws GenerateException {
367 if (anzSpielRunden < 1) {
368 return;
369 }
370
371 int letzteDatenzeile = getSpielerSpalte().getLetzteMitDatenZeileInSpielerNrSpalte();
372 int validSpalteNr = validateSpalte();
373 String formulaSortError = "LEN(TRIM(INDIRECT(ADDRESS(ROW();" + (validSpalteNr + 1) + "))))>0";
374
375 int ersteSpalteVertikaleErgebnisse = SpielrundeSheetKonstanten.ERSTE_SPALTE_VERTIKALE_ERGEBNISSE;
376 int spielrundeSheetErsteDatenzeile = SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE;
377 Position erstePos = Position.from(ersteSpalteVertikaleErgebnisse, spielrundeSheetErsteDatenzeile);
378 Position letztePosPlusPunkte = Position.from(SpielrundeSheetKonstanten.SPALTE_VERTIKALE_ERGEBNISSE_PLUS,
379 1000 + spielrundeSheetErsteDatenzeile);
380 String suchMatrixPlusPunkte = erstePos.getAddressWith$() + ":" + letztePosPlusPunkte.getAddressWith$();
381 String verweisAufSpalteSpielerNr = "INDIRECT(ADDRESS(ROW();" + (SPIELER_NR_SPALTE + 1) + ";4))";
382
383 int geradeColor = getKonfigurationSheet().getRanglisteHintergrundFarbeGerade();
384 int ungeradeColor = getKonfigurationSheet().getRanglisteHintergrundFarbeUnGerade();
385
386 for (int spielRunde = 1; spielRunde <= anzSpielRunden; spielRunde++) {
387 SheetRunner.testDoCancelTask();
388
389 String formulaSheetName = "$'"
390 + aktuelleSpielrundeSheet.getSheetName(getSpieltagNr(), SpielRundeNr.from(spielRunde)) + "'.";
391 String isnaFormula = "ISNA(VLOOKUP(" + verweisAufSpalteSpielerNr + ";" + formulaSheetName
392 + suchMatrixPlusPunkte + ";2;0))";
393
394 int plusSpalte = ERSTE_SPIELRUNDE_SPALTE + ((spielRunde - 1) * ANZAHL_SPALTEN_IN_SPIELRUNDE);
395 RangePosition rundenRange = RangePosition.from(plusSpalte, ERSTE_DATEN_ZEILE, plusSpalte + 1,
396 letzteDatenzeile);
397
398 SheetHelper.faerbeZeilenAbwechselnd(this, rundenRange, geradeColor, ungeradeColor);
399 ConditionalFormatHelper cfHelper = ConditionalFormatHelper.from(this, rundenRange).clear();
400 cfHelper.formula1(formulaSortError).operator(ConditionOperator.FORMULA)
401 .style(new FehlerStyle()).applyAndDoReset();
402 cfHelper.formula1("AND(ISEVEN(ROW());" + isnaFormula + ")").isFormula()
403 .style(new NichtGespieltHintergrundFarbeGeradeStyle(nichtGespieltGeradeFarbe)).applyAndDoReset();
404 cfHelper.formula1("AND(ISODD(ROW());" + isnaFormula + ")").isFormula()
405 .style(new NichtGespieltHintergrundFarbeUnGeradeStyle(nichtGespieltUnGeradeFarbe)).applyAndDoReset();
406 }
407 }
