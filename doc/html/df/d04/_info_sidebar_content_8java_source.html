267 String getPluginVersion() {
268 var context = getCurrentSpreadsheet().getxContext();
269 String installiert;
270 boolean updateVerfuegbar = false;
271 String neu = null;
272 try {
273 var service = ReleaseUpdateService.get();
274 installiert = service.getInstallierteVersion()
275 .orElse(ExtensionsHelper.from(context).getVersionNummer());
276 updateVerfuegbar = service.isUpdateVerfuegbar()
277 || GlobalProperties.get().isNewVersionCheckImmerTrue();
278 if (updateVerfuegbar) {
279 neu = service.getNeuesteVersionTag().orElse(null);
280 }
281 } catch (IllegalStateException e) {
282 installiert = ExtensionsHelper.from(context).getVersionNummer();
283 }
284 if (installiert == null) {
285 return "–";
286 }
287 if (updateVerfuegbar && neu != null) {
288 return I18n.get("sidebar.info.version.neu", installiert, neu);
289 }
290 return I18n.get("sidebar.info.version", installiert);
291 }

305 private void versionTextFarbeAktualisieren(XControl control) {
306 if (control == null) {
307 return;
308 }
309 try {
310 var modell = control.getModel();
311 if (isVersionUpdateVerfuegbar()) {
312 var props = Lo.qi(XPropertySet.class, modell);
313 if (props != null) {
314 props.setPropertyValue("TextColor", Integer.valueOf(VERSION_UPDATE_TEXT_COLOR));
315 }
316 } else {
317 var state = Lo.qi(com.sun.star.beans.XPropertyState.class, modell);
318 if (state != null) {
319 state.setPropertyToDefault("TextColor");
320 }
321 }
322 } catch (Exception e) {
323 logger.error("Fehler beim Setzen der Versions-Textfarbe", e);
324 }
325 }
