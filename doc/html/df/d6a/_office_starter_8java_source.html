85 logger.info("Loading Office...");

89 } else {
90 socketContext(); // connects to office via a socket

95 logger.error(errorMsg);
96 throw new RuntimeException(errorMsg);
97 }
98
99 // get the remote office service manager
100 mcFactory = xComponentContext.getServiceManager();
101 if (mcFactory == null) {
102 String errorMsg = "Office Service Manager is unavailable";
103 logger.error(errorMsg);
104 throw new RuntimeException(errorMsg);
105 }

108 xDesktop = Lo.createInstanceMCF(XDesktop.class, "com.sun.star.frame.Desktop", mcFactory, xComponentContext);

112 throw new RuntimeException(errorMsg);

116 xComponentLoader = Lo.qi(XComponentLoader.class, xDesktop);

134 // https://forum.openoffice.org/en/forum/viewtopic.php?f=44&t=1014

136 private void socketContext() {
137 try {
138 ArrayList<String> cmd = new ArrayList<String>();
139 // requires soffice to be in Linux PATH env var.
140 cmd.add(SOFFICE_BIN);
141 if (this.headless) {
142 cmd.add("--headless");
143 }
144 cmd.add("--accept=socket,host=localhost,port=" + SOCKET_PORT + ";urp;");
145 if (userInstallation != null) {
146 cmd.add("-env:UserInstallation=" + userInstallation);
147 }
148 cmd.add("--norestore");
149 cmd.add("--nofirststartwizard");
150
151 logger.info(String.join(",", cmd));
152
153 ProcessBuilder pb = new ProcessBuilder(cmd);
154 pb.inheritIO();
155 sofficeProcess = pb.start();
156 if (sofficeProcess != null) {
157 logger.info("Office process created");
158 }
159
160 // Create a local Component Context
161 XComponentContext localContext = Bootstrap.createInitialComponentContext(null);
162
163 // Get the local service manager
164 XMultiComponentFactory localFactory = localContext.getServiceManager();
165
166 // connect to Office via its socket
167 XConnector connector = Lo.qi(XConnector.class,
168 localFactory.createInstanceWithContext("com.sun.star.connection.Connector", localContext));
169
170 com.sun.star.connection.XConnection connection = verbindeMitRetry(connector);
171
172 // create a bridge to Office via the socket
173 XBridgeFactory bridgeFactory = Lo.qi(XBridgeFactory.class,
174 localFactory.createInstanceWithContext("com.sun.star.bridge.BridgeFactory", localContext));
175
176 // create a nameless bridge with no instance provider
177 XBridge bridge = bridgeFactory.createBridge("socketBridgeAD", "urp", connection, null);
178 this.bridgeComponent = Lo.qi(XComponent.class, bridge);
179
180 // get the remote service manager
181 XMultiComponentFactory serviceManager = Lo.qi(XMultiComponentFactory.class,
182 bridge.getInstance("StarOffice.ServiceManager"));
183
184 // retrieve Office's remote component context as a property
185 XPropertySet props = Lo.qi(XPropertySet.class, serviceManager);
186 // initObject);
187 Object defaultContext = props.getPropertyValue("DefaultContext");
188
189 // get the remote interface XComponentContext
190 xComponentContext = Lo.qi(XComponentContext.class, defaultContext);
191 } catch (java.lang.Exception e) {
192 logger.error("Unable to socket connect to Office", e);
193 }
194 }

202 private com.sun.star.connection.XConnection verbindeMitRetry(XConnector connector) throws InterruptedException {
203 long deadline = System.nanoTime() + TimeUnit.SECONDS.toNanos(CONNECT_TIMEOUT_SEKUNDEN);
204 ExecutorService connectExec = Executors.newSingleThreadExecutor();
205 try {
206 java.lang.Exception letzterFehler = null;
207 while (true) {
208 Future<com.sun.star.connection.XConnection> versuch = connectExec
209 .submit(() -> connector.connect("socket,host=localhost,port=" + SOCKET_PORT));
210 try {
211 return versuch.get(CONNECT_VERSUCH_TIMEOUT_SEKUNDEN, TimeUnit.SECONDS);
212 } catch (ExecutionException e) {
213 // soffice lauscht noch nicht (Connection refused) – warten und erneut versuchen
214 letzterFehler = e;
215 } catch (TimeoutException e) {
216 versuch.cancel(true);
217 letzterFehler = e;
218 }
219 if (System.nanoTime() >= deadline) {
220 throw new RuntimeException(
221 "Timeout/Fehler beim Verbinden mit LibreOffice auf Port " + SOCKET_PORT, letzterFehler);
222 }
223 Thread.sleep(CONNECT_RETRY_PAUSE_MILLIS);
224 }
225 } finally {
226 connectExec.shutdownNow();
227 }
228 }

231 public void closeOffice() {
232 logger.info("Closing Office");
233 if (xDesktop == null) {
234 logger.error("No office connection found");
235 killSofficeProcess();
236 return;
237 }
238
239 if (isOfficeTerminated.get()) {
240 logger.warn("Office has already been requested to terminate");
241 return;
242 }
243
244 int numTries = 1;
245 try {
246 while (!isOfficeTerminated.get() && (numTries < 4)) {
247 Thread.sleep(200);
248 isOfficeTerminated.set(tryToTerminate(numTries));
249 numTries++;
250 }
251 } catch (InterruptedException e) {
252 }
253
254 // Falls UNO-Terminate nicht geklappt hat oder Prozess noch läuft: forcibly killen
255 // und warten bis Port 8100 frei ist (Voraussetzung für nächsten startup())
256 killSofficeProcess();
257 }
258
259 private void killSofficeProcess() {
260 if (sofficeProcess == null) {
261 return;
262 }
263 if (sofficeProcess.isAlive()) {
264 logger.info("soffice-Prozess läuft noch, wird beendet");
265 sofficeProcess.destroy();
266 try {
267 boolean exited = sofficeProcess.waitFor(5, TimeUnit.SECONDS);
268 if (!exited) {
269 logger.warn("soffice reagiert nicht auf destroy(), forcible kill");
270 sofficeProcess.destroyForcibly();
271 sofficeProcess.waitFor(3, TimeUnit.SECONDS);
272 }
273 } catch (InterruptedException e) {
274 Thread.currentThread().interrupt();
275 }
276 }
277 sofficeProcess = null;
278 }
279
280 private boolean tryToTerminate(int numTries) {
281 try {
282 boolean isDead = xDesktop.terminate();
283 if (isDead) {
284 if (numTries > 1)
285 logger.info(numTries + ". Office terminated");
286 else
287 logger.info("Office terminated");
288 } else
289 logger.error(numTries + ". Office failed to terminate");
290 return isDead;
291 } catch (com.sun.star.lang.DisposedException e) {
292 logger.info("Office link disposed");
293 return true;
294 } catch (java.lang.Exception e) {
295 logger.error("Termination exception: " + e.getMessage(), e);
296 return false;
297 }
