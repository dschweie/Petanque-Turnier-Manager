135 public synchronized void starten(long dauerSekunden, String bezeichnung, int port, int hintergrundFarbe) {
136 if (dauerSekunden <= 0) {
137 logger.warn("starten() mit ungültiger Dauer aufgerufen: {}", dauerSekunden);
138 return;
139 }
140 stoppeTickTask();
141 this.bezeichnung = bezeichnung != null ? bezeichnung : "";
142 this.hintergrundFarbeHex = "#" + String.format("%06x", hintergrundFarbe & 0xFFFFFF);
143 this.snoozed = false;
144 try {
145 WebServerManager.get().timerServerBesorgen(port);
146 } catch (IOException e) {
147 logger.error("Timer-Webserver konnte nicht gestartet werden auf Port {}: {}", port, e.getMessage(), e);
148 MessageBox.from(xContext, MessageBoxTypeEnum.ERROR_OK)
149 .caption(I18n.get("timer.dialog.titel"))
150 .message(I18n.get("timer.fehler.webserver.port.belegt", port))
151 .show();
152 }
153
154 endNanos = System.nanoTime() + dauerSekunden * 1_000_000_000L;
155 zustand = TimerZustand.LAEUFT;
156 tickTask = executor.scheduleAtFixedRate(this::tick, 0, 1, TimeUnit.SECONDS);
157
158 logger.info("Timer gestartet: {} Sekunden, Port {}", dauerSekunden, port);
159 }

162 public synchronized void pauseOderFortsetzen() {
163 switch (zustand) {
164 case LAEUFT -> {
165 restNanos = endNanos - System.nanoTime();
166 if (restNanos < 0) restNanos = 0;
167 stoppeTickTask();
168 zustand = TimerZustand.PAUSIERT;
169 emittiere(aktuellerState());
170 logger.debug("Timer pausiert, restNanos={}", restNanos);
171 }
172 case PAUSIERT -> {
173 endNanos = System.nanoTime() + restNanos;
174 zustand = TimerZustand.LAEUFT;
175 tickTask = executor.scheduleAtFixedRate(this::tick, 0, 1, TimeUnit.SECONDS);
176 logger.debug("Timer fortgesetzt");
177 }
178 default -> logger.debug("pauseOderFortsetzen() im Zustand {} ignoriert", zustand);
179 }

187 public synchronized void zeitAnpassen(long deltaSekunden) {
188 if (zustand == TimerZustand.INAKTIV || zustand == TimerZustand.BEENDET) {
189 return;
190 }
191 if (zustand == TimerZustand.LAEUFT) {
192 endNanos = Math.max(System.nanoTime(), endNanos + deltaSekunden * 1_000_000_000L);
193 } else {
194 restNanos = Math.max(0, restNanos + deltaSekunden * 1_000_000_000L);
195 }
196 emittiere(aktuellerState());
197 logger.debug("Zeit angepasst um {} s, neuer Zustand: {}", deltaSekunden, aktuellerState().anzeige());
198 }

233 private void tick() {
234 long verbleibend;
235 TimerZustand neuerZustand;
236
237 synchronized (this) {
238 verbleibend = (endNanos - System.nanoTime()) / 1_000_000_000L;
239 if (verbleibend <= 0) {
240 verbleibend = 0;
241 neuerZustand = TimerZustand.BEENDET;
242 stoppeTickTask();
243 zustand = TimerZustand.BEENDET;
244 } else {
245 neuerZustand = TimerZustand.LAEUFT;
246 }
247 }
248
249 emittiere(new TimerState(formatiere(verbleibend), verbleibend, neuerZustand, bezeichnung, hintergrundFarbeHex, snoozed));
250 }

298 public static long parseDauer(String mmss) {
299 if (mmss == null || !mmss.matches("^\\d{1,2}:[0-5]\\d$")) {
300 throw new IllegalArgumentException("Ungültiges Format: " + mmss);
301 }
302 var teile = mmss.split(":");
303 long minuten = Long.parseLong(teile[0]);
304 long sekunden = Long.parseLong(teile[1]);
305 long gesamt = minuten * 60 + sekunden;
306 if (gesamt <= 0) {
307 throw new IllegalArgumentException("Dauer muss größer als 0 sein");
308 }
309 return gesamt;
310 }
