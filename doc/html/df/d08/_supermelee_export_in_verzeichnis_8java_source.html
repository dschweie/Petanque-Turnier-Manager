34 @Override
35 protected ExportErgebnis exportiereInVerzeichnis(Path zielVerzeichnis) throws GenerateException {
36 processBox().info(I18n.get("export.info.pdf"));
37
38 var ws = getWorkingSpreadsheet();
39 var xDoc = ws.getWorkingSpreadsheetDocument();
40 var konfiguration = new SuperMeleeKonfigurationSheet(ws);
41 var spieltagRanglisteSheet = new SpieltagRanglisteSheet(ws, SpielTagNr.from(1));
42 int anzahlSpieltage = spieltagRanglisteSheet.countNumberOfRanglisten();
43
44 String turnierlogoUrl = StringUtils.strip(konfiguration.getTurnierlogoUrl());
45 String turniername = StringUtils.strip(konfiguration.getKopfZeileMitte());
46
47 if (StringUtils.isEmpty(turnierlogoUrl)) {
48 processBox().info(I18n.get("export.warnung.turnierlogo.fehlt"));
49 } else {
50 processBox().info(I18n.get("export.info.turnierlogo", turnierlogoUrl));
51 }
52
53 List<Path> exportierteDateien = new ArrayList<>();
54 List<String> spieltagSheetNamen = new ArrayList<>();
55 List<String> spieltagSchluessel = new ArrayList<>();
56 List<String> spieltagPdfUrls = new ArrayList<>();
57 List<String> spieltagTitel = new ArrayList<>();
58
59 for (int nr = 1; nr <= anzahlSpieltage; nr++) {
60 var schluessel = SheetMetadataHelper.schluesselSpieltagRangliste(nr);
61 var xSheet = SheetMetadataHelper.findeSheetUndHeile(xDoc, schluessel, SheetNamen.spieltagRangliste(nr));
62 if (xSheet == null) {
63 continue;
64 }
65 var sheetName = Lo.qi(XNamed.class, xSheet).getName();
66 var titel = I18n.get("export.supermelee.spieltag", nr);
67 var pdf = exportierePdfAusHtml(sheetName, titel, zielVerzeichnis);
68 if (pdf != null) {
69 exportierteDateien.add(pdf);
70 }
71 spieltagSheetNamen.add(sheetName);
72 spieltagSchluessel.add(schluessel);
73 spieltagTitel.add(titel);
74 spieltagPdfUrls.add(buildPdfUrl(pdf));
75 }
76
77 var endranglisteSchluessel = SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_ENDRANGLISTE;
78 var xEndrangliste = SheetMetadataHelper.findeSheetUndHeile(xDoc, endranglisteSchluessel,
79 SheetNamen.endrangliste());
80 String endranglisteSheetName = xEndrangliste != null ? Lo.qi(XNamed.class, xEndrangliste).getName() : null;
81 Path pdfEndrangliste = endranglisteSheetName != null
82 ? exportierePdfAusHtml(endranglisteSheetName, I18n.get("export.supermelee.nav.endrangliste"), zielVerzeichnis)
83 : null;
84 if (pdfEndrangliste != null) {
85 exportierteDateien.add(pdfEndrangliste);
86 }
87
88 processBox().info(I18n.get("export.info.html"));
89 var htmlExport = exportiereHtml(zielVerzeichnis, turniername, turnierlogoUrl,
90 endranglisteSheetName, pdfEndrangliste, spieltagSheetNamen, spieltagSchluessel,
91 spieltagTitel, spieltagPdfUrls);
92 htmlExport.addTo(exportierteDateien);
93
94 return new ExportErgebnis(exportierteDateien);
95 }

97 private HtmlExportErgebnis exportiereHtml(Path zielVerzeichnis, String turniername, String logoUrl,
98 String endranglisteSheetName, Path pdfEndrangliste,
99 List<String> sheetNamen, List<String> schluessel, List<String> titel, List<String> pdfUrls)
100 throws GenerateException {
101 List<ExportHtmlSeite.Section> sections = new ArrayList<>();
102 String endranglisteNavSheetName = endranglisteSheetName != null ? endranglisteSheetName : SheetNamen.endrangliste();
103 sections.add(new ExportHtmlSeite.Section(SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_ENDRANGLISTE,
104 I18n.get("export.supermelee.nav.endrangliste"),
105 endranglisteNavSheetName, buildPdfUrl(pdfEndrangliste)));
106
107 for (int i = 0; i < sheetNamen.size(); i++) {
108 sections.add(new ExportHtmlSeite.Section(schluessel.get(i), titel.get(i),
109 sheetNamen.get(i), pdfUrls.get(i)));
110 }
111
112 return exportiereHtml(zielVerzeichnis, "SuperMelee.html",
113 StringUtils.defaultIfBlank(turniername, TurnierSystem.SUPERMELEE.getBezeichnung()),
114 logoUrl, sections);
115 }
