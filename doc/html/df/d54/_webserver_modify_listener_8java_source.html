127 private void refreshTask() {
128 try {
129 do {
130 dirty.set(false);
131 dirtySeitMs = 0L;
132 var current = ws;
133 if (current == null) {
134 // ws kann zwischenzeitlich abgemeldet worden sein – Welle als verarbeitet markieren.
135 break;
136 }
137 try {
138 WebServerManager.get().sseRefreshSenden(current);
139 } catch (RuntimeException e) {
140 logger.debug("SSE-Refresh fehlgeschlagen: {}", e.getMessage());
141 }
142 // Falls während refresh wieder dirty gesetzt wurde, nochmal durch.
143 } while (dirty.get());
144 } finally {
145 scheduled.set(false);
146 // Race-Fenster zwischen letztem dirty.get()==false und scheduled=false:
147 // ein modified() könnte exakt hier ankommen und sehen, dass bereits ein Task
148 // pending ist (scheduled==true). Nach dem Reset auf false muss daher nochmal
149 // geprüft werden, sonst bleibt dirty stehen.
150 if (dirty.get()) {
151 scheduleIfNeeded();
152 }
153 }
154 }
