162 public boolean vollstaendigAktualisieren() throws GenerateException {
163 XSpreadsheet xSheet = getXSpreadSheet();
164 if (xSheet == null) {
165 logger.warn("Kaskaden-KO Meldeliste nicht gefunden");
166 return false;
167 }
168 stringsBesinigen(xSheet);
169 teamnummernVergeben(xSheet);
170 aktivDefaultSetzen(xSheet);
171 pruefeAufDoppelteTeamNr(xSheet);
172 nachNrSortieren(xSheet);
173 upDateSheet();
174 return true;
175 }

177 private void aktivDefaultSetzen(XSpreadsheet xSheet) throws GenerateException {
178 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
179 if (letzteZeile < ERSTE_DATEN_ZEILE) {
180 return;
181 }
182 int vornameSpalte = delegate.getVornameSpalte(0);
183 int aktivSpalte = delegate.getAktivSpalte();
184
185 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
186 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
187 if (vorname == null || vorname.isBlank()) {
188 continue;
189 }
190 int aktiv = getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
191 if (aktiv <= 0) {
192 getSheetHelper().setNumberValueInCell(
193 NumberCellValue.from(xSheet, Position.from(aktivSpalte, zeile))
194 .setValue(AKTIV_WERT_NIMMT_TEIL));
195 }
196 }
197 }

199 private void teamnummernVergeben(XSpreadsheet xSheet) throws GenerateException {
200 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
201 if (letzteZeile < ERSTE_DATEN_ZEILE) {
202 return;
203 }
204
205 int nrSpalte = getNrSpalte();
206 int vornameSpalte = delegate.getVornameSpalte(0);
207
208 RangePosition sortRange = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, delegate.getAktivSpalte(),
209 letzteZeile);
210 SortHelper.from(this, sortRange).spalteToSort(nrSpalte).abSteigendSortieren().doSort();
211
212 int letztNr = Math.max(0,
213 getSheetHelper().getIntFromCell(xSheet, Position.from(nrSpalte, ERSTE_DATEN_ZEILE)));
214
215 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
216 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
217 if (vorname == null || vorname.isBlank()) {
218 continue;
219 }
220 int nr = getSheetHelper().getIntFromCell(xSheet, Position.from(nrSpalte, zeile));
221 if (nr <= 0) {
222 letztNr++;
223 getSheetHelper().setNumberValueInCell(
224 NumberCellValue.from(xSheet, Position.from(nrSpalte, zeile)).setValue(letztNr));
225 }
226 }
227 }

235 delegate.getAktivSpalte(), letzteZeile);
236 SortHelper.from(this, range).spalteToSort(getNrSpalte()).aufSteigendSortieren(true).doSort();
237 }
