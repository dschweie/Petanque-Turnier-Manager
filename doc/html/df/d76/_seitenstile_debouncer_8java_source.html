75 public static void aktualisiereSeitenstileDebounced(WorkingSpreadsheet ws) {
76 if (ws == null) {
77 return;
78 }
79 XSpreadsheetDocument doc = ws.getWorkingSpreadsheetDocument();
80 if (doc == null) {
81 return;
82 }
83
84 synchronized (FUTURE_PRO_DOKUMENT) {
85 ScheduledFuture<?> alt = FUTURE_PRO_DOKUMENT.remove(doc);
86 if (alt != null) {
87 alt.cancel(false);
88 }
89
90 AtomicReference<ScheduledFuture<?>> futureRef = new AtomicReference<>();
91 ScheduledFuture<?> neu = SCHEDULER.schedule(
92 () -> posteSeitenstilAktualisierung(ws, doc, futureRef.get()),
93 DEBOUNCE_MS,
94 TimeUnit.MILLISECONDS);
95 futureRef.set(neu);
96 FUTURE_PRO_DOKUMENT.put(doc, neu);
97 }
98 }

100 private static void posteSeitenstilAktualisierung(WorkingSpreadsheet ws, XSpreadsheetDocument doc,
101 ScheduledFuture<?> future) {
102 synchronized (FUTURE_PRO_DOKUMENT) {
103 if (FUTURE_PRO_DOKUMENT.get(doc) == future) {
104 FUTURE_PRO_DOKUMENT.remove(doc);
105 }
106 }
107 LoMainThread.post(ws.getxContext(), () -> {
108 if (istDisposed(doc)) {
109 logger.debug("PageStyle-Update übersprungen – Dokument bereits disposed");
110 return;
111 }
112 aktualisiereSeitenstileJetzt(ws);
113 });
114 }

120 public static void aktualisiereSeitenstileJetzt(WorkingSpreadsheet ws) {
121 if (ws == null) {
122 return;
123 }
124 synchronized (UPDATE_LOCK) {
125 try {
126 KonfigurationSheetRegistry.fuerAktivesDokument(ws)
127 .ifPresent(BaseKonfigurationSheet::seitenstileAktualisieren);
128 } catch (RuntimeException e) {
129 logger.warn("PageStyle-Aktualisierung fehlgeschlagen", e);
130 }
131 }
132 }

134 private static boolean istDisposed(XSpreadsheetDocument doc) {
135 XComponent comp = UnoRuntime.queryInterface(XComponent.class, doc);
136 if (comp == null) {
137 return true;
138 }
139 try {
140 // Probe-Call: bei disposedem Dokument wirft jede UNO-Methode DisposedException
141 doc.getSheets();
142 return false;
143 } catch (DisposedException ignored) {
144 return true;
145 } catch (RuntimeException e) {
146 logger.debug("Dokument-Lebendigkeit nicht prüfbar, behandle als disposed", e);
147 return true;
148 }
