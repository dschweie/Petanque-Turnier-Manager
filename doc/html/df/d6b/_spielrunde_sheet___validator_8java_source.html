82 public void validateSpieltag(SpielTagNr spielTagNr) throws GenerateException {
83 processBoxinfo("processbox.validate.spieltag", spielTagNr.getNr());
84
85 meldeliste.setSpielTag(spielTagNr);
86 List<Spielrunde> spielrunden = new ArrayList<>();
87
88 for (int spielrunde = 1; spielrunde < 99; spielrunde++) {
89 SpielRundeNr spielRundeNr = SpielRundeNr.from(spielrunde);
90 List<Team> teams = spielrundeEinlesen(spielTagNr, spielRundeNr);
91
92 if (teams != null && teams.size() > 0) {
93 spielrunden.add(new Spielrunde(spielTagNr, spielRundeNr, teams));
94 } else {
95 break;
96 }
97 }
98
99 SpielerMeldungen alleMeldungen = meldeliste.getAlleMeldungen();
100 processBoxinfo("processbox.meldungen.anzahl", alleMeldungen.size());
101 processBoxinfo("processbox.spielrunden.anzahl", spielrunden.size());
102
103 validateDoppelteTeams(alleMeldungen, spielrunden);
104 processBoxinfo("processbox.fertig.status");
105 }

107 private void validateDoppelteTeams(SpielerMeldungen alleMeldungen, List<Spielrunde> spielrunden)
108 throws GenerateException {
109 int fehlerCntr = 0;
110 int spielrundeCntr = 0;
111 for (Spielrunde spielrunde : spielrunden) {
112 spielrundeCntr++;
113 int teamCntr = 0;
114 for (Team team : spielrunde.teamlist) {
115 teamCntr++;
116 for (Spieler spielerToValidate : team.spieler()) {
117 Spieler spielerAusMeldung = alleMeldungen.findSpielerByNr(spielerToValidate.getNr());
118 if (spielerAusMeldung == null) {
119 throw new GenerateException(I18n.get("error.spieler.null"));
120 }
121
122 int warbereitsimTeamMitNr = validateWarImTeam(spielerAusMeldung, spielerToValidate, team);
123 if (warbereitsimTeamMitNr > 0) {
124 ProcessBox.from().fehler(
125 "Doppelte Auslosung gefunden in Spieltag: " + meldeliste.getSpielTag().getNr()
126 + ", Spielrunde: " + spielrundeCntr + ", TeamNr: " + teamCntr + ", Spieler: "
127 + spielerToValidate.getNr() + " hat bereits zusammen gespielt mit "
128 + warbereitsimTeamMitNr);
129 fehlerCntr++;
130 }
131 for (Spieler spielerausTeam : team.spieler()) {
132 if (!spielerToValidate.equals(spielerausTeam)) {
133 spielerAusMeldung.addWarImTeamMitWennNichtVorhanden(spielerausTeam);
134 }
135 }
136 }
137 }
138 }
139
140 if (fehlerCntr > 0) {
141 throw new GenerateException(I18n.get("error.spielrunde.doppelte.auslosung", fehlerCntr));
142 }
143 }

145 private int validateWarImTeam(Spieler spielerAusMeldung, Spieler spielerToValidate, Team team) {
146 for (Spieler spielerausTeam : team.spieler()) {
147 if (!spielerToValidate.equals(spielerausTeam)) {
148 if (spielerAusMeldung.warImTeamMit(spielerausTeam)) {
149 return spielerausTeam.getNr();
150 }
151 }
152 }
153 return 0;
154 }

156 private List<Team> spielrundeEinlesen(SpielTagNr spielTagNr, SpielRundeNr spielRundeNr) throws GenerateException {
157 String sheetNamen = getSheetName(spielTagNr, spielRundeNr);
158
159 List<Team> teamList = new ArrayList<>();
160 XSpreadsheet spieltag = getSheetHelper().findByName(sheetNamen);
161
162 if (spieltag == null) {
163 return teamList;
164 }
165
166 getSheetHelper().setActiveSheet(spieltag);
167
168 Position posSpielrNr = Position.from(ERSTE_SPIELERNR_SPALTE, ERSTE_DATEN_ZEILE);
169
170 try {
171 int teamCntr = 1;
172 HashSet<Integer> spielrNrSet = new HashSet<>();
173 for (int zeileCntr = 0; zeileCntr < 999; zeileCntr++) {
174 Integer paarungCntr = getSheetHelper().getIntFromCell(spieltag,
175 Position.from(posSpielrNr).spaltePlus(-1));
176
177 if (paarungCntr < 1) {
178 break;
179 }
180 Team teamA = Team.from(teamCntr++);
181 for (int spalteOffsetTeamA = 0; spalteOffsetTeamA < 3; spalteOffsetTeamA++) {
182 nextSpielerInTeam(spieltag, posSpielrNr, spielrNrSet, teamA);
183 }
184 teamList.add(teamA);
185
186 Team teamB = Team.from(teamCntr++);
187 for (int spalteOffsetTeamB = 0; spalteOffsetTeamB < 3; spalteOffsetTeamB++) {
188 nextSpielerInTeam(spieltag, posSpielrNr, spielrNrSet, teamB);
189 }
190
191 teamList.add(teamB);
192
193 posSpielrNr.spalte(ERSTE_SPIELERNR_SPALTE).zeilePlusEins();
194 }
195 } catch (AlgorithmenException e) {
196 logger.error(e.getMessage(), e);
197 throw new GenerateException(e.getMessage());
198 }
199 return teamList;
200 }
201
202 private void nextSpielerInTeam(XSpreadsheet spieltag, Position posSpielrNr, HashSet<Integer> spielrNrSet,
203 Team team) throws GenerateException, AlgorithmenException {
204
205 Integer spielerNrAusSheet = getSheetHelper().getIntFromCell(spieltag, posSpielrNr);
206 if (spielerNrAusSheet > 0) {
207 if (spielrNrSet.contains(spielerNrAusSheet)) {
208 throw new GenerateException(I18n.get("error.spieler.doppelt", spielerNrAusSheet));
209 }
210 spielrNrSet.add(spielerNrAusSheet);
211 team.addSpielerWennNichtVorhanden(Spieler.from(spielerNrAusSheet));
212 }
213 posSpielrNr.spaltePlusEins();
214 }
