72 public void upDateRanglisteSpalte() throws GenerateException {
73
74 iRanglisteSheet.processBoxinfo("processbox.rangliste.aktualisieren");
75 int letzteZeile = getIRanglisteSheet().getLetzteMitDatenZeileInSpielerNrSpalte();
76 int ersteZeile = getIRanglisteSheet().getErsteDatenZiele();
77
78 if (letzteZeile < ersteZeile) {
79 return;
80 }
81
82 // =WENN(ZEILE()=4;1;WENN(UND(
83 // INDIREKT(ADRESSE(ZEILE()-1;12;4))=INDIREKT(ADRESSE(ZEILE();12;4));
84 // INDIREKT(ADRESSE(ZEILE()-1;14;4))=INDIREKT(ADRESSE(ZEILE();14;4));
85 // INDIREKT(ADRESSE(ZEILE()-1;17;4))=INDIREKT(ADRESSE(ZEILE();17;4));
86 // INDIREKT(ADRESSE(ZEILE()-1;15;4))=INDIREKT(ADRESSE(ZEILE();15;4)));
87 // INDIREKT(ADRESSE(ZEILE()-1;SPALTE();4));ZEILE()-3))
88
89 String ranglisteAdressPlusEinPlatzIndiekt = "INDIRECT(ADDRESS(ROW()-1;COLUMN();4))";
90
91 List<Position> ranglisteSpalten = iRanglisteSheet.getRanglisteSpalten();
92
93 String formula = "=IF(ROW()=" + (ersteZeile + 1) + ";1;IF(AND("
94 + ranglisteSpalten.stream().map(this::indirectFormula).collect(Collectors.joining(";"))
95 + ");" + ranglisteAdressPlusEinPlatzIndiekt + ";ROW()-" + ersteZeile + "))";
96
97 // Formel als Block in alle Zeilen der Platz-Spalte schreiben.
98 // fillAuto würde die Hintergrundfarbe der Quellzelle (Zebra) in alle Zielzellen kopieren
99 // und so die zuvor aufgetragene Zebra-Färbung der Platz-Spalte überschreiben.
100 RangePosition platzRange = RangePosition.from(rangListeSpalte, ersteZeile, rangListeSpalte, letzteZeile);
101 int anzahlZeilen = letzteZeile - ersteZeile + 1;
102 String[][] formulas = new String[anzahlZeilen][1];
103 for (int i = 0; i < anzahlZeilen; i++) {
104 SheetRunner.testDoCancelTask();
105 formulas[i][0] = formula;
106 }
107 try {
108 XCellRange xCellRange = getSheet().getCellRangeByPosition(platzRange.getStartSpalte(),
109 platzRange.getStartZeile(), platzRange.getEndeSpalte(), platzRange.getEndeZeile());
110 XCellRangeFormula xRangeFormula = Lo.qi(XCellRangeFormula.class, xCellRange);
111 xRangeFormula.setFormulaArray(formulas);
112 } catch (com.sun.star.lang.IndexOutOfBoundsException e) {
113 throw new GenerateException("Platz-Spalte konnte nicht beschrieben werden: " + e.getMessage());
114 }
115
116 // Platz-Spalte: fett + dicke rechte Linie
117 getSheetHelper().setPropertiesInRange(getSheet(), platzRange,
118 CellProperties.from()
119 .setCharWeight(FontWeight.BOLD)
120 .setBorder(BorderFactory.from().allThin().boldLn().forTop().forRight().toBorder()));
121 }
122
129 private String indirectFormula(Position pos) {
130 return "INDIRECT(ADDRESS(ROW()-1;" + (pos.getSpalte() + 1) + ";4))=" + "INDIRECT(ADDRESS(ROW();" + (pos.getSpalte() + 1) + ";4))";
131 }
