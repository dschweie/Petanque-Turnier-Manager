336 private void gespieltenRundenEinlesen(SpielerMeldungen aktiveMeldungen, SpielTagNr spielTagNr, int abSpielrunde,
337 int bisSpielrunde) throws GenerateException {
338 int spielrunde = 1;
339
340 if (bisSpielrunde < abSpielrunde || bisSpielrunde < 1) {
341 return;
342 }
343 if (abSpielrunde > 1) {
344 spielrunde = abSpielrunde;
345 }
346
347 sheet.processBoxinfo("processbox.meldungen.gespielter.runden.einlesen", spielTagNr.getNr());
348 var xDoc = sheet.getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
349 int defaultTeamGroesse = konfigurationSheet.getSuperMeleeMode() == SuperMeleeMode.Doublette ? 2 : 3;
350
351 for (; spielrunde <= bisSpielrunde; spielrunde++) {
352 SheetRunner.testDoCancelTask();
353 // Iterations-Lookup: Metadaten-first (überlebt Umbenennung), Fallback auf Namen
354 XSpreadsheet xsheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
355 SheetMetadataHelper.schluesselSupermeleeSpielrunde(spielTagNr.getNr(), spielrunde),
356 getSheetName(spielTagNr, SpielRundeNr.from(spielrunde)));
357
358 if (xsheet == null) {
359 continue;
360 }
361 Position pospielerNr = Position.from(ERSTE_SPIELERNR_SPALTE, ERSTE_DATEN_ZEILE);
362
363 boolean zeileIstLeer = false;
364 int maxcntr = 999;
365 while (!zeileIstLeer && maxcntr > 0) {
366 maxcntr--;
367 Team team1 = Team.from(1);
368 Team team2 = Team.from(2);
369 for (int teamCntr = 1; teamCntr <= 2; teamCntr++) {
370 Team team = (teamCntr == 1) ? team1 : team2;
371 for (int spielerCntr = 1; spielerCntr <= 3; spielerCntr++) {
372 pospielerNr.spalte(ERSTE_SPIELERNR_SPALTE + ((teamCntr - 1) * 3) + spielerCntr - 1);
373 int spielerNr = sheet.getSheetHelper().getIntFromCell(xsheet, pospielerNr);
374 if (spielerNr > 0) {
375 Spieler spieler = aktiveMeldungen.findSpielerByNr(spielerNr);
376 if (spieler != null) {
377 try {
378 team.addSpielerWennNichtVorhanden(spieler);
379 } catch (AlgorithmenException e) {

381 throw new GenerateException(I18n.get("error.spielrunde.einlesen"));
