116 @Override
117 public void doRun() throws GenerateException {
118 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.POULE)
119 .prefix(getLogPrefix()).validate()) {
120 return;
121 }
122
123 getSheetHelper().removeAllSheetsExclude();
124
125 // Meldeliste mit Standardparametern anlegen (kein Dialog)
126 meldeListe.createMeldelisteWithParams(TEST_FORMATION, true, true);
127
128 testNamenEinfuegen();
129
130 // Teamnummern vergeben und Meldeliste aktualisieren
131 new PouleMeldeListeSheetUpdate(getWorkingSpreadsheet()).doRun();
132 }

134 private void testNamenEinfuegen() throws GenerateException {
135 int anzSpielerProTeam = getKonfigurationSheet().getMeldeListeFormation().getAnzSpieler();
136 boolean teamnameAktiv = getKonfigurationSheet().isMeldeListeTeamnameAnzeigen();
137 boolean vereinsnameAktiv = getKonfigurationSheet().isMeldeListeVereinsnameAnzeigen();
138 var spieler = testnamenLoader.listeMitSpielerTestNamen(anzTeams * anzSpielerProTeam);
139
140 var data = new RangeData();
141 for (int team = 0; team < anzTeams; team++) {
142 testDoCancelTask();
143 var zeile = data.addNewRow();
144 if (teamnameAktiv) {
145 zeile.newString("Team " + (team + 1));
146 }
147 for (int s = 0; s < anzSpielerProTeam; s++) {
148 var stn = spieler.get(team * anzSpielerProTeam + s);
149 zeile.newString(stn.vorname());
150 zeile.newString(stn.nachname());
151 if (vereinsnameAktiv) {
152 zeile.newString("Verein " + ((team % 5) + 1));
153 }
154 }
155 zeile.newEmpty(); // Setzposition
156 zeile.newInt(PouleListeDelegate.AKTIV_WERT_NIMMT_TEIL);
157 }
158
159 var xSheet = meldeListe.getXSpreadSheet();
160 var startPos = Position.from(1, ERSTE_DATEN_ZEILE);
161 RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
162 data.getRangePosition(startPos)).setDataInRange(data);
163
164 meldeListe.upDateSheet();
165 }
