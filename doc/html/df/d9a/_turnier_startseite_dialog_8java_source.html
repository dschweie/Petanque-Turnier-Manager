212 private boolean speichernUndAnwenden() {
213 boolean aktiv = leseCheckBox();
214 String portText = leseFeld(CTRL_PORT);
215 String zoomText = leseFeld(CTRL_ZOOM);
216 String logo = leseFeld(CTRL_LOGO);
217 String beschreibung = leseFeld(CTRL_BESCHREIBUNG);
218
219 int port;
220 try {
221 port = Integer.parseInt(portText.trim());
222 if (port < 1 || port > 65535) {
223 throw new NumberFormatException("Port außerhalb des Bereichs");
224 }
225 } catch (NumberFormatException e) {
226 zeigeFehler(I18n.get("konfiguration.startseite.fehler.port"));
227 return false;
228 }
229
230 int zoom;
231 try {
232 zoom = Integer.parseInt(zoomText.trim());
233 if (zoom < 10 || zoom > 500) {
234 throw new NumberFormatException("Zoom außerhalb des Bereichs");
235 }
236 } catch (NumberFormatException e) {
237 zeigeFehler(I18n.get("webserver.composite.konfig.fehler.zoom.ungueltig"));
238 return false;
239 }
240
241 String animation = ANIMATION_KEYS[Math.max(0, leseListBoxIndex(CTRL_ANIMATION))];
242
243 GlobalProperties.get().speichernStartseite(port, aktiv, zoom);
244 var docProps = new DocumentPropertiesHelper(currentSpreadsheet);
245 docProps.setStringProperty(DOC_PROP_TURNIERLOGO_URL, logo);
246 docProps.setStringProperty(DOC_PROP_TURNIERBESCHREIBUNG, beschreibung);
247 docProps.setStringProperty(DOC_PROP_BESCHREIBUNG_ANIMATION, animation);
248 docProps.setIntProperty(DOC_PROP_BESCHREIBUNG_TEXTFARBE, textfarbeInt);
249 WebServerManager.get().konfigurationGeaendert();
250 logger.info("Turnier-Startseite gespeichert: aktiv={}, Port={}, Zoom={}%, Textfarbe=#{}",
251 aktiv, port, zoom, String.format("%06x", textfarbeInt & 0xFFFFFF));
252 return true;
253 }

441 private void oeffneDateiAuswahl() {
442 try {
443 XFilePicker3 picker = FilePicker.createWithMode(xContext,
444 TemplateDescription.FILEOPEN_SIMPLE);
445 picker.setTitle(I18n.get("konfiguration.startseite.logo.picker.titel"));
446 picker.appendFilter(I18n.get("konfiguration.startseite.logo.picker.filter"),
447 "*.png;*.jpg;*.jpeg;*.gif;*.svg;*.webp");
448
449 String aktuell = leseFeld(CTRL_LOGO);
450 if (aktuell != null && !aktuell.isBlank() && !aktuell.startsWith("http")) {
451 try {
452 var pfad = java.nio.file.Path.of(aktuell);
453 var eltern = pfad.getParent();
454 if (eltern != null && java.nio.file.Files.isDirectory(eltern)) {
455 picker.setDisplayDirectory(eltern.toUri().toURL().toExternalForm());
456 }
457 var dateiname = pfad.getFileName();
458 if (dateiname != null) {
459 picker.setDefaultName(dateiname.toString());
460 }
461 } catch (java.net.MalformedURLException | RuntimeException ignored) {
462 // ungültiger Pfad → Default-Verzeichnis
463 }
464 }
465
466 if (picker.execute() != ExecutableDialogResults.OK) {
467 return;
468 }
469 String[] dateien = picker.getFiles();
470 if (dateien.length == 0) {
471 return;
472 }
473 setzeFeld(CTRL_LOGO, urlAlsPfad(dateien[0]));
474 } catch (RuntimeException e) {
475 logger.error("Fehler bei Datei-Auswahl", e);
476 }
477 }

483 return url;
484 }
