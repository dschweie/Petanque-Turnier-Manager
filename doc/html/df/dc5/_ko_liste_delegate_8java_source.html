185 private void renderKonfigurationsZuZellen(XSpreadsheet xKonfigSheet) throws GenerateException {
186 int headerFarbe = Integer.parseInt(BasePropertiesSpalte.HEADER_BACK_COLOR.replace("#", ""), 16);
187
188 // Kopfzeile
189 sheet.getSheetHelper().setStringValueInCell(
190 StringCellValue.from(xKonfigSheet, Position.from(0, 0), "Eigenschaft")
191 .setCharWeight(FontWeight.BOLD)
192 .setCellBackColor(headerFarbe));
193 sheet.getSheetHelper().setStringValueInCell(
194 StringCellValue.from(xKonfigSheet, Position.from(1, 0), "Wert")
195 .setCharWeight(FontWeight.BOLD)
196 .setCellBackColor(headerFarbe));
197
198 List<ConfigProperty<?>> props = de.petanqueturniermanager.ko.konfiguration.KoPropertiesSpalte.KONFIG_PROPERTIES;
199 int zeile = 1;
200 for (ConfigProperty<?> prop : props) {
201 String beschreibung = prop.getDescription() != null ? prop.getDescription() : prop.getKey();
202 // Nur erste Zeile der Beschreibung (keine Zeilenumbrüche)
203 beschreibung = beschreibung.split("[\r\n]")[0];
204
205 String wert;
206 if (prop.getType() == ConfigPropertyType.COLOR) {
207 int farbwert = konfigurationSheet.getKoPropertiesSpalte().readIntProperty(prop.getKey());
208 wert = String.format("#%06X", farbwert & 0xFFFFFF);
209 } else if (prop.getType() == ConfigPropertyType.INTEGER) {
210 wert = String.valueOf(konfigurationSheet.getKoPropertiesSpalte().readIntProperty(prop.getKey()));
211 } else {
212 wert = konfigurationSheet.getKoPropertiesSpalte().readStringProperty(prop.getKey());
213 }
214
215 sheet.getSheetHelper().setStringValueInCell(
216 StringCellValue.from(xKonfigSheet, Position.from(0, zeile), beschreibung));
217 sheet.getSheetHelper().setStringValueInCell(
218 StringCellValue.from(xKonfigSheet, Position.from(1, zeile), wert));
219 zeile++;
220 }
221 }

371 private void formatDatenSpalten() throws GenerateException {
372 sheet.processBoxinfo("processbox.ko.meldeliste.formatieren");
373 Formation formation = konfigurationSheet.getMeldeListeFormation();
374 int anzSpieler = formation.getAnzSpieler();
375 int letzteDatenZeile = getLetzteDatenZeileUseMin();
376
377 // Zeilenfarben hier nochmals holen, damit sie nach den Fehler-Bedingungen angehängt
378 // werden können (Priorität: Fehler > Zeilenfarbe).
379 MeldungenHintergrundFarbeGeradeStyle farbeGerade = konfigurationSheet.getMeldeListeHintergrundFarbeGeradeStyle();
380 MeldungenHintergrundFarbeUnGeradeStyle farbeUngerade = konfigurationSheet
381 .getMeldeListeHintergrundFarbeUnGeradeStyle();
382
383 // Nr-Spalte: doppelte Linie rechts
384 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
385 getTeamNrSpalte(), letzteDatenZeile);
386 RangeHelper.from(sheet, nrRange).setRangeProperties(
387 RangeProperties.from().setBorder(
388 BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
389
390 // Bedingte Formatierung Nr-Spalte: Text=rot, Duplikat=rot, außerhalb [1,999]=rot,
391 // danach Zeilenfarbe (niedrigste Priorität)
392 String kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
393 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
394 ConditionalFormatHelper.from(sheet, nrRange).clear()
395 .formulaIsText().styleIsFehler().applyAndDoReset()
396 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
397 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
398 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset()
399 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
400 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
401
402 // Teamname-Spalte (optional)
403 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
404 RangePosition teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, 1, letzteDatenZeile);
405 RangeHelper.from(sheet, teamnameRange).setRangeProperties(
406 RangeProperties.from().setBorder(
407 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
408 }
409
410 // Spieler-Spalten (Vorname + Nachname [+ Vereinsname])
411 for (int s = 0; s < anzSpieler; s++) {
412 int ersteSpielerSpalte = getVornameSpalte(s);
413 int letzteSpielerSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
414 ? getVereinsnameSpalte(s)
415 : getNachnameSpalte(s);
416 RangePosition spielerRange = RangePosition.from(ersteSpielerSpalte, ERSTE_DATEN_ZEILE,
417 letzteSpielerSpalte, letzteDatenZeile);
418 RangeHelper.from(sheet, spielerRange).setRangeProperties(
419 RangeProperties.from().setBorder(
420 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
421 }
422
423 // RNG-Spalte
424 RangePosition rngRange = RangePosition.from(getRanglisteSpalte(), ERSTE_DATEN_ZEILE,
425 getRanglisteSpalte(), letzteDatenZeile);
426 RangeHelper.from(sheet, rngRange).setRangeProperties(
427 RangeProperties.from().setBorder(
428 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
429
430 // Bedingte Formatierung RNG-Spalte (Pflichtfeld):
431 // - Text-Wert → rot
432 // - Zeile hat Spielernamen aber RNG ist leer (≤ 0) → rot
433 // - Duplikat (Wert > 0, aber mehrfach vorhanden) → rot
434 int vornameSpalteNr = getVornameSpalte(0) + 1; // ADDRESS ist 1-basiert
435 String vornameRef = "INDIRECT(ADDRESS(ROW();" + vornameSpalteNr + "))";
436 String kondRngLeer = "AND(" + vornameRef + "<>\"\";" + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<=0)";
437 String kondDoppeltRng = "AND(" + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ">0;COUNTIF("
438 + Position.from(getRanglisteSpalte(), 0).getSpalteAddressWith$() + ";"
439 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1)";
440 ConditionalFormatHelper.from(sheet, rngRange).clear()
441 .formulaIsText().styleIsFehler().applyAndDoReset()
442 .formula1(kondRngLeer).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
443 .formula1(kondDoppeltRng).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
444 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
445 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
446
447 // Aktiv-Spalte: nur Formatierung (kein Default-Wert – wird nur bei neuen Sheets gesetzt)
448 RangePosition aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE,
449 getAktivSpalte(), letzteDatenZeile);
450 RangeHelper.from(sheet, aktivRange).setRangeProperties(
451 RangeProperties.from().centerJustify()
452 .setBorder(BorderFactory.from().allThin().toBorder()));
453
454 // Bedingte Formatierung Aktiv-Spalte: rot wenn nicht leer AND nicht 1 AND nicht 2
455 String kondAktivUngueltig = "AND(NOT(ISBLANK(" + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "));"
456 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>1;"
457 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>2)";
458 ConditionalFormatHelper.from(sheet, aktivRange).clear()
459 .formula1(kondAktivUngueltig).operator(ConditionOperator.FORMULA)
460 .styleIsFehler().applyAndDoReset()
461 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
462 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
463
464 int aktivSpalte = getAktivSpalte();
465 EditierbaresZelleFormatHelper.anwenden(sheet, RangePosition.from(1, ERSTE_DATEN_ZEILE, aktivSpalte, letzteDatenZeile));
466 }

485 TeamMeldungen getAlleMeldungen() throws GenerateException {
486 XSpreadsheet xSheet = sheet.getXSpreadSheet();
487 int vornameSpalte = getVornameSpalte(0);
488 int letzteZeile = letzteZeileMitDaten(xSheet);
489 TeamMeldungen meldungen = new TeamMeldungen();
490 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
491 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
492 if (vorname == null || vorname.isEmpty()) {
493 continue;
494 }
495 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
496 if (nr <= 0) {
497 continue;

697 void pruefeAufDoppelteTeamNr(XSpreadsheet xSheet) throws GenerateException {
698 int letzteZeile = letzteZeileMitDaten(xSheet);
699 if (letzteZeile < ERSTE_DATEN_ZEILE) {
700 return;
701 }
702 int vornameSpalte = getVornameSpalte(0);
703 Map<Integer, List<Integer>> alleNrn = new LinkedHashMap<>();
704 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
705 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
706 if (vorname == null || vorname.isEmpty()) {
707 continue;
708 }
709 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
710 if (nr <= 0) {
711 continue;
712 }
713 alleNrn.computeIfAbsent(nr, k -> new ArrayList<>()).add(zeile);
714 }
715 Map<Integer, List<Integer>> duplikate = new LinkedHashMap<>();
716 for (Map.Entry<Integer, List<Integer>> entry : alleNrn.entrySet()) {
717 if (entry.getValue().size() > 1) {
718 duplikate.put(entry.getKey(), entry.getValue());
719 }
720 }
721 if (duplikate.isEmpty()) {
722 return;
723 }
724 StringBuilder sb = new StringBuilder("Meldeliste wurde nicht aktualisiert.\nDoppelte Startnummern:");
725 for (Map.Entry<Integer, List<Integer>> entry : duplikate.entrySet()) {
726 sb.append("\nNr. ").append(entry.getKey()).append(": Zeilen ");
727 List<Integer> zeilen = entry.getValue();
728 for (int i = 0; i < zeilen.size(); i++) {
729 if (i > 0) {
730 sb.append(", ");
731 }
732 sb.append(zeilen.get(i) + 1); // 0-basiert → 1-basiert für Benutzer
733 }
734 }
735 throw new GenerateException(sb.toString());
736 }
