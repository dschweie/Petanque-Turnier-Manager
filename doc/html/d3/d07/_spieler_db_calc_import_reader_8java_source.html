56 @Override
57 public ImportRohdaten read(ImportRequest request) throws SpielerDbException {
58 Path datei = request.source();
59 if (!Files.isRegularFile(datei)) {
60 throw new SpielerDbException("ODS-Datei fehlt: " + datei);
61 }
62 EnumSet<ExportEntity> entities = request.entities();
63
64 XSpreadsheetDocument doc = oeffneVersteckt(datei);
65 try {
66 XSpreadsheets sheets = doc.getSheets();
67
68 List<RohSpieler> spieler = entities.contains(ExportEntity.SPIELER)
69 ? leseSpieler(sheets)
70 : List.of();
71 List<RohVerein> vereine = entities.contains(ExportEntity.VEREINE)
72 ? leseVereine(sheets)
73 : List.of();
74 List<RohLabel> labels = entities.contains(ExportEntity.LABELS)
75 ? leseLabels(sheets)
76 : List.of();
77 List<RohSpielerLabel> junction =
78 entities.contains(ExportEntity.SPIELER) && entities.contains(ExportEntity.LABELS)
79 ? leseJunction(sheets)
80 : List.of();
81
82 return new ImportRohdaten(spieler, vereine, labels, junction);
83 } finally {
84 OfficeDocumentHelper.closeDoc(doc);
85 }
86 }

100 XComponent comp = loader.loadComponentFromURL(url, "_blank", 0,

102 XSpreadsheetDocument doc = Lo.qi(XSpreadsheetDocument.class, comp);

104 throw new SpielerDbException("ODS-Datei ist kein Calc-Dokument: " + datei);

107 } catch (com.sun.star.uno.Exception | MalformedURLException e) {
108 throw new SpielerDbException("ODS-Öffnen fehlgeschlagen: " + datei, e);

112 private static List<RohSpieler> leseSpieler(XSpreadsheets sheets) throws SpielerDbException {
113 XSpreadsheet sheet = holePflichtSheet(sheets, "Spieler");
114 validiereHeader(sheet, "Spieler", "nr", "vorname", "nachname",
115 "vereinNr", "vereinName", "lizenznr");
116 List<RohSpieler> erg = new ArrayList<>();
117 for (int row = 1; row < MAX_ZEILEN; row++) {
118 String vorname = leseString(sheet, 1, row);
119 String nachname = leseString(sheet, 2, row);
120 if (vorname.isEmpty() && nachname.isEmpty()
121 && leseString(sheet, 0, row).isEmpty()) {
122 break;
123 }
124 erg.add(new RohSpieler(
125 leseInteger(sheet, 0, row),
126 vorname,
127 nachname,
128 leseInteger(sheet, 3, row),
129 leerAlsNull(leseString(sheet, 4, row)),
130 leerAlsNull(leseString(sheet, 5, row))));
131 }
132 return erg;
133 }

165 XSpreadsheet sheet = holePflichtSheet(sheets, "SpielerLabels");
166 validiereHeader(sheet, "SpielerLabels", "spielerNr", "labelNr");

170 Integer lNr = leseInteger(sheet, 1, row);

175 throw new SpielerDbException("Junction-Sheet 'SpielerLabels' Zeile "

177 }
178 erg.add(new RohSpielerLabel(sNr, lNr));

199 private static void validiereHeader(XSpreadsheet sheet, String sheetName, String... erwartet)

201 for (int i = 0; i < erwartet.length; i++) {

210 private static String leseString(XSpreadsheet sheet, int col, int row) {
211 try {
212 XCell cell = sheet.getCellByPosition(col, row);
213 XText text = Lo.qi(XText.class, cell);
214 return text == null ? "" : text.getString();
215 } catch (com.sun.star.lang.IndexOutOfBoundsException e) {
216 logger.debug("Zelle ({}, {}) nicht zugreifbar: {}", col, row, e.getMessage());
217 return "";
218 }

221 @Nullable
222 private static Integer leseInteger(XSpreadsheet sheet, int col, int row) {
223 String s = leseString(sheet, col, row).strip();
224 if (s.isEmpty()) {
225 return null;
226 }
227 try {
228 return Integer.valueOf(s);
229 } catch (NumberFormatException e) {
230 // Calc liefert numerische Zellen als „1234" oder „1234,0" je nach Locale.
231 try {
232 return (int) Double.parseDouble(s.replace(',', '.'));
233 } catch (NumberFormatException ex) {
234 return null;
235 }
236 }
