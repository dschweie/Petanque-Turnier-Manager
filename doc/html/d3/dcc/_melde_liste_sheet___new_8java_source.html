161 public void createMeldelisteWithParams(SuperMeleeMode mode) throws GenerateException {
162 var neuesSheet = NewSheet.from(this, SheetNamen.meldeliste(), METADATA_SCHLUESSEL)
163 .pos(DefaultSheetPos.MELDELISTE).tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).hideGrid().setActiv()
164 .setDocVersionWhenNew().create();
165 if (neuesSheet.isDidCreate()) {
166 getKonfigurationSheet().setSuperMeleeMode(mode);
167 SpielTagNr spielTag1 = new SpielTagNr(1);
168 setSpielTag(spielTag1);
169 getKonfigurationSheet().setAktiveSpieltag(spielTag1);
170 getKonfigurationSheet().setAktiveSpielRunde(SpielRundeNr.from(1));
171 upDateSheet();
172 }

240 @Override
241 protected void doRun() throws GenerateException {
242 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.SUPERMELEE)
243 .prefix(getLogPrefix()).validate()) {
244 return;
245 }
246
247 // Dialog zuerst – bei Abbruch keine Änderungen am Dokument
248 Optional<SupermeleeStartDialog.StartParameter> param;
249 try {
250 param = SupermeleeStartDialog.from(getWorkingSpreadsheet()).show();
251 } catch (com.sun.star.uno.Exception e) {
252 String errMsg = I18n.get("error.dialog.startdialog", e.getMessage());
253 logger.error(errMsg, e);
254 throw new GenerateException(errMsg);
255 }
256
257 if (param.isEmpty()) {
258 return; // Benutzer hat abgebrochen – keine Dokument-Änderungen
259 }
260
261 // Erst nach Bestätigung: TurnierSystem + Page Styles setzen
262 getKonfigurationSheet().update();
263
264 getSheetHelper().removeAllSheetsExclude();
265 createMeldelisteWithParams(param.get().triplette() ? SuperMeleeMode.Triplette : SuperMeleeMode.Doublette);
266 }
