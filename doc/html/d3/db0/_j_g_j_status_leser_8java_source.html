47 private JGJTurnierSchritt zaehleSpiele(XSpreadsheet sheet) {
48 int hrGesamt = 0;
49 int hrGespielt = 0;
50 int rrGesamt = 0;
51 int rrGespielt = 0;
52
53 try {
54 for (int zeile = JGJSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE; zeile < MAX_ZEILEN; zeile++) {
55 XCell nrZelle = sheet.getCellByPosition(JGJSpielPlanSheet.SPIEL_NR_SPALTE, zeile);
56 if (nrZelle.getType() == CellContentType.EMPTY) {
57 break;
58 }
59 var xText = Lo.qi(XText.class, nrZelle);
60 String nr = xText != null ? xText.getString() : "";
61 boolean istHinrunde = nr.startsWith(JGJSpielPlanSheet.NR_HINRUNDE_PREFIX);
62 boolean istRueckrunde = nr.startsWith(JGJSpielPlanSheet.NR_RUECKRUNDE_PREFIX);
63 if (!istHinrunde && !istRueckrunde) {
64 continue;
65 }
66
67 boolean gespielt = sheet.getCellByPosition(JGJSpielPlanSheet.SPIELE_A_SPALTE, zeile)
68 .getType() != CellContentType.EMPTY;
69
70 if (istHinrunde) {
71 hrGesamt++;
72 if (gespielt) hrGespielt++;
73 } else {
74 rrGesamt++;
75 if (gespielt) rrGespielt++;
76 }
77 }
78 } catch (IndexOutOfBoundsException e) {
79 logger.error("Fehler beim Lesen des JGJ-Spielplans", e);
80 }
81 return new JGJTurnierSchritt(true, hrGespielt, hrGesamt, rrGespielt, rrGesamt);
82 }
