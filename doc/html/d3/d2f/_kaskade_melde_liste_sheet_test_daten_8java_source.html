88 private void teamNamenEinfuegen() throws GenerateException {
89 var konfiguration = delegate.getKonfigurationSheet();
90 int anzSpieler = konfiguration.getMeldeListeFormation().getAnzSpieler();
91 boolean teamnameAktiv = konfiguration.isMeldeListeTeamnameAnzeigen();
92 boolean vereinsnameAktiv = konfiguration.isMeldeListeVereinsnameAnzeigen();
93 var spieler = testnamenLoader.listeMitSpielerTestNamen(anzTeams * anzSpieler);
94
95 var data = new RangeData();
96 for (int i = 0; i < anzTeams; i++) {
97 testDoCancelTask();
98 var zeile = data.addNewRow();
99 if (teamnameAktiv) {
100 zeile.newString("Team " + (i + 1));
101 }
102 for (int s = 0; s < anzSpieler; s++) {
103 var stn = spieler.get(i * anzSpieler + s);
104 zeile.newString(stn.vorname());
105 zeile.newString(stn.nachname());
106 if (vereinsnameAktiv) {
107 zeile.newString("Verein " + ((i % 5) + 1));
108 }
109 }
110 zeile.newEmpty(); // Setzposition
111 zeile.newInt(KaskadeListeDelegate.AKTIV_WERT_NIMMT_TEIL);
112 }
113
114 var xSheet = meldeListeNew.getXSpreadSheet();
115 var startPos = Position.from(1, KaskadeListeDelegate.ERSTE_DATEN_ZEILE);
116 RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
117 data.getRangePosition(startPos)).setDataInRange(data);
118 }
