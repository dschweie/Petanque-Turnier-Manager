49 public FormeSpielrunde generiereSpielrunde() {
50 FormeSpielrunde formeSpielrunde = new FormeSpielrunde(1);
51 doppelteGespieltePaarungenVorhanden = false;
52 doppelteGespieltePaarungen.clear();
53
54 ArrayList<Team> teamRangListeWork = teamRangListe.getCloneTeamListe();
55 int haelfte = teamRangListeWork.size() / 2;
56
57 for (int cntr = 0; cntr < haelfte; cntr++) {
58 Team teamA = teamRangListeWork.getFirst();
59 Team teamB = sucheGegnerFuer(teamA, teamRangListeWork);
60
61 if (teamB != null) {
62 formeSpielrunde.addPaarungWennNichtVorhanden(new TeamPaarung(teamA, teamB));
63 teamRangListeWork.remove(teamB);
64 } else {
65 TauschTeams tausch = kanntauschenMit(teamA, teamRangListeWork, formeSpielrunde);
66 if (tausch != null) {
67 Team freigegenerB = tausch.teamPaarungTausch().getB();
68 tausch.teamPaarungTausch().setB(tausch.teamAusRangliste());
69 teamRangListeWork.remove(tausch.teamAusRangliste());
70 formeSpielrunde.addPaarungWennNichtVorhanden(new TeamPaarung(teamA, freigegenerB));
71 } else {
72 // kein valider Gegner und kein Tausch möglich → Doppel-Paarung
73 teamB = teamRangListeWork.getLast();
74 doppelteGespieltePaarungenVorhanden = true;
75 doppelteGespieltePaarungen.add(teamA.getNr() + ":" + teamB.getNr());
76 formeSpielrunde.addPaarungWennNichtVorhanden(new TeamPaarung(teamA, teamB));
77 teamRangListeWork.remove(teamB);
78 }
79 }
80 teamRangListeWork.remove(teamA);
81 }
82
83 return formeSpielrunde;

94 Team sucheGegnerFuer(final Team teamA, final List<Team> teamRangListeWork) {

110 TauschTeams kanntauschenMit(final Team teamA, final List<Team> teamRangListeWork, final FormeSpielrunde formeSpielrunde) {
111 List<Team> reversedRangliste = Lists.reverse(teamRangListeWork);
112
113 for (Team teamBAusSpielrunde : formeSpielrunde.getBTeams()) {
114 if (!teamA.hatAlsGegner(teamBAusSpielrunde)) {
115 TeamPaarung paarungTausch = formeSpielrunde.findTeamPaarung(teamBAusSpielrunde);
116 for (Team teamAusRangliste : reversedRangliste) {
117 if (!teamA.equals(teamAusRangliste) && !paarungTausch.getA().hatAlsGegner(teamAusRangliste)) {
118 return new TauschTeams(teamAusRangliste, paarungTausch);
119 }
120 }
121 }
122 }
123 return null;
124 }
