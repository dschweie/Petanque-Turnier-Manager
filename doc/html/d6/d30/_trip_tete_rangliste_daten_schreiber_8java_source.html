50 var spielPlanSheet = new TripTeteSpielPlanSheet(workingSpreadsheet);

52 .leseBegegnungen();
53
54 var rangliste = new TripTeteRangliste();
55 meldungen.teams().forEach(rangliste::addTeam);
56 begegnungen.forEach(rangliste::addBegegnung);

58 List<TripTeteTeamErgebnis> sortiert = rangliste.getRangliste();
59 Map<Integer, String> namen = namenMap();
60
61 RangeData rangeData = new RangeData();
62 int rang = 1;
63 for (int i = 0; i < sortiert.size(); i++) {
64 if (i > 0 && sortiert.get(i).compareTo(sortiert.get(i - 1)) != 0) {
65 rang = i + 1;
66 }
67 TripTeteTeamErgebnis ergebnis = sortiert.get(i);
68 int teamNr = ergebnis.getTeam().getNr();
69 RowData row = rangeData.addNewRow();
70 row.newInt(teamNr);
71 row.newString(namen.getOrDefault(teamNr, ""));
72 row.newInt(rang);
73 row.newInt(ergebnis.getBegegnungenGewonnen());
74 row.newInt(ergebnis.getBegegnungenVerloren());
75 row.newInt(ergebnis.getPartienGewonnen());
76 row.newInt(ergebnis.getPartienVerloren());
77 row.newInt(ergebnis.getPartienGewonnen() - ergebnis.getPartienVerloren());
78 row.newInt(ergebnis.getSpielPunktePlus());
79 row.newInt(ergebnis.getSpielPunkteMinus());
80 row.newInt(ergebnis.getSpielPunkteDiff());
81 row.newInt(ergebnis.getBegegnungenGespielt());
82 }
83
84 Position startPos = Position.from(TripTeteRanglisteSheet.TEAM_NR_SPALTE,
85 TripTeteRanglisteSheet.ERSTE_DATEN_ZEILE);
86 RangeHelper.from(ranglisteSheet, rangeData.getRangePosition(startPos)).setDataInRange(rangeData);
87 }
