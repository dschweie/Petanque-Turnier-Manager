68 public static TeilnehmerStatus ermitteln(WorkingSpreadsheet ws) {
69 if (ws == null) {
70 return new TeilnehmerStatus(0, 0);
71 }
72 try {
73 TurnierSystem ts = new DocumentPropertiesHelper(ws).getTurnierSystemAusDocument();
74 if (ts == null || ts == TurnierSystem.KEIN) {
75 return new TeilnehmerStatus(0, 0);
76 }
77 SheetHelper sh = new SheetHelper(ws);
78 XSpreadsheet sheet = sh.findByName(SheetNamen.meldeliste());
79 if (sheet == null) {
80 return new TeilnehmerStatus(0, 0);
81 }
82 int aktivSpalte = ermittleAktivSpalte(sh, sheet, ts, ws);
83 return zaehlen(sh, sheet, aktivSpalte);
84 } catch (RuntimeException e) {
85 logger.warn("Teilnehmerzahl konnte nicht ermittelt werden", e);
86 return new TeilnehmerStatus(0, 0);
87 }
88 }

97 private static int ermittleAktivSpalte(SheetHelper sh, XSpreadsheet sheet, TurnierSystem ts,
98 WorkingSpreadsheet ws) {
99 String gesuchterHeader = gesuchterHeader(ts, ws);
100 if (gesuchterHeader == null || gesuchterHeader.isBlank()) {
101 return -1;
102 }
103 for (int spalte = 0; spalte < MAX_HEADER_SPALTEN; spalte++) {
104 String headerText = sh.getTextFromCell(sheet,
105 Position.from(spalte, MeldeListeKonstanten.ZWEITE_HEADER_ZEILE));
106 if (headerText != null && headerText.equals(gesuchterHeader)) {
107 return spalte;
108 }
109 }
110 return -1;
111 }

113 private static String gesuchterHeader(TurnierSystem ts, WorkingSpreadsheet ws) {
114 if (ts == TurnierSystem.SUPERMELEE) {
115 try {
116 SpielTagNr spieltagNr = new SuperMeleeKonfigurationSheet(ws).getAktiveSpieltag();
117 return I18n.get("column.header.spieltag.nr", spieltagNr.getNr());
118 } catch (RuntimeException e) {
119 logger.debug("Aktiver Spieltag (Supermelee) konnte nicht ermittelt werden: {}", e.getMessage());
120 return null;
121 }
122 }
123 return I18n.get("column.header.aktiv");
124 }

126 private static TeilnehmerStatus zaehlen(SheetHelper sh, XSpreadsheet sheet, int aktivSpalte) {
127 int angemeldet = 0;
128 int aktiv = 0;
129 int leereZeilenInFolge = 0;
130 for (int zeile = MeldeListeKonstanten.ERSTE_DATEN_ZEILE; zeile < MAX_ZEILEN; zeile++) {
131 if (zeileHatNamen(sh, sheet, zeile)) {
132 leereZeilenInFolge = 0;
133 angemeldet++;
134 if (aktivSpalte >= 0
135 && sh.getIntFromCell(sheet, Position.from(aktivSpalte, zeile)) == WERT_AKTIV) {
136 aktiv++;
137 }
138 } else if (++leereZeilenInFolge >= MAX_LEERE_ZEILEN_IN_FOLGE) {
139 break;
140 }
141 }
142 // Fallback wenn Aktiv-Spalte nicht gefunden: Aktiv = Angemeldet (verhindert „0"-Anzeige bei
143 // ungewöhnlichem Header-Layout / Sprachwechsel im Dokument).
144 return new TeilnehmerStatus(angemeldet, aktivSpalte >= 0 ? aktiv : angemeldet);
145 }
146
147 private static boolean zeileHatNamen(SheetHelper sh, XSpreadsheet sheet, int zeile) {
148 return !istLeer(sh.getTextFromCell(sheet, Position.from(VORNAME_SPALTE, zeile)))
149 || !istLeer(sh.getTextFromCell(sheet, Position.from(NACHNAME_SPALTE, zeile)));
150 }

153 return text == null || text.isBlank();
