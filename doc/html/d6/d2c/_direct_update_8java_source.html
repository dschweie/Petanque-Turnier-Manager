50 ExtensionsHelper extensionsHelper = ExtensionsHelper.from(getxContext());
51
52 processBoxinfo("processbox.direkt.aktualisierung.start");
53 processBoxinfo("processbox.installierte.version", extensionsHelper.getVersionNummer());
54
55 ReleaseInfo release;
56 try {
57 release = ReleaseUpdateService.get().getAktuellesRelease().orElse(null);
58 } catch (IllegalStateException e) {
59 release = null;
60 }
61 if (release == null) {
62 processBox().fehler("Keine " + DownloadExtension.EXTENSION_FILE_SUFFIX
63 + " Datei zum Download vorhanden.");
64 return;
65 }
66 var oxtAsset = release.findeAsset(
67 DownloadExtension.EXTENSION_FILE_PREFIX,
68 DownloadExtension.EXTENSION_FILE_SUFFIX).orElse(null);
69 if (oxtAsset == null) {
70 processBox().fehler("Keine " + DownloadExtension.EXTENSION_FILE_SUFFIX
71 + " Datei zum Download vorhanden.");
72 return;
73 }
74 URL downloadURL;
75 try {
76 downloadURL = URI.create(oxtAsset.downloadUrl()).toURL();
77 } catch (IllegalArgumentException | MalformedURLException e) {
78 processBox().fehler("Ungültige Download-URL: " + e.getMessage());
79 return;
80 }
81
82 processBoxinfo("processbox.lade.herunter", downloadURL.toString());
83
84 try {
85 Path tempDir = Files.createTempDirectory("ptm_update");
86 Path tempFile = tempDir.resolve(oxtAsset.name());
87
88 FileUtils.copyURLToFile(downloadURL, tempFile.toFile(), 30000, 30000);
89 processBoxinfo("processbox.download.abgeschlossen.installation");
90
91 var xShell = Lo.createInstanceMCF(
92 XSystemShellExecute.class,
93 "com.sun.star.system.SystemShellExecute",
94 getxContext().getServiceManager(),
95 getxContext());
96 if (xShell == null) {
97 throw new GenerateException("SystemShellExecute nicht verfügbar.");
98 }
99 xShell.execute(tempFile.toUri().toString(), "", (short) SystemShellExecuteFlags.URIS_ONLY);
100
101 processBoxinfo("processbox.neue.version.installiert");
102 MessageBox.from(getxContext(), MessageBoxTypeEnum.INFO_OK)
103 .caption(I18n.get("msg.caption.aktualisierung.abgeschlossen"))
104 .message(I18n.get("msg.text.aktualisierung.abgeschlossen"))
105 .show();
106 } catch (Exception e) {
107 logger.error(e.getMessage(), e);
108 processBox().fehler(e.getMessage());
109 }
110 }
