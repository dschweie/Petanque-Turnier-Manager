169 private Formation leseFormation(XControlContainer xcc) {
170 if (istRadioGewaehlt(xcc, "radioTete")) {
171 return Formation.TETE;
172 }
173 if (istRadioGewaehlt(xcc, "radioDoublette")) {
174 return Formation.DOUBLETTE;
175 }
176 return Formation.TRIPLETTE;
177 }

183 private boolean istRadioGewaehlt(XControlContainer xcc, String name) {
184 XControl ctrl = xcc.getControl(name);
185 if (ctrl == null) {
186 return false;
187 }
188 XRadioButton radio = Lo.qi(XRadioButton.class, ctrl);
189 return radio != null && radio.getState();
190 }
191
192 private boolean leseCheckBoxZustand(XControlContainer xcc, String name) {
193 XControl ctrl = xcc.getControl(name);
194 if (ctrl == null) {
195 return false;
196 }
197 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
198 return cb != null && cb.getState() == 1;
199 }
