156 @VisibleForTesting
157 List<List<Integer>> splitSortBloecke(int[] sortSpaltenToSplit) {
158
159 final int chunkSize = 3;
160 final AtomicInteger counter = new AtomicInteger();
161
162 Map<Integer, List<Integer>> groups = Arrays.stream(sortSpaltenToSplit).boxed()
163 .collect(Collectors.groupingBy(s -> counter.getAndIncrement() / chunkSize));
164
165 List<List<Integer>> bloecke = new ArrayList<>(groups.values());
166
167 Collections.reverse(bloecke);
168
169 return bloecke;
170 }

186 private void doSortMax3Spalten(List<Integer> sortMax3Spalten) throws GenerateException {
187
188 checkNotNull(sortMax3Spalten);
189 checkArgument(sortMax3Spalten.size() > 0);
190
191 // lo 7 ist das noch so
192 checkArgument(sortMax3Spalten.size() < 4); // max 3 spalten
193
194 XCellRange xCellRangeToSort = RangeHelper.from(xSpreadsheet, workingSpreadsheetDocument, rangePositionToSort)
195 .getCellRange();
196
197 if (xCellRangeToSort == null) {
198 return;
199 }
200
201 XSortable xSortable = Lo.qi(XSortable.class, xCellRangeToSort);
202 TableSortField[] sortFields = new TableSortField[sortMax3Spalten.size()];
203
204 int idx = 0;
205 for (Integer spalteToSort : sortMax3Spalten) {
206 TableSortField sortField = new TableSortField();
207 sortField.Field = spalteToSort; // 0 = erste spalte
208 sortField.IsAscending = aufSteigendSortieren;
209 sortField.IsCaseSensitive = caseSensitive;
210 sortFields[idx] = sortField;
211 idx++;
212 }
213
214 PropertyValue[] aSortDesc = new PropertyValue[2];
215 PropertyValue propVal = new PropertyValue();
216 propVal.Name = "SortFields";
217 propVal.Value = sortFields;
218 aSortDesc[0] = propVal;
219
220 // specifies if cell formats are moved with the contents they belong to.
221 propVal = new PropertyValue();
222 propVal.Name = "BindFormatsToContent";
223 propVal.Value = bindFormatsToContent;
224 aSortDesc[1] = propVal;
225
226 xSortable.sort(aSortDesc);
227 }
