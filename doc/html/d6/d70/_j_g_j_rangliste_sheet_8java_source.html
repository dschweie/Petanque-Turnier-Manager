251 private List<TeamStats> berechneUndSortiere(TeamMeldungen aktiveMeldungen) throws GenerateException {
252 Map<Integer, int[]> statsRaw = leseSpielplanStats(aktiveMeldungen);
253 List<TeamStats> daten = new ArrayList<>();
254 for (Team team : aktiveMeldungen.teams()) {
255 int[] s = statsRaw.getOrDefault(team.getNr(), new int[4]);
256 daten.add(new TeamStats(team.getNr(), s[0], s[1], s[2], s[3]));
257 }
258 daten.sort(Comparator.comparingInt(TeamStats::spielePlus).reversed()
259 .thenComparing(Comparator.comparingInt(TeamStats::spielPunkteDiff).reversed())
260 .thenComparing(Comparator.comparingInt(TeamStats::spielPunktePlus).reversed()));
261 return daten;
262 }

264 private Map<Integer, int[]> leseSpielplanStats(TeamMeldungen aktiveMeldungen) throws GenerateException {
265 Map<Integer, int[]> statsMap = new HashMap<>();
266 for (Team team : aktiveMeldungen.teams()) {
267 statsMap.put(team.getNr(), new int[4]);
268 }
269
270 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
271 XSpreadsheet spielplanSheet = SheetMetadataHelper.findeSheetUndHeile(
272 xDoc, SheetMetadataHelper.SCHLUESSEL_JGJ_SPIELPLAN, JGJSpielPlanSheet.sheetName());
273 if (spielplanSheet == null) {
274 return statsMap;
275 }
276
277 int startZeile = JGJSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE;
278 RangeData teamNrData = RangeHelper.from(spielplanSheet, xDoc,
279 RangePosition.from(JGJSpielPlanSheet.TEAM_A_NR_SPALTE, startZeile,
280 JGJSpielPlanSheet.TEAM_B_NR_SPALTE, startZeile + MAX_SPIELPLAN_ZEILEN))
281 .getDataFromRange();
282
283 RangeData spielpunkteData = RangeHelper.from(spielplanSheet, xDoc,
284 RangePosition.from(JGJSpielPlanSheet.SPIELPNKT_A_SPALTE, startZeile,
285 JGJSpielPlanSheet.SPIELPNKT_B_SPALTE, startZeile + MAX_SPIELPLAN_ZEILEN))
286 .getDataFromRange();
287
288 for (int i = 0; i < teamNrData.size(); i++) {
289 RowData teamNrZeile = teamNrData.get(i);
290 if (teamNrZeile.size() < 2) {
291 break;
292 }
293 int nrA = teamNrZeile.get(0).getIntVal(0);
294 if (nrA <= 0) {
295 continue; // Gruppen-Header-Zeile oder Ende
296 }
297 int nrB = teamNrZeile.get(1).getIntVal(0);
298
299 if (nrB <= 0) {
300 var freispielStats = statsMap.computeIfAbsent(nrA, k -> new int[4]);
301 freispielStats[0]++;
302 RowData freispielPunkte = spielpunkteData.get(i);
303 freispielStats[2] += freispielPunkte.size() > 0 ? freispielPunkte.get(0).getIntVal(0) : 0;
304 freispielStats[3] += freispielPunkte.size() > 1 ? freispielPunkte.get(1).getIntVal(0) : 0;
305 continue;
306 }
307
308 RowData punkteZeile = spielpunkteData.get(i);
309 int pktA = punkteZeile.size() > 0 ? punkteZeile.get(0).getIntVal(0) : 0;
310 int pktB = punkteZeile.size() > 1 ? punkteZeile.get(1).getIntVal(0) : 0;
311
312 if (pktA <= 0 && pktB <= 0) {
313 continue;
314 }
315
316 statsMap.computeIfAbsent(nrA, k -> new int[4])[2] += pktA;
317 statsMap.computeIfAbsent(nrA, k -> new int[4])[3] += pktB;
318 statsMap.computeIfAbsent(nrB, k -> new int[4])[2] += pktB;
319 statsMap.computeIfAbsent(nrB, k -> new int[4])[3] += pktA;
320
321 if (pktA > pktB) {
322 statsMap.computeIfAbsent(nrA, k -> new int[4])[0]++;
323 statsMap.computeIfAbsent(nrB, k -> new int[4])[1]++;
324 } else if (pktB > pktA) {
325 statsMap.computeIfAbsent(nrB, k -> new int[4])[0]++;
326 statsMap.computeIfAbsent(nrA, k -> new int[4])[1]++;
327 }
328 }
329 return statsMap;
330 }

425 private void insertDatenAlsWerte(XSpreadsheet sheet, List<TeamStats> sortiert,
426 Map<Integer, String> teamNamen, int startZeile, boolean mitPlatz) throws GenerateException {
427 if (sortiert.isEmpty()) {
428 return;
429 }
430
431 int letzteZeile = startZeile + sortiert.size() - 1;
432
433 // Block 1: Nr, Name (+ optional Platz)
434 RangeData block1 = new RangeData();
435 for (int i = 0; i < sortiert.size(); i++) {
436 TeamStats stats = sortiert.get(i);
437 RowData row = block1.addNewRow();
438 row.newInt(stats.teamNr());
439 row.newString(teamNamen.getOrDefault(stats.teamNr(), ""));
440 if (mitPlatz) {
441 row.newInt(i + 1);
442 }
443 }
444 int block1EndSpalte = mitPlatz ? PLATZ_SPALTE : TEAM_NAME_SPALTE;
445 RangeHelper.from(this, RangePosition.from(TEAM_NR_SPALTE, startZeile, block1EndSpalte, letzteZeile))
446 .setDataInRange(block1);
447
448 // Block 2: Spiele+, Spiele-, SpieleΔ, SpPunkte+, SpPunkte-, SpPunkteΔ
449 RangeData block2 = new RangeData();
450 for (TeamStats stats : sortiert) {
451 RowData row = block2.addNewRow();
452 row.newInt(stats.spielePlus());
453 row.newInt(stats.spieleMinus());
454 row.newInt(stats.spielDiff());
455 row.newInt(stats.spielPunktePlus());
456 row.newInt(stats.spielPunkteMinus());
457 row.newInt(stats.spielPunkteDiff());
458 }
459 RangeHelper.from(this, block2.getRangePosition(Position.from(SPIELE_PLUS_SPALTE, startZeile)))
460 .setDataInRange(block2);
461
462 // Nr-Spalte: grau + doppelte rechte Linie + zentriert
463 getSheetHelper().setPropertiesInRange(sheet,
464 RangePosition.from(TEAM_NR_SPALTE, startZeile, TEAM_NR_SPALTE, letzteZeile),
465 CellProperties.from()
466 .margin(MeldeListeKonstanten.CELL_MARGIN)
467 .setCharColor(ColorHelper.CHAR_COLOR_GRAY_SPIELER_NR)
468 .setBorder(BorderFactory.from().allThin().doubleLn().forRight().toBorder())
469 .centerJustify());
470
471 // Name-Spalte: linksbündig
472 getSheetHelper().setPropertiesInRange(sheet,
473 RangePosition.from(TEAM_NAME_SPALTE, startZeile, TEAM_NAME_SPALTE, letzteZeile),
474 CellProperties.from().margin(MeldeListeKonstanten.CELL_MARGIN).setAllThinBorder().setHoriJustify(CellHoriJustify.LEFT));
475 }
