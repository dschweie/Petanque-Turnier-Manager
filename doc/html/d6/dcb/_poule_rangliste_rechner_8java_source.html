45 public List<PouleTeamErgebnis> sortiere(List<PouleTeamErgebnis> ergebnisse) {
46 if (ergebnisse.isEmpty()) {
47 return List.of();
48 }
49 int[][] paarungen = bauePaarungen(ergebnisse);
50 int[][] siege = baueSiege(ergebnisse);
51 int[][] spielPunkte = baueSpielPunkte(ergebnisse);
52
53 // Phase 1: stabile Basissortierung mit totalem Tiebreaker (teamNr)
54 Comparator<PouleTeamErgebnis> basisKomparator =
55 Comparator.comparingInt(PouleTeamErgebnis::siege).reversed()
56 .thenComparingInt(PouleTeamErgebnis::niederlagen)
57 .thenComparing(
58 Comparator.comparingInt(PouleTeamErgebnis::punkteDifferenz).reversed())
59 .thenComparingInt(PouleTeamErgebnis::teamNr);
60
61 var sortiert = new ArrayList<>(ergebnisse.stream().sorted(basisKomparator).toList());
62
63 // Phase 2: Direktvergleich für Zweier-Gruppen verbessern
64 verbessereDurchDirektvergleich(sortiert, paarungen, siege, spielPunkte);
65
66 return List.copyOf(sortiert);
67 }

73 private void verbessereDurchDirektvergleich(List<PouleTeamErgebnis> sortiert,
74 int[][] paarungen, int[][] siege, int[][] spielPunkte) {
75 int n = sortiert.size();
76 int start = 0;
77 while (start < n) {
78 int end = start + 1;
79 while (end < n && sindGleichrangig(sortiert.get(start), sortiert.get(end))) {
80 end++;
81 }
82 if (end - start == 2) {
83 var a = sortiert.get(start);
84 var b = sortiert.get(end - 1);
85 int komparatorWert = new Direktvergleich(a.teamNr(), b.teamNr(), paarungen, siege, spielPunkte)
86 .calc().toKomparatorWert();
87 if (komparatorWert > 0) {
88 sortiert.set(start, b);
89 sortiert.set(end - 1, a);
90 }
91 }
92 start = end;
93 }
94 }

96 private boolean sindGleichrangig(PouleTeamErgebnis a, PouleTeamErgebnis b) {
97 return a.siege() == b.siege()

107 private int[][] bauePaarungen(List<PouleTeamErgebnis> ergebnisse) {
108 int anzZeilen = zaehleSpiele(ergebnisse);
109 int[][] paarungen = new int[anzZeilen][2];
110 int idx = 0;
111 for (var ergebnis : ergebnisse) {
112 for (var spiel : ergebnis.spielErgebnisse()) {
113 paarungen[idx][0] = ergebnis.teamNr();
114 paarungen[idx][1] = spiel.gegnerNr();
115 idx++;
116 }
117 }
118 return paarungen;
119 }

125 private int[][] baueSiege(List<PouleTeamErgebnis> ergebnisse) {
126 int anzZeilen = zaehleSpiele(ergebnisse);
127 int[][] siege = new int[anzZeilen][2];
128 int idx = 0;
129 for (var ergebnis : ergebnisse) {
130 for (var spiel : ergebnis.spielErgebnisse()) {
131 siege[idx][0] = spiel.eigeneSpunkte() > spiel.gegnerSpunkte() ? 1 : 0;
132 siege[idx][1] = spiel.gegnerSpunkte() > spiel.eigeneSpunkte() ? 1 : 0;
133 idx++;
134 }
135 }
136 return siege;
137 }
