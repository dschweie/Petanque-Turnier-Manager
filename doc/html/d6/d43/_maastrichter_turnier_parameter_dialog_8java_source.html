260 private Formation readFormation(XControlContainer xcc) {
261 return switch (readListBoxSelected(xcc, "lstFormation")) {
262 case 1 -> Formation.DOUBLETTE;
263 case 2 -> Formation.TRIPLETTE;
264 default -> Formation.TETE;
265 };
266 }
267
268 private static short formationIndex(Formation formation) {
269 return switch (formation) {
270 case DOUBLETTE -> 1;
271 case TRIPLETTE -> 2;
272 default -> 0;
273 };
274 }

276 private short readListBoxSelected(XControlContainer xcc, String name) {
277 XControl ctrl = xcc.getControl(name);
278 if (ctrl == null) return 0;
279 XListBox lb = Lo.qi(XListBox.class, ctrl);
280 return lb != null ? lb.getSelectedItemPos() : 0;
281 }
282
283 private boolean readCheckBoxState(XControlContainer xcc, String name) {
284 XControl ctrl = xcc.getControl(name);
285 if (ctrl == null) return false;
286 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
287 return cb != null && cb.getState() == 1;
288 }
