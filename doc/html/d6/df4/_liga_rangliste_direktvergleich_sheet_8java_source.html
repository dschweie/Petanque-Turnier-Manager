102 private void upDateSheet() throws GenerateException {
103 meldeListe.upDateSheet();
104 TeamMeldungen alleMeldungen = meldeListe.getAlleMeldungen();
105 this.ligaSpielPlan = new LigaSpielPlan(alleMeldungen);
106
107 getxCalculatable().enableAutomaticCalculation(false); // speed up
108 if (!alleMeldungen.isValid()) {
109 processBoxinfo("processbox.abbruch");
110 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK).caption(I18n.get("msg.caption.liga.spielplan"))
111 .message(I18n.get("msg.text.ungueltige.anzahl.meldungen")).show();
112 return;
113 }
114
115 if (!NewSheet.from(this, SheetNamen.direktvergleich(), METADATA_SCHLUESSEL)
116 .pos(DefaultSheetPos.LIGA_DIREKTEVERGLEICH).setForceCreate(true).setActiv()
117 .hideGrid().tabColor(konfigurationSheet.getDirektvergleichTabFarbe()).create().isDidCreate()) {
118 ProcessBox.from().info("Abbruch vom Benutzer, Liga Direktvergleich wurde nicht erstellt");
119 return;
120 }
121 meldungenSpalte.alleAktiveUndAusgesetzteMeldungenAusmeldelisteEinfuegen(meldeListe);
122 int headerBackColor = getKonfigurationSheet().getRanglisteHeaderFarbe();
123 meldungenSpalte.insertHeaderInSheet(headerBackColor);
124
125 dateneinfuegen(alleMeldungen);
126 formatData();
127 addConditionalFormuleForDirektVergleichReturnCode();
128 StringCellValue lastPos = addFooter();
129 printBereichDefinieren(lastPos.getPos());
130 }

163 private void dateneinfuegen(TeamMeldungen alleMeldungen) throws GenerateException {
164 processBoxinfo("processbox.direktvergleich.einfuegen");
165
166 Position startTeamNrPos = Position.from(ERSTE_SPALTE_DIREKTVERGLEICH, TEAM_NR_HEADER_ZEILE);
167 int headerBackColor = getKonfigurationSheet().getRanglisteHeaderFarbe();
168 StringCellValue htmNr = StringCellValue.from(getXSpreadSheet()).setPos(startTeamNrPos)
169 .setCellBackColor(headerBackColor).centerJustify().setAllThinBorder();
170
171 for (IMeldung<Team> mld : alleMeldungen.getMeldungenSortedByNr()) {
172 htmNr.setValue(mld.getNr());
173 getSheetHelper().setStringValueInCell(htmNr);
174 htmNr.spaltePlusEins();
175 }
176
177 // formula einfuegen
178 StringCellValue formula = StringCellValue.from(getXSpreadSheet())
179 .setCharWeight(com.sun.star.awt.FontWeight.BOLD);
180 StringCellValue xStr = StringCellValue.from(getXSpreadSheet()).setValue("x");
181 String spielplanBegegnungenVerweis = ligaSpielPlanVerweis(LigaSpielPlanSheet.TEAM_A_NR_SPALTE);
182 String spielplanSpieleVerweis = ligaSpielPlanVerweis(LigaSpielPlanSheet.SPIELE_A_SPALTE);
183 String spielplanSpielPunkteVerweis = ligaSpielPlanVerweis(LigaSpielPlanSheet.SPIELPNKT_A_SPALTE);
184 for (IMeldung<Team> mldA : alleMeldungen.getMeldungenSortedByNr()) {
185 for (IMeldung<Team> mldB : alleMeldungen.getMeldungenSortedByNr()) {
186 if (mldA.getNr() != mldB.getNr()) {
187 String formuleStr = direktVergleichFormula(mldA.getNr(), mldB.getNr(), spielplanBegegnungenVerweis,
188 spielplanSpieleVerweis, spielplanSpielPunkteVerweis);
189 formula.setPos(startTeamNrPos).zeilePlus(mldA.getNr()).spaltePlus(mldB.getNr() - 1)
190 .setComment(mldA.getNr() + ":" + mldB.getNr()).setValue(formuleStr);
191 getSheetHelper().setFormulaInCell(formula);
192 } else {
193 xStr.setPos(startTeamNrPos).zeilePlus(mldA.getNr()).spaltePlus(mldB.getNr() - 1);
194 getSheetHelper().setStringValueInCell(xStr);
195 }
196 }
197 }

200 private String ligaSpielPlanVerweis(int startSpalte) throws GenerateException {
201 int anzZeilen = (ligaSpielPlan.anzRunden() * 2) * ligaSpielPlan.anzBegnungenProRunde();
202 Position startBegegnungenPos = Position.from(startSpalte, LigaSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE);
203 return "$'" + LigaSpielPlanSheet.sheetName() + "'." + startBegegnungenPos.getAddressWith$() + ":"
204 + startBegegnungenPos.spaltePlusEins().zeilePlus(anzZeilen - 1).getAddressWith$();
205 }

242 private StringCellValue addFooter() throws GenerateException {
243 processBoxinfo("processbox.fusszeile.einfuegen");
244 Position startPos = Position.from(TEAM_NR_SPALTE, footerStartLinePos() + 1);
245 StringCellValue direktvergleichResultCode = StringCellValue.from(getXSpreadSheet()).setPos(startPos)
246 .setCharHeight(8);
247 StringCellValue direktvergleichResultVal = StringCellValue.from(direktvergleichResultCode).spaltePlusEins()
248 .setHoriJustify(com.sun.star.table.CellHoriJustify.LEFT);
249
250 List<DirektvergleichResult> sortedList = DirektvergleichResult.stream()
251 .sorted(Comparator.comparingInt(DirektvergleichResult::getCode)).collect(Collectors.toList());
252
253 for (DirektvergleichResult drslt : sortedList) {
254 direktvergleichResultCode.setValue(drslt.getCode());
255 direktvergleichResultVal.setValue(drslt.getAnzeigeText());
256 getSheetHelper().setStringValueInCell(direktvergleichResultCode);
257 getSheetHelper().setStringValueInCell(direktvergleichResultVal);
258 direktvergleichResultCode.zeilePlusEins();
259 direktvergleichResultVal.zeilePlusEins();
260 }
261 return direktvergleichResultCode.zeilePlus(-1); // letzte zeile
262 }
