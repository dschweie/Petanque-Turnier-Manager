200 protected SpielrundenAkkumulation gespieltenRundenEinlesen(TeamMeldungen aktiveMeldungen, int abSpielrunde,
201 int bisSpielrunde) throws GenerateException {
202
203 Map<Integer, int[]> rawPointsMap = new HashMap<>(); // teamNr → [0]=eigenePunkte, [1]=kassiertePunkte
204 Map<Integer, List<Integer>> gegnerMap = new HashMap<>();
205 Set<Integer> freilosTeams = new HashSet<>();
206 Map<Integer, Integer> wertungProTeam = new HashMap<>();
207
208 for (Team team : aktiveMeldungen.teams()) {
209 rawPointsMap.put(team.getNr(), new int[2]);
210 gegnerMap.put(team.getNr(), new ArrayList<>());
211 wertungProTeam.put(team.getNr(), 0);
212 }
213
214 int siegaufschlag = new FormuleX().getSiegaufschlag(Math.max(bisSpielrunde, 1));
215
216 if (bisSpielrunde >= abSpielrunde && bisSpielrunde >= 1) {
217 int spielrunde = Math.max(abSpielrunde, 1);
218 processBoxinfo("processbox.gespielte.runden.einlesen", spielrunde, bisSpielrunde);
219 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
220
221 for (; spielrunde <= bisSpielrunde; spielrunde++) {
222 SheetRunner.testDoCancelTask();
223 XSpreadsheet sheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
224 getSpielrundeSchluessel(spielrunde), getSheetName(SpielRundeNr.from(spielrunde)));
225 if (sheet == null) {
226 continue;
227 }
228 leseRundeEin(sheet, aktiveMeldungen, rawPointsMap, gegnerMap, freilosTeams,
229 wertungProTeam, siegaufschlag);
230 }
231 }
232
233 List<FormuleXErgebnis> ergebnisse = new ArrayList<>();
234 for (Team team : aktiveMeldungen.teams()) {
235 int[] pts = rawPointsMap.getOrDefault(team.getNr(), new int[2]);
236 List<Integer> gegnerNrn = gegnerMap.getOrDefault(team.getNr(), new ArrayList<>());
237 boolean hatteFreilos = freilosTeams.contains(team.getNr());
238 ergebnisse.add(new FormuleXErgebnis(team.getNr(), pts[0], pts[1], gegnerNrn, hatteFreilos));
239 }
240 return new SpielrundenAkkumulation(ergebnisse, wertungProTeam);
241 }

243 private void leseRundeEin(XSpreadsheet sheet, TeamMeldungen aktiveMeldungen, Map<Integer, int[]> rawPointsMap,
244 Map<Integer, List<Integer>> gegnerMap, Set<Integer> freilosTeams,
245 Map<Integer, Integer> wertungProTeam, int siegaufschlag) throws GenerateException {
246 RangePosition readRange = RangePosition.from(TEAM_A_SPALTE, ERSTE_DATEN_ZEILE, ERG_TEAM_B_SPALTE,
247 ERSTE_DATEN_ZEILE + 999);
248 RangeData rowsData = RangeHelper
249 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange).getDataFromRange();
250
251 for (RowData row : rowsData) {
252 if (row.size() < 2) {
253 break;
254 }
255 int nrA = row.get(0).getIntVal(0); // TEAM_A (relativ: 0)
256 if (nrA <= 0) {
257 break;
258 }
259 Team teamA = aktiveMeldungen.getTeam(nrA);
260 if (teamA == null) {
261 continue;
262 }
263
264 int nrB = row.get(1).getIntVal(0); // TEAM_B (relativ: 1)
265 if (nrB <= 0) {
266 freilosTeams.add(nrA);
267 wertungProTeam.merge(nrA, 126, Integer::sum);
268 continue;
269 }
270 Team teamB = aktiveMeldungen.getTeam(nrB);
271 if (teamB == null) {
272 continue;
273 }
274
275 gegnerMap.computeIfAbsent(nrA, k -> new ArrayList<>()).add(nrB);
276 gegnerMap.computeIfAbsent(nrB, k -> new ArrayList<>()).add(nrA);
277
278 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0; // ERG_TEAM_A (relativ: 2)
279 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0; // ERG_TEAM_B (relativ: 3)
280
281 rawPointsMap.computeIfAbsent(nrA, k -> new int[2])[0] += ergA;
282 rawPointsMap.computeIfAbsent(nrA, k -> new int[2])[1] += ergB;
283 rawPointsMap.computeIfAbsent(nrB, k -> new int[2])[0] += ergB;
284 rawPointsMap.computeIfAbsent(nrB, k -> new int[2])[1] += ergA;
285
286 if (ergA > ergB) {
287 wertungProTeam.merge(nrA, siegaufschlag + ergA + (ergA - ergB), Integer::sum);
288 wertungProTeam.merge(nrB, ergB, Integer::sum);
289 } else if (ergB > ergA) {
290 wertungProTeam.merge(nrB, siegaufschlag + ergB + (ergB - ergA), Integer::sum);
291 wertungProTeam.merge(nrA, ergA, Integer::sum);
292 } else if (ergA > 0) {
293 // Unentschieden mit erfassten Punkten: beide bekommen Verlierer-Formel
294 wertungProTeam.merge(nrA, ergA, Integer::sum);
295 wertungProTeam.merge(nrB, ergB, Integer::sum);
296 }
297 }

305 protected boolean neueSpielrunde(TeamMeldungen meldungen, SpielRundeNr neueSpielrundeNr,
306 SpielrundenAkkumulation akkumulierung, boolean force) throws GenerateException {
307 checkNotNull(meldungen);
308
309 processBoxinfo("processbox.neue.spielrunde", neueSpielrundeNr.getNr());
310 processBoxinfo("processbox.anzahl.meldungen", meldungen.size());
311
312 if (!NewSheet.from(this, getSheetName(getSpielRundeNr()), getSpielrundeSchluessel(getSpielRundeNr().getNr()))
313 .pos(DefaultSheetPos.SCHWEIZER_WORK).setForceCreate(force).setActiv().hideGrid().create()
314 .isDidCreate()) {
315 ProcessBox.from().info(I18n.get("formulex.spielrunde.abbruch"));
316 return false;
317 }
318
319 getKonfigurationSheet().setAktiveSpielRunde(getSpielRundeNr());
320
321 FormuleX formuleX = new FormuleX();
322 List<TeamPaarung> paarungen;
323
324 if (neueSpielrundeNr.getNr() == 1) {
325 paarungen = formuleX.ersteRunde(meldungen.teams());
326 } else {
327 Map<Integer, Integer> wertungProTeam = akkumulierung.wertungProTeam();
328 List<FormuleXErgebnis> sortiert = formuleX.sortiereNachWertung(akkumulierung.ergebnisse(),
329 e -> wertungProTeam.getOrDefault(e.teamNr(), 0));
330 paarungen = formuleX.weitereRunde(sortiert);
331 }
332
333 teamPaarungenEinfuegen(paarungen);
334 datenErsteSpalte();
335 datenformatieren();
336 fehlerSpalteFormatieren();
337 header();
338 trennlinienSetzen();
339 druckBereichSetzen();
340
341 return true;
342 }

344 private void teamPaarungenEinfuegen(List<TeamPaarung> paarungen) throws GenerateException {
345 if (paarungen == null) {
346 return;
347 }
348 int freispielPlus = getKonfigurationSheet().getFreispielPunktePlus();
349 int freispielMinus = getKonfigurationSheet().getFreispielPunkteMinus();
350 RangeData rangeData = new RangeData();
351

353 SheetRunner.testDoCancelTask();

355 row.add(new CellData(teamPaarung.getA().getNr()));
356 if (teamPaarung.hasB()) {
357 row.add(new CellData(teamPaarung.getB().getNr()));
358 } else {
359 // Freilos: kein Gegner; ERG-Spalten mit Freispiel-Property-Werten vorbelegen
360 row.add(new CellData(""));
361 row.add(new CellData(freispielPlus));
362 row.add(new CellData(freispielMinus));
363 }
364 }
365
366 Position startPos = Position.from(TEAM_A_SPALTE, ERSTE_DATEN_ZEILE);
367 RangeHelper.from(this, rangeData.getRangePosition(startPos)).setDataInRange(rangeData);
368 }

432 public static List<RangePosition> ermittleEditierbareErgebnisRanges(XSpreadsheet sheet,
433 XSpreadsheetDocument xDoc, int letzteZeile) throws GenerateException {
434 Set<Integer> freilosZeilen = ermittleFreilosZeilen(sheet, xDoc, letzteZeile);
435 List<RangePosition> ranges = new ArrayList<>();
436 int blockStart = -1;
437 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
438 if (freilosZeilen.contains(zeile)) {
439 if (blockStart >= 0) {
440 ranges.add(RangePosition.from(ERG_TEAM_A_SPALTE, blockStart, ERG_TEAM_B_SPALTE, zeile - 1));
441 blockStart = -1;
442 }
443 } else if (blockStart < 0) {
444 blockStart = zeile;
445 }
446 }
447 if (blockStart >= 0) {
448 ranges.add(RangePosition.from(ERG_TEAM_A_SPALTE, blockStart, ERG_TEAM_B_SPALTE, letzteZeile));
449 }
450 return ranges;
451 }

453 private static Set<Integer> ermittleFreilosZeilen(XSpreadsheet sheet,
454 XSpreadsheetDocument xDoc, int letzteZeile) throws GenerateException {
455 if (letzteZeile < ERSTE_DATEN_ZEILE) {
456 return Set.of();
457 }
458 RangeData teamBData = RangeHelper
459 .from(sheet, xDoc, RangePosition.from(TEAM_B_SPALTE, ERSTE_DATEN_ZEILE, TEAM_B_SPALTE, letzteZeile))
460 .getDataFromRange();
461 Set<Integer> freilos = new HashSet<>();
462 for (int i = 0; i < teamBData.size(); i++) {
463 RowData row = teamBData.get(i);
464 int nrB = (row.size() > 0) ? row.get(0).getIntVal(0) : 0;
465 if (nrB <= 0) {
466 freilos.add(ERSTE_DATEN_ZEILE + i);
467 }
468 }
469 return freilos;
470 }

504 private void fehlerSpalteFormatieren() throws GenerateException {
505 XSpreadsheet sheet = getXSpreadSheet();
506 Position letztePos = letztePositionRechtsUnten();
507 if (letztePos == null) {
508 return;
509 }
510 int letzteZeile = letztePos.getZeile();
511
512 getSheetHelper().setColumnWidth(sheet, Position.from(FEHLER_SPALTE, ERSTE_HEADER_ZEILE), 1800);
513
