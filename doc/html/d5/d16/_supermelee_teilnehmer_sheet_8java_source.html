99 protected void befuelleTeilnehmerDaten() throws GenerateException {
100 meldeliste.setSpielTag(getSpielTagNr());
101
102 processBoxinfo("processbox.spieltag.meldungen.einlesen", getSpielTagNr().getNr());
103 SpielerMeldungen aktiveUndAusgesetzt = meldeliste.getAktiveUndAusgesetztMeldungen();
104
105 List<TeilnehmerEintrag> eintraege = new ArrayList<>(aktiveUndAusgesetzt.size());
106 if (aktiveUndAusgesetzt.size() > 0) {
107 Map<Integer, String> spielerNamen = leseSpielerNamenAusMeldeliste();
108 for (Spieler spieler : aktiveUndAusgesetzt.getSpielerList()) {
109 int nr = spieler.getNr();
110 // Anzeige ist bereits "Nachname, Vorname" → dient zugleich als Sortierschlüssel.
111 String anzeigeName = spielerNamen.getOrDefault(nr, "");
112 eintraege.add(new TeilnehmerEintrag(nr, "", anzeigeName, anzeigeName));
113 }
114 eintraege.sort(konfigurationSheet.getTeilnehmerListeSortModus().comparator());
115 }
116
117 processBoxinfo("processbox.spieltag.meldungen.einfuegen", getSpielTagNr().getNr(), eintraege.size());
118
119 TeilnehmerSheetBuilder builder = TeilnehmerSheetBuilder.from(this)
120 .daten(eintraege)
121 .teamnameAktiv(false)
122 .maxProBlock(konfigurationSheet.getMaxAnzTeilnehmerInSpalte())
123 .headerFarbe(konfigurationSheet.getMeldeListeHeaderFarbe())
124 .zebraFarben(konfigurationSheet.getMeldeListeHintergrundFarbeGerade(),
125 konfigurationSheet.getMeldeListeHintergrundFarbeUnGerade())
126 .aufbauen();
127
128 int letzteSpalte = builder.getLetzteDatenSpalte();
129 int footerZeile = builder.getLetzteDatenZeile() + 1;
130 StringCellValue footer = StringCellValue.from(getXSpreadSheet(), Position.from(0, footerZeile))
131 .setValue(I18n.get("teilnehmer.footer.anzahl", eintraege.size()))
132 .setEndPosMergeSpalte(letzteSpalte).setCharWeight(FontWeight.BOLD).setCharHeight(12)
133 .setShrinkToFit(true);
134 getSheetHelper().setStringValueInCell(footer);
135
136 SuperMeleeTeamRechner teamRechner = new SuperMeleeTeamRechner(eintraege.size());
137 footer.zeilePlusEins().setValue(I18n.get("teilnehmer.footer.teams",
138 teamRechner.getAnzDoublette(), teamRechner.getAnzTriplette()));
139 getSheetHelper().setStringValueInCell(footer);
140 footer.zeilePlusEins().setValue(I18n.get("teilnehmer.footer.bahnen", teamRechner.getAnzBahnen()));
141 getSheetHelper().setStringValueInCell(footer);
142
143 builder.freezeUndPrintbereich(footer.getPos().getZeile());
144 }

146 private Map<Integer, String> leseSpielerNamenAusMeldeliste() throws GenerateException {
147 Map<Integer, String> result = new HashMap<>();
148 XSpreadsheet xMeldeliste = meldeliste.getXSpreadSheet();
149 if (xMeldeliste == null) {
150 return result;
151 }
152 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
153 var spalte = meldeliste.getMeldungenSpalte();
154 int ersteNameSpalte = spalte.getErsteMeldungNameSpalte();
155 int letzteNameSpalte = spalte.getLetzteMeldungNameSpalte();
156
157 RangeData data = RangeHelper.from(xMeldeliste, xDoc,
158 RangePosition.from(MELDELISTE_NR_SPALTE, MELDELISTE_ERSTE_DATEN_ZEILE,
159 letzteNameSpalte, MELDELISTE_ERSTE_DATEN_ZEILE + 999)).getDataFromRange();
160 for (RowData row : data) {
161 if (row.isEmpty()) {
162 break;
163 }
164 int nr = row.get(0).getIntVal(0);
165 if (nr <= 0) {
166 break;
167 }
168 String name = kombiniereName(row, ersteNameSpalte, letzteNameSpalte);
169 result.put(nr, name);
170 }
171 return result;
172 }

175 private static String kombiniereName(RowData row, int ersteNameSpalte, int letzteNameSpalte) {
176 if (ersteNameSpalte == letzteNameSpalte) {
177 String name = row.size() > ersteNameSpalte ? row.get(ersteNameSpalte).getStringVal() : "";
178 return name != null ? name.trim() : "";
179 }
180 String vorname = row.size() > ersteNameSpalte && row.get(ersteNameSpalte).getStringVal() != null
181 ? row.get(ersteNameSpalte).getStringVal().trim() : "";
182 String nachname = row.size() > letzteNameSpalte && row.get(letzteNameSpalte).getStringVal() != null
183 ? row.get(letzteNameSpalte).getStringVal().trim() : "";
184 if (nachname.isEmpty()) {
185 return vorname;
186 }
187 if (vorname.isEmpty()) {
188 return nachname;
189 }
190 return nachname + ", " + vorname;
191 }
