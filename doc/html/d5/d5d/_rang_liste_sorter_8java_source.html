79 public void insertSortValidateSpalte(boolean isVisible) throws GenerateException {
80
81 int letzteDatenZeile = getIRangliste().getLetzteMitDatenZeileInSpielerNrSpalte();
82 int ersteDatenZiele = getIRangliste().getErsteDatenZiele();
83
84 XSpreadsheet sheet = getIRangliste().getXSpreadSheet();
85
86 // formula zusammenbauen
87 // --------------------------------------------------------------------------
88
89 List<Position> ranglisteSpalten = getIRangliste().getRanglisteSpalten();
90
91 StringBuffer formulaBuff = new StringBuffer();
92
93 // =WENN(L4>L3;"X1";"") &
94 // WENN(UND(L4=L3;N4>N3);"X3";"") &
95 // WENN(UND(L4=L3;N4=N3;Q4>Q3);"X4";"") &
96 // WENN(UND(L4=L3;N4=N3;Q4=Q3;O4>O3);"X5";"")
97
98 formulaBuff.append("IF(" + compareFormula(ranglisteSpalten.get(0), ">") + ";\"X1\";\"\")");
99 for (int ranglisteCntr = 1; ranglisteCntr < ranglisteSpalten.size(); ranglisteCntr++) {
100 // ----------------
101 formulaBuff.append(" & IF(AND(");
102 formulaBuff.append(compareFormula(ranglisteSpalten.get(0), "="));
103 int restSpalte;
104 for (restSpalte = 1; restSpalte < ranglisteCntr; restSpalte++) {
105 formulaBuff.append(";");
106 formulaBuff.append(compareFormula(ranglisteSpalten.get(restSpalte), "="));
107 }
108
109 formulaBuff.append(";");
110 formulaBuff.append(compareFormula(ranglisteSpalten.get(restSpalte), ">"));
111 formulaBuff.append(")");
112 formulaBuff.append(";\"X" + (ranglisteCntr + 2) + "\";\"\")");
113 }
114
115 // ----------------
116 // + " & IF(AND("
117 // + compareFormula(summeSpielGewonnenZelle1,"=")
118 // + ";" + compareFormula(summeSpielDiffZelle1,">")
119 // + ")"
120 // + ";\"X2\";\"\")"
121 // // ----------------
122 // + " & IF(AND("
123 // + compareFormula(summeSpielGewonnenZelle1,"=")
124 // + ";" + compareFormula(summeSpielDiffZelle1,"=")
125 // + ";" + compareFormula(punkteDiffZelle1,">")
126 // + ")"
127 // + ";\"X3\";\"\")"
128 // // ----------------
129 // + " & IF(AND("
130 // + compareFormula(summeSpielGewonnenZelle1,"=")
131 // + ";" + compareFormula(summeSpielDiffZelle1,"=")
132 // + ";" + compareFormula(punkteDiffZelle1,"=")
133 // + ";" + compareFormula(punkteGewonnenZelle1,">")
134 // + ")"
135 // + ";\"X4\";\"\")"
136
137 // Erste Daten-Zeile immer leer: Die Formel vergleicht mit der Zeile darüber.
138 // Ohne Guard würde Zeile 1 mit dem Header verglichen (z.B. gemergter Wert = 0).
139 String formel = "IF(ROW()=" + (ersteDatenZiele + 1) + ";\"\";(" + formulaBuff + "))";
140
141 FillAutoPosition fillAutoPosition = FillAutoPosition.from(Position.from(validateSpalte(), ersteDatenZiele))
142 .zeile(letzteDatenZeile);
143 getSheetHelper().setFormulaInCell(
144 StringCellValue.from(sheet, Position.from(validateSpalte(), ersteDatenZiele))
145 .setValue(formel)
146 .setFillAuto(fillAutoPosition));
147
148 // Alle Nummer Bold
149 getSheetHelper().setPropertiesInRange(sheet,
150 RangePosition.from(validateSpalte(), ersteDatenZiele, validateSpalte(), letzteDatenZeile),
151 CellProperties.from().setCharWeight(FontWeight.BOLD).setCharColor(ColorHelper.CHAR_COLOR_RED));
152 // --------------------------------------------------------------------------
153 // Header am ende, wegen Bug ? Auto Fill und ausgeblendete Spalte
154 ColumnProperties columnProperties = ColumnProperties.from()
155 .setWidth(MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH).setHoriJustify(CellHoriJustify.CENTER)
156 .isVisible(isVisible);
157 StringCellValue validateHeader = StringCellValue
158 .from(sheet, Position.from(validateSpalte(), ersteDatenZiele - 1)).addColumnProperties(columnProperties)
159 .setValue("");
160
161 if (isVisible) {
162 validateHeader.setComment(I18n.get("rangliste.comment.validate.spalte"));
163 }
164
165 getSheetHelper().setStringValueInCell(validateHeader);
166 // --------------------------------------------------------------------------
167 }

187 public void isErrorInSheet() throws GenerateException {
188 int letzteDatenZeile = getIRangliste().getLetzteMitDatenZeileInSpielerNrSpalte();
189 int ersteDatenZiele = getIRangliste().getErsteDatenZiele();
190 XSpreadsheet sheet = getIRangliste().getXSpreadSheet();
191 int validateSpalte = validateSpalte();
192 Position valSpalteStart = Position.from(validateSpalte, ersteDatenZiele);
193 Position valSpalteEnd = Position.from(validateSpalte, letzteDatenZeile);
194
195 for (int zeile = valSpalteStart.getZeile(); zeile <= valSpalteEnd.getZeile(); zeile++) {
196 String text = getSheetHelper().getTextFromCell(sheet, Position.from(validateSpalte, zeile));
197 if (StringUtils.isNoneBlank(text)) {
198 // error
199 throw new GenerateException(I18n.get("error.spieltagrangliste.zeile", zeile));
200 }
201 }

204 public void doSort() throws GenerateException {
205
206 int ersteSpalte = getIRangliste().getErsteSpalte();
207 int ersteDatenZiele = getIRangliste().getErsteDatenZiele();
208 int letzteSpalte = getIRangliste().getLetzteSpalte();
209 int letzteDatenZeile = getIRangliste().getLetzteMitDatenZeileInSpielerNrSpalte();
210
211 getIRangliste().calculateAll(); // zum sortieren werte kalkulieren
212 List<Position> ranglisteSpalten = getIRangliste().getRanglisteSpalten();
213
214 int[] sortSpalten = new int[ranglisteSpalten.size()];
215
216 int idx = 0;
217 // erste sort spalte = erste spalte in range
218 for (Position ranglistePos : ranglisteSpalten) {
219 // Position im Sheet
220 // Sortpos muss minus offset ersteSpalte sein
221 sortSpalten[idx] = ranglistePos.getSpalte() - ersteSpalte;
222 idx++;
223 }
224 RangePosition toSortRange = RangePosition.from(ersteSpalte, ersteDatenZiele, letzteSpalte, letzteDatenZeile);
225 sortHelper(toSortRange, sortSpalten);
226 }

228 private void sortHelper(RangePosition toSortRange, int[] sortSpalten) throws GenerateException {
229 SortHelper.from(getIRangliste(), toSortRange).abSteigendSortieren().spaltenToSort(sortSpalten).doSort();
230 }
