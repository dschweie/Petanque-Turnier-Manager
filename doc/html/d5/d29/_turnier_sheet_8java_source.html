197 public int getSheetPosition() {
198 XSpreadsheets xSpreadsheets = workingSpreadsheet.getWorkingSpreadsheetDocument().getSheets();
199 XIndexAccess xIndexAccess = Lo.qi(XIndexAccess.class, xSpreadsheets);
200 int anzSheets = xIndexAccess.getCount();
201 String thisSheetName = getName();
202 int retIdx = -1;
203
204 try {
205 for (int index = 0; index < anzSheets; index++) {
206 XSpreadsheet xSpreadsheet = Lo.qi(XSpreadsheet.class, xIndexAccess.getByIndex(index));
207 XNamed xsheetname = Lo.qi(XNamed.class, xSpreadsheet);
208 if (Objects.equals(thisSheetName, xsheetname.getName())) {
209 retIdx = index;
210 break;
211 }
212 }
213 } catch (IndexOutOfBoundsException | WrappedTargetException e) {
214 logger.error(e.getMessage(), e);
215 }
216
217 return retIdx;
218 }
