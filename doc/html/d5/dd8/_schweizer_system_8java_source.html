55 public List<SchweizerTeamErgebnis> sortiereNachAuswertungskriterien(List<SchweizerTeamErgebnis> ergebnisse) {
56 // BHZ (Buchholz) für jedes Team berechnen: Summe der Siege aller Gegner
57 Map<Integer, Integer> bhzMap = berechneBuchholz(ergebnisse);
58
59 // FBHZ (Feinbuchholz) berechnen: Summe der BHZ-Werte aller Gegner
60 Map<Integer, Integer> fbhzMap = berechneFeinbuchholz(ergebnisse, bhzMap);
61
62 Comparator<SchweizerTeamErgebnis> comparator = Comparator
63 .comparingInt(SchweizerTeamErgebnis::siege).reversed()
64 .thenComparing(Comparator.comparingInt(
65 (SchweizerTeamErgebnis e) -> bhzMap.getOrDefault(e.teamNr(), 0)).reversed())
66 .thenComparing(Comparator.comparingInt(
67 (SchweizerTeamErgebnis e) -> fbhzMap.getOrDefault(e.teamNr(), 0)).reversed())
68 .thenComparing(Comparator.comparingInt(SchweizerTeamErgebnis::punktedifferenz).reversed());
69
70 return ergebnisse.stream().sorted(comparator).collect(Collectors.toList());
71 }

140 public List<TeamPaarung> ersteRunde(List<Team> teams) {
141 boolean freiSpiel = IsEvenOrOdd.isOdd(teams.size());
142 int letzteMeldungNr = freiSpiel ? teams.size() + 1 : teams.size();
143 int anzTeamPaarungen = letzteMeldungNr / 2;
144
145 // first shuffle
146 List<Team> shuffled = new ArrayList<>(teams);
147 Collections.shuffle(shuffled, RandomSource.asJavaRandom());
148
149 // now sort nach Setzpos
150 List<Team> sortedTeamList = shuffled.stream()
151 .sorted((m1, m2) -> Integer.compare(m1.getSetzPos(), m2.getSetzPos())).toList();
152
153 // split into 2 List Team A/Team B
154 List<List<Team>> partition = ListUtils.partition(sortedTeamList, anzTeamPaarungen);
155
156 // merge A+B list together in one list TeamPaarung
157 List<Team> listATeams = partition.get(0);
158 List<Team> listBTeams = partition.get(1);
159 int listBSize = listBTeams.size();
160 return IntStream.range(0, anzTeamPaarungen).mapToObj(i -> {
161 Team teamA = listATeams.get(i);
162 Team teamB = (((anzTeamPaarungen - i) <= listBSize) ? listBTeams.get(anzTeamPaarungen - i - 1) : null);
163 if (teamB == null) {
164 teamA.setHatteFreilos(true);
165 }
166 return new TeamPaarung(teamA, teamB).addGegner();
167 }).sorted((tp1, tp2) -> {
168 // Freilos an letzte Stelle
169 if (tp1.getB() == null) {
170 return 1;
171 } else if (tp2.getB() == null) {
172 return -1;
173 }
174 return Integer.compare(tp1.getA().getNr(), tp2.getA().getNr());
175 }).toList();
176 }

206 public List<TeamPaarung> weitereRunde(List<Team> teams, List<SchweizerTeamErgebnis> ergebnisse) {
207 boolean freiSpiel = IsEvenOrOdd.isOdd(teams.size());
208 List<TeamPaarung> teamPaarungList = new ArrayList<>();
209
210 // sicher gehen dass hatgegner flag auf false
211 teams.forEach(team -> team.setHatGegner(false));
212
213 // zuerst freilos vergeben
214 if (freiSpiel) {
215 Team freilosTeam = IntStream.range(0, teams.size())
216 .mapToObj(i -> teams.get(teams.size() - i - 1))
217 .filter(team -> !team.isHatteFreilos()).findFirst().orElse(teams.get(0));
218 freilosTeam.setHatteFreilos(true);
219 freilosTeam.setHatGegner(true);
220 teamPaarungList.add(new TeamPaarung(freilosTeam));
221 }
222
223 // Sieggruppen bauen und innerhalb paaren
224 Map<Integer, Integer> siegeProTeam = ergebnisse.stream()
225 .collect(Collectors.toMap(SchweizerTeamErgebnis::teamNr, SchweizerTeamErgebnis::siege));
226
227 List<Team> aktivTeams = teams.stream().filter(t -> !t.isHatGegner()).toList();
228 List<List<Team>> gruppen = baueSiegeGruppen(aktivTeams, siegeProTeam);
229 paareGruppen(gruppen, teamPaarungList, teams);
230
231 return teamPaarungList.stream().sorted((tp1, tp2) -> {
232 // Freilos an letzter Stelle
233 if (tp1.isFreilos()) {
234 return 1;
235 } else if (tp2.isFreilos()) {
236 return -1;
237 }
238 return 0;
239 }).toList();
240 }

253 @VisibleForTesting
254 List<List<Team>> baueSiegeGruppen(List<Team> sortierteTeams, Map<Integer, Integer> siegeProTeam) {
255 List<List<Team>> gruppen = new ArrayList<>();
256 List<Team> aktuelleGruppe = null;
257 int letztesSiege = Integer.MIN_VALUE;
258
259 for (Team team : sortierteTeams) {
260 int siege = siegeProTeam.getOrDefault(team.getNr(), 0);
261 if (aktuelleGruppe == null || siege != letztesSiege) {
262 aktuelleGruppe = new ArrayList<>();
263 gruppen.add(aktuelleGruppe);
264 letztesSiege = siege;
265 }
266 aktuelleGruppe.add(team);
267 }
268
269 // Ungerade Gruppen: schwächstes Team in nächsttiefere Gruppe floaten (Carry-Over)
270 for (int i = 0; i < gruppen.size() - 1; i++) {
271 List<Team> gruppe = gruppen.get(i);
272 if (gruppe.size() % 2 != 0) {
273 Team floatTeam = gruppe.removeLast();
274 gruppen.get(i + 1).addFirst(floatTeam); // Float-Team wird zuerst gepaart
275 }
276 }
277 gruppen.removeIf(List::isEmpty);
278 return gruppen;
279 }

290 private void paareGruppen(List<List<Team>> gruppen, List<TeamPaarung> teamPaarungList, List<Team> alleTeams) {
291 for (List<Team> gruppe : gruppen) {
292 for (Team team : gruppe) {
293 if (!team.isHatGegner()) {
294 List<Team> restTeams = gruppe.stream().filter(t -> !t.isHatGegner() && !team.equals(t)).toList();
295 if (!restTeams.isEmpty()) {
296 Team gegner = findeGegner(team, restTeams);
297 if (gegner != null) {
298 teamPaarungList.add(new TeamPaarung(team, gegner).addGegner().setHatGegner());
299 } else {
300 // alle in der Gruppe bereits Gegner – Tausch versuchen
301 TeamPaarung invalid = new TeamPaarung(team, restTeams.getFirst());
302 TeamPaarung kannTauschenMit = kannTauschenMit(invalid, teamPaarungList, alleTeams);
303 if (kannTauschenMit != null) {
304 tauschenTeamsInPaarung(invalid, kannTauschenMit);
305 restTeams.getFirst().addGegner(team);
306 }
307 teamPaarungList.add(invalid.addGegner().setHatGegner());
308 }
309 }
310 }
311 }
312 }

315 @VisibleForTesting
316 boolean tauschenTeamsInPaarung(TeamPaarung paarA, TeamPaarung paarB) {
317 if (!paarA.hasB() || !paarB.hasB()) {
318 return false;
319 }
320
321 boolean didChange = false;
322
323 // tausche B: A1:B2 <-> A2:B1
324 if (!paarA.getA().hatAlsGegner(paarB.getB()) && !paarB.getA().hatAlsGegner(paarA.getB())) {
325 paarA.removeGegner();
326 paarB.removeGegner();
327 Team A_Bteam = paarA.getB();
328 paarA.setB(paarB.getB());
329 paarB.setB(A_Bteam);
330 didChange = true;
331 }
332 // tausche A+B: A1:A2 <-> B1:B2
333 if (!didChange && !paarA.getA().hatAlsGegner(paarB.getA()) && !paarB.getB().hatAlsGegner(paarA.getB())) {
334 paarA.removeGegner();
335 paarB.removeGegner();
336 Team A_Bteam = paarA.getB();
337 paarA.setB(paarB.getA());
338 paarB.setA(A_Bteam);
339 didChange = true;
340 }
341
342 if (didChange) {
343 paarA.setHatGegner();
344 paarA.addGegner();
345 paarB.setHatGegner();
346 paarB.addGegner();
347 }
348
349 return didChange;
350 }

360 @VisibleForTesting
361 TeamPaarung kannTauschenMit(TeamPaarung invalidTeamPaarung, List<TeamPaarung> paarungen, List<Team> teamListe) {
362
363 Team teamAInvalid = invalidTeamPaarung.getA();
364 Team teamBInvalid = invalidTeamPaarung.getB();
365 if (teamBInvalid == null) {
366 return null;
367 }
368
369 // team suchen zum tauschen
370 List<Team> reverseTeamliste = Lists.reverse(teamListe); // original wird nicht verändert
371
372 Team tauschTeam = reverseTeamliste.stream().filter(teamRev -> {
373 return teamRev.isHatGegner() && !teamRev.equals(teamAInvalid) && !teamRev.equals(teamBInvalid);
374 }).filter(teamRev2 -> {
375 // ist a- oder b-Team?
376 boolean isA = isATeam(teamRev2, paarungen);
377 // Gegner vom potenziellen Tauschteam suchen
378 Team gegnerVonteamRev2 = findGegnerAusTeamPaarungen(teamRev2, paarungen);
379
380 if (gegnerVonteamRev2 == null) {
381 return false;
382 }
383
384 if (isA) {
385 // kann gegen Spielen, A Invalid kann getauscht werden
386 return teamRev2.hatAlsGegner(teamBInvalid) && !gegnerVonteamRev2.hatAlsGegner(teamAInvalid);
387 }
388 // kann gegen Spielen, B Invalid kann getauscht werden
389 return !teamRev2.hatAlsGegner(teamAInvalid) && !gegnerVonteamRev2.hatAlsGegner(teamBInvalid);
390 }).findFirst().orElse(null);
391
392 if (tauschTeam != null) {
393 return paarungen.stream().filter(paarung -> paarung.isInPaarung(tauschTeam)).findFirst().orElse(null);
394 }
395 return null;
396 }
