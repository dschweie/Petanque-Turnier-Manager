185 private void renderKonfigurationsZuZellen(XSpreadsheet xKonfigSheet) throws GenerateException {
186 int headerFarbe = Integer.parseInt(BasePropertiesSpalte.HEADER_BACK_COLOR.replace("#", ""), 16);
187
188 sheet.getSheetHelper().setStringValueInCell(
189 StringCellValue.from(xKonfigSheet, Position.from(0, 0), "Eigenschaft")
190 .setCharWeight(FontWeight.BOLD)
191 .setCellBackColor(headerFarbe));
192 sheet.getSheetHelper().setStringValueInCell(
193 StringCellValue.from(xKonfigSheet, Position.from(1, 0), "Wert")
194 .setCharWeight(FontWeight.BOLD)
195 .setCellBackColor(headerFarbe));
196
197 List<ConfigProperty<?>> props = KaskadePropertiesSpalte.KONFIG_PROPERTIES;
198 int zeile = 1;
199 for (ConfigProperty<?> prop : props) {
200 String beschreibung = prop.getDescription() != null ? prop.getDescription() : prop.getKey();
201 beschreibung = beschreibung.split("[\r\n]")[0];
202
203 String wert;
204 if (prop.getType() == ConfigPropertyType.COLOR) {
205 int farbwert = konfigurationSheet.getKaskadePropertiesSpalte().readIntProperty(prop.getKey());
206 wert = String.format("#%06X", farbwert & 0xFFFFFF);
207 } else if (prop.getType() == ConfigPropertyType.INTEGER) {
208 wert = String.valueOf(konfigurationSheet.getKaskadePropertiesSpalte().readIntProperty(prop.getKey()));
209 } else {
210 wert = konfigurationSheet.getKaskadePropertiesSpalte().readStringProperty(prop.getKey());
211 }
212
213 sheet.getSheetHelper().setStringValueInCell(
214 StringCellValue.from(xKonfigSheet, Position.from(0, zeile), beschreibung));
215 sheet.getSheetHelper().setStringValueInCell(
216 StringCellValue.from(xKonfigSheet, Position.from(1, zeile), wert));
217 zeile++;
218 }

358 private void formatDatenSpalten() throws GenerateException {
359 sheet.processBoxinfo("processbox.ko.meldeliste.formatieren");
360 Formation formation = konfigurationSheet.getMeldeListeFormation();
361 int anzSpieler = formation.getAnzSpieler();
362 int letzteDatenZeile = getLetzteDatenZeileUseMin();
363
364 MeldungenHintergrundFarbeGeradeStyle farbeGerade = konfigurationSheet.getMeldeListeHintergrundFarbeGeradeStyle();
365 MeldungenHintergrundFarbeUnGeradeStyle farbeUngerade = konfigurationSheet
366 .getMeldeListeHintergrundFarbeUnGeradeStyle();
367
368 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
369 getTeamNrSpalte(), letzteDatenZeile);
370 RangeHelper.from(sheet, nrRange).setRangeProperties(
371 RangeProperties.from().setBorder(
372 BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
373
374 String kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
375 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
376 ConditionalFormatHelper.from(sheet, nrRange).clear()
377 .formulaIsText().styleIsFehler().applyAndDoReset()
378 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
379 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
380 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset()
381 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
382 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
383
384 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
385 RangePosition teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, 1, letzteDatenZeile);
386 RangeHelper.from(sheet, teamnameRange).setRangeProperties(
387 RangeProperties.from().setBorder(
388 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
389 }
390
391 for (int s = 0; s < anzSpieler; s++) {
392 int ersteSpielerSpalte = getVornameSpalte(s);
393 int letzteSpielerSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
394 ? getVereinsnameSpalte(s)
395 : getNachnameSpalte(s);
396 RangePosition spielerRange = RangePosition.from(ersteSpielerSpalte, ERSTE_DATEN_ZEILE,
397 letzteSpielerSpalte, letzteDatenZeile);
398 RangeHelper.from(sheet, spielerRange).setRangeProperties(
399 RangeProperties.from().setBorder(
400 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
401 }
402
403 RangePosition spRange = RangePosition.from(getSetzPositionSpalte(), ERSTE_DATEN_ZEILE,
404 getSetzPositionSpalte(), letzteDatenZeile);
405 RangeHelper.from(sheet, spRange).setRangeProperties(
406 RangeProperties.from().setBorder(
407 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
408
409 // SP ist optional (leer oder 0 = kein Setzstatus). Gleiche SP-Werte sind erlaubt und beabsichtigt:
410 // Teams mit identischer SP werden in Runde 1 nicht gegeneinander ausgelost.
411 ConditionalFormatHelper.from(sheet, spRange).clear()
412 .formulaIsText().styleIsFehler().applyAndDoReset()
413 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
414 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
415
416 RangePosition aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE,
417 getAktivSpalte(), letzteDatenZeile);
418 RangeHelper.from(sheet, aktivRange).setRangeProperties(
419 RangeProperties.from().centerJustify()
420 .setBorder(BorderFactory.from().allThin().toBorder()));
421
422 String kondAktivUngueltig = "AND(NOT(ISBLANK(" + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "));"
423 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>1;"
424 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>2)";
425 ConditionalFormatHelper.from(sheet, aktivRange).clear()
426 .formula1(kondAktivUngueltig).operator(ConditionOperator.FORMULA)
427 .styleIsFehler().applyAndDoReset()
428 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
429 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
430
431 EditierbaresZelleFormatHelper.anwenden(sheet, RangePosition.from(1, ERSTE_DATEN_ZEILE, getAktivSpalte(), letzteDatenZeile));
432 }

451 TeamMeldungen getAktiveMeldungen() throws GenerateException {
452 sheet.processBoxinfo("processbox.ko.meldeliste.einlesen");
453 XSpreadsheet xSheet = sheet.getXSpreadSheet();
454 int vornameSpalte = getVornameSpalte(0);
455 int spSpalte = getSetzPositionSpalte();
456 int aktivSpalte = getAktivSpalte();
457 int letzteZeile = letzteZeileMitDaten(xSheet);
458
459 TeamMeldungen meldungen = new TeamMeldungen();
460 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
461 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
462 if (vorname == null || vorname.isEmpty()) {
463 continue;
464 }
465 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
466 if (nr <= 0) {
467 continue;
468 }
469 int aktiv = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
470 if (aktiv == AKTIV_WERT_NIMMT_TEIL) {
471 int setzPos = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(spSpalte, zeile));
472 meldungen.addTeamWennNichtVorhanden(Team.from(nr).setSetzPos(setzPos));
473 }
474 }
475 return meldungen;
476 }

486 TeamMeldungen getMeldungenSortiertNachSetzposition() throws GenerateException {
487 sheet.processBoxinfo("processbox.ko.meldeliste.sortieren");
488 XSpreadsheet xSheet = sheet.getXSpreadSheet();
489 int vornameSpalte = getVornameSpalte(0);
490 int spSpalte = getSetzPositionSpalte();
491 int aktivSpalte = getAktivSpalte();
492 int letzteZeile = letzteZeileMitDaten(xSheet);
493
494 record TeamZeile(int nr, int sp) {}
495 List<TeamZeile> alleTeams = new ArrayList<>();
496
497 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
498 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
499 if (vorname == null || vorname.isEmpty()) {
500 continue;
501 }
502 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
503 if (nr <= 0) {
504 continue;
505 }
506 int aktiv = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
507 if (aktiv != AKTIV_WERT_NIMMT_TEIL) {
508 continue;
509 }
510 int sp = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(spSpalte, zeile));
511 alleTeams.add(new TeamZeile(nr, sp));
512 }
513
514 // Gruppen nach SP-Wert: SP > 0 = eigene Gruppe, SP = 0 = ungesetzte Teams
515 Map<Integer, List<TeamZeile>> setzGruppen = new LinkedHashMap<>();
516 List<TeamZeile> ungesetzte = new ArrayList<>();
517 for (TeamZeile t : alleTeams) {
518 if (t.sp() > 0) {
519 setzGruppen.computeIfAbsent(t.sp(), k -> new ArrayList<>()).add(t);
520 } else {
521 ungesetzte.add(t);
522 }
523 }
524 // Innerhalb jeder Gruppe nach Teamnummer sortieren (deterministisch)

616 void pruefeAufDoppelteTeamNr(XSpreadsheet xSheet) throws GenerateException {
617 int letzteZeile = letzteZeileMitDaten(xSheet);
618 if (letzteZeile < ERSTE_DATEN_ZEILE) {
619 return;
620 }
621 int vornameSpalte = getVornameSpalte(0);
622 Map<Integer, List<Integer>> alleNrn = new LinkedHashMap<>();
623 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
624 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
625 if (vorname == null || vorname.isEmpty()) {
626 continue;
627 }
628 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
629 if (nr <= 0) {
630 continue;
631 }
632 alleNrn.computeIfAbsent(nr, k -> new ArrayList<>()).add(zeile);
633 }
634 Map<Integer, List<Integer>> duplikate = new LinkedHashMap<>();
635 for (Map.Entry<Integer, List<Integer>> entry : alleNrn.entrySet()) {
636 if (entry.getValue().size() > 1) {
637 duplikate.put(entry.getKey(), entry.getValue());
638 }
639 }
640 if (duplikate.isEmpty()) {
641 return;
642 }
643 var sb = new StringBuilder("Meldeliste wurde nicht aktualisiert.\nDoppelte Startnummern:");
644 for (Map.Entry<Integer, List<Integer>> entry : duplikate.entrySet()) {
645 sb.append("\nNr. ").append(entry.getKey()).append(": Zeilen ");
646 List<Integer> zeilen = entry.getValue();
647 for (int i = 0; i < zeilen.size(); i++) {
648 if (i > 0) {
649 sb.append(", ");
650 }
651 sb.append(zeilen.get(i) + 1);
652 }
653 }
654 throw new GenerateException(sb.toString());
655 }
