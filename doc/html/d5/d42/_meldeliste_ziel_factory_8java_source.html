40 public static Optional<MeldelisteZiel> fuerAktivesSheet(WorkingSpreadsheet ws) {
41 TurnierSystem ts = new DocumentPropertiesHelper(ws).getTurnierSystemAusDocument();
42 if (ts == null || ts == TurnierSystem.KEIN) {
43 return Optional.empty();
44 }
45 Optional<MeldelisteLayout> layoutOpt = leseLayout(ts, ws);
46 if (layoutOpt.isEmpty()) {
47 return Optional.empty();
48 }
49 MeldelisteLayout l = layoutOpt.get();
50 return SheetMeldelisteAdapter.fuer(ws, SheetNamen.meldeliste(), ts,
51 l.formation(), l.teamnameAktiv(), l.vereinsnameAktiv());
52 }

60 public static Optional<TurnierSystem> nichtUnterstuetztesSystem(WorkingSpreadsheet ws) {

83 private static Optional<MeldelisteLayout> leseLayout(TurnierSystem ts, WorkingSpreadsheet ws) {
84 try {
85 return switch (ts) {
86 case SUPERMELEE -> Optional.of(new MeldelisteLayout(Formation.MELEE, false, false));
87 case KO -> {
88 KoKonfigurationSheet k = new KoKonfigurationSheet(ws);

90 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
91 }
92 case SCHWEIZER -> {
93 SchweizerKonfigurationSheet k = new SchweizerKonfigurationSheet(ws);
94 yield Optional.of(new MeldelisteLayout(k.getMeldeListeFormation(),
95 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
96 }
97 case MAASTRICHTER -> {
98 MaastrichterKonfigurationSheet k = new MaastrichterKonfigurationSheet(ws);
99 yield Optional.of(new MeldelisteLayout(k.getMeldeListeFormation(),
100 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
101 }
102 case JGJ -> {
103 JGJKonfigurationSheet k = new JGJKonfigurationSheet(ws);
104 yield Optional.of(new MeldelisteLayout(k.getMeldeListeFormation(),
105 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
106 }
107 case POULE -> {
108 PouleKonfigurationSheet k = new PouleKonfigurationSheet(ws);
109 yield Optional.of(new MeldelisteLayout(k.getMeldeListeFormation(),
110 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
111 }
112 case KASKADE -> {
113 KaskadeKonfigurationSheet k = new KaskadeKonfigurationSheet(ws);
114 yield Optional.of(new MeldelisteLayout(k.getMeldeListeFormation(),
115 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
116 }
117 case FORMULEX -> {
118 FormuleXKonfigurationSheet k = new FormuleXKonfigurationSheet(ws);
119 yield Optional.of(new MeldelisteLayout(k.getMeldeListeFormation(),
120 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
121 }
122 case TRIPTETE -> {
123 TripTeteKonfigurationSheet k = new TripTeteKonfigurationSheet(ws);
124 yield Optional.of(new MeldelisteLayout(Formation.TRIPLETTE,
125 k.isMeldeListeTeamnameAnzeigen(), k.isMeldeListeVereinsnameAnzeigen()));
126 }
127 case LIGA, KEIN -> Optional.empty();
128 };
129 } catch (RuntimeException e) {
130 logger.warn("Konfig-Layout für {} konnte nicht gelesen werden", ts, e);
131 return Optional.empty();
