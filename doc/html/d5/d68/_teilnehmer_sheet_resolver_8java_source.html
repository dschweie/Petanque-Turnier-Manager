61 @Override
62 public Optional<XSpreadsheet> resolve(WorkingSpreadsheet ws) {
63 var doc = ws.getWorkingSpreadsheetDocument();
64 if (doc == null) {
65 return Optional.empty();
66 }
67 try {
68 // 1. Supermelee: per-Spieltag-Schlüssel des aktiven Spieltags
69 int spieltagNr = ermittleAktivenSpieltag(ws);
70 if (spieltagNr > 0) {
71 String schluessel = SheetMetadataHelper.schluesselSupermeleeTeilnehmer(spieltagNr);
72 Optional<XSpreadsheet> ueberMeta = SheetMetadataHelper.findeSheet(doc, schluessel);
73 if (ueberMeta.isPresent()) {
74 letzterSpieltagNr = spieltagNr;
75 logger.debug("Teilnehmer-Sheet (Supermelee Spieltag {}) über Metadaten gefunden", spieltagNr);
76 return ueberMeta;
77 }
78 Optional<XSpreadsheet> ueberName = sucheNachFallbackNamen(doc, spieltagNr, schluessel);
79 if (ueberName.isPresent()) {
80 letzterSpieltagNr = spieltagNr;
81 return ueberName;
82 }
83 }
84
85 // 2. Single-Sheet-Systeme: globaler Teilnehmer-Schlüssel
86 Optional<XSpreadsheet> single = SheetMetadataHelper.findeSheet(doc, SheetMetadataHelper.SCHLUESSEL_TEILNEHMER);
87 if (single.isPresent()) {
88 letzterSpieltagNr = 0;
89 logger.debug("Teilnehmer-Sheet (Single) über Metadaten gefunden");
90 return single;
91 }
92
93 // 3. Letzter Fallback: Single-Sheet über bekannte Sheet-Namen
94 var sheets = doc.getSheets();
95 for (String fallbackName : SINGLE_FALLBACK_NAMEN) {
96 if (sheets.hasByName(fallbackName)) {
97 XSpreadsheet sheet = UnoRuntime.queryInterface(
98 XSpreadsheet.class, sheets.getByName(fallbackName));
99 if (sheet != null) {
100 try {
101 SheetMetadataHelper.schreibeSheetMetadaten(doc, sheet,
102 SheetMetadataHelper.SCHLUESSEL_TEILNEHMER);
103 logger.debug("Teilnehmer-Sheet geheilt mit Fallback-Name '{}'", fallbackName);
104 } catch (Exception e) {
105 logger.warn("Konnte Metadaten nicht schreiben für Teilnehmer-Sheet '{}': {}",
106 fallbackName, e.getMessage());
107 }
108 letzterSpieltagNr = 0;
109 return Optional.of(sheet);
110 }
111 }
112 }
113
114 logger.warn("Teilnehmer-Sheet nicht gefunden (weder Supermelee-Spieltag {} noch Single-Schlüssel noch Fallback-Namen)",
115 spieltagNr);
116 return Optional.empty();
117 } catch (Exception e) {
118 logger.error("Fehler beim Auflösen des Teilnehmer-Sheets", e);
119 return Optional.empty();
120 }
121 }

123 private int ermittleAktivenSpieltag(WorkingSpreadsheet ws) {
124 try {

129 return 0;
