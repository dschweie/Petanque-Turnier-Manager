73 static Optional<String> zeigen(WorkingSpreadsheet ws, String host) throws GenerateException {
74 var future = new CompletableFuture<Optional<String>>();
75 LoMainThread.post(ws.getxContext(), () -> {
76 try {
77 var dialog = new PasswortEingabeDialog(ws.getxContext(), host, ws);
78 dialog.erstelleUndAusfuehren();
79 Optional<String> ergebnis = dialog.passwort != null && !dialog.passwort.isBlank()
80 ? Optional.of(dialog.passwort)
81 : Optional.empty();
82 ergebnis.ifPresent(pw -> {
83 if (dialog.speichern) {
84 GlobalProperties.get().setUploadPasswort(host, pw);
85 }
86 });
87 future.complete(ergebnis);
88 } catch (Exception e) {
89 logger.error("Fehler im Passwort-Dialog", e);
90 future.completeExceptionally(e);
91 }
92 });
93 try {
94 return future.get();
95 } catch (InterruptedException e) {
96 Thread.currentThread().interrupt();
97 return Optional.empty();
98 } catch (ExecutionException e) {
99 Throwable cause = e.getCause();
100 throw new GenerateException(cause != null ? cause.getMessage() : e.getMessage());
101 }
102 }

150 private void beimOkGeklickt() {
151 if (xcc == null || xDialog == null) {
152 return;
153 }
154 var passwortCtrl = xcc.getControl("txtPasswort");
155 if (passwortCtrl != null) {
156 var tc = Lo.qi(XTextComponent.class, passwortCtrl);
157 passwort = tc != null ? tc.getText() : "";
158 }
159 var checkCtrl = xcc.getControl("chkSpeichern");
160 if (checkCtrl != null) {
161 var cb = Lo.qi(XCheckBox.class, checkCtrl);
162 speichern = cb != null && cb.getState() == 1;
163 }
164 xDialog.endExecute();
165 }
