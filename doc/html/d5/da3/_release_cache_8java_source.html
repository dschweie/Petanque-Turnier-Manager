104 void schreibeMitZeitstempel(ReleaseInfo release, Instant cachedAt) throws IOException {
105 Objects.requireNonNull(release, "release");
106 Objects.requireNonNull(cachedAt, "cachedAt");
107 var parent = cacheDatei.getParent();
108 if (parent != null) {
109 Files.createDirectories(parent);
110 }
111 var tmp = (parent != null ? parent : Paths.get("."))
112 .resolve(dateiname() + ".tmp");
113 var huelle = new CacheHuelle(cachedAt, release);
114 try (BufferedWriter writer = Files.newBufferedWriter(tmp)) {
115 gson.toJson(huelle, CacheHuelle.class, writer);
116 }
117 try {
118 Files.move(tmp, cacheDatei,
119 StandardCopyOption.ATOMIC_MOVE,
120 StandardCopyOption.REPLACE_EXISTING);
121 } catch (AtomicMoveNotSupportedException e) {
122 logger.debug("ATOMIC_MOVE nicht unterstützt – Fallback auf REPLACE_EXISTING", e);
123 Files.move(tmp, cacheDatei, StandardCopyOption.REPLACE_EXISTING);
124 } finally {
125 Files.deleteIfExists(tmp);
126 }

133 public Optional<ReleaseInfo> ladeWennFrisch(Duration maxAlter) {
134 Objects.requireNonNull(maxAlter, "maxAlter");
135 var huelle = laden().orElse(null);
136 if (huelle == null) {
137 return Optional.empty();
138 }
139 var alter = Duration.between(huelle.cachedAt, Instant.now());
140 if (alter.isNegative() || alter.compareTo(maxAlter) <= 0) {
141 return Optional.of(huelle.release);
142 }
143 logger.debug("Cache-Eintrag ist {} alt (max {}) – als stale verworfen", alter, maxAlter);
144 return Optional.empty();
145 }

152 public Optional<ReleaseInfo> ladeUnabhaengigVomAlter() {

156 private Optional<CacheHuelle> laden() {
157 if (!(Files.exists(cacheDatei) && Files.isReadable(cacheDatei))) {
158 return Optional.empty();
159 }
160 try (BufferedReader reader = Files.newBufferedReader(cacheDatei)) {
161 var huelle = gson.fromJson(reader, CacheHuelle.class);
162 if (huelle == null || huelle.cachedAt == null || huelle.release == null) {
163 korruptUmbenennen("unvollstaendige-struktur");
164 return Optional.empty();
165 }
166 return Optional.of(huelle);
167 } catch (JsonSyntaxException e) {
168 logger.warn("Cache-Datei {} enthält ungültiges JSON – wird als korrupt markiert", cacheDatei, e);
169 korruptUmbenennen("json-syntax");
170 return Optional.empty();
171 } catch (IOException e) {
172 logger.warn("Cache-Datei {} konnte nicht gelesen werden", cacheDatei, e);
173 return Optional.empty();
174 } catch (RuntimeException e) {
175 // Record-Compact-Constructor wirft NullPointerException/IllegalArgumentException
176 // wenn Pflichtfelder im JSON fehlen → wie Korruption behandeln.
177 logger.warn("Cache-Datei {} hat unerwartete Struktur – wird als korrupt markiert", cacheDatei, e);
178 korruptUmbenennen("struktur-mismatch");
179 return Optional.empty();
180 }
181 }
182
183 private void korruptUmbenennen(String grund) {
184 var ziel = cacheDatei.resolveSibling(
185 dateiname() + ".corrupt-" + Instant.now().toEpochMilli());
186 try {
187 Files.move(cacheDatei, ziel, StandardCopyOption.REPLACE_EXISTING);
188 logger.info("Korrupte Cache-Datei nach {} verschoben (Grund: {})", ziel, grund);
189 } catch (IOException ioe) {
190 logger.warn("Konnte korrupte Cache-Datei {} nicht umbenennen", cacheDatei, ioe);
191 }

211 private static final class InstantTypeAdapter extends TypeAdapter<Instant> {
212 @Override
213 public void write(JsonWriter out, @Nullable Instant value) throws IOException {
214 if (value == null) {
215 out.nullValue();
216 return;
217 }
218 out.value(DateTimeFormatter.ISO_INSTANT.format(value));
219 }
220
221 @Override
222 public @Nullable Instant read(JsonReader in) throws IOException {
223 if (in.peek() == JsonToken.NULL) {
224 in.nextNull();
225 return null;
226 }
227 var raw = in.nextString();
228 try {
229 return Instant.parse(raw);
230 } catch (DateTimeParseException e) {
231 throw new JsonSyntaxException("Ungültiger Instant-Wert: " + raw, e);
