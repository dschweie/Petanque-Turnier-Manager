140 @Override
141 protected void doRun() throws GenerateException {
142 setSpielTag(getKonfigurationSheet().getAktiveSpieltag());
143 SpielRundeNr aktuelleSpielrunde = getKonfigurationSheet().getAktiveSpielRunde();
144 setSpielRundeNr(aktuelleSpielrunde);
145 getMeldeListe().upDateSheet();
146 SpielerMeldungen aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
147
148 if (!canStart(aktiveMeldungen)) {
149 return;
150 }
151
152 int historieAb = getKonfigurationSheet().getSpielRundeNeuAuslosenAb();
153 int historieBis = aktuelleSpielrunde.getNr() - 1;
154 gespieltenRundenEinlesen(aktiveMeldungen, historieAb, historieBis);
155 if (neueSpielrunde(aktiveMeldungen, aktuelleSpielrunde, historieAb, historieBis)) {
156 new SpielrundeSheet_Validator(getWorkingSpreadsheet()).validateSpieltag(getSpielTag());
157 getSheetHelper().setActiveSheet(getXSpreadSheet());
158 }

161 public SpielerSpielrundeErgebnisList ergebnisseEinlesen() throws GenerateException {
162 SpielerSpielrundeErgebnisList spielerSpielrundeErgebnisse = new SpielerSpielrundeErgebnisList();
163 XSpreadsheet xsheet = getXSpreadSheet();
164
165 if (xsheet == null) {
166 return spielerSpielrundeErgebnisse;
167 }
168
169 Position spielerNrPos = Position.from(ERSTE_SPIELERNR_SPALTE, ERSTE_DATEN_ZEILE);
170
171 int tCntr = 0;
172 boolean spielerZeilevorhanden = true;
173 while (spielerZeilevorhanden && tCntr++ < 999) {
174 spielerZeilevorhanden = false;
175
176 for (int spalteCntr = 1; spalteCntr <= 3; spalteCntr++) {
177 int spielerNr = getSheetHelper().getIntFromCell(xsheet, spielerNrPos);
178 if (spielerNr > 0) {
179 spielerZeilevorhanden = true;
180 spielerSpielrundeErgebnisse.add(SpielerSpielrundeErgebnis.from(getSpielRundeNr(), spielerNr,
181 spielerNrPos, ERSTE_SPALTE_ERGEBNISSE, SpielRundeTeam.A));
182 }
183 spielerNrPos.spaltePlusEins();
184 }
185
186 for (int spalteCntr = 1; spalteCntr <= 3; spalteCntr++) {
187 int spielerNr = getSheetHelper().getIntFromCell(xsheet, spielerNrPos);
188 if (spielerNr > 0) {
189 spielerZeilevorhanden = true;
190 spielerSpielrundeErgebnisse.add(SpielerSpielrundeErgebnis.from(getSpielRundeNr(), spielerNr,
191 spielerNrPos, ERSTE_SPALTE_ERGEBNISSE, SpielRundeTeam.B));
192 }
193 spielerNrPos.spaltePlusEins();
194 }
195 spielerNrPos.zeilePlusEins().spalte(ERSTE_SPIELERNR_SPALTE);
196 }
197 return spielerSpielrundeErgebnisse;
198 }

200 public int countNumberOfSpielRundenSheets(SpielTagNr spieltag) throws GenerateException {
201 checkNotNull(spieltag);
202 int anz = 0;
203 int anzSheets = getSheetHelper().getAnzSheets();
204 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
205 for (int rdnCntr = 1; rdnCntr <= anzSheets; rdnCntr++) {
206 XSpreadsheet xsheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
207 SheetMetadataHelper.schluesselSupermeleeSpielrunde(spieltag.getNr(), rdnCntr),
208 getSheetName(spieltag, SpielRundeNr.from(rdnCntr)));
209 if (xsheet != null) {
210 anz++;
211 }
212 }
213 return anz;
