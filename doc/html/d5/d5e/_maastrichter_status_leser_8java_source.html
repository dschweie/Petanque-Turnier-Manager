40 public MaastrichterTurnierSchritt liesStatus() {

42
43 int hoechsteVorrunde = 0;
44 XSpreadsheet hoechsteVorrundeSheet = null;
45 for (int n = 1; n < 1000; n++) {
46 var schluessel = SheetMetadataHelper.schluesselMaastrichterVorrunde(n);
47 var sheetOpt = SheetMetadataHelper.findeSheet(xDoc, schluessel);
48 if (sheetOpt.isEmpty()) {

52 hoechsteVorrundeSheet = sheetOpt.get();

63 var schluessel = SheetMetadataHelper.schluesselMaastrichterFinalrunde(b);
64 if (SheetMetadataHelper.findeSheet(xDoc, schluessel).isPresent()) {
65 vorhandeneFinalrundenKeys.add(schluessel);
66 }
67 }
68
69 if (vorhandeneFinalrundenKeys.isEmpty()) {
70 return new MaastrichterTurnierSchritt(true, hoechsteVorrunde, zaehler[0], zaehler[1], false, false);
71 }
72

74 return new MaastrichterTurnierSchritt(true, hoechsteVorrunde, zaehler[0], zaehler[1], true, beendet);

77 private int[] zaehleVorrundeSpiele(XSpreadsheet sheet) {
78 int gesamt = 0;
79 int gespielt = 0;
80 try {
81 for (int zeile = SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE; zeile < MAX_ZEILEN; zeile++) {
82 var teamZelle = sheet.getCellByPosition(SchweizerAbstractSpielrundeSheet.TEAM_A_SPALTE, zeile);
83 if (teamZelle.getType() == CellContentType.EMPTY) {
84 break;
85 }
86 gesamt++;
87 var ergZelle = sheet.getCellByPosition(SchweizerAbstractSpielrundeSheet.ERG_TEAM_A_SPALTE, zeile);
88 if (ergZelle.getType() == CellContentType.VALUE) {
89 gespielt++;
90 }
91 }
92 } catch (IndexOutOfBoundsException e) {
93 logger.error("Fehler beim Lesen der Maastrichter-Vorrunde-Spiele", e);
94 }
95 return new int[] { gespielt, gesamt };

98 private boolean pruefeAlleFinalrundenBeendet(XSpreadsheetDocument xDoc, List<String> finalrundenKeys) {
99 for (var finalrundeKey : finalrundenKeys) {
100 var finalrundeSheet = SheetMetadataHelper.findeSheet(xDoc, finalrundeKey);
101 if (finalrundeSheet.isEmpty() || !pruefeFinalrundeBeendet(xDoc, finalrundeSheet.get(), finalrundeKey)) {
102 return false;
103 }
104 }
105 return true;
106 }

108 private boolean pruefeFinalrundeBeendet(XSpreadsheetDocument xDoc, XSpreadsheet sheet, String finalrundeKey) {
109 var scoreKey = SheetMetadataHelper.scoreSchluessel(finalrundeKey);
110 var scoreText = SheetMetadataHelper.leseScoreText(xDoc, scoreKey);
111 if (scoreText == null || !scoreText.startsWith(SCORE_PRAEFIX)) {
112 return false;
113 }
114 var positionenStr = scoreText.substring(SCORE_PRAEFIX.length());
115 var positionen = positionenStr.split(POSITIONS_TRENNZEICHEN);
116 try {
117 for (var pos : positionen) {
118 if (pos.isBlank()) {
119 continue;
120 }
121 var teile = pos.split(KOORDINATEN_TRENNZEICHEN);
122 if (teile.length < 2) {
123 continue;
124 }
125 int spalte = Integer.parseInt(teile[0].trim());
126 int zeile = Integer.parseInt(teile[1].trim());
127 var zelle = sheet.getCellByPosition(spalte, zeile);
128 if (zelle.getType() != CellContentType.VALUE) {
129 return false;
130 }
131 }
132 } catch (Exception e) {
133 logger.error("Fehler beim Prüfen der Maastrichter-Finalrunden-Scores", e);
134 return false;
135 }
136 return positionen.length > 0;
137 }
