28 public short show(String sTitle, String sMessage) {
29 short result = 0;
30 try {
31 XWindow xParent = DocumentHelper.getCurrentFrame(xContext).getContainerWindow();
32
33 // // get access to the office toolkit environment
34 XToolkit xKit = Lo.qi(XToolkit.class,
35 xContext.getServiceManager().createInstanceWithContext("com.sun.star.awt.Toolkit", xContext));
36 //
37 // describe the info box in its parameters
38 WindowDescriptor aDescriptor = new com.sun.star.awt.WindowDescriptor();
39 aDescriptor.WindowServiceName = "infobox";
40 aDescriptor.Bounds = new Rectangle(0, 0, 300, 200);
41 aDescriptor.WindowAttributes = WindowAttribute.BORDER | WindowAttribute.MOVEABLE
42 | WindowAttribute.CLOSEABLE;
43 aDescriptor.Type = WindowClass.MODALTOP;
44 aDescriptor.ParentIndex = 1;
45 aDescriptor.Parent = Lo.qi(XWindowPeer.class, xParent);
46 //
47 // // create the info box window
48 XWindowPeer xPeer = xKit.createWindow(aDescriptor);
49 XMessageBox xInfoBox = Lo.qi(XMessageBox.class, xPeer);
50 if (xInfoBox == null) {
51 return 0;
52 }
53 // // fill it with all given information and show it
54 xInfoBox.setCaptionText(sTitle);
55 xInfoBox.setMessageText(sMessage);
56
57 result = xInfoBox.execute();
58
59 } catch (java.lang.Throwable exIgnore) {
60 // ignore any problem, which can occur here.
61 // It's not really a bug for this example job, if
62 // it's message could not be printed out!
63 }
64 return result;
65 }
