77 private void validateSpielSystem() throws GenerateException {
78 // Property im Document vorhanden ?
79 DocumentPropertiesHelper docPropHelper = new DocumentPropertiesHelper(getWorkingSpreadsheet());
80 int spielsystem = docPropHelper.getIntProperty(BasePropertiesSpalte.KONFIG_PROP_NAME_TURNIERSYSTEM,
81 TurnierSystem.KEIN.getId());
82 if (spielsystem > 0) { // 0 = Kein
83 TurnierSystem turnierSystemAusDocument = TurnierSystem.findById(spielsystem);
84 TurnierSystem turnierSystemAusSheet = getTurnierSystem();
85 if (turnierSystemAusDocument != null && turnierSystemAusSheet.getId() != turnierSystemAusDocument.getId()) {
86 de.petanqueturniermanager.helper.msgbox.ProcessBox.from()
87 .fehler("Tabellenkalkulationsdokument wurde mit Turniersystem " + turnierSystemAusDocument
88 + " erstellt.");
89 throw new GenerateException(I18n.get("error.turniersystem.stimmt.nicht", getTurnierSystem(), turnierSystemAusDocument));
90 }
91 }

103 private void validateDocErstelltMitVersion() throws GenerateException {
104 DocumentPropertiesHelper docPropHelper = new DocumentPropertiesHelper(getWorkingSpreadsheet());
105
106 TurnierSystem spielsystem = docPropHelper.getTurnierSystemAusDocument();
107
108 // haben wir ein Turnier Dokument ?
109 if (spielsystem != TurnierSystem.KEIN) {
110 // Property im Document vorhanden ?
111 String docVersion = docPropHelper.getStringProperty(BasePropertiesSpalte.KONFIG_PROP_ERSTELLT_MIT_VERSION,
112 "unbekannt");
113 if ("unbekannt".equals(docVersion)) {
114 DocumentHelper.setDocErstelltMitVersion(getWorkingSpreadsheet());
115 return;
116 }
117 String pluginVersionNummer = ExtensionsHelper.from(getxContext()).getVersionNummer();
118 boolean versionDoMatch = false;
119 if (docVersion != null) {
120 versionDoMatch = Objects.equals(docVersion, pluginVersionNummer);
121 }
122
123 if (!versionDoMatch) {
124 getLogger().debug("Dokument Erstellt mit Version = " + docVersion);
125 getLogger().debug("Plugin Version = " + pluginVersionNummer);
126
127 String errMsg = "Dokument-Version '" + docVersion + "' stimmt nicht mit Plugin-Version '"
128 + pluginVersionNummer + "' überein";
129
130 getLogger().warn("Das Turnier-Dokument wurde mit einer anderen PTM Plugin-Version erstellt");
131 MessageBoxResult answer = MessageBox.from(getxContext(), MessageBoxTypeEnum.WARN_YES_NO_CANCEL)
132 .caption(errMsg)
133 .message(I18n.get("msg.text.version.unterschied"))
134 .show();
135
136 if (answer == MessageBoxResult.CANCEL) {
137 throw new GenerateException(errMsg);
138 }
139 if (answer == MessageBoxResult.YES) {
140 DocumentHelper.setDocErstelltMitVersion(getWorkingSpreadsheet());
141 }
142 }
143 }
