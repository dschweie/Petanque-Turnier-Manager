72 int ersterSpielerOffset = teamnameAktiv ? 2 : 1;
73 int spaltenProSpieler = vereinsnameAktiv ? 3 : 2;
74 int maxSpalte = ersterSpielerOffset + anzSpieler * spaltenProSpieler - 1;
75
76 var xDoc = sheet.getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
77 RangeData data = RangeHelper.from(xSheet, xDoc,
78 RangePosition.from(0, ersteDatenZeile, maxSpalte, ersteDatenZeile + LESE_BIS_ZEILE_OFFSET))
79 .getDataFromRange();
80
81 for (RowData row : data) {
82 if (row.isEmpty()) {
83 break;

86 if (nr <= 0) {
87 break;
88 }
89 spielerNamen.put(nr, bauSpielerNamenZusammen(row, ersterSpielerOffset, spaltenProSpieler));
90 sortNamen.put(nr, leseSortNachname(row, ersterSpielerOffset));
91 if (teamnameAktiv && row.size() > 1) {
92 String teamname = row.get(1).getStringVal();
93 teamnamen.put(nr, teamname != null ? teamname : "");
94 }
95 }

100 private static String leseSortNachname(RowData row, int ersterSpielerOffset) {
101 int vorSpalte = ersterSpielerOffset;
102 int nachSpalte = vorSpalte + 1;
103 String vorname = vorSpalte < row.size() ? row.get(vorSpalte).getStringVal() : "";
104 String nachname = nachSpalte < row.size() ? row.get(nachSpalte).getStringVal() : "";
105 String nn = nachname != null ? nachname.trim() : "";
106 return !nn.isEmpty() ? nn : (vorname != null ? vorname.trim() : "");
107 }

109 private String bauSpielerNamenZusammen(RowData row, int ersterSpielerOffset, int spaltenProSpieler) {
110 StringBuilder sb = new StringBuilder();
111 for (int s = 0; s < anzSpieler; s++) {
112 int vorSpalte = ersterSpielerOffset + s * spaltenProSpieler;
113 int nachSpalte = vorSpalte + 1;
114 String vorname = vorSpalte < row.size() ? row.get(vorSpalte).getStringVal() : "";
115 String nachname = nachSpalte < row.size() ? row.get(nachSpalte).getStringVal() : "";
116 String spielerName = baueSpielerName(vorname, nachname);
117 if (!spielerName.isEmpty()) {
118 if (sb.length() > 0) {
119 sb.append(" / ");
120 }
121 sb.append(spielerName);
122 }
123 }
124 return sb.toString();
125 }

127 private static String baueSpielerName(String vorname, String nachname) {
128 String vn = vorname != null ? vorname.trim() : "";
129 String nn = nachname != null ? nachname.trim() : "";
130 if (vn.isEmpty() && nn.isEmpty()) {
131 return "";
132 }
133 if (vn.isEmpty()) {
134 return nn;
135 }
136 if (nn.isEmpty()) {
137 return vn;
138 }
139 return vn + " " + nn;
140 }
