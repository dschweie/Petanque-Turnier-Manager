131 @Override
132 protected void doRun() throws GenerateException {
133 getxCalculatable().enableAutomaticCalculation(false);
134
135 var anzahlKaskaden = konfigurationSheet.getAnzahlKaskaden();
136 var aktiveRunde = konfigurationSheet.getAktiveKaskadenRunde();
137
138 if (konfigurationSheet.isKoFelderErstellt()) {
139 processBoxinfo("kaskade.spielrunde.ko.felder.bereits.erstellt");
140 return;
141 }
142
143 if (aktiveRunde >= anzahlKaskaden) {
144 processBoxinfo("kaskade.spielrunde.alle.kaskaden.abgeschlossen", anzahlKaskaden);
145 // KaskadeKoFeldSheet erstellt zuerst intern die Gruppenrangliste, dann alle KO-Felder
146 new KaskadeKoFeldSheet(getWorkingSpreadsheet()).doRun();
147 konfigurationSheet.setKoFelderErstellt(true);
148 return;
149 }
150
151 if (aktiveRunde > 0 && !alleErgebnisseEingetragen(aktiveRunde)) {
152 var aktuelleSheet = getXSpreadSheet();
153 if (aktuelleSheet != null) {
154 getSheetHelper().setActiveSheet(aktuelleSheet);
155 }
156 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
157 .caption(I18n.get("msg.caption.naechste.runde.nicht.moeglich"))
158 .message(I18n.get("kaskade.spielrunde.fehler.ergebnisse.fehlen", aktiveRunde))
159 .show();
160 return;
161 }
162
163 // Hook: Gruppen-Rangliste auf den Stand der gerade beendeten Runde bringen,
164 // bevor die nächste Kaskadenrunde generiert wird (analog Maastrichter/Supermelee).
165 if (aktiveRunde >= 1) {
166 new KaskadenRanglistenAktualisierer(getWorkingSpreadsheet(), getSheetHelper()).aktualisiereRanglisten();
167 }
168
169 var naechsteRundeNr = aktiveRunde + 1;
170 processBoxinfo("processbox.kaskade.naechste.runde", naechsteRundeNr);
171
172 if (naechsteRundeNr == 1) {
173 meldeListe.vollstaendigAktualisieren();
174 } else {
175 meldeListe.upDateSheet();
176 }
177 var meldungenNachSP = meldeListe.getMeldungenSortiertNachSetzposition();
178
179 if (meldungenNachSP.size() < 4) {
180 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
181 .caption(I18n.get("msg.caption.aktuelle.spielrunde.fehler"))
182 .message(I18n.get("kaskade.spielrunde.fehler.zu.wenige.teams"))
183 .show();
184 return;
185 }
186
187 boolean freispielGewonnen = konfigurationSheet.getFreispielPunktePlus()
188 > konfigurationSheet.getFreispielPunkteMinus();
189 var plan = KaskadenKoRundenPlaner.berechne(meldungenNachSP.size(), anzahlKaskaden, freispielGewonnen);
190 var runde = plan.kaskadeRunden().get(naechsteRundeNr - 1);
191
192 aktuelleRundeNr = naechsteRundeNr;
193 var sheetName = SheetNamen.kaskadenRunde(naechsteRundeNr);
194 var schluessel = SheetMetadataHelper.schluesselKaskadenRunde(naechsteRundeNr);
195
196 var erstellt = NewSheet.from(this, sheetName, schluessel)
197 .pos(DefaultSheetPos.KASKADE_WORK)
198 .setForceCreate(isForceOk())
199 .tabColor(konfigurationSheet.getSpielrundeTabFarbe())
200 .setActiv()
201 .hideGrid()
202 .create()
203 .isDidCreate();
204
205 if (!erstellt) {
206 processBoxinfo("schweizer.spielrunde.abbruch");
207 return;
208 }
209
210 spielrundeEintragen(runde, naechsteRundeNr == 1, plan, meldungenNachSP);
211 headerFormatieren(naechsteRundeNr);
212 SheetFreeze.from(getTurnierSheet()).anzZeilen(ERSTE_DATEN_ZEILE).doFreeze();
213 datenFormatieren(runde);
214 druckBereichSetzen();
215
216 konfigurationSheet.setAktiveKaskadenRunde(naechsteRundeNr);
217 }

223 private void spielrundeEintragen(KaskadenKoRunde runde, boolean ersteRunde,
224 KaskadenKoRundenPlan plan, TeamMeldungen meldungenNachSP) throws GenerateException {
225 processBoxinfo("processbox.neue.spielrunde", runde.rundenNr());
226
227 int freispielPlus = konfigurationSheet.getFreispielPunktePlus();
228 int freispielMinus = konfigurationSheet.getFreispielPunkteMinus();
229
230 var teamNrMap = teamNrMapErstellen(ersteRunde, plan, runde.rundenNr(), meldungenNachSP);
231
232 var datenBlock = new RangeData();
233 var gruppeBlocks = new ArrayList<GruppeBlock>();
234 int laufendeNr = 1;
235 int aktuelleDatenZeile = ERSTE_DATEN_ZEILE;
236
237 for (var gruppenRunde : runde.gruppenRunden()) {
238 SheetRunner.testDoCancelTask();
239 int gruppeStart = aktuelleDatenZeile;
240
241 for (var spiel : gruppenRunde.spielPaare()) {
242 var zeile = datenBlock.addNewRow();
243 zeile.add(new CellData(laufendeNr));
244 zeile.add(teamCellData(gruppenRunde.pfad(), spiel.positionA(), teamNrMap));
245 zeile.add(teamCellData(gruppenRunde.pfad(), spiel.positionB(), teamNrMap));
246 zeile.add(new CellData(""));
247 zeile.add(new CellData(""));
248 laufendeNr++;
249 aktuelleDatenZeile++;
250 }
251
252 if (gruppenRunde.anzFreilose() > 0) {
253 var freilosZeile = datenBlock.addNewRow();
254 freilosZeile.add(new CellData(laufendeNr));
255 freilosZeile.add(teamCellData(gruppenRunde.pfad(), gruppenRunde.anzTeams(), teamNrMap));
256 freilosZeile.add(new CellData("")); // kein Gegner = Freilos
257 freilosZeile.add(new CellData(freispielPlus)); // ERG_TEAM_A vorbelegen
258 freilosZeile.add(new CellData(freispielMinus)); // ERG_TEAM_B vorbelegen
259 laufendeNr++;
260 aktuelleDatenZeile++;
261 }
262
263 gruppeBlocks.add(new GruppeBlock(gruppenRunde.pfad(), gruppeStart, aktuelleDatenZeile - 1));
264 }
265
266 var datenStart = Position.from(SPIEL_NR_SPALTE, ERSTE_DATEN_ZEILE);
267 RangeHelper.from(this, datenBlock.getRangePosition(datenStart)).setDataInRange(datenBlock);
268
269 gruppenLabelsSchreiben(gruppeBlocks);
270 }

278 private Map<String, List<Integer>> teamNrMapErstellen(boolean ersteRunde, KaskadenKoRundenPlan plan,
279 int rundenNr, TeamMeldungen meldungenNachSP) throws GenerateException {
280 if (ersteRunde) {
281 var auslosung = new ArrayList<>(meldungenNachSP.teams());
282 Collections.shuffle(auslosung, RandomSource.asJavaRandom());
283 return Map.of("", auslosung.stream().map(Team::getNr).toList());
284 }
285 return new KaskadeRundenErgebnisLeser(getWorkingSpreadsheet()).ladeZwischenPositionen(rundenNr, plan);
286 }

294 private CellData teamCellData(String pfad, int position, Map<String, List<Integer>> teamNrMap) {
295 if (teamNrMap != null) {
296 var teams = teamNrMap.get(pfad);
297 if (teams != null && position >= 1 && position <= teams.size()) {
298 return new CellData(teams.get(position - 1));
299 }
300 }
301 var label = pfad.isEmpty() ? String.valueOf(position) : pfad + "-" + position;
302 return new CellData(label);
303 }

465 private void gruppentrennlinienSetzen(KaskadenKoRunde runde) throws GenerateException {
466 var sheet = getXSpreadSheet();
467 int aktuellZeile = ERSTE_DATEN_ZEILE;
468 var gruppen = runde.gruppenRunden();
469
470 for (int i = 0; i < gruppen.size() - 1; i++) {
471 var gruppe = gruppen.get(i);
472 int anzZeilen = gruppe.spielPaare().size() + gruppe.anzFreilose();
473 int trennZeile = aktuellZeile + anzZeilen - 1;
474
475 var trennRange = RangePosition.from(GRUPPE_SPALTE, trennZeile, ERG_TEAM_B_SPALTE, trennZeile);
476 getSheetHelper().setPropertyInRange(sheet, trennRange, TABLE_BORDER2,
477 BorderFactory.from().allThin().boldLn().forBottom().toBorder());
478
479 aktuellZeile += anzZeilen;
480 }
