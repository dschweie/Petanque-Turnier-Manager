158 @Override
159 protected void doRun() throws GenerateException {
160 XSpreadsheet xSheet = getXSpreadSheet();
161 if (xSheet == null) {
162 logger.warn("K.-O. Meldeliste nicht gefunden");
163 return;
164 }
165 stringsBesinigen(xSheet);
166 teamnummernVergeben(xSheet);
167 aktivDefaultSetzen(xSheet);
168 pruefeAufDoppelteTeamNr(xSheet);
169 nachRangSortieren(xSheet);
170 upDateSheet();
171 }

176 private void aktivDefaultSetzen(XSpreadsheet xSheet) throws GenerateException {
177 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
178 if (letzteZeile < ERSTE_DATEN_ZEILE) {
179 return;
180 }
181 int vornameSpalte = delegate.getVornameSpalte(0);
182 int aktivSpalte = delegate.getAktivSpalte();
183
184 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
185 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
186 if (vorname == null || vorname.isBlank()) {
187 continue;
188 }
189 int aktiv = getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
190 if (aktiv <= 0) {
191 getSheetHelper().setNumberValueInCell(
192 NumberCellValue.from(xSheet, Position.from(aktivSpalte, zeile))
193 .setValue(AKTIV_WERT_NIMMT_TEIL));
194 }
195 }
196 }

201 private void teamnummernVergeben(XSpreadsheet xSheet) throws GenerateException {
202 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
203 if (letzteZeile < ERSTE_DATEN_ZEILE) {
204 return;
205 }
206
207 int nrSpalte = getNrSpalte();
208 int vornameSpalte = delegate.getVornameSpalte(0);
209
210 // Höchste vorhandene Nr lesen (absteigend sortieren, dann erste Zeile lesen)
211 RangePosition sortRange = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, delegate.getAktivSpalte(),
212 letzteZeile);
213 SortHelper.from(this, sortRange).spalteToSort(nrSpalte).abSteigendSortieren().doSort();
214
215 int letztNr = Math.max(0,
216 getSheetHelper().getIntFromCell(xSheet, Position.from(nrSpalte, ERSTE_DATEN_ZEILE)));
217
218 // Fehlende Nummern vergeben (nr <= 0 = leer/kein Wert)
219 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
220 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
221 if (vorname == null || vorname.isBlank()) {
222 continue;
223 }
224 int nr = getSheetHelper().getIntFromCell(xSheet, Position.from(nrSpalte, zeile));
225 if (nr <= 0) {
226 letztNr++;
227 getSheetHelper().setNumberValueInCell(
228 NumberCellValue.from(xSheet, Position.from(nrSpalte, zeile)).setValue(letztNr));
229 }
230 }
231 }

236 private void nachRangSortieren(XSpreadsheet xSheet) throws GenerateException {
237 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
238 if (letzteZeile < ERSTE_DATEN_ZEILE) {
239 return;
240 }
241 RangePosition range = RangePosition.from(getNrSpalte(), ERSTE_DATEN_ZEILE,
242 delegate.getAktivSpalte(), letzteZeile);
243 SortHelper.from(this, range).spalteToSort(delegate.getRanglisteSpalte())
244 .aufSteigendSortieren(true).doSort();
245 }
