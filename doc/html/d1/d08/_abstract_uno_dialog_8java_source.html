37 protected final void erstelleUndAusfuehren() throws com.sun.star.uno.Exception {

40 // Dialog-Modell anlegen
41 Object dialogModel = mcf.createInstanceWithContext(
42 "com.sun.star.awt.UnoControlDialogModel", xContext);
43 XPropertySet dlgProps = Lo.qi(XPropertySet.class, dialogModel);
44 dlgProps.setPropertyValue("PositionX", Integer.valueOf(50));

46 dlgProps.setPropertyValue("Width", Integer.valueOf(getBreite()));
47 dlgProps.setPropertyValue("Height", Integer.valueOf(getHoehe()));
48 dlgProps.setPropertyValue("Moveable", Boolean.TRUE);
49 dlgProps.setPropertyValue("Sizeable", Boolean.valueOf(istVeraenderbar()));
50 dlgProps.setPropertyValue("Title", getTitel());
51
52 // Dialog-Control anlegen
53 Object dialog = mcf.createInstanceWithContext(
54 "com.sun.star.awt.UnoControlDialog", xContext);
55 XControl xControl = Lo.qi(XControl.class, dialog);
56 xControl.setModel(Lo.qi(XControlModel.class, dialogModel));
57 XDialog xDialog = Lo.qi(XDialog.class, dialog);
58
59 // Toolkit + Peer erzeugen
60 Object toolkit = mcf.createInstanceWithContext(
61 "com.sun.star.awt.Toolkit", xContext);
62 XToolkit xToolkit = Lo.qi(XToolkit.class, toolkit);
63 XWindow xWindow = Lo.qi(XWindow.class, xControl);
64 xWindow.setVisible(false);
65 // Parent-Peer = Container-Window des aufrufenden LO-Frames. Damit erkennt der
66 // Window-Manager den Dialog als Transient/Child des Frames; nach Schließen kehrt
67 // der Tastatur-Focus deterministisch zum aufrufenden Dokument zurück.
68 // Ohne Parent (null) entscheidet der WM rein anhand der Stack-Order – bei mehreren
69 // offenen Calc-Fenstern springt der Focus dann auf das zuletzt aktive andere Fenster.
70 xControl.createPeer(xToolkit, holeParentPeer());
71 XWindowPeer windowPeer = xControl.getPeer();
72
73 // Steuerelemente durch Unterklasse hinzufügen
74 erstelleFelder(
75 mcf,
76 Lo.qi(XMultiServiceFactory.class, dialogModel),
77 Lo.qi(XNameContainer.class, dialogModel),
78 xToolkit, windowPeer, dlgProps, xDialog);
79
80 // Dialog ausführen – 1 = OK, 0 = Abbrechen/X
81 short ergebnis = xDialog.execute();
82 if (ergebnis == 1) {
83 try {
84 beiOkGeklickt();
85 } catch (com.sun.star.uno.Exception e) {
86 throw e;
87 } catch (Exception e) {
88 throw new com.sun.star.uno.Exception(e.getMessage(), e);
89 }
90 }
91
92 Lo.qi(XComponent.class, dialog).dispose();
93 }
