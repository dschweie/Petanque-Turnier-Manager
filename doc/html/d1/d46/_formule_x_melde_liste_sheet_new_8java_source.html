121 public void createMeldelisteWithParams(Formation formation, boolean teamnameAnzeigen,
122 boolean vereinsnameAnzeigen, int anzahlRunden) throws GenerateException {
123 logger.info("Erstelle Formule X Meldeliste: formation={}, teamname={}, verein={}, runden={}",
124 formation, teamnameAnzeigen, vereinsnameAnzeigen, anzahlRunden);
125
126 var neuesSheet = NewSheet.from(this, SheetNamen.meldeliste(), METADATA_SCHLUESSEL)
127 .pos(DefaultSheetPos.MELDELISTE).hideGrid().tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).setDocVersionWhenNew().create();
128 if (neuesSheet.isDidCreate()) {
129 getKonfigurationSheet().setMeldeListeFormation(formation);
130 getKonfigurationSheet().setMeldeListeTeamnameAnzeigen(teamnameAnzeigen);
131 getKonfigurationSheet().setMeldeListeVereinsnameAnzeigen(vereinsnameAnzeigen);
132 getKonfigurationSheet().setAnzahlRunden(anzahlRunden);
133 upDateSheet();
134 }
135 }
136
137 @Override
138 protected void doRun() throws GenerateException {
139 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.FORMULEX)
140 .prefix(getLogPrefix()).validate()) {
141 return;
142 }
143
144 Optional<FormuleXTurnierParameterDialog.TurnierParameter> param;
145 try {
146 param = FormuleXTurnierParameterDialog.from(getWorkingSpreadsheet())
147 .anzeigen(Formation.TRIPLETTE, true, false, 4);
148 } catch (com.sun.star.uno.Exception e) {
149 String errMsg = I18n.get("error.dialog.parameterdialog", e.getMessage());
150 logger.error(errMsg, e);
151 throw new GenerateException(errMsg);
152 }
153
154 if (param.isEmpty()) {
155 return;
156 }
157
158 getKonfigurationSheet().update();
159 getSheetHelper().removeAllSheetsExclude();
160 createMeldelisteWithParams(
161 param.get().formation(),
162 param.get().teamnameAnzeigen(),
163 param.get().vereinsnameAnzeigen(),
164 param.get().anzahlRunden());
165
166 logger.info("Formule X Meldeliste erstellt.");
167 }
