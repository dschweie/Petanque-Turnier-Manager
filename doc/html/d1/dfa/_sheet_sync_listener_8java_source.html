218 private void registriereSelectionChangeListener(XModel xModel, XSpreadsheetDocument xDoc) {
219 var controller = xModel.getCurrentController();
220 if (controller == null) return;
221 boolean istSpreadsheetView = Lo.qi(XSpreadsheetView.class, controller) != null;
222 logger.debug("registriereSelectionChangeListener: controller-typ={}",
223 istSpreadsheetView ? "ScTabViewShell" : "Druckvorschau/Sonstige");
224 if (!istSpreadsheetView) return;
225 XSelectionSupplier selSupplier = Lo.qi(XSelectionSupplier.class, controller);
226 if (selSupplier == null) return;
227
228 selSupplier.addSelectionChangeListener(new XSelectionChangeListener() {
229 private XSpreadsheet letztesSheet = null;

231 @Override
232 public void selectionChanged(EventObject e) {
233 try {
234 XSpreadsheetView view = Lo.qi(XSpreadsheetView.class, e.Source);
235 if (view == null) return;
236
237 XSpreadsheet aktuellesSheet = view.getActiveSheet();
238 if (aktuellesSheet == null) return;
239
240 XSpreadsheet vorherigesSheet = letztesSheet;
241 letztesSheet = aktuellesSheet;
242
243 if (aktuellesSheet == vorherigesSheet) return;
244
245 boolean istAufZielSheet = zielSheetMatch.test(xDoc, aktuellesSheet);
246 boolean warAufZielSheet = vorherigesSheet != null
247 && zielSheetMatch.test(xDoc, vorherigesSheet);
248
249 if (istAufZielSheet && !warAufZielSheet && !SheetRunner.isRunning()
250 && istPassendesDokument(xDoc)) {
251 if (SheetRunner.consumeSelectionChangeSuppression()) {
252 logger.trace("selectionChanged: Unterdrückt – ausgelöst durch setActiveSheet()");
253 return;
254 }
255 plane(xDoc, aktuellesSheet, "selectionChanged");
256 }
257 } catch (RuntimeException t) {
258 logger.error("Fehler im SelectionChangeListener", t);
259 }
260 }

360 private void pruefeUndStarte(XSpreadsheetDocument xDoc, XSpreadsheet sheet,
361 EingabeSignatur signatur, String key, String grund, int versuch) {
362 long startNs = System.nanoTime();
363 long hashStartNs = startNs;
364 SignaturErgebnis ergebnis = signatur.berechne(xDoc, versuch);
365 long hashDauerMs = (System.nanoTime() - hashStartNs) / 1_000_000L;
366 switch (ergebnis) {
367 case SignaturErgebnis.Ok ok -> handleOk(xDoc, sheet, key, grund, ok.hash());
368 case SignaturErgebnis.SheetFehlt fehlt -> handleSheetFehlt(xDoc, sheet, key, fehlt);
369 case SignaturErgebnis.TransientFehler te ->
370 handleTransient(xDoc, sheet, signatur, key, grund, te);
371 case SignaturErgebnis.PermanenterFehler pe -> logger.warn(
372 "Sheet-Sync-Signatur fehlgeschlagen (permanent, key={}): {}", key, pe.grund(),
373 pe.cause());
374 }
375 long gesamtMs = (System.nanoTime() - startNs) / 1_000_000L;
376 PerfLog.log(logger, "[WORKER-TIMING] SheetSyncListener.pruefeUndStarte key={} system={} hash={} ms gesamt={} ms ergebnis={} thread={}",
377 key, erwartesTurnierSystem, hashDauerMs, gesamtMs,
378 ergebnis.getClass().getSimpleName(), Thread.currentThread().getName());
379 }

381 private void handleOk(XSpreadsheetDocument xDoc, XSpreadsheet sheet, String key,
382 String grund, String hash) {
383 boolean force = verbrauchForce(xDoc, key);
384 Optional<String> gespeichert = SheetSyncSignaturStore.ladeHash(xDoc, key);
385 if (!force && gespeichert.isPresent() && gespeichert.get().equals(hash)) {
386 SheetSyncSignaturStore.aktualisiereVerifyZeit(xDoc, key);
387 logger.debug("Sheet-Sync übersprungen (Hash unverändert, key={})", key);
388 return;
389 }
390 if (SheetRunner.isRunning()) {
391 logger.debug("Sheet-Sync ausgelassen – SheetRunner läuft (key={})", key);
392 return;
393 }
394 String effektiverGrund = force ? "forcedRevalidation"
395 : (gespeichert.isEmpty() ? "erstaufbau" : "hashMismatch");
396 logger.debug("Sheet-Sync START – key={}, trigger={}, grund={}",
397 key, grund, effektiverGrund);
398 // Hash vor dem asynchronen Runner speichern: stimmt mit den jetzt gelesenen Eingaben
399 // überein. Sollte der Rebuild scheitern, wird beim nächsten Trigger der dann frische
400 // Hash neu verglichen.
401 SheetSyncSignaturStore.speichereNachRebuild(xDoc, key, hash, effektiverGrund);
402 runnerFactory.apply(new WorkingSpreadsheet(xContext, xDoc), sheet).startSilent();
403 }

405 private void handleSheetFehlt(XSpreadsheetDocument xDoc, XSpreadsheet sheet, String key,
406 SignaturErgebnis.SheetFehlt fehlt) {
407 if (!fehlt.erwartet()) {
408 logger.debug("Sheet-Sync-Quelle '{}' fehlt (optional) – skip", fehlt.stabileId());
409 return;
410 }
411 if (SheetSyncSignaturStore.recoveryBereitsVersucht(xDoc, key)) {
412 logger.debug("Sheet-Sync-Quelle '{}' fehlt erwartet, Recovery bereits versucht – skip",
413 fehlt.stabileId());
414 return;
415 }
416 if (SheetRunner.isRunning()) {
417 return;
418 }
419 logger.warn("Sheet-Sync-Quelle '{}' fehlt erwartet (key={}) – Recovery-Rebuild",
420 fehlt.stabileId(), key);
421 SheetSyncSignaturStore.markiereRecoveryVersucht(xDoc, key);
422 runnerFactory.apply(new WorkingSpreadsheet(xContext, xDoc), sheet).startSilent();
423 }

425 private void handleTransient(XSpreadsheetDocument xDoc, XSpreadsheet sheet,
426 EingabeSignatur signatur, String key, String grund,
427 SignaturErgebnis.TransientFehler te) {
428 if (te.versuch() < SheetSyncDebouncer.MAX_RETRY) {
429 int naechster = te.versuch() + 1;
430 logger.debug("Sheet-Sync-Signatur transient fehlgeschlagen (key={}, versuch={}): {} – Retry",
431 key, te.versuch(), te.grund());
432 SheetSyncDebouncer.get().scheduleRetry(xDoc, key, naechster,
433 () -> pruefeUndStarte(xDoc, sheet, signatur, key, grund, naechster));
434 } else {
435 logger.warn("Sheet-Sync-Signatur transient fehlgeschlagen (key={}, versuch={}): {} – "
436 + "forceNextCheck gesetzt", key, te.versuch(), te.grund(), te.cause());
437 markiereForce(xDoc, key);
438 }
