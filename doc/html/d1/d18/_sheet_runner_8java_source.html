169 @Override
170 public final void run() {
171 boolean laueftJetzt = koordinatorVorgekoppelt || !koordinator.getAndSetLaeuft(true);
172 if (laueftJetzt) {
173 logger.debug("Start SheetRunner");
174 long runStartNs = System.nanoTime();
175 logger.trace("[FOKUS-TRACE] SheetRunner.run START class={} workingDoc={} silentBackground={} thread={}",
176 this.getClass().getSimpleName(),
177 de.petanqueturniermanager.comp.ProtocolHandler.beschreibeDokument(
178 workingSpreadsheet.getWorkingSpreadsheetDocument()),
179 silentBackground, Thread.currentThread().getName());
180 PerfLog.log(logger, "[WORKER-TIMING] SheetRunner.run START class={} thread={}",
181 this.getClass().getSimpleName(), Thread.currentThread().getName());
182 registerDisposingListener();
183 koordinator.setRunner(this);
184 koordinator.benachrichtigeListener(); // Menü deaktivieren
185 boolean isFehler = false;
186
187 // Vor-Lauf-Modified-Zustand erfassen, BEVOR der ControllerLock (Paint-Lock) und
188 // irgendwelche Schreibvorgänge laufen. Für transparente Läufe wird er unten – nach
189 // dem unlockControllers() – wiederhergestellt (siehe istModifiedFlagTransparent()).
190 final boolean docModifiedVorRun = istModifiedFlagTransparent() && istDokumentModified();
191
192 // Renderpfad des Calc-Dokuments für die Dauer des Laufs sperren: LO
193 // unterdrückt das Repaint zwischen den (potentiell hunderten) UNO-
194 // Property-Writes. Auf Windows mit D3D-Backend ist das die Schutzklammer
195 // gegen native Renderer-Crashes (scfiltlo / D3DScreenUpdateManager,
196 // nachgewiesen via Minidump beim Anwender). Der Lock umschließt
197 // bewusst auch das finally – endCommandScope() macht den Großteil
198 // der Protect/Unprotect- und CellStyle-Last und muss ebenfalls unter
199 // dem Lock laufen.
200 try (ControllerLock _ = ControllerLock.lock(workingSpreadsheet.getWorkingSpreadsheetDocument())) {
201 try {
202 if (koordinatorVorgekoppelt && !silentBackground) {
203 processBox().run(); // Nur Menü-Aktionen: ProcessBox animieren und sichtbar halten
204 }
205 if (turnierSystem != TurnierSystem.KEIN && isUpdateKonfigurationSheetBeforeDoRun()
206 && isDocumentAlive()) {
207 updateKonfigurationSheet();
208 }
209 // Lazy-Unprotect-Scope öffnen: ein einziges entsperren/schuetzen pro Kommando
210 // statt mehrfaches Toggle in jeder Sub-Operation. Echte Entsperrung passiert
211 // erst beim ersten Style-/CF-Trigger (ConditionalFormatHelper / RangeHelper.clearRange).
212 if (turnierSystem != TurnierSystem.KEIN && TurnierModus.get().istAktiv()) {
213 BlattschutzRegistry.fuer(turnierSystem)
214 .ifPresent(k -> BlattschutzManager.get().beginCommandScope(k, workingSpreadsheet));
215 }
216 if (isDocumentAlive()) {
217 doRun();
218 WebServerManager.get().sseRefreshSenden(workingSpreadsheet);
219 // Während des Runners eingetroffene Modify-Events wurden vom Listener
220 // zwar als dirty markiert, aber nicht eingeplant. Hier nachholen,
221 // damit kein Benutzer-Event verloren geht.
222 WebServerManager.get().getModifyListener().markDirtyAndSchedule();
223 }
224 } catch (DisposedException e) {
225 documentDisposed = true;
226 logger.debug("Dokument disposed während SheetRunner – sauberer Abbruch", e);
227 } catch (GenerateException e) {
228 handleGenerateException(e);
229 } catch (Exception e) {
230 isFehler = true;
231 processBox().fehler(I18n.get("processbox.interner.fehler", e.getClass().getName())).fehler(e.getMessage())
232 .fehler(I18n.get("processbox.log.hinweis"));
233 getLogger().error(e.getMessage(), e);
234 } finally {
235 koordinator.setLaeuft(false); // Immer an erste stelle diesen flag zurück
236 // Lazy-Unprotect-Scope schließen: wenn unterwegs entsperrt wurde, jetzt einmal schützen.
237 // Idempotent (No-Op falls kein Scope offen war), eigene try/finally-Robustheit im Manager.
238 BlattschutzManager.get().endCommandScope();
239 koordinator.setRunner(null);
240 koordinator.benachrichtigeListener(); // Menü reaktivieren
241 if (documentDisposed) {
242 logger.debug("SheetRunner-Cleanup: Dokument bereits disposed, keine UI-Updates");
243 } else if ((koordinatorVorgekoppelt && !silentBackground) || isFehler) {
244 // Menü-Aktion oder Fehler: ProcessBox-Fenster sichtbar zeigen
245 if (isFehler) {
246 processBox().visible().fehler(I18n.get("processbox.fehler.status")).ready();
247 } else {
248 processBox().visibleWennAutomatisch().info(I18n.get("processbox.fertig.status")).ready();
249 }
250 } else {
251 // Listener-ausgelöst oder silent-Background: nur in ProcessBox loggen, Fenster NICHT aufpoppen
252 processBox().info(I18n.get("processbox.fertig.status"));
253 }
254 if (isDocumentAlive()) {
255 try {
256 getxCalculatable().enableAutomaticCalculation(true); // falls abgeschaltet wurde
257 } catch (DisposedException e) {
258 documentDisposed = true;
259 logger.debug("Dokument disposed bei enableAutomaticCalculation", e);
260 }
261 }
262 }
263 }
264 // Während des Laufs koaleszierte TurnierEvents jetzt einmal feuern.
265 // isRunning() ist hier bereits false (im finally gesetzt), der Dispatch
266 // erfolgt über AsyncCallback auf den LO-Main-Thread und kollidiert daher
267 // nicht mit den Aufräumarbeiten dieses Worker-Threads.
268 if (isDocumentAlive()) {
269 try {
270 PetanqueTurnierMngrSingleton.flushPendingTurnierEvent();
271 } catch (RuntimeException e) {
272 getLogger().warn("Fehler beim flushPendingTurnierEvent: " + e.getMessage(), e);
273 }
274 }
275 if (isDocumentAlive()) {
276 try {
277 autoSave();
278 if (!isFehler && backupDocumentAfterRun) {
279 backUpDocument("2"); // after run
280 }
281 } catch (DisposedException e) {
282 documentDisposed = true;
283 logger.debug("Dokument disposed während Post-Run-Operationen", e);
284 } catch (Exception e) {
285 getLogger().warn("Fehler bei Post-Run-Operationen: " + e.getMessage(), e);
286 }
287 }
288 unregisterDisposingListener();
289
290 if (istModifiedFlagTransparent() && isDocumentAlive()) {
291 // Der ControllerLock ist hier bereits freigegeben → das beim unlockControllers()
292 // nachgeholte, Paint-Lock-aufgeschobene SetDocumentModified() wurde angewandt.
293 // Erst JETZT lässt sich der Vor-Lauf-Zustand verlässlich wiederherstellen, damit
294 // ein reiner Formatier-/Tab-Wechsel-Lauf das Dokument nicht als geändert markiert.
295 setzeDokumentModified(docModifiedVorRun);
296 }
297
298 // Hinweis: ehemals folgte hier ein fokussiereArbeitsDokument()-Aufruf,
299 // der via XFrame.activate() + XTopWindow.toFront() aus dem Worker-Thread
300 // den Fokus auf das Arbeits-Dokument zurückbringen sollte. Auf Wayland
301 // führt dieser UNO-Call auf Frame-/Top-Window-Objekte vom Nicht-Main-
302 // Thread reproduzierbar zu LO-Crashes (siehe Vorfall-Logs vom 22.05.2026).
303 // Stabilität wird höher gewichtet als die Fokus-Korrektur — die Folge
304 // (Fokus springt nach Sheet-Operationen auf das älteste sichtbare
305 // Calc-Fenster, nicht auf das bearbeitete Doc) ist als bekannte
306 // Einschränkung in README.md dokumentiert.
307
308 long gesamtMs = (System.nanoTime() - runStartNs) / 1_000_000L;
309 PerfLog.log(logger, "[WORKER-TIMING] SheetRunner.run ENDE class={} dauer={} ms fehler={}",
310 this.getClass().getSimpleName(), gesamtMs, isFehler);
311
312 } else {
313 MessageBox.from(getxContext(), MessageBoxTypeEnum.WARN_OK)
314 .caption(I18n.get("msg.caption.aktive.verarbeitung"))
315 .message(I18n.get("msg.text.verarbeitung.laeuft")).show();
316 }
317 }

459 private void autoSave() {
460 if (istModifiedFlagTransparent()) {
461 // Transparente Läufe (reiner Spielplan-Formatierer beim Tab-Wechsel) dürfen das
462 // Dokument niemals speichern – sie verändern den Speicher-Zustand nicht.
463 return;
464 }
465 if (!GlobalProperties.get().isAutoSave()) {
466 return;
467 }
468 // Nur speichern, wenn der Lauf das Dokument tatsächlich verändert hat. Idempotente bzw.
469 // reine Infrastruktur-Läufe (z.B. der Spielplan-Formatierer oder ein Sheet-Sync-Verify
470 // beim Tab-Wechsel) lassen das Modified-Flag über ohneModifiedFlag() auf false – dann gibt
471 // es nichts zu sichern und eine Phantom-Speicherung bei jedem Tab-Wechsel unterbleibt.
472 if (!istDokumentModified()) {
473 return;
474 }
475 BackUp.from(workingSpreadsheet).doSave();
476 }

506 private void setzeDokumentModified(boolean modified) {
507 XModifiable xModifiable = Lo.qi(XModifiable.class, workingSpreadsheet.getWorkingSpreadsheetDocument());
508 if (xModifiable == null || xModifiable.isModified() == modified) {
509 return;
510 }
511 try {
512 xModifiable.setModified(modified);
513 } catch (com.sun.star.beans.PropertyVetoException e) {
514 logger.warn("Modified-Flag konnte nicht auf {} gesetzt werden", modified, e);
515 }
