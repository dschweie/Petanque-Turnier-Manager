69 @Override
70 protected void doRun() throws GenerateException {
71 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.MAASTRICHTER)
72 .prefix(getLogPrefix()).validate()) {
73 return;
74 }
75
76 Optional<MaastrichterTurnierParameterDialog.TurnierParameter> param;
77 try {
78 param = MaastrichterTurnierParameterDialog.from(getWorkingSpreadsheet()).show(
79 Formation.DOUBLETTE, false, false, SpielplanTeamAnzeige.NR,
80 konfigurationSheet.getRankingModus(),
81 konfigurationSheet.getAnzVorrunden(),
82 konfigurationSheet.getSpielbaumTeamAnzeige(),
83 konfigurationSheet.getSpielbaumSpielbahn(),
84 konfigurationSheet.isSpielbaumSpielUmPlatz3(),
85 konfigurationSheet.getGruppenGroesse(),
86 konfigurationSheet.getMinLetzteGruppeGroesse(),
87 konfigurationSheet.getMaastrichterGruppenModus());
88 } catch (Exception e) {
89 logger.error("{} Fehler beim Anzeigen des Parameterdialogs: {}", e.getMessage(), e);
90 throw new GenerateException("Fehler beim Anzeigen des Parameterdialogs: " + e.getMessage());
91 }
92
93 if (param.isEmpty()) {
94 return;
95 }
96
97 // TurnierSystem + Page Styles setzen
98 konfigurationSheet.update();
99
100 getSheetHelper().removeAllSheetsExclude();
101
102 konfigurationSheet.setRankingModus(param.get().rankingModus);
103 konfigurationSheet.setAnzVorrunden(param.get().anzVorrunden);
104 konfigurationSheet.setAktiveSpielRunde(SpielRundeNr.from(1));
105 konfigurationSheet.setSpielbaumTeamAnzeige(param.get().spielbaumTeamAnzeige);
106 konfigurationSheet.setSpielbaumSpielbahn(param.get().spielbaumSpielbahn);
107 konfigurationSheet.setSpielbaumSpielUmPlatz3(param.get().spielUmPlatz3);
108 konfigurationSheet.setGruppenGroesse(param.get().gruppenGroesse);
109 konfigurationSheet.setMinLetzteGruppeGroesse(param.get().minLetzteGruppeGroesse);
110 konfigurationSheet.setMaastrichterGruppenModus(param.get().gruppenModus);
111
112 erstelleMeldeliste(param.get().formation, param.get().teamnameAnzeigen, param.get().vereinsnameAnzeigen,
113 param.get().spielplanTeamAnzeige);
114 }

120 public void erstelleMeldeliste(Formation formation, boolean teamnameAnzeigen, boolean vereinsnameAnzeigen,
121 SpielplanTeamAnzeige spielplanTeamAnzeige) throws GenerateException {
122 var neuesSheet = NewSheet.from(this, SheetNamen.meldeliste(), METADATA_SCHLUESSEL)
123 .pos(DefaultSheetPos.MELDELISTE).hideGrid().tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).setDocVersionWhenNew().create();
124 if (neuesSheet.isDidCreate()) {
125 konfigurationSheet.setMeldeListeFormation(formation);
126 konfigurationSheet.setMeldeListeTeamnameAnzeigen(teamnameAnzeigen);
127 konfigurationSheet.setMeldeListeVereinsnameAnzeigen(vereinsnameAnzeigen);
128 konfigurationSheet.setSpielplanTeamAnzeige(spielplanTeamAnzeige);
129 // Layout via MaastrichterMeldeListeSheetUpdate aufbauen (Format identisch mit Schweizer)
130 new MaastrichterMeldeListeSheetUpdate(getWorkingSpreadsheet()).upDateSheet();
131 }
132 }
