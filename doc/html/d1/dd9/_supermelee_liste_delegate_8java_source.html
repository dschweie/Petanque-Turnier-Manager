116 int countAnzSpieltageInMeldeliste() throws GenerateException {
117 int anzSpieltage = 0;
118 int ersteSpieltagspalteSpalte = meldeListeHelper.ersteSpieltagSpalte();
119 Position posHeader = Position.from(ersteSpieltagspalteSpalte, ZWEITE_HEADER_ZEILE);
120
121 for (int spielTagcntr = 1; spielTagcntr < 90; spielTagcntr++) {
122 String header = sheet.getSheetHelper().getTextFromCell(sheet.getXSpreadSheet(), posHeader);
123 if (StringUtils.isNotBlank(header) && header.contains(spielTagHeader(SpielTagNr.from(spielTagcntr)))) {
124 anzSpieltage++;
125 } else {
126 break;
127 }
128 posHeader.spaltePlusEins();
129 }
130 return anzSpieltage;
131 }
