41 for (int n = 1; n < 1000; n++) {

43 var sheetOpt = SheetMetadataHelper.findeSheet(xDoc, schluessel);

48 hoechsteRundeSheet = sheetOpt.get();

54

56 int gespielt = zaehler[0];
57 int gesamt = zaehler[1];
58
59 boolean beendet = hoechsteRunde >= anzahlRunden && gesamt > 0 && gespielt == gesamt;
60 return new FormuleXTurnierSchritt(true, hoechsteRunde, gespielt, gesamt, anzahlRunden, beendet);

63 private int[] zaehleSpiele(XSpreadsheet sheet) {

68 var teamZelle = sheet.getCellByPosition(FormuleXAbstractSpielrundeSheet.TEAM_A_SPALTE, zeile);
69 if (teamZelle.getType() == CellContentType.EMPTY) {
70 break;
71 }
72 var teamBZelle = sheet.getCellByPosition(FormuleXAbstractSpielrundeSheet.TEAM_B_SPALTE, zeile);
73 if (teamBZelle.getType() == CellContentType.EMPTY) {
74 continue; // Freilos: kein Gegner, kein Ergebnis nötig
75 }
76 gesamt++;
77 var ergZelle = sheet.getCellByPosition(FormuleXAbstractSpielrundeSheet.ERG_TEAM_A_SPALTE, zeile);
78 if (ergZelle.getType() == CellContentType.VALUE) {
79 gespielt++;
80 }
81 }
82 } catch (IndexOutOfBoundsException e) {
83 logger.error("Fehler beim Lesen der Formule-X-Spielrunden-Spiele", e);
84 }
85 return new int[] { gespielt, gesamt };
