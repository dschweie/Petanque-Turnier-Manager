235 private void fuegeZeileHinzu() {
236 try {
237 leseZeilenDatenAusControls();
238 var tempIdx = new int[]{-1};
239 Consumer<CompositeViewEintragRoh> callback = e -> {
240 if (tempIdx[0] == -1) {
241 eintraege.add(e);
242 tempIdx[0] = eintraege.size() - 1;
243 } else {
244 eintraege.set(tempIdx[0], e);
245 }
246 speichernUndAktualisieren();
247 };
248 var detailDialog = new CompositeViewDetailDialog(
249 xContext, null, berechneNaechstenFreienPort(), komboBoxItems, callback);
250 var neuerEintrag = detailDialog.zeigen();
251 if (neuerEintrag != null && tempIdx[0] == -1) {
252 // OK ohne vorheriges Anwenden: normaler Add-Pfad
253 eintraege.add(neuerEintrag);
254 }
255 aktualisiereZeilenArea();
256 } catch (com.sun.star.uno.Exception e) {
257 logger.error("Fehler beim Hinzufügen eines Composite Views: {}", e.getMessage(), e);
258 }

261 private void bearbeiteZeile(int idx) {
262 try {
263 leseZeilenDatenAusControls();
264 var eintrag = eintraege.get(idx);
265 Consumer<CompositeViewEintragRoh> callback = geaendert -> {
266 eintraege.set(idx, geaendert);
267 speichernUndAktualisieren();
268 };
269 var detailDialog = new CompositeViewDetailDialog(
270 xContext, eintrag, eintrag.port(), komboBoxItems, callback);
271 var geaenderterEintrag = detailDialog.zeigen();
272 if (geaenderterEintrag != null) {
273 eintraege.set(idx, geaenderterEintrag); // idempotent falls Callback schon gesetzt hat
274 }
275 aktualisiereZeilenArea();
276 } catch (com.sun.star.uno.Exception e) {
277 logger.error("Fehler beim Bearbeiten des Composite Views: {}", e.getMessage(), e);
278 }

314 private void leseZeilenDatenAusControls() {
315 for (int i = 0; i < eintraege.size(); i++) {
316 XControl aktivCtrl = xcc.getControl("aktivRow_" + i);
317 if (aktivCtrl == null) break;
318 boolean aktiv = Lo.qi(XCheckBox.class, aktivCtrl).getState() == 1;
319 var alt = eintraege.get(i);
320 eintraege.set(i, new CompositeViewEintragRoh(alt.port(), alt.name(), aktiv, alt.zoom(), alt.mitHeaderFooter(), alt.layoutJson(), alt.panels()));
321 }

324 private void validiereEintraege() throws UngueltigeEingabeException {
325 Set<Integer> bekannte = new HashSet<>();
326 for (int i = 0; i < eintraege.size(); i++) {
327 var e = eintraege.get(i);
328 int nr = i + 1;
329 if (e.port() == 0) {
330 throw new UngueltigeEingabeException(
331 I18n.get("webserver.composite.konfig.fehler.port.leer", nr));
332 }
333 if (e.port() < 1 || e.port() > 65535) {
334 throw new UngueltigeEingabeException(
335 I18n.get("webserver.composite.konfig.fehler.port.ungueltig", nr, e.port()));
336 }
337 if (!bekannte.add(e.port())) {
338 throw new UngueltigeEingabeException(
339 I18n.get("webserver.composite.konfig.fehler.port.duplikat", e.port()));
340 }
341 if (e.panels().isEmpty()) {
342 throw new UngueltigeEingabeException(
343 I18n.get("webserver.composite.konfig.fehler.kein.panel"));
344 }
345 }
