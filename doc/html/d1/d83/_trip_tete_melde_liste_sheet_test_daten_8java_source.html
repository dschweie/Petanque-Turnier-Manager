74 TurnierSheet.from(meldelisteSheet, getWorkingSpreadsheet()).setActiv();

76 List<Triplett> testTeams = listeMitTestTripletts();

87
88 int zeile = ersteDatenZeile;
89 int cntr = 0;
90 for (Triplett team : testTeams) {
91 schreibeSpieler(meldelisteSheet, zeile, vornameSpalte0, nachnameSpalte0, team.sp1Vorname(), team.sp1Nachname());
92 schreibeSpieler(meldelisteSheet, zeile, vornameSpalte1, nachnameSpalte1, team.sp2Vorname(), team.sp2Nachname());
93 schreibeSpieler(meldelisteSheet, zeile, vornameSpalte2, nachnameSpalte2, team.sp3Vorname(), team.sp3Nachname());
94 getSheetHelper().setNumberValueInCell(
95 NumberCellValue.from(meldelisteSheet, Position.from(aktivSpalte, zeile)).setValue(AKTIV_NIMMT_TEIL));
96 zeile++;
97 if (++cntr >= anzTeams) {
98 break;
99 }
100 }
101 meldeListe.upDateSheet();
102 }
