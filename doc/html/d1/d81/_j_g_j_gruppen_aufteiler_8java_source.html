31 public static List<TeamMeldungen> teileInGruppen(TeamMeldungen meldungen, int gruppengroesse) {
32 List<Team> teams = meldungen.teams();
33 if (teams.isEmpty() || gruppengroesse <= 0) {
34 return List.of(kopiere(teams));

43 private static List<TeamMeldungen> blockweise(List<Team> teams, int gruppengroesse) {
44 List<TeamMeldungen> gruppen = new ArrayList<>();
45 for (int i = 0; i < teams.size(); i += gruppengroesse) {
46 TeamMeldungen gruppe = new TeamMeldungen();
47 for (int j = i; j < Math.min(i + gruppengroesse, teams.size()); j++) {

55 private static List<TeamMeldungen> mitSetzPositionen(List<Team> teams, int gruppengroesse) {

100 private static int kleinsteFreieGruppe(List<TeamMeldungen> gruppen, Set<Integer> bereitsBelegt) {
101 int besterIdx = -1;
102 int besteGroesse = Integer.MAX_VALUE;
103 for (int i = 0; i < gruppen.size(); i++) {
