107 public void upDateSheet() throws GenerateException {
108 meldeListe.upDateSheet();
109
110 getxCalculatable().enableAutomaticCalculation(false);
111 try {
112 TeamMeldungen meldungen = meldeListe.getAlleMeldungen();
113 if (!meldungen.isValid()) {
114 processBoxinfo("processbox.abbruch");
115 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
116 .caption(I18n.get("msg.caption.triptete.rangliste"))
117 .message(I18n.get("msg.text.ungueltige.anzahl.meldungen")).show();
118 return;
119 }
120
121 if (!NewSheet.from(this, SheetNamen.rangliste(), METADATA_SCHLUESSEL)
122 .pos(DefaultSheetPos.TRIPTETE_ENDRANGLISTE).setForceCreate(true).setActiv()
123 .hideGrid().tabColor(konfigurationSheet.getRanglisteTabFarbe()).create().isDidCreate()) {
124 ProcessBox.from().info("Abbruch vom Benutzer, Trip-Tête-Rangliste wurde nicht erstellt");
125 return;
126 }
127
128 insertHeader();
129 TripTeteRanglisteDatenSchreiber.from(this, meldeListe, getWorkingSpreadsheet()).schreibeDaten();
130 for (int spalte = TEAM_NR_SPALTE; spalte <= LETZTE_SPALTE; spalte++) {
131 getSheetHelper().setOptimaleBreitePlusMarge(getXSpreadSheet(), spalte, SheetHelper.OPTIMALE_BREITE_MARGE);
132 }
133 insertFooter(meldungen.size());
134 formatieren(meldungen.size());
135 printBereichDefinieren(meldungen.size());
136 SheetFreeze.from(getTurnierSheet()).anzZeilen(ERSTE_DATEN_ZEILE).anzSpalten(3).doFreeze();
137 } finally {
138 getxCalculatable().enableAutomaticCalculation(true);
139 }
140 }

142 private void insertHeader() throws GenerateException {
143 int headerBackColor = konfigurationSheet.getRanglisteHeaderFarbe();
144
145 BorderFactory borderFact = BorderFactory.from().allThin().boldLn().forBottom();
146 TableBorder2 brdDaten = borderFact.toBorder();
147 CellProperties headerProp = CellProperties.from().setAllThinBorder().margin(MARGIN).centerJustify()
148 .setCellBackColor(headerBackColor).setShrinkToFit(true);
149 ColumnProperties colSchmal = ColumnProperties.from().setWidth(900).centerJustify();
150
151 // Nr – überspannt Header-Zeilen 1 und 2 (analog Liga-MeldungenSpalte)
152 getSheetHelper().setStringValueInCell(
153 StringCellValue.from(getXSpreadSheet(), Position.from(TEAM_NR_SPALTE, 1))
154 .setColumnProperties(colSchmal)
155 .setValue(I18n.get("column.header.nr"))
156 .setCellBackColor(headerBackColor)
157 .setBorder(BorderFactory.from(borderFact).doubleLn().forRight().toBorder())
158 .setHoriJustify(CellHoriJustify.CENTER)
159 .setVertJustify(CellVertJustify2.CENTER)
160 .setEndPosMergeZeilePlus(1)
161 .setShrinkToFit(true));
162
163 // Name – überspannt Header-Zeilen 1 und 2, identisch wie Nr
164 getSheetHelper().setStringValueInCell(
165 StringCellValue.from(getXSpreadSheet(), Position.from(NAME_SPALTE, 1))
166 .setColumnProperties(colSchmal)
167 .setValue(I18n.get("column.header.name"))
168 .setCellBackColor(headerBackColor)
169 .setBorder(brdDaten)
170 .setHoriJustify(CellHoriJustify.CENTER)
171 .setEndPosMergeZeilePlus(1)
172 .setShrinkToFit(true));
173
174 // Platz – überspannt alle 3 Header-Zeilen, hochkant, dicke rechte Linie
175 getSheetHelper().setStringValueInCell(
176 StringCellValue.from(getXSpreadSheet(), Position.from(RANG_SPALTE, 0))
177 .setColumnProperties(colSchmal)
178 .setValue(I18n.get("column.header.platz"))
179 .setRotate90()
180 .setVertJustify(CellVertJustify2.CENTER)
181 .setCellBackColor(headerBackColor)
182 .setBorder(BorderFactory.from(borderFact).forRight().toBorder())
183 .setEndPosMergeZeilePlus(2)
184 .setShrinkToFit(true)
185 .setCharWeight(FontWeight.BOLD));
186
187 // ---- Header-Zeile 3 (unterste): +/- Δ Labels ----
188 RangeData zeile3Data = new RangeData();
189 RowData zeile3 = zeile3Data.addNewRow();
190 zeile3.newString("+"); zeile3.newString("-"); // Punkte
191 zeile3.newString("+"); zeile3.newString("-"); zeile3.newString("Δ"); // Spiele + Diff
192 zeile3.newString("+"); zeile3.newString("-"); zeile3.newString("Δ"); // Spielpunkte + Diff
193
194 RangeHelper.from(this, zeile3Data.getRangePosition(Position.from(ERSTE_SUMMEN_SPALTE, 2)))
195 .setDataInRange(zeile3Data)
196 .setRangeProperties(RangeProperties.from().centerJustify().setBorder(brdDaten)
197 .setCellBackColor(headerBackColor).margin(MARGIN).setShrinkToFit(true));
198
199 // ---- Header-Zeile 2: Punkte / Spiele / Spielpunkte ----
200 StringCellValue header2val = StringCellValue.from(getXSpreadSheet())
201 .setPos(Position.from(ERSTE_SUMMEN_SPALTE, 1))
202 .setEndPosMergeSpaltePlus(1).setCellProperties(headerProp);
203
204 header2val.setValue(I18n.get("column.header.punkte"));
205 getSheetHelper().setStringValueInCell(header2val);
206 header2val.spaltePlus(2);
207
208 header2val.setValue(I18n.get("column.header.spiele")).setEndPosMergeSpaltePlus(2);
209 getSheetHelper().setStringValueInCell(header2val);
210 header2val.spaltePlus(3);
211
212 header2val.setValue(I18n.get("column.header.spielpunkte")).setEndPosMergeSpaltePlus(2);
213 getSheetHelper().setStringValueInCell(header2val);
214
215 // ---- Header-Zeile 1 (oberste): Summen + Begegn. ----
216 StringCellValue header1val = StringCellValue.from(getXSpreadSheet())
217 .setPos(Position.from(ERSTE_SUMMEN_SPALTE, 0))
218 .setEndPosMergeSpaltePlus(7).setCellProperties(headerProp);
219 header1val.setValue(I18n.get("column.header.summen"));
220 getSheetHelper().setStringValueInCell(header1val);
221
222 // Begegn. – überspannt alle 3 Zeilen, hochkant
223 TableBorder2 begegnungenBrd = BorderFactory.from(borderFact).doubleLn().forLeft().toBorder();
224 getSheetHelper().setStringValueInCell(
225 StringCellValue.from(getXSpreadSheet()).setPos(Position.from(BEGEGNUNGEN_SPALTE, 0))
226 .setValue(I18n.get("column.header.begegn"))
227 .setRotate90()
228 .setEndPosMergeZeilePlus(2)

231 .setCellBackColor(headerBackColor)
232 .setShrinkToFit(true)
233 .setComment(I18n.get("comment.begegnungen")));

245 private void formatieren(int anzTeams) throws GenerateException {
246 int letzteDatenZeile = ERSTE_DATEN_ZEILE + anzTeams - 1;
247 RangePosition daten = RangePosition.from(TEAM_NR_SPALTE, ERSTE_DATEN_ZEILE, LETZTE_SPALTE, letzteDatenZeile);
248 RangeHelper.from(this, daten).setRangeProperties(
249 RangeProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forTop().toBorder())
250 .centerJustify().margin(MARGIN).setShrinkToFit(true));
251
252 RangePosition begegnungenDaten = RangePosition.from(BEGEGNUNGEN_SPALTE, ERSTE_DATEN_ZEILE, BEGEGNUNGEN_SPALTE, letzteDatenZeile);
253 RangeHelper.from(this, begegnungenDaten).setRangeProperties(
254 RangeProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forTop().doubleLn().forLeft().toBorder()));
255
256 // Nr-Spalte: durchgehend rechte doppelte Linie (von Row 0 bis letzte Datenzeile)
257 RangeHelper.from(this, RangePosition.from(TEAM_NR_SPALTE, 0, TEAM_NR_SPALTE, letzteDatenZeile))
258 .setRangeProperties(RangeProperties.from()
259 .setBorder(BorderFactory.from().doubleLn().forRight().toBorder()));
260
261 // Nr-Spalte: Datenzellen explizit zentriert (horizontal + vertikal)
262 getSheetHelper().setPropertiesInRange(getXSpreadSheet(),
263 RangePosition.from(TEAM_NR_SPALTE, ERSTE_DATEN_ZEILE, TEAM_NR_SPALTE, letzteDatenZeile),
264 CellProperties.from().centerJustify());
265
266 // Platz-Spalte: fett + dicke rechte Linie
267 getSheetHelper().setPropertiesInRange(getXSpreadSheet(),
268 RangePosition.from(RANG_SPALTE, ERSTE_DATEN_ZEILE, RANG_SPALTE, letzteDatenZeile),
269 CellProperties.from()
270 .setCharWeight(FontWeight.BOLD)
271 .setBorder(BorderFactory.from().allThin().boldLn().forTop().forRight().toBorder()));
272
273 Integer farbeGerade = konfigurationSheet.getRanglisteHintergrundFarbeGerade();
274 Integer farbeUngerade = konfigurationSheet.getRanglisteHintergrundFarbeUnGerade();
275 RanglisteGeradeUngeradeFormatHelper.from(this, daten).geradeFarbe(farbeGerade)
276 .ungeradeFarbe(farbeUngerade).apply();
277 }
