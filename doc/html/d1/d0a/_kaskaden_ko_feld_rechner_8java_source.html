117 private static List<FeldKnoten> spalteFelderAuf(List<FeldKnoten> felder, boolean freispielGewonnen) {
118 var neueFelder = new ArrayList<FeldKnoten>(felder.size() * 2);
119 for (var feld : felder) {
120 int n = feld.anzTeams();
121 int sieger = freispielGewonnen ? (n + 1) / 2 : n / 2;
122 int verlierer = n - sieger;
123 neueFelder.add(new FeldKnoten(feld.pfad() + "S", sieger));
124 neueFelder.add(new FeldKnoten(feld.pfad() + "V", verlierer));
125 }
126 return neueFelder;
127 }

133 private static List<KaskadenKoFeldInfo> erstelleFeldInfos(List<FeldKnoten> feldListe) {
134 var ergebnis = new ArrayList<KaskadenKoFeldInfo>(feldListe.size());
135 for (int i = 0; i < feldListe.size(); i++) {
