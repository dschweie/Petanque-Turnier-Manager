39 public static List<ConfigProperty<?>> getKonfigProperties(WorkingSpreadsheet currentSpreadsheet) {
40 try {
41
42 TurnierSystem turnierSystem = getTurnierSystem(currentSpreadsheet);
43 if (turnierSystem != null) {
44 // leider oldschool fest verdrathen, weil reflection nicht funktioniert :-(
45 switch (turnierSystem) {
46 case LIGA:
47 return LigaPropertiesSpalte.KONFIG_PROPERTIES;
48 case MAASTRICHTER:
49 return MaastrichterPropertiesSpalte.KONFIG_PROPERTIES;
50 case SUPERMELEE:
51 return SuperMeleePropertiesSpalte.KONFIG_PROPERTIES;
52 case SCHWEIZER:
53 return SchweizerPropertiesSpalte.KONFIG_PROPERTIES;
54 case JGJ:
55 return JGJPropertiesSpalte.KONFIG_PROPERTIES;
56 case KO:
57 return KoPropertiesSpalte.KONFIG_PROPERTIES;
58 case POULE:
59 return PoulePropertiesSpalte.KONFIG_PROPERTIES;
60 case KASKADE:
61 return KaskadePropertiesSpalte.KONFIG_PROPERTIES;
62 case FORMULEX:
63 return FormuleXPropertiesSpalte.KONFIG_PROPERTIES;
64 case TRIPTETE:
65 return TripTetePropertiesSpalte.KONFIG_PROPERTIES;
66 default:
67 logger.error("TurnierSystem ungültig " + turnierSystem.getBezeichnung());
68 break;
69 }
70 }
71 } catch (Exception e) {
72 logger.error(e.getMessage(), e);
73 }
74 return null;
75 }

81 private static TurnierSystem getTurnierSystem(WorkingSpreadsheet currentSpreadsheet) {
82 DocumentPropertiesHelper docPropHelper = new DocumentPropertiesHelper(currentSpreadsheet);

84 TurnierSystem.KEIN.getId());
85 TurnierSystem turnierSystem = TurnierSystem.findById(turniersystemId);
86 return turnierSystem;
