193 private List<LigaTeamStats> berechneUndSortiere(TeamMeldungen aktiveMeldungen,
194 Map<Integer, String> teamNamen) throws GenerateException {
195 Map<Integer, LigaTeamStats> statsMap = leseSpielplanStats(aktiveMeldungen, teamNamen);
196 List<LigaTeamStats> daten = new ArrayList<>();
197 for (Team team : aktiveMeldungen.teams()) {
198 daten.add(statsMap.getOrDefault(team.getNr(), new LigaTeamStats(team.getNr(),
199 teamNamen.getOrDefault(team.getNr(), ""), 0, 0, 0, 0, 0, 0)));
200 }
201 daten.sort(Comparator.comparingInt(LigaTeamStats::punktePlus).reversed()
202 .thenComparing(Comparator.comparingInt(LigaTeamStats::spielePlus).reversed())
203 .thenComparing(Comparator.comparingInt(LigaTeamStats::spPunktePlus).reversed())
204 .thenComparing(Comparator.comparingInt(LigaTeamStats::spPunkteDiff).reversed()));
205 return daten;
206 }

208 private Map<Integer, LigaTeamStats> leseSpielplanStats(TeamMeldungen aktiveMeldungen,
209 Map<Integer, String> teamNamen) throws GenerateException {
210 Map<Integer, LigaTeamStats> statsMap = new HashMap<>();
211 for (Team team : aktiveMeldungen.teams()) {
212 statsMap.put(team.getNr(), new LigaTeamStats(team.getNr(),
213 teamNamen.getOrDefault(team.getNr(), ""), 0, 0, 0, 0, 0, 0));
214 }
215
216 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
217 XSpreadsheet spielplanSheet = SheetMetadataHelper.findeSheetUndHeile(
218 xDoc, SheetMetadataHelper.SCHLUESSEL_LIGA_SPIELPLAN, LigaSpielPlanSheet.LEGACY_SHEET_NAMEN);
219 if (spielplanSheet == null) {
220 return statsMap;
221 }
222
223 int startZeile = LigaSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE;
224 RangeData teamNrData = RangeHelper.from(spielplanSheet, xDoc,
225 RangePosition.from(LigaSpielPlanSheet.TEAM_A_NR_SPALTE, startZeile,
226 LigaSpielPlanSheet.TEAM_B_NR_SPALTE, startZeile + MAX_SPIELPLAN_ZEILEN))
227 .getDataFromRange();
228
229 // PUNKTE_A(8) bis SPIELPNKT_B(13) = 6 Spalten
230 RangeData ergebnisData = RangeHelper.from(spielplanSheet, xDoc,
231 RangePosition.from(LigaSpielPlanSheet.PUNKTE_A_SPALTE, startZeile,
232 LigaSpielPlanSheet.SPIELPNKT_B_SPALTE, startZeile + MAX_SPIELPLAN_ZEILEN))
233 .getDataFromRange();
234
235 int freispielSpPktPlus = konfigurationSheet.getFreispielPunktePlus();
236 int freispielSpPktMinus = konfigurationSheet.getFreispielPunkteMinus();
237
238 for (int i = 0; i < teamNrData.size(); i++) {
239 RowData teamNrZeile = teamNrData.get(i);
240 if (teamNrZeile.size() < 2) {
241 break;
242 }
243 int nrA = teamNrZeile.get(0).getIntVal(0);
244 if (nrA <= 0) {
245 continue;
246 }
247 int nrB = teamNrZeile.get(1).getIntVal(0);
248 RowData erg = ergebnisData.get(i);
249 if (erg.size() < 6) {
250 continue;
251 }
252 int pktA = erg.get(0).getIntVal(0);
253 int pktB = erg.get(1).getIntVal(0);
254 int spA = erg.get(2).getIntVal(0);
255 int spB = erg.get(3).getIntVal(0);
256 int spPktA = erg.get(4).getIntVal(0);
257 int spPktB = erg.get(5).getIntVal(0);
258
259 if (nrB <= 0) {
260 // Freispiel: immer ein Sieg für Team A; Spielpunkte aus Konfiguration
261 statsMap.computeIfPresent(nrA, (k, s) -> s.plus(1, 0, spA, 0, freispielSpPktPlus, freispielSpPktMinus));
262 continue;
263 }
264
265 if (pktA <= 0 && pktB <= 0 && spA <= 0 && spB <= 0 && spPktA <= 0 && spPktB <= 0) {
266 continue; // noch nicht gespielt
267 }
268
269 statsMap.computeIfPresent(nrA, (k, s) -> s.plus(pktA, pktB, spA, spB, spPktA, spPktB));
270 statsMap.computeIfPresent(nrB, (k, s) -> s.plus(pktB, pktA, spB, spA, spPktB, spPktA));
271 }
272 return statsMap;
273 }

275 private void schreibeDatenAlsWerte(XSpreadsheet sheet, List<LigaTeamStats> sortiert) throws GenerateException {
276 if (sortiert.isEmpty()) {
277 return;
278 }
279 int letzteZeile = ERSTE_DATEN_ZEILE + sortiert.size() - 1;
280
281 // Block 1: Nr, Name
282 RangeData block1 = new RangeData();
283 for (LigaTeamStats s : sortiert) {
284 RowData row = block1.addNewRow();
285 row.newInt(s.teamNr());
286 row.newString(s.teamName());
287 }
288 RangeHelper.from(this, RangePosition.from(TEAM_NR_SPALTE, ERSTE_DATEN_ZEILE, TEAM_NAME_SPALTE, letzteZeile))
289 .setDataInRange(block1);
290
291 // Block 2: Statistik (9 Spalten)
292 RangeData block2 = new RangeData();
293 for (LigaTeamStats s : sortiert) {
294 RowData row = block2.addNewRow();
295 row.newInt(s.punktePlus());
296 row.newInt(s.punkteMinus());
297 row.newInt(s.spielePlus());
298 row.newInt(s.spieleMinus());
299 row.newInt(s.spieleDiff());
300 row.newInt(s.spPunktePlus());
301 row.newInt(s.spPunkteMinus());
302 row.newInt(s.spPunkteDiff());
303 row.newInt(s.begegnungen());
304 }
305 RangeHelper.from(this, block2.getRangePosition(Position.from(PUNKTE_PLUS_SPALTE, ERSTE_DATEN_ZEILE)))
306 .setDataInRange(block2);
307
308 // Nr-Spalte: grau + doppelte rechte Linie + zentriert
309 getSheetHelper().setPropertiesInRange(sheet,
310 RangePosition.from(TEAM_NR_SPALTE, ERSTE_DATEN_ZEILE, TEAM_NR_SPALTE, letzteZeile),
311 CellProperties.from()
312 .setCharColor(ColorHelper.CHAR_COLOR_GRAY_SPIELER_NR)
313 .setBorder(BorderFactory.from().allThin().doubleLn().forRight().toBorder())
314 .margin(MARGIN)
315 .centerJustify());
316
317 // Name-Spalte: linksbündig
318 getSheetHelper().setPropertiesInRange(sheet,
319 RangePosition.from(TEAM_NAME_SPALTE, ERSTE_DATEN_ZEILE, TEAM_NAME_SPALTE, letzteZeile),
320 CellProperties.from().setAllThinBorder().setHoriJustify(CellHoriJustify.LEFT).margin(MARGIN));
321
322 // Statistik-Spalten: zentriert, thin border, bold-top
323 getSheetHelper().setPropertiesInRange(sheet,
324 RangePosition.from(RANGLISTE_SPALTE, ERSTE_DATEN_ZEILE, BEGEGNUNGEN_SPALTE, letzteZeile),
325 CellProperties.from().setAllThinBorder()
326 .setBorder(BorderFactory.from().allThin().boldLn().forTop().toBorder())
327 .setHoriJustify(CellHoriJustify.CENTER).margin(MARGIN));
328
329 // Nr-Spalte: durchgängig doppelte rechte Linie
330 getSheetHelper().setPropertiesInRange(sheet,
331 RangePosition.from(TEAM_NR_SPALTE, ERSTE_DATEN_ZEILE - 2, TEAM_NR_SPALTE, letzteZeile),
332 CellProperties.from().setBorder(BorderFactory.from().doubleLn().forRight().toBorder()));
333
334 // Platz-Spalte: fett + durchgängig dicke rechte Linie
335 getSheetHelper().setPropertiesInRange(sheet,
336 RangePosition.from(RANGLISTE_SPALTE, ERSTE_DATEN_ZEILE - 2, RANGLISTE_SPALTE, letzteZeile),
337 CellProperties.from()
338 .setCharWeight(com.sun.star.awt.FontWeight.BOLD)
339 .setBorder(BorderFactory.from().boldLn().forRight().toBorder()));
340
341 // Begegnungen-Spalte: durchgängig dicke linke Linie
342 getSheetHelper().setPropertiesInRange(sheet,
343 RangePosition.from(BEGEGNUNGEN_SPALTE, ERSTE_DATEN_ZEILE - 2, BEGEGNUNGEN_SPALTE, letzteZeile),
344 CellProperties.from().setBorder(BorderFactory.from().boldLn().forLeft().toBorder()));
345
346 // Doppelte Trennlinien: Spiele-Block (col 5) und SpPunkte-Block (col 8)
347 for (int spalte : new int[] { SPIELE_PLUS_SPALTE, SP_PUNKTE_PLUS_SPALTE }) {
348 getSheetHelper().setPropertiesInRange(sheet,
349 RangePosition.from(spalte, ERSTE_DATEN_ZEILE - 2, spalte, letzteZeile),
350 CellProperties.from().setBorder(BorderFactory.from().doubleLn().forLeft().toBorder()));
351 }
352 }

399 private void insertHeader(XSpreadsheet sheet) throws GenerateException {
400 int headerFarbe = konfigurationSheet.getRanglisteHeaderFarbe();
401 TableBorder2 borderBottom = BorderFactory.from().allThin().boldLn().forBottom().toBorder();
402
403 // Zeile 0: Nr, Name – je 2 Zeilen hoch
404 for (int col : new int[] { TEAM_NR_SPALTE, TEAM_NAME_SPALTE }) {
405 getSheetHelper().setStringValueInCell(StringCellValue
406 .from(sheet, Position.from(col, 0),
407 col == TEAM_NR_SPALTE ? I18n.get("column.header.nr") : I18n.get("column.header.name"))
408 .setCellProperties(CellProperties.from()
409 .setBorder(BorderFactory.from().allThin().boldLn().forBottom().toBorder())
410 .setCellBackColor(headerFarbe).margin(MARGIN).centerJustify().setShrinkToFit(true))
411 .setEndPosMergeZeilePlus(1));
412 }
413
414 // Zeile 0: Rang – 2 Zeilen hoch, 90°, fett
415 getSheetHelper().setStringValueInCell(StringCellValue
416 .from(sheet, Position.from(RANGLISTE_SPALTE, 0), I18n.get("column.header.platz"))
417 .setCellProperties(CellProperties.from().setAllThinBorder()
418 .setCellBackColor(headerFarbe).margin(MARGIN).centerJustify().setShrinkToFit(true)
419 .setCharWeight(com.sun.star.awt.FontWeight.BOLD)
420 .setBorder(BorderFactory.from().allThin().boldLn().forBottom().forRight().toBorder()))
421 .setEndPosMergeZeilePlus(1).setRotate90());
422
423 // Zeile 0: Punkte-Gruppe (2 Spalten breit)
424 getSheetHelper().setStringValueInCell(StringCellValue
425 .from(sheet, Position.from(PUNKTE_PLUS_SPALTE, 0), I18n.get("column.header.punkte"))
426 .setCellProperties(CellProperties.from().setAllThinBorder()
427 .setCellBackColor(headerFarbe).margin(MARGIN).centerJustify().setShrinkToFit(true))
428 .setEndPosMergeSpaltePlus(1));
429
430 // Zeile 0: Spiele-Gruppe (3 Spalten breit)
431 getSheetHelper().setStringValueInCell(StringCellValue
432 .from(sheet, Position.from(SPIELE_PLUS_SPALTE, 0), I18n.get("column.header.spiele"))
433 .setCellProperties(CellProperties.from().setAllThinBorder()
434 .setCellBackColor(headerFarbe).margin(MARGIN).centerJustify().setShrinkToFit(true))
435 .setEndPosMergeSpaltePlus(2));
436
437 // Zeile 0: Spielpunkte-Gruppe (3 Spalten breit)
438 getSheetHelper().setStringValueInCell(StringCellValue
439 .from(sheet, Position.from(SP_PUNKTE_PLUS_SPALTE, 0), I18n.get("column.header.spielpunkte"))
440 .setCellProperties(CellProperties.from().setAllThinBorder()
441 .setCellBackColor(headerFarbe).margin(MARGIN).centerJustify().setShrinkToFit(true))
442 .setEndPosMergeSpaltePlus(2));
443
444 // Zeile 0: Begegnungen – 2 Zeilen hoch, 90°
445 getSheetHelper().setStringValueInCell(StringCellValue
446 .from(sheet, Position.from(BEGEGNUNGEN_SPALTE, 0), I18n.get("column.header.begegn"))
447 .setCellProperties(CellProperties.from().setAllThinBorder()
448 .setCellBackColor(headerFarbe).margin(MARGIN).centerJustify().setShrinkToFit(true))
449 .setEndPosMergeZeilePlus(1).setRotate90());
450
451 // Zeile 1: Sub-Header +/-/Δ
452 RangeData subHeader = new RangeData();
453 RowData subRow = subHeader.addNewRow();
454 subRow.newString("+");
455 subRow.newString("-");
456 subRow.newString("+");
457 subRow.newString("-");
458 subRow.newString("Δ");
459 subRow.newString("+");
460 subRow.newString("-");
461 subRow.newString("Δ");
462 RangeHelper.from(this, subHeader.getRangePosition(Position.from(PUNKTE_PLUS_SPALTE, 1)))
463 .setDataInRange(subHeader)
464 .setRangeProperties(RangeProperties.from().centerJustify()
465 .setBorder(borderBottom).setCellBackColor(headerFarbe).margin(MARGIN).setShrinkToFit(true));
466 }
