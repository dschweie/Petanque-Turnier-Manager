165 }

168 // 5. Peer erzeugen (instantiiert alle Child-Controls aus dem Modell)
169 Object toolkit = xMCF.createInstanceWithContext("com.sun.star.awt.Toolkit", context);
170 XToolkit xToolkit = Lo.qi(XToolkit.class, toolkit);
171 XWindow xWindow = Lo.qi(XWindow.class, xControl);
172 xWindow.setVisible(false);
173 xControl.createPeer(xToolkit, null);
174
175 // 6. Dialog anzeigen (blockiert bis endExecute oder Fenster-Schliessen)
176 xDialog.execute();
177
178 Optional<TurnierParameter> result = Optional.empty();
179 if (okPressed) {
180 Formation formation = readFormation(xcc);
181 boolean teamnameAnzeigen = readCheckBoxState(xcc, "cbTeamname");
182 boolean vereinsnameAnzeigen = readCheckBoxState(xcc, "cbVereinsname");
183 SpielplanTeamAnzeige spielplanAnzeige = readListBoxSelected(xcc, "lstSpielplan") == 1
184 ? SpielplanTeamAnzeige.NAME : SpielplanTeamAnzeige.NR;
185 SchweizerRankingModus rankingModus = readListBoxSelected(xcc, "lstRankingModus") == 1
186 ? SchweizerRankingModus.OHNE_BUCHHOLZ : SchweizerRankingModus.MIT_BUCHHOLZ;
187 result = Optional.of(
188 new TurnierParameter(formation, teamnameAnzeigen, vereinsnameAnzeigen, spielplanAnzeige, rankingModus));
189 }
190
191 Lo.qi(XComponent.class, dialog).dispose();
192 ProcessBox.from().visibleWennAutomatisch();
193
194 return result;
195 }

201 private Formation readFormation(XControlContainer xcc) {
202 return switch (readListBoxSelected(xcc, "lstFormation")) {
203 case 1 -> Formation.DOUBLETTE;
204 case 2 -> Formation.TRIPLETTE;
205 default -> Formation.TETE;
206 };
207 }
208
209 private static short formationIndex(Formation formation) {
210 return switch (formation) {
211 case DOUBLETTE -> 1;
212 case TRIPLETTE -> 2;
213 default -> 0;
214 };
215 }

217 private short readListBoxSelected(XControlContainer xcc, String name) {
218 XControl ctrl = xcc.getControl(name);
219 if (ctrl == null) {
220 return 0;
221 }
222 XListBox lb = Lo.qi(XListBox.class, ctrl);
223 return lb != null ? lb.getSelectedItemPos() : 0;
224 }

226 private boolean readCheckBoxState(XControlContainer xcc, String name) {
227 XControl ctrl = xcc.getControl(name);
228 if (ctrl == null) {
229 return false;
230 }
231 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
232 return cb != null && cb.getState() == 1;
233 }
