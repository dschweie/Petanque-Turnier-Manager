191 .setHeaderLeft(getKopfZeileLinks())

193 .setHeaderRight(getKopfZeileRechts()).create();
194
195 for (int spieltagCntr = 1; spieltagCntr <= MAX_SPIELTAG; spieltagCntr++) {
196 String propNameKey = SuperMeleePropertiesSpalte.PROP_SPIELTAG_KOPFZEILE(spieltagCntr);
197 String spielTagKopfzeile = propertiesSpalte.readStringProperty(propNameKey);
198 if (spielTagKopfzeile != null) {
199 PageStyleHelper.from(this, SpielTagNr.from(spieltagCntr)).initDefaultFooter()
200 .setFooterCenter(getFusszeileMitte()).setFooterLeft(getFusszeileLinks())
201 .setHeaderCenter(spielTagKopfzeile).create();
202 }
