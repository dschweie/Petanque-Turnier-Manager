53 public static Optional<InstallierteVersion> ermitteln(XComponentContext context) {
54 Objects.requireNonNull(context, "context");
55 try {
56 XPackageInformationProvider provider = PackageInformationProvider.get(context);
57 if (provider == null) {
58 logger.debug("XPackageInformationProvider nicht verfügbar");
59 return Optional.empty();
60 }
61 return ausExtensionListe(provider.getExtensionList());
62 } catch (RuntimeException e) {
63 logger.debug("Fehler beim Ermitteln der installierten Version: {}", e.getMessage());
64 return Optional.empty();
65 }

72 static Optional<InstallierteVersion> ausExtensionListe(String[][] extensions) {
73 if (extensions == null) {
74 logger.debug("Extension-Liste ist null");
75 return Optional.empty();
76 }
77 for (String[] eintrag : extensions) {
78 if (eintrag == null || eintrag.length < 2) {
79 continue;
80 }
81 if (EXTENSION_ID.equals(eintrag[0])) {
82 var versionsString = eintrag[1];
83 if (versionsString == null || versionsString.isBlank()) {
84 logger.debug("Extension-Eintrag {} hat leeren Versions-Slot", EXTENSION_ID);
85 return Optional.empty();
86 }
87 return Optional.of(new InstallierteVersion(versionsString));
88 }
89 }
90 logger.debug("Extension-ID {} nicht in der Liste gefunden", EXTENSION_ID);
91 return Optional.empty();
92 }
