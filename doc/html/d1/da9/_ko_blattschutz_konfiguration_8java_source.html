76 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, KoKonfigurationSheet konfigSheet,
77 List<SheetSchutzInfo> infos) {
78 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_KO_MELDELISTE).ifPresent(sheet -> {
79 try {
80 int aktivSpalte = berechneAktivSpalte(konfigSheet);
81 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
82 RangePosition.from(1, MELDELISTE_ERSTE_DATEN_ZEILE,
83 aktivSpalte, MeldungenSpalte.MAX_ANZ_MELDUNGEN))));
84 } catch (GenerateException e) {
85 logger.warn("Editierbare Meldeliste-Spalten konnten nicht berechnet werden: {}", e.getMessage(), e);
86 }
87 });
88 }
89
90 private void sammleTurnierbaumSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
91 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
92 SheetMetadataHelper.SCHLUESSEL_KO_TURNIERBAUM_PREFIX);
93 for (var turnierbaumKey : schluessel) {
94 SheetMetadataHelper.findeSheet(xDoc, turnierbaumKey).ifPresent(sheet -> {
95 var scoreKey = SheetMetadataHelper.scoreSchluessel(turnierbaumKey);
96 var encoded = SheetMetadataHelper.leseScoreText(xDoc, scoreKey);
97 var editierbareBereiche = (encoded != null)
98 ? KoTurnierbaumSheet.decodeScoreBereiche(encoded)
99 : List.<RangePosition>of();
100 if (editierbareBereiche.isEmpty()) {
101 infos.add(SheetSchutzInfo.vollGesperrt(sheet));
102 } else {
103 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, editierbareBereiche));
104 }
105 });
106 }
107 }

113 private int berechneAktivSpalte(KoKonfigurationSheet konfigSheet) throws GenerateException {
114 int spaltenProSpieler = konfigSheet.isMeldeListeVereinsnameAnzeigen() ? 3 : 2;
115 int ersterOffset = konfigSheet.isMeldeListeTeamnameAnzeigen() ? 2 : 1;
116 int letzteDataSpalte = ersterOffset + konfigSheet.getMeldeListeFormation().getAnzSpieler() * spaltenProSpieler - 1;
117 return letzteDataSpalte + 2; // +1=RNG, +2=Aktiv
118 }
