212 public StringCellValue addFooter() throws GenerateException {
213
214 int ersteFooterZeile = spielerSpalte.naechsteFreieDatenZeileInSpielerNrSpalte();
215 int anzMerge = getLetzteSpalte(); // SPIELER_NR_SPALTE = 0, daher direkt letzteSpalte
216 StringCellValue stringVal = StringCellValue
217 .from(getSheet(), Position.from(spielerSpalte.getSpielerNrSpalte(), ersteFooterZeile))
218 .setHoriJustify(CellHoriJustify.LEFT).setCharHeight(8);
219
220 String anzSpielerFormula = "\"Anzahl Spieler: \" & " + spielerSpalte.formulaCountSpieler();
221 stringVal.setEndPosMergeSpaltePlus(anzMerge)
222 .addCellProperty(ICommonProperties.IS_TEXT_WRAPPED, Boolean.TRUE)
223 .addRowProperty("OptimalHeight", Boolean.TRUE);
224 getSheetHelper().setFormulaInCell(stringVal.setValue(anzSpielerFormula));
225
226 SuprMleEndranglisteSortMode suprMleEndranglisteSortMode = getSuprMleEndranglisteSortMode();
227
228 String sortInfoString = "Reihenfolge zur Ermittlung der Platzierung: 1. Spiele +, 2. Spiele Δ, 3. Punkte Δ, 4. Punkte +";
229 if (SuprMleEndranglisteSortMode.ANZTAGE == suprMleEndranglisteSortMode) {
230 sortInfoString = "Reihenfolge zur Ermittlung der Platzierung: 1. Spiele +,2. Anz Spieltage, 3. Spiele Δ, 4. Punkte Δ, 5. Punkte +";
231 }
232 getSheetHelper().setStringValueInCell(stringVal.zeilePlusEins()
233 .setEndPosMergeSpaltePlus(anzMerge)
234 .addCellProperty(ICommonProperties.IS_TEXT_WRAPPED, Boolean.TRUE)
235 .addRowProperty("OptimalHeight", Boolean.TRUE)
236 .setValue(sortInfoString));
237
238 return stringVal;
239 }
