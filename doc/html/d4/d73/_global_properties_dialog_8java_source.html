128 @Override
129 protected void beiOkGeklickt() throws Exception {
130 GlobalProperties gp = GlobalProperties.get();
131 String gewaehlterLevel = (cmbLogLevel != null) ? cmbLogLevel.getText() : "";
132 gp.speichern(
133 cbAutosave != null && cbAutosave.getState() == 1,
134 cbBackup != null && cbBackup.getState() == 1,
135 cbNewVersion != null && cbNewVersion.getState() == 1,
136 cbProzessBoxAutomatisch != null && cbProzessBoxAutomatisch.getState() == 1,
137 cbProzessBoxSchliessen != null && cbProzessBoxSchliessen.getState() == 1,
138 cbPerformanceLogging != null && cbPerformanceLogging.getState() == 1,
139 gewaehlterLevel
140 );
141 try {
142 ReleaseUpdateService.get().loeseListenerAus();
143 } catch (IllegalStateException e) {
144 // Service nie initialisiert – ok.
145 }
146 // ProcessBox bewusst NICHT anzeigen/togglen — der Konfig-Dialog soll
147 // weder beim Öffnen noch beim Schließen die ProcessBox einblenden.
148 // Die neue isProzessBoxAutomatischAnzeigen-Einstellung greift erst beim
149 // nächsten SheetRunner-Lauf.
150 logger.info("Plugin-Konfiguration gespeichert");
151 }
