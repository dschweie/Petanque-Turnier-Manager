376 private static void sammelePraefixSchluessel(XSpreadsheetDocument xDoc, String prefix,
377 String stabileIdPrefix, List<SignaturQuelle> ziel, QuellenFabrik fabrik) {
378 String[] schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc, prefix);
379 List<String> sortiert = new ArrayList<>(Arrays.asList(schluessel));
380 sortiert.sort(Comparator.comparing(SignaturQuellen::numerischerSortierschluessel)
381 .thenComparing(Comparator.naturalOrder()));
382 for (String s : sortiert) {
383 String suffix = s.startsWith(prefix) ? s.substring(prefix.length()) : s;
384 String stabileId = stabileIdPrefix + entferneSchluesselSuffix(suffix);
385 ziel.add(fabrik.erzeugen(stabileId, s));
386 }
387 }
388
393 private static long numerischerSortierschluessel(String name) {
394 long nummer = 0;
395 boolean gefunden = false;
396 for (int i = 0; i < name.length(); i++) {
397 char c = name.charAt(i);
398 if (c >= '0' && c <= '9') {
399 nummer = nummer * 10 + (c - '0');
400 gefunden = true;
401 } else if (gefunden) {
402 return nummer;
403 }
404 }
405 return gefunden ? nummer : Long.MAX_VALUE;
406 }
407
409 private static String entferneSchluesselSuffix(String suffix) {
410 if (suffix.endsWith(SheetMetadataHelper.SCHLUESSEL_SUFFIX)) {
411 return suffix.substring(0,
412 suffix.length() - SheetMetadataHelper.SCHLUESSEL_SUFFIX.length());
413 }
414 return suffix;
415 }
