48 AddConfigElementsToWindow addConfigElementsToWindow) throws com.sun.star.uno.Exception {
49 List<ConfigProperty<?>> konfigProperties = KonfigurationSingleton.getKonfigProperties(currentSpreadsheet);
50 if (konfigProperties == null) {
51 return;
52 }
53 var tabFarbProps = konfigProperties.stream()
54 .filter(p -> p.getType() == ConfigPropertyType.COLOR && p.isTabFarbe())
55 .collect(Collectors.toList());
56 if (tabFarbProps.isEmpty()) {
57 return;
58 }
59 var trennlinie = GuiFactory.createHLine(param, new Rectangle(0, 0, 190, 4), null);
60 layout.addLayout(new ControlLayout(trennlinie), 1);
61 tabFarbProps.forEach(addConfigElementsToWindow::addPropToPanel);
62 }
