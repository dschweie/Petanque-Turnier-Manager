80 @Override
81 protected void doRun() throws GenerateException {
82 getxCalculatable().enableAutomaticCalculation(false);
83 processBoxinfo("processbox.kaskade.ko.felder.erstellen");
84
85 meldeListe.upDateSheet();
86 int gesamtTeams = meldeListe.getMeldungenSortiertNachSetzposition().size();
87
88 boolean freispielGewonnen = konfigurationSheet.getFreispielPunktePlus()
89 > konfigurationSheet.getFreispielPunkteMinus();
90 var plan = KaskadenKoRundenPlaner.berechne(
91 gesamtTeams, konfigurationSheet.getAnzahlKaskaden(), freispielGewonnen);
92
93 var belegungen = new KaskadeRundenErgebnisLeser(getWorkingSpreadsheet()).ladeFeldBelegungen(plan);
94
95 // Gruppenrangliste immer zuerst erstellen (Voraussetzung für KO-Felder)
96 var gruppenRangliste = new KaskadeGruppenRanglisteSheet(getWorkingSpreadsheet());
97 gruppenRangliste.setForceOk(isForceOk());
98 gruppenRangliste.doRun();
99
100 for (var belegung : belegungen) {
101 SheetRunner.testDoCancelTask();
102 feldSheetErstellen(belegung);
103 }
104 }

110 private void feldSheetErstellen(KaskadenFeldBelegung belegung) throws GenerateException {
111 var feld = belegung.feld();
112 if (feld.gesamtTeams() < 2) {
113 LOGGER.info("Feld {} hat weniger als 2 Teams ({}), wird übersprungen.",
114 feld.bezeichner(), feld.gesamtTeams());
115 return;
116 }
117
118 processBoxinfo("processbox.kaskade.ko.plan.erstellen", feld.bezeichner());
119 letzterBezeichner = feld.bezeichner();
120
121 var gruppeTeams = feldBelegungZuTeamMeldungen(belegung.teamNrs());
122 var sheetName = SheetNamen.kaskadenFeld(feld.bezeichner());
123 var schluessel = SheetMetadataHelper.schluesselKaskadenFeld(feld.bezeichner());
124 var konfig = new KaskadeKoBracketKonfigAdapter(konfigurationSheet);
125
126 new KoTurnierbaumSheet(getWorkingSpreadsheet())
127 .erstelleGruppeBracket(gruppeTeams, sheetName, DefaultSheetPos.KASKADE_FELDER, konfig, schluessel);
128 }
129
130 private static TeamMeldungen feldBelegungZuTeamMeldungen(List<Integer> teamNrs) {
131 var meldungen = new TeamMeldungen();
132 teamNrs.forEach(nr -> meldungen.addTeamWennNichtVorhanden(Team.from(nr)));
133 return meldungen;
134 }
