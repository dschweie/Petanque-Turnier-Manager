77 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, WorkingSpreadsheet ws,
78 List<SheetSchutzInfo> infos) {
79 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_FORMULEX_MELDELISTE).ifPresent(sheet -> {
80 try {
81 var konfigSheet = new FormuleXKonfigurationSheet(ws);
82 int aktivSpalte = berechneAktivSpalte(konfigSheet);
83 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
84 RangePosition.from(1, MELDELISTE_ERSTE_DATEN_ZEILE,
85 aktivSpalte, MeldungenSpalte.MAX_ANZ_MELDUNGEN))));
86 } catch (GenerateException e) {
87 LOGGER.warn("Editierbare Meldeliste-Spalten konnten nicht berechnet werden: {}", e.getMessage(), e);
88 }
89 });
90 }
91
92 private void sammleSpielrundenSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
93 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
94 SheetMetadataHelper.SCHLUESSEL_FORMULEX_SPIELRUNDE_PREFIX);
95 for (var key : schluessel) {
96 SheetMetadataHelper.findeSheet(xDoc, key).ifPresent(sheet -> {
97 try {
98 int letzteZeile = ermittleLetzteSpielrundeZeile(sheet);
99 List<RangePosition> editierbar = FormuleXAbstractSpielrundeSheet
100 .ermittleEditierbareErgebnisRanges(sheet, xDoc, letzteZeile);
101 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, editierbar));
102 } catch (GenerateException e) {
103 LOGGER.warn("Editierbare Spielrunde-Bereiche konnten nicht berechnet werden: {}",
104 e.getMessage(), e);
105 }
106 });
107 }
108 }
109
110 private void sammleVollGesperrteSheets(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
111 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_FORMULEX_RANGLISTE)
112 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
113 }

115 private int ermittleLetzteSpielrundeZeile(com.sun.star.sheet.XSpreadsheet sheet) {
116 int letzteZeile = FormuleXAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE;
117 try {
118 for (int zeile = FormuleXAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE;
119 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
120 XCell xCell = sheet.getCellByPosition(FormuleXAbstractSpielrundeSheet.TEAM_A_SPALTE, zeile);
121 if (CellContentType.EMPTY.equals(xCell.getType())) {
122 break;
123 }
124 letzteZeile = zeile;
125 }
126 } catch (Exception e) {
127 LOGGER.warn("Letzte Spielrunde-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
128 }
129 return letzteZeile;
130 }

132 private int berechneAktivSpalte(FormuleXKonfigurationSheet konfigSheet) throws GenerateException {
133 int spaltenProSpieler = konfigSheet.isMeldeListeVereinsnameAnzeigen() ? 3 : 2;
134 int ersterOffset = konfigSheet.isMeldeListeTeamnameAnzeigen() ? 2 : 1;
135 int letzteDataSpalte = ersterOffset
136 + konfigSheet.getMeldeListeFormation().getAnzSpieler() * spaltenProSpieler - 1;
137 return letzteDataSpalte + 2; // +1=SP, +2=Aktiv
138 }
