171 int anzSpieltage = countAnzSpieltageInMeldeliste();
172 getKonfigurationSheet().setAktiveSpieltag(SpielTagNr.from(anzSpieltage + 1));
173 setSpielTag(getKonfigurationSheet().getAktiveSpieltag());
174 getKonfigurationSheet().setAktiveSpielRunde(SpielRundeNr.from(1));
175
176 // Blattschutz entfernen: clearContents wird von LO auf gesperrten Zellen lautlos ignoriert.
177 // upDateSheet() → formatDaten() setzt den Schutz danach korrekt neu.
178
179 RangePosition cleanUpRange = RangePosition.from(aktuelleSpieltagSpalte(), ERSTE_HEADER_ZEILE,
180 aktuelleSpieltagSpalte(), MeldungenSpalte.MAX_ANZ_MELDUNGEN);
181 RangeHelper.from(this, cleanUpRange).clearRange();
182 upDateSheet();
183 getxCalculatable().calculateAll();
184 }
