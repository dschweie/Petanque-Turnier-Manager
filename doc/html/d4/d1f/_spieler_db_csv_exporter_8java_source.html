51 public void export(SpielerDbExportData data, ExportRequest request) throws SpielerDbException {
52 if (!request.entities().contains(ExportEntity.SPIELER)) {
53 throw new SpielerDbException(
54 "Flache CSV exportiert Spieler+Verein — Spieler-Entity muss im Scope sein");
55 }
56 Path zielDatei = request.target();
57 Path elternVerzeichnis = zielDatei.toAbsolutePath().getParent();
58 if (elternVerzeichnis != null) {
59 try {
60 Files.createDirectories(elternVerzeichnis);
61 } catch (IOException e) {
62 throw new SpielerDbException("CSV-Zielverzeichnis nicht anlegbar: "
63 + elternVerzeichnis, e);
64 }
65 }
66
67 List<SpielerMitVerein> sortiert = sortiere(data.spieler());
68
69 try (BufferedWriter writer = oeffne(zielDatei);
70 ICSVWriter csv = new CSVWriterBuilder(writer)
71 .withSeparator(SpielerDbCsvFormat.SEPARATOR)
72 .withLineEnd(System.lineSeparator())
73 .build()) {
74 writer.write(SpielerDbCsvFormat.formatMarkerZeile() + System.lineSeparator());
75 csv.writeNext(SpielerDbCsvFormat.HEADER, false);
76 for (SpielerMitVerein s : sortiert) {
77 csv.writeNext(zuZeile(s), false);
78 }
79 } catch (IOException e) {
80 throw new SpielerDbException("CSV-Export fehlgeschlagen: " + zielDatei, e);
81 }
82 }

84 private static List<SpielerMitVerein> sortiere(List<SpielerMitVerein> spieler) {
85 // Collator für stabile, locale-konsistente Reihenfolge (Umlaute richtig).
86 Collator collator = Collator.getInstance(Locale.ROOT);
87 collator.setStrength(Collator.SECONDARY); // case-insensitiv
88 Comparator<String> stringComparator = Comparator.nullsFirst(collator::compare);
89 Comparator<SpielerMitVerein> cmp = Comparator
90 .<SpielerMitVerein, String>comparing(SpielerMitVerein::nachname, stringComparator)
91 .thenComparing(SpielerMitVerein::vorname, stringComparator)
92 .thenComparing(SpielerMitVerein::vereinName, stringComparator);
93 List<SpielerMitVerein> kopie = new ArrayList<>(spieler);
94 kopie.sort(cmp);
95 return kopie;
96 }

98 private static String[] zuZeile(SpielerMitVerein s) {
99 return new String[] {
100 s.vorname(),
101 s.nachname(),
102 s.vereinName() == null ? "" : s.vereinName(),
103 s.lizenznr() == null ? "" : s.lizenznr()
104 };
105 }

111 return new BufferedWriter(w);
