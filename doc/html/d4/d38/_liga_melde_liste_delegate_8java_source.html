95 private void renameSpielPlanSheet() throws GenerateException {
96 String oldName = "Liga Spielplan";
97 sheet.processBoxinfo("processbox.pruefe.ob.vorhanden", oldName);
98 if (sheet.getSheetHelper().findByName(oldName) != null) {
99 MessageBoxResult answer = MessageBox.from(sheet.getxContext(), MessageBoxTypeEnum.WARN_YES_NO)
100 .caption(I18n.get("msg.caption.tabelle", oldName))
101 .message(I18n.get("msg.text.tabelle.umbenennen.frage", oldName, LigaSpielPlanSheet.sheetName()))
102 .show();
103
104 if (answer == MessageBoxResult.YES) {
105 XSpreadsheet spielplan = sheet.getSheetHelper().findByName(oldName);
106 if (sheet.getSheetHelper().reNameSheet(spielplan, LigaSpielPlanSheet.sheetName())) {
107 MessageBox.from(sheet.getxContext(), MessageBoxTypeEnum.INFO_OK)
108 .caption(I18n.get("msg.caption.tabelle", oldName))
109 .message(I18n.get("msg.text.tabelle.umbenannt.ok", oldName, LigaSpielPlanSheet.sheetName()))
110 .show();
111 } else {
112 MessageBox.from(sheet.getxContext(), MessageBoxTypeEnum.ERROR_OK)
113 .caption(I18n.get("msg.caption.tabelle", oldName))
114 .message(I18n.get("msg.text.tabelle.umbenannt.fehler", oldName, LigaSpielPlanSheet.sheetName()))
115 .show();
116 throw new GenerateException(I18n.get("error.sheet.umbenennen", oldName, LigaSpielPlanSheet.sheetName()));
117 }
118 } else {
119 throw new GenerateException(I18n.get("error.sheet.umbenennen", oldName, LigaSpielPlanSheet.sheetName()));
120 }
121 }

203 private void infoSpaltenHeaderSchreiben(int headerBackColor) throws GenerateException {
204 record InfoSpalte(int spalte, String i18nKey, int breite) {}
205 var spalten = List.of(
206 new InfoSpalte(MANNSCHAFTSFUEHRER_SPALTE, "liga.meldeliste.column.header.mannschaftsfuehrer", MANNSCHAFTSFUEHRER_WIDTH),
207 new InfoSpalte(EMAIL_SPALTE, "liga.meldeliste.column.header.email", EMAIL_WIDTH),
208 new InfoSpalte(TELEFON_SPALTE, "liga.meldeliste.column.header.telefon", TELEFON_WIDTH),
209 new InfoSpalte(STARTSPIELZEIT_SPALTE, "liga.meldeliste.column.header.startspielzeit", STARTSPIELZEIT_WIDTH),
210 new InfoSpalte(ADRESSE_SPALTE, "liga.meldeliste.column.header.adresse", ADRESSE_WIDTH));
211
212 for (var info : spalten) {
213 var colProps = ColumnProperties.from().setWidth(info.breite())
214 .setHoriJustify(CellHoriJustify.CENTER).setVertJustify(CellVertJustify2.CENTER)
215 .margin(MeldeListeKonstanten.CELL_MARGIN);
216 var celVal = StringCellValue
217 .from(sheet.getXSpreadSheet(), Position.from(info.spalte(), ZWEITE_HEADER_ZEILE),
218 I18n.get(info.i18nKey()))
219 .addColumnProperties(colProps)
220 .setBorder(BorderFactory.from().allThin().boldLn().forTop().toBorder())
221 .setCellBackColor(headerBackColor)
222 .setVertJustify(CellVertJustify2.CENTER)
223 .setShrinkToFit(true);
224 sheet.getSheetHelper().setStringValueInCell(celVal);
225 }

276 Map<Integer, String> leseTeamNamenMap() throws GenerateException {
277 Map<Integer, String> result = new HashMap<>();
278 XSpreadsheet mlSheet = sheet.getXSpreadSheet();
279 if (mlSheet == null) {
280 return result;
281 }
282 var xDoc = sheet.getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
283 RangeData data = RangeHelper.from(mlSheet, xDoc,
284 RangePosition.from(SPIELER_NR_SPALTE, ERSTE_DATEN_ZEILE, SPIELER_NR_SPALTE + 1, ERSTE_DATEN_ZEILE + 999))
285 .getDataFromRange();
286 for (RowData row : data) {
287 if (row.isEmpty()) {
288 break;
289 }
290 int nr = row.get(0).getIntVal(0);
291 if (nr <= 0) {
292 continue;
293 }
294 String name = row.size() > 1 ? row.get(1).getStringVal().trim() : "";
295 result.put(nr, name);
296 }
297 return result;
298 }
