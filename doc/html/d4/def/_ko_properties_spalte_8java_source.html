336 public static int normalisiereMinLetzteGruppeGroesse(int wert) {
337 if (wert <= 0) {
338 return DEFAULT_MIN_LETZTE_GRUPPE_GROESSE;
339 }
340 for (Integer erlaubt : ERLAUBTE_MIN_LETZTE_GRUPPEN_GROESSEN) {
341 if (wert <= erlaubt) {
342 return erlaubt;
343 }
344 }
345 return ERLAUBTE_MIN_LETZTE_GRUPPEN_GROESSEN.getLast();
346 }

348 public static int normalisiereMinLetzteGruppeGroesse(String wert) {
349 if (wert == null || wert.isBlank()) {
350 return DEFAULT_MIN_LETZTE_GRUPPE_GROESSE;
351 }
352 try {
353 return normalisiereMinLetzteGruppeGroesse((int) Math.round(Double.parseDouble(wert.trim())));
354 } catch (NumberFormatException e) {
355 return DEFAULT_MIN_LETZTE_GRUPPE_GROESSE;
356 }
