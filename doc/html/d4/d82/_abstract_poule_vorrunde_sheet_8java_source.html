244 var teams = poule.teams();

279 vorrundeStartZeile + 3);

283 verliererFormel(rSieger), siegerFormel(rVerlierer),
284 vorrundeStartZeile + 4);

410 var ergA = Position.from(SPALTE_ERG_A, refZeile).getAddress();

415 + "IF(" + ergA + ">" + ergB + ";" + teamA + ";" + teamB + ");\"\")";

450 private void schreibeFehlerFormel(XSpreadsheet xSheet, int zeile, boolean mitTeamGuard)
451 throws GenerateException {
452 var ergA = Position.from(SPALTE_ERG_A, zeile).getAddress();
453 var ergB = Position.from(SPALTE_ERG_B, zeile).getAddress();
454 var leer = "AND(ISBLANK(" + ergA + ");ISBLANK(" + ergB + "))";
455 var gueltig = "AND(" + ergA + "<14;" + ergB + "<14;" + ergA + ">-1;" + ergB + ">-1;" + ergA + "<>" + ergB + ")";
456 var fehlerText = "\"" + I18n.get("poule.vorrunde.fehler.formel") + "\"";
457
458 String formel;
459 if (mitTeamGuard) {
460 var teamA = Position.from(SPALTE_TEAM_A_NR, zeile).getAddress();
461 var teamB = Position.from(SPALTE_TEAM_B_NR, zeile).getAddress();
462 formel = "IF(OR(NOT(ISNUMBER(" + teamA + "));NOT(ISNUMBER(" + teamB + ")));\"\";IF("
463 + leer + ";\"\";IF(" + gueltig + ";\"\";" + fehlerText + ")))";
464 } else {
465 formel = "IF(" + leer + ";\"\";IF(" + gueltig + ";\"\";" + fehlerText + "))";
466 }
467
468 getSheetHelper().setFormulaInCell(
469 StringCellValue.from(xSheet, Position.from(SPALTE_FEHLER, zeile), formel)
470 .setCharColor(ColorHelper.CHAR_COLOR_RED)
471 .setCharWeight(FontWeight.BOLD)
472 .setHoriJustify(CellHoriJustify.CENTER));
473 }
