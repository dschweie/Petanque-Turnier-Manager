88 Integer cellNumTeamA = getSheetHelper().getIntFromCell(vorRunden, Position.from(spalteCnt, zeileCntr));
89 if (cellNumTeamA > 0) {

91 var teamAInListe = teamList.stream().filter(teamA::equals).findFirst().orElse(null);
92 if (teamAInListe != null) {
93 Integer cellNumTeamB = getSheetHelper().getIntFromCell(vorRunden, Position.from(spalteCnt + 1, zeileCntr));
94 if (cellNumTeamB > 0) {
95 var teamB = Team.from(cellNumTeamB);
96 teamList.stream().filter(teamB::equals).findFirst()
