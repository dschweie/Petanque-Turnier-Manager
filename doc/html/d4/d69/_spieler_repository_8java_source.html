201 public SpielerDatensatz insert(SpielerDatensatz neu) throws SpielerDbException {
202 String vorname = neu.vorname().strip();
203 String nachname = neu.nachname().strip();
204 if (vorname.isEmpty() || nachname.isEmpty()) {
205 throw new SpielerDbException("Vorname und Nachname sind Pflichtfelder");
206 }
207 String lizenz = normLizenz(neu.lizenznr());
208 String sql = "INSERT INTO SPIELER (VORNAME, NACHNAME, VEREIN_NR, LIZENZNR) "
209 + "VALUES (?, ?, ?, ?)";
210 boolean vorigerAutoCommit = true;
211 try {
212 vorigerAutoCommit = connection.getAutoCommit();
213 connection.setAutoCommit(false);
214 int neueNr;
215 try (PreparedStatement ps = connection.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
216 ps.setString(1, vorname);
217 ps.setString(2, nachname);
218 setNullableInt(ps, 3, neu.vereinNr());
219 setNullableString(ps, 4, lizenz);
220 ps.executeUpdate();
221 try (ResultSet keys = ps.getGeneratedKeys()) {
222 if (!keys.next()) {
223 throw new SpielerDbException("Insert lieferte keinen Schlüssel");
224 }
225 neueNr = keys.getInt(1);
226 }
227 }
228 schreibeLabelZuweisungen(neueNr, neu.labelNrs());
229 connection.commit();
230 return new SpielerDatensatz(neueNr, vorname, nachname,
231 neu.vereinNr(), neu.labelNrs(), lizenz);
232 } catch (SQLException e) {
233 rollbackLeise();
234 if (istNameDuplikat(e)) {
235 throw new NameDuplikatException("Spieler mit gleichem Vor-/Nachnamen "
236 + "und Verein existiert bereits: " + vorname + " " + nachname, e);
237 }
238 if (istUniqueVerletzung(e)) {
239 throw new LizenzDuplikatException("Lizenznummer existiert bereits: " + lizenz, e);
240 }
241 if (istFkVerletzung(e)) {
242 throw new SpielerDbException(
243 "Verein-/Label-Referenz ungültig: vereinNr=" + neu.vereinNr()
244 + ", labelNrs=" + neu.labelNrs(), e);
245 }
246 throw new SpielerDbException("Spieler-Insert fehlgeschlagen", e);
247 } finally {
248 wiederherstellenAutoCommit(vorigerAutoCommit);
249 }

440 private static boolean istUniqueVerletzung(SQLException e) {
441 String raw = e.getMessage();
442 if (raw == null) {
443 return false;
444 }
445 String msg = raw.toLowerCase(Locale.ROOT);
446 // SQLite-Format: "[SQLITE_CONSTRAINT_UNIQUE] ... UNIQUE constraint failed: ..."
447 if (msg.contains("unique constraint failed") || msg.contains("sqlite_constraint_unique")) {
448 return true;
449 }
450 // HSQLDB-Style-Fallback (für alte/andere Backends)
451 String state = e.getSQLState();
452 return state != null && state.startsWith("23") && msg.contains("unique");
453 }
454
455 private static boolean istFkVerletzung(SQLException e) {
456 String raw = e.getMessage();
457 if (raw == null) {
458 return false;
459 }
460 String msg = raw.toLowerCase(Locale.ROOT);
461 if (msg.contains("foreign key constraint failed") || msg.contains("sqlite_constraint_foreignkey")) {
462 return true;
463 }
464 String state = e.getSQLState();
465 return state != null && state.startsWith("23")
466 && (msg.contains("foreign") || msg.contains("integrity"));
467 }
