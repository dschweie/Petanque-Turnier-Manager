162 public void testNamenEinfuegen() throws GenerateException {
163 meldeListe.setSpielTag(SpielTagNr.from(1));
164 // Meldeliste neu anlegen falls sie nicht existiert (z.B. nach removeAllSheetsExclude)
165 if (meldeListe.getXSpreadSheet() == null) {
166 meldeListe.createMeldelisteWithParams(konfigurationSheet.getSuperMeleeMode());
167 }
168 XSpreadsheet meldelisteSheet = meldeListe.getXSpreadSheet();
169 getSheetHelper().setActiveSheet(meldelisteSheet);
170
171 List<SpielerTestname> testNamen = testnamenLoader.listeMitSpielerTestNamen(ANZ_TESTNAMEN);
172
173 int vornameSpalte = meldeListe.getMeldungenSpalte().getErsteMeldungNameSpalte();
174 int nachnameSpalte = meldeListe.getMeldungenSpalte().getLetzteMeldungNameSpalte();
175 Position posVorname = Position.from(vornameSpalte, MeldeListeKonstanten.ERSTE_DATEN_ZEILE - 1);
176 Position posNachname = Position.from(nachnameSpalte, MeldeListeKonstanten.ERSTE_DATEN_ZEILE - 1);
177 Position posSpielerNr = Position.from(MeldeListeKonstanten.SPIELER_NR_SPALTE,
178 MeldeListeKonstanten.ERSTE_DATEN_ZEILE - 1);
179 NumberCellValue spielrNr = NumberCellValue.from(meldelisteSheet, posSpielerNr);
180 StringCellValue nameCelVal = StringCellValue.from(meldelisteSheet, posVorname);
181
182 for (int spielerCntr = 0; spielerCntr < testNamen.size(); spielerCntr++) {
183 SheetRunner.testDoCancelTask();
184 posVorname.zeilePlusEins();
185 posNachname.zeile(posVorname.getZeile());
186 String textFromCell = getSheetHelper().getTextFromCell(meldelisteSheet, posVorname);
187
188 if (StringUtils.isNotEmpty(textFromCell)) {
189 throw new GenerateException(I18n.get("error.testdaten.daten.vorhanden"));
190 }
191
192 SpielerTestname tn = testNamen.get(spielerCntr);
193 getSheetHelper().setStringValueInCell(nameCelVal.setPos(posVorname).setValue(tn.vorname()));
194 if (vornameSpalte != nachnameSpalte) {
195 getSheetHelper().setStringValueInCell(nameCelVal.setPos(posNachname).setValue(tn.nachname()));
196 }
197
198 spielrNr.zeile(posVorname.getZeile());
199 int randomNum = RandomSource.nextInt(0, 3);
200 if (randomNum == 1) { // nur die einser eintragen
201 // zum test spielrnr vorgeben, mix in nr erreichen
202 getSheetHelper().setNumberValueInCell(spielrNr.setValue((double) spielerCntr + 1));
203 } else {
204 // andere Nummer leer
205 getSheetHelper().setStringValueInCell(StringCellValue.from(spielrNr).setValue(""));
206 }
207 }
208
209 meldeListe.upDateSheet();
210 }
