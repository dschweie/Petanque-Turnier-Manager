67 public static XSpreadsheet bereitstellen(WorkingSpreadsheet ws) throws GenerateException {
68 SheetHelper sh = new SheetHelper(ws);
69 String sheetName = SheetNamen.spielerDbVorlage();
70 XSpreadsheet vorhandenes = sh.findByName(sheetName);
71 if (vorhandenes != null) {
72 sh.setActiveSheet(vorhandenes);
73 return vorhandenes;
74 }
75
76 short pos = anhaengePosition(sh);
77 XSpreadsheet sheet = sh.newIfNotExist(sheetName, pos, TAB_FARBE);
78 if (sheet == null) {
79 throw new GenerateException(I18n.get("error.tabelle.nicht.vorhanden", sheetName));
80 }
81
82 schreibeHeader(ws, sheet);
83 formatiereHeader(sheet);
84 setzeSpaltenbreiten(sh, sheet);
85 sh.setActiveSheet(sheet);
86 freezeKopfzeile(sheet, ws);
87 return sheet;
88 }

111 SPALTE_FEHLERURSACHE, HEADER_ZEILE);
112 RangeHelper.from(sheet, ws.getWorkingSpreadsheetDocument(), kopfPos).setDataInRange(rd);

115 private static void formatiereHeader(XSpreadsheet sheet) {
116 try {
117 XCellRange header = sheet.getCellRangeByPosition(SPALTE_VORNAME, HEADER_ZEILE,
118 SPALTE_FEHLERURSACHE, HEADER_ZEILE);
119 XPropertySet props = Lo.qi(XPropertySet.class, header);
