123 @Override
124 protected void doRun() throws GenerateException {
125 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.POULE)
126 .prefix(getLogPrefix()).validate()) {
127 return;
128 }
129
130 Optional<PouleTurnierParameterDialog.TurnierParameter> param;
131 try {
132 param = PouleTurnierParameterDialog.from(getWorkingSpreadsheet())
133 .anzeigen(Formation.TRIPLETTE, false, false);
134 } catch (Exception e) {
135 String errMsg = I18n.get("error.dialog.parameterdialog", e.getMessage());
136 logger.error(errMsg, e);
137 throw new GenerateException(errMsg);
138 }
139
140 if (param.isEmpty()) {
141 return;
142 }
143
144 getKonfigurationSheet().update();
145 getSheetHelper().removeAllSheetsExclude();
146 createMeldelisteWithParams(param.get().formation(), param.get().teamnameAnzeigen(), param.get().vereinsnameAnzeigen());
147 }

153 public void createMeldelisteWithParams(Formation formation, boolean teamnameAnzeigen, boolean vereinsnameAnzeigen)
154 throws GenerateException {
155 logger.info("Erstelle Poule-Meldeliste: formation={}, teamname={}, verein={}", formation, teamnameAnzeigen,
156 vereinsnameAnzeigen);
157
158 var neuesSheet = NewSheet.from(this, SheetNamen.pouleMeldeliste(), METADATA_SCHLUESSEL)
159 .pos(DefaultSheetPos.MELDELISTE).hideGrid().tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).setDocVersionWhenNew().create();
160 if (neuesSheet.isDidCreate()) {
161 getKonfigurationSheet().setMeldeListeFormation(formation);
162 getKonfigurationSheet().setMeldeListeTeamnameAnzeigen(teamnameAnzeigen);
163 getKonfigurationSheet().setMeldeListeVereinsnameAnzeigen(vereinsnameAnzeigen);
164 upDateSheet();
165 }
166 }
