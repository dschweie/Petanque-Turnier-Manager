43 @Override
44 public void doRun() throws GenerateException {
45 XSpreadsheetDocument doc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
46 ReentrancyState state = STATES.computeIfAbsent(doc, k -> new ReentrancyState());
47 if (!state.running.compareAndSet(false, true)) {
48 state.dirty.set(true);
49 logger.debug("RanglisteUpdate bereits aktiv – Dirty-Rerun vorgemerkt");
50 return;
51 }
52 try {
53 do {
54 state.dirty.set(false);
55 updateIntern();
56 } while (state.dirty.get());
57 } finally {
58 state.running.set(false);
59 }
60 }

62 private void updateIntern() throws GenerateException {
63 // SpielTagNr aus dem Refresh-Parameter ins Delegate übertragen –
64 // sonst NPE in getSpieltagNr() (delegate.spielTagNr wird sonst nur
65 // via generate() gesetzt, und das überspringen wir im Update-Pfad).
66 SpielTagNr spielTagNr = getSpielTagFuerRefresh();
67 if (spielTagNr == null) {
68 spielTagNr = getKonfigurationSheet().getAktiveSpieltag();
69 }
70 setSpieltagNr(spielTagNr);
71
72 XSpreadsheet sheet = getXSpreadSheet();
73 if (sheet == null) {
74 logger.debug("Spieltag-Rangliste-Sheet fehlt – vollständiger Erstaufbau");
75 new SpieltagRanglisteSheet(getWorkingSpreadsheet(), getSpieltagNr()).doRun();
76 return;
77 }
78 processBoxinfo("processbox.rangliste.aktualisieren");
79
80 int anzSpielRunden = getAktuelleSpielrundeSheet().countNumberOfSpielRundenSheets(getSpieltagNr());
81 if (anzSpielRunden < 1) {
82 return;
83 }
84
85 int spielerAnzahl = getSpielerSpalte().getSpielerNrList().size();
86 RanglisteUpdateHelper.loescheDatenzeilen(this, sheet, spielerAnzahl);
87
88 berechnungUndSchreiben(sheet, anzSpielRunden);
89 getxCalculatable().calculate();
90 getRangListeSorter().doSort();
91 getRangListeSpalte().upDateRanglisteSpalte();
92 }
