33 private void renderColgroup(TabelleModel model, StringBuilder sb) {
34 sb.append("<colgroup>");
35 var spaltenBreiten = model.getSpaltenBreiten();
36 for (int c = 0; c < model.getSpalten(); c++) {
37 Integer breite = spaltenBreiten.get(c);
38 if (istVersteckt(breite)) {
39 sb.append("<col style=\"display:none\" />");
40 } else {
41 int px = zuPx(breite != null ? breite : SPALTENBREITE_FALLBACK);
42 sb.append("<col style=\"width:").append(px).append("px\" />");
43 }
44 }
45 sb.append("</colgroup>");
46 }

48 private void renderKopfUndRumpf(TabelleModel model, StringBuilder sb) {
49 int kopfZeilenAnzahl = model.getKopfZeilenAnzahl();
50 var gitter = model.getGitter();
51
52 if (kopfZeilenAnzahl > 0) {
53 sb.append("<thead>");

61 for (int r = kopfZeilenAnzahl; r < gitter.size(); r++) {
62 renderZeile(model, gitter.get(r), r, sb);
63 }
64 sb.append("</tbody>");

67 private void renderZeile(TabelleModel model, List<String> zeile, int zeilenIdx, StringBuilder sb) {
68 Integer hoehe = model.getZeilenHoehen().get(zeilenIdx);
69 int px = zuPx(hoehe != null ? hoehe : ZEILENHOEHE_FALLBACK);
70 sb.append("<tr style=\"height:").append(px).append("px\">");
71
72 var spaltenBreiten = model.getSpaltenBreiten();
73 for (int c = 0; c < zeile.size(); c++) {
74 if (istVersteckt(spaltenBreiten.get(c))) {
75 continue;
76 }
77 String id = zeile.get(c);
78 if (id == null) {
79 continue; // Merge-Slave
80 }
81 ZelleModel zelle = model.getZellen().get(id);
82 if (zelle == null) {
83 sb.append("<td></td>"); // catch-Fall aus TabellenMapper
84 } else {
85 renderZelle(zelle, sb);
86 }
87 }
88
89 sb.append("</tr>");
90 }

92 private void renderZelle(ZelleModel zelle, StringBuilder sb) {
93 StyleModel s = zelle.stil();
94 sb.append("<td");
95 if (s.colspan() > 1) {
96 sb.append(" colspan=\"").append(s.colspan()).append("\"");
97 }
98 if (s.rowspan() > 1) {
99 sb.append(" rowspan=\"").append(s.rowspan()).append("\"");
100 }
101 sb.append(" style=\"").append(buildCss(s)).append("\"");
102 sb.append(">");
103 sb.append(StringEscapeUtils.escapeHtml4(zelle.wert() != null ? zelle.wert() : ""));
104 sb.append("</td>");
105 }

107 private String buildCss(StyleModel s) {
108 var css = new StringBuilder();
109 css.append("overflow:hidden;");
110
111 if (s.hintergrundfarbe() != null) {
112 css.append("background-color:").append(s.hintergrundfarbe()).append(";");
113 }
114 if (s.schriftfarbe() != null) {
115 css.append("color:").append(s.schriftfarbe()).append(";");
116 }
117 if (s.fett()) {
118 css.append("font-weight:bold;");
119 }
120 if (s.kursiv()) {
121 css.append("font-style:italic;");
122 }
123 if (s.schriftart() != null) {
124 css.append("font-family:").append(bereinigeFontFamily(s.schriftart())).append(";");
125 }
126 if (s.schriftgroesse() > 0) {
127 css.append("font-size:").append(formatPt(s.schriftgroesse())).append("pt;");
128 }
129 if (s.ausrichtung() != null) {
130 css.append("text-align:").append(s.ausrichtung()).append(";");
131 }
132 String va = s.vertikaleAusrichtung() != null ? s.vertikaleAusrichtung() : "top";
133 css.append("vertical-align:").append(va).append(";");
134 css.append(s.zeilenumbruch() ? "white-space:normal;" : "white-space:nowrap;");
135
136 int rot = s.rotationGrad();
137 if (rot == 90 || rot == 270) {
138 css.append("writing-mode:vertical-rl;");
139 } else if (rot != 0) {
140 css.append("transform:rotate(").append(rot).append("deg);");
141 }
142
143 if (s.linienOben() != null) {
144 css.append("border-top:").append(s.linienOben()).append(";");
145 }
146 if (s.linienUnten() != null) {
147 css.append("border-bottom:").append(s.linienUnten()).append(";");
148 }
149 if (s.linienLinks() != null) {
150 css.append("border-left:").append(s.linienLinks()).append(";");
151 }
152 if (s.linienRechts() != null) {
153 css.append("border-right:").append(s.linienRechts()).append(";");
154 }
155
156 return css.toString();
157 }
