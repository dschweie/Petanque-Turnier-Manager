80 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, WorkingSpreadsheet ws,
81 List<SheetSchutzInfo> infos) {
82 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_SCHWEIZER_MELDELISTE).ifPresent(sheet -> {
83 try {
84 var konfigSheet = new SchweizerKonfigurationSheet(ws);
85 int aktivSpalte = berechneAktivSpalte(konfigSheet);
86 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
87 RangePosition.from(1, MELDELISTE_ERSTE_DATEN_ZEILE,
88 aktivSpalte, MeldungenSpalte.MAX_ANZ_MELDUNGEN))));
89 } catch (GenerateException e) {
90 logger.warn("Editierbare Meldeliste-Spalten konnten nicht berechnet werden: {}", e.getMessage(), e);
91 }
92 });
93 }
94
95 private void sammleSpielrundenSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
96 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
97 SheetMetadataHelper.SCHLUESSEL_SCHWEIZER_SPIELRUNDE_PREFIX);
98 for (var key : schluessel) {
99 SheetMetadataHelper.findeSheet(xDoc, key).ifPresent(sheet ->
100 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet,
101 List.of(berechneSpielrundeErgebnisBereich(sheet)))));
102 }
103 }
104
105 private void sammleVollGesperrteSheets(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
106 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_SCHWEIZER_RANGLISTE)
107 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
108 }
109
115 private RangePosition berechneSpielrundeErgebnisBereich(XSpreadsheet sheet) {
116 return RangePosition.from(
117 SchweizerAbstractSpielrundeSheet.ERG_TEAM_A_SPALTE,
118 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE,
119 SchweizerAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE,
120 ermittleLetzteSpielrundeZeile(sheet));
121 }
122
123 private int ermittleLetzteSpielrundeZeile(XSpreadsheet sheet) {
124 int letzteZeile = SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE;
125 try {
126 for (int zeile = SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE;
127 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
128 XCell xCell = sheet.getCellByPosition(SchweizerAbstractSpielrundeSheet.TEAM_A_SPALTE, zeile);
129 if (CellContentType.EMPTY.equals(xCell.getType())) {
130 break;
131 }
132 letzteZeile = zeile;
133 }
134 } catch (Exception e) {
135 logger.warn("Letzte Spielrunde-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
136 }
137 return letzteZeile;
138 }

144 private int berechneAktivSpalte(SchweizerKonfigurationSheet konfigSheet) throws GenerateException {
145 int spaltenProSpieler = konfigSheet.isMeldeListeVereinsnameAnzeigen() ? 3 : 2;
146 int ersterOffset = konfigSheet.isMeldeListeTeamnameAnzeigen() ? 2 : 1;
147 int letzteDataSpalte = ersterOffset + konfigSheet.getMeldeListeFormation().getAnzSpieler() * spaltenProSpieler - 1;
148 return letzteDataSpalte + 2; // +1=SP, +2=Aktiv
149 }
