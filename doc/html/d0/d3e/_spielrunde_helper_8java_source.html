72 public void datenErsteSpalte(SpielrundeSpielbahn spielrundeSpielbahnFlagAusKonfig, int erstZeile, int letzteZeile,
73 int nrSpalte, int headerZeile, int headerZeile2, Integer headerColor) throws GenerateException {

75 sheet.processBoxinfo("processbox.erste.spalte.daten.einfuegen");
76
77 Position posHeaderZelle1 = Position.from(nrSpalte, headerZeile);

79 Position posErsteDatenZelle = Position.from(nrSpalte, erstZeile);
80 Position posLetztDatenZelle = Position.from(nrSpalte, letzteZeile);
81 RangePosition rangeErsteSpalte = RangePosition.from(posErsteDatenZelle, posLetztDatenZelle);
82
83 // header
84 // -------------------------
85 // spalte paarungen Nr oder Spielbahn-Nummer
86 // -------------------------
87 ColumnProperties columnProperties = ColumnProperties.from().setVertJustify(CellVertJustify2.CENTER)
88 .setHoriJustify(CellHoriJustify.CENTER);
89 if (spielrundeSpielbahnFlagAusKonfig == SpielrundeSpielbahn.X) {
90 if (mitHeader) {
91 StringCellValue headerValue = StringCellValue.from(sheet, posHeaderZelle1).setRotateAngle(0)
92 .setVertJustify(CellVertJustify2.BOTTOM).centerHoriJustify().setCellBackColor(headerColor)
93 .setCharHeight(headerCharHeight).setEndPosMergeZeilePlus(1).setValue("#").setShrinkToFit(true)
94 .setBorder(BorderFactory.from().allThin().boldLn().forBottom().toBorder());
95 sheet.getSheetHelper().setStringValueInCell(headerValue);
96 }
97
98 columnProperties.setWidth(500); // Paarungen cntr
99 sheet.getSheetHelper().setColumnProperties(sheet, nrSpalte, columnProperties);
100 } else {
101 // Spielbahn Spalte header
102 if (mitHeader) {
103 StringCellValue headerValue = StringCellValue.from(sheet, posHeaderZelle1).setRotateAngle(27000)
104 .setVertJustify(CellVertJustify2.CENTER).setHoriJustify(CellHoriJustify.CENTER)
105 .setCellBackColor(headerColor).setCharHeight(headerCharHeight).setEndPosMergeZeilePlus(1)
106 .setValue(I18n.get("column.header.bahn")).setShrinkToFit(true)
107 .setBorder(BorderFactory.from().allThin().boldLn().forBottom().toBorder())
108 .setComment(I18n.get("comment.spielbahn"));
109 sheet.getSheetHelper().setStringValueInCell(headerValue);
110 }
111 columnProperties.setWidth(900);
112 sheet.getSheetHelper().setColumnProperties(sheet, nrSpalte, columnProperties);
113 }
114
115 RangePosition nbrRangeOhneHeader = RangePosition.from(posErsteDatenZelle, posLetztDatenZelle);
116 sheet.getSheetHelper().setPropertiesInRange(sheet, nbrRangeOhneHeader,
117 CellProperties.from().setCharHeight(nrCharHeight).setBorder(BorderFactory.from().allThin().toBorder()));
118
119 int anzPaarungen = letzteZeile - erstZeile + 1;
120
121 // Daten
122 if (spielrundeSpielbahnFlagAusKonfig == SpielrundeSpielbahn.X
123 || spielrundeSpielbahnFlagAusKonfig == SpielrundeSpielbahn.N) {
124 StringCellValue formulaCellValue = StringCellValue.from(sheet, posErsteDatenZelle);
125 formulaCellValue.setValue("=ROW()-" + erstZeile).setFillAutoDown(letzteZeile);
126 sheet.getSheetHelper().setFormulaInCell(formulaCellValue);
127 } else if (spielrundeSpielbahnFlagAusKonfig == SpielrundeSpielbahn.R) {
128 // R = Spielbahn -> random x
129 // anzahl paarungen ?
130 ArrayList<Integer> bahnnummern = new ArrayList<>();
131 // fill
132 for (int i = 1; i <= anzPaarungen; i++) {
133 bahnnummern.add(i);
134 }
135 // mishen
136 Collections.shuffle(bahnnummern, RandomSource.asJavaRandom());
137 StringCellValue stringCellValue = StringCellValue.from(sheet, posErsteDatenZelle);
138 for (Integer bahnnr : bahnnummern) {
139 SheetRunner.testDoCancelTask();
140 if (bahnnr > 0) { // es kann sein das wir lücken haben, = teampaarungen ohne bahnnummer
141 stringCellValue.setValue(bahnnr);
142 sheet.getSheetHelper().setStringValueInCell(stringCellValue);
143 }
144 stringCellValue.zeilePlusEins();
145 }
146 } else if (spielrundeSpielbahnFlagAusKonfig == SpielrundeSpielbahn.L) {
147 Position itteratepos = Position.from(posErsteDatenZelle); // orginal nicht verändern
148 // leere spalte
149 for (int i = 1; i <= anzPaarungen; i++) {
150 sheet.getSheetHelper().clearValInCell(sheet.getXSpreadSheet(), itteratepos);
151 itteratepos.zeilePlusEins();
152 }
153 }
154
155 // Formatiere Daten Spalte
156 // erste Spalte, mit prüfung auf doppelte nummer
157
158 FehlerStyle fehlerStyle = new FehlerStyle();
159 String conditionfindDoppelt = "COUNTIF(" + posErsteDatenZelle.getSpalteAddressWith$() + ";"
160 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
161 String conditionNotEmpty = ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>\"\"";
162 String formulaFindDoppelteSpielrNr = "AND(" + conditionfindDoppelt + ";" + conditionNotEmpty + ")";
163 ConditionalFormatHelper.from(sheet, rangeErsteSpalte).clear().formula1(formulaFindDoppelteSpielrNr)
164 .operator(ConditionOperator.FORMULA).style(fehlerStyle).applyAndDoReset();
165 SheetHelper.faerbeZeilenAbwechselnd(sheet, rangeErsteSpalte,
166 spielrundeHintergrundFarbeGeradeStyle.getFarbe(),
167 spielrundeHintergrundFarbeUnGeradeStyle.getFarbe());
168
169 }
