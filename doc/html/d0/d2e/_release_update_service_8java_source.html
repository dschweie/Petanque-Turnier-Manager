144 private void starteInitialenRefresh() {
145 // Komplett asynchron auf dem Background-Executor: InstallierteVersion.ermitteln
146 // und ReleaseCache.lade* sind File-/UNO-IO, die nicht auf dem LO-Main-Thread
147 // im Plugin-Init laufen dürfen (sonst weißes Calc-Fenster beim Start).
148 executor.execute(this::initialerRefreshInternalMitTiming);
149 }

151 private void initialerRefreshInternalMitTiming() {
152 long startNs = System.nanoTime();
153 try {
154 initialerRefreshInternal();
155 } finally {
156 long dauerMs = (System.nanoTime() - startNs) / 1_000_000L;
157 PerfLog.log(logger, "[STARTUP-TIMING] ReleaseUpdateService initialerRefresh (background): {} ms", dauerMs);
158 }
159 }

161 private void initialerRefreshInternal() {
162 if (installierteVersion.isEmpty()) {
163 installierteVersion = InstallierteVersion.ermitteln(context).map(InstallierteVersion::raw);
164 }
165 var frisch = cache.ladeWennFrisch(CACHE_TTL);
166 if (frisch.isPresent()) {
167 aktuellesRelease = frisch;
168 aktualisiereStatusAusCache(frisch.get());
169 return;
170 }
171 // älterer Cache als Übergangsanzeige akzeptieren
172 var alt = cache.ladeUnabhaengigVomAlter();
173 if (alt.isPresent()) {
174 aktuellesRelease = alt;
175 aktualisiereStatusAusCache(alt.get());
176 }
177 plane(false);
178 }

254 public synchronized void dispose() {
255 if (disposed) {
256 return;
257 }
258 disposed = true;
259 executor.shutdownNow();
260 try {
261 if (!executor.awaitTermination(2, TimeUnit.SECONDS)) {
262 logger.debug("Executor wurde nicht innerhalb von 2s beendet");
263 }
264 } catch (InterruptedException e) {
265 Thread.currentThread().interrupt();
266 }
267 }

269 private void plane(boolean force) {
270 if (disposed || executor.isShutdown()) {
271 return;
272 }
273 try {
274 executor.execute(() -> fuehreRefreshAus(force));
275 } catch (java.util.concurrent.RejectedExecutionException e) {
276 logger.debug("Refresh-Task abgewiesen (Executor herunterfahren?)");
277 }
278 }
279
280 private void fuehreRefreshAus(boolean force) {
281 if (!force) {
282 var frisch = cache.ladeWennFrisch(CACHE_TTL);
283 if (frisch.isPresent()) {
284 aktuellesRelease = frisch;
285 aktualisiereStatusAusCache(frisch.get());
286 benachrichtigeListener();
287 return;
288 }
289 }
290 setzeStatus(UpdateStatus.LAEUFT);
291 benachrichtigeListener();
292 for (int versuch = 0; versuch < retryBackoffs.size(); versuch++) {
293 if (Thread.currentThread().isInterrupted() || disposed) {
294 return;
295 }
296 var release = client.ladeLetztesRelease();
297 if (release.isPresent()) {
298 aktuellesRelease = release;
299 try {
300 cache.schreibe(release.get());
301 } catch (java.io.IOException e) {
302 logger.warn("Konnte Cache nicht schreiben: {}", e.getMessage());
303 }
304 aktualisiereStatusAusRelease(release.get());
305 benachrichtigeListener();
306 return;
307 }
308 if (versuch + 1 < retryBackoffs.size()) {
309 if (!warte(retryBackoffs.get(versuch + 1))) {
310 return;
311 }
312 }
313 }
314 // alle Retries fehlgeschlagen – Status zurücksetzen, gecachten Release-Stand
315 // (falls vorhanden) trotzdem als Anzeige stehen lassen.
316 setzeStatus(UpdateStatus.NICHT_VERFUEGBAR);
317 benachrichtigeListener();
318 }

334 private void aktualisiereStatusAusRelease(ReleaseInfo release) {
335 var inst = installierteVersion.orElse(null);
336 var verfuegbar = release.tagName();
337 if (release.prerelease()) {
338 setzeStatus(UpdateStatus.KEIN_UPDATE);
339 return;
340 }
341 if (VersionVergleicher.istNeuer(inst, verfuegbar)) {
342 setzeStatus(UpdateStatus.UPDATE_VERFUEGBAR);
343 } else {
344 setzeStatus(UpdateStatus.KEIN_UPDATE);
345 }
346 }
