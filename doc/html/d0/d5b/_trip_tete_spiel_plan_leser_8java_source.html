40 public List<TripTeteBegegnungErgebnis> leseBegegnungen() throws GenerateException {
41 List<TripTeteBegegnungErgebnis> ergebnisse = new ArrayList<>();
42 XSpreadsheet sheet = spielplanSheet.getXSpreadSheet();
43 int letzteZeile = TripTeteSpielPlanSheet.ERSTE_DATEN_ZEILE + MAX_SCAN_ZEILEN;
44
45 for (int zeile = TripTeteSpielPlanSheet.ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
46 int teamANr = liesInt(sheet, TripTeteSpielPlanSheet.TEAM_A_NR_SPALTE, zeile);

55 private Optional<TripTeteBegegnungErgebnis> leseBegegnung(XSpreadsheet sheet, int zeile) {
56 int teamANr = liesInt(sheet, TripTeteSpielPlanSheet.TEAM_A_NR_SPALTE, zeile);
57 if (teamANr < 0) {
58 return Optional.empty();
59 }
60 int teamBNr = liesInt(sheet, TripTeteSpielPlanSheet.TEAM_B_NR_SPALTE, zeile);
61 if (teamBNr <= 0) {
62 return Optional.empty(); // Freilos
63 }
64 int triA = liesInt(sheet, TripTeteSpielPlanSheet.TRI_A_SPALTE, zeile);
65 if (triA < 0) {
66 return Optional.empty(); // Ergebnis noch nicht eingetragen
67 }
68 int triB = liesInt(sheet, TripTeteSpielPlanSheet.TRI_B_SPALTE, zeile);
69 int douA = liesInt(sheet, TripTeteSpielPlanSheet.DOU_A_SPALTE, zeile);
70 int douB = liesInt(sheet, TripTeteSpielPlanSheet.DOU_B_SPALTE, zeile);
71 int teteA = liesInt(sheet, TripTeteSpielPlanSheet.TETE_A_SPALTE, zeile);
72 int teteB = liesInt(sheet, TripTeteSpielPlanSheet.TETE_B_SPALTE, zeile);
73 if (triB < 0 || douA < 0 || douB < 0 || teteA < 0 || teteB < 0) {
74 return Optional.empty();
75 }
76 return Optional.of(
77 new TripTeteBegegnungErgebnis(Team.from(teamANr), Team.from(teamBNr))
78 .setPartieErgebnis(TripTetePartie.TRIPLETTE, new SpielErgebnis(triA, triB))
79 .setPartieErgebnis(TripTetePartie.DOUBLETTE, new SpielErgebnis(douA, douB))
80 .setPartieErgebnis(TripTetePartie.TETE, new SpielErgebnis(teteA, teteB)));
81 }
