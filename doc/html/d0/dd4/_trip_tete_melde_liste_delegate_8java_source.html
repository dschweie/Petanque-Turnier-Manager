144 private void sortiereUndNummeriere() throws GenerateException {
145 XSpreadsheet xSheet = sheet.getXSpreadSheet();
146 SheetHelper sh = sheet.getSheetHelper();
147 int letzte = letzteZeileMitDaten(xSheet) + MIN_ANZAHL_MELDUNGEN_ZEILEN;
148
149 int naechsteNr = berechneNaechsteFreieNr(sh, xSheet, letzte);
150
151 for (int z = ERSTE_DATEN_ZEILE_OVERRIDE; z <= letzte; z++) {
152 String vorname = sh.getTextFromCell(xSheet, Position.from(getVornameSpalte(0), z));
153 if (vorname == null || vorname.isBlank()) {
154 sh.clearValInCell(xSheet, Position.from(getTeamNrSpalte(), z));
155 } else {
156 int vorhandeneNr = sh.getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), z));
157 if (vorhandeneNr <= 0) {
158 sh.setNumberValueInCell(NumberCellValue.from(xSheet, Position.from(getTeamNrSpalte(), z))
159 .setValue(naechsteNr++));
160 }
161 }
162 }
163
164 int letzteNachNummern = letzteZeileMitDaten(xSheet);
165 if (letzteNachNummern >= ERSTE_DATEN_ZEILE_OVERRIDE) {
166 RangePosition sortRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE_OVERRIDE,
167 getAktivSpalte(), letzteNachNummern);
168 SortHelper.from(sheet, sortRange).spalteToSort(getTeamNrSpalte()).aufSteigendSortieren(true).doSort();
169 }
170 }

297 private void formatDatenSpalten() throws GenerateException {
298 int letzteDatenZeile = getLetzteDatenZeileUseMin();
299
300 MeldungenHintergrundFarbeGeradeStyle farbeGerade = konfigurationSheet.getMeldeListeHintergrundFarbeGeradeStyle();
301 MeldungenHintergrundFarbeUnGeradeStyle farbeUngerade = konfigurationSheet.getMeldeListeHintergrundFarbeUnGeradeStyle();
302
303 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE_OVERRIDE,
304 getTeamNrSpalte(), letzteDatenZeile);
305 RangeHelper.from(sheet, nrRange).setRangeProperties(
306 RangeProperties.from().setBorder(
307 BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
308
309 String kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
310 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
311 ConditionalFormatHelper.from(sheet, nrRange).clear()
312 .formulaIsText().styleIsFehler().applyAndDoReset()
313 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
314 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
315 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset()
316 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
317 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
318
319 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
320 RangePosition teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE_OVERRIDE, 1, letzteDatenZeile);
321 RangeHelper.from(sheet, teamnameRange).setRangeProperties(
322 RangeProperties.from().setBorder(
323 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
324 }
325
326 int anzSpieler = Formation.TRIPLETTE.getAnzSpieler();
327 for (int s = 0; s < anzSpieler; s++) {
328 int ersteSpielerSpalte = getVornameSpalte(s);
329 int letzteSpielerSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
330 ? getVereinsnameSpalte(s)
331 : getNachnameSpalte(s);
332 RangePosition spielerRange = RangePosition.from(ersteSpielerSpalte, ERSTE_DATEN_ZEILE_OVERRIDE,
333 letzteSpielerSpalte, letzteDatenZeile);
334 RangeHelper.from(sheet, spielerRange).setRangeProperties(
335 RangeProperties.from().setBorder(
336 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
337 }
338
339 RangePosition aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE_OVERRIDE,
340 getAktivSpalte(), letzteDatenZeile);
341 RangeHelper.from(sheet, aktivRange).setRangeProperties(
342 RangeProperties.from().centerJustify()
343 .setBorder(BorderFactory.from().allThin().toBorder()));
344
345 String kondAktivUngueltig = "AND(NOT(ISBLANK(" + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "));"
346 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>1)";
347 ConditionalFormatHelper.from(sheet, aktivRange).clear()
348 .formula1(kondAktivUngueltig).operator(ConditionOperator.FORMULA)
349 .styleIsFehler().applyAndDoReset()
350 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
351 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
352
353 EditierbaresZelleFormatHelper.anwenden(sheet,
354 RangePosition.from(1, ERSTE_DATEN_ZEILE_OVERRIDE, getAktivSpalte(), letzteDatenZeile));
355 }

399 private String leseAnzeigename(SheetHelper sh, XSpreadsheet xSheet, int zeile, boolean teamnameAktiv)
400 throws GenerateException {
401 if (teamnameAktiv) {
402 String tn = sh.getTextFromCell(xSheet, Position.from(getTeamnameSpalte(), zeile));
403 return tn != null ? tn.strip() : "";
404 }
405 var sb = new StringBuilder();
406 for (int s = 0; s < Formation.TRIPLETTE.getAnzSpieler(); s++) {
407 String vn = sh.getTextFromCell(xSheet, Position.from(getVornameSpalte(s), zeile));
408 String nn = sh.getTextFromCell(xSheet, Position.from(getNachnameSpalte(s), zeile));
409 vn = vn != null ? vn.strip() : "";
410 nn = nn != null ? nn.strip() : "";
411 if (vn.isEmpty() && nn.isEmpty()) {
412 continue;
413 }
414 if (!sb.isEmpty()) {
415 sb.append(" / ");
416 }
417 sb.append(vn).append(" ").append(nn);
418 }
419 return sb.toString().strip();
420 }

461 List<Integer> getSpielerNrList() throws GenerateException {
462 XSpreadsheet xSheet = sheet.getXSpreadSheet();
463 int letzte = letzteZeileMitDaten(xSheet);
464 List<Integer> liste = new ArrayList<>();
465 if (letzte < ERSTE_DATEN_ZEILE_OVERRIDE) {
466 return liste;
467 }
468 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE_OVERRIDE,
469 getTeamNrSpalte(), letzte);
470 RangeData data = RangeHelper.from(xSheet, sheet.getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), nrRange)
471 .getDataFromRange();
472 for (RowData row : data) {
473 int nr = row.get(0).getIntVal(-1);
474 if (nr > 0) {
475 liste.add(nr);
476 }
477 }
478 Collections.sort(liste);
479 return liste;
480 }
