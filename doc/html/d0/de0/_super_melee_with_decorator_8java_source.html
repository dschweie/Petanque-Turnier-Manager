98 @Override
99 public MeleeSpielRunde neueSpielrundeTripletteMode( int rndNr,
100 SpielerMeldungen meldungen,
101 boolean nurDoublette)
102 throws AlgorithmenException
103 {
104 List<Team> grid = new ArrayList<>();
105 checkNotNull(meldungen, "Meldungen = null");
106 checkArgument((7 != meldungen.size()),"Mit 7 Spielern lässt sich keine Spielrunde generieren.");
107 checkArgument((3 < meldungen.size()), "Es werden mindestens vier Spieler benötigt, um eine Spielrunde zu generieren.");
108
109 this.updateParticipants(meldungen.spieler());
110
111 while (grid.isEmpty())
112 {
113 List<Team> encounter = null;
114
115 if(this.generator.hasNext())
116 { // Der Generator kennt eine Permutation, die gute Paarung sein kann
117 int[] permutation = this.generator.getNext();
118 int index = permutation.length - 1;
119
120 do {
121 switch((index+1) % 6)
122 {
123 case 1, 2, 3, 4: // Es muss mindestens mit einem Doublette gespielt werden
124 encounter = this.tryDoubletteVSDoublette(this.participants.get(permutation[index]), this.participants.get(permutation[index-1]), this.participants.get(permutation[index-2]), this.participants.get(permutation[index-3]));
125 if(null != encounter)
126 { // Die Partie kann nach derzeitigem Stand gespielt werden.
127 grid.addAll(encounter);
128 index -= 4;
129 }
130 break;
131 case 5: // Es muss genau einmal Doublette gegen Triplette gespielt werden
132 encounter = this.tryDoubletteVSTriplette(this.participants.get(permutation[index]), this.participants.get(permutation[index-1]), this.participants.get(permutation[index-2]), this.participants.get(permutation[index-3]), this.participants.get(permutation[index-4]));
133 if(null != encounter)
134 { // Die Partie kann nach derzeitigem Stand gespielt werden.
135 grid.addAll(encounter);
136 index -=5;
137 }
138 break;
139 default: // Es werden Triplettes hinzugefügt
140 encounter = this.tryTripletteVSTriplette(this.participants.get(permutation[index]), this.participants.get(permutation[index-1]), this.participants.get(permutation[index-2]), this.participants.get(permutation[index-3]), this.participants.get(permutation[index-4]), this.participants.get(permutation[index-5]));
141 if(null != encounter)
142 { // Die Partie kann nach derzeitigem Stand gespielt werden.
143 grid.addAll(encounter);
144 index -= 6;
145 }
146 }
147 } while((0 < index) && (null != encounter));
148
149 if(null == encounter)
150 { // In diesem Fall war mindestens eine Partie nicht passend.
151 grid.clear();
152 }
153 }
154 else
155 { // Ende der Permutationen ist erreicht
156 if(this.ruleset.isRulesetSuspendable())
157 {
158 this.ruleset.suspendFirstRule(); // Regelwerk aufweichen
159 this.resetPermutationGenerator(); // wieder von vorne suchen
160 grid.clear(); // keine Lösung gefunden
161 } else {
162 // Mit dem Regelwerk ist kann keine Begegnung generiert werden
163 String spieltagPraefix = spieltagNrFuerLog > 0 ? "Spieltag " + spieltagNrFuerLog + ", " : "";
164 logger.warn("{}Spielrunde {}: Alle möglichen Spielerkombinationen ausgeschöpft ({} Spieler).",
165 spieltagPraefix, rndNr, this.participants.size());
166 throw new AlgorithmenException(
167 "Keine gültige Spielrunde für Runde " + rndNr + " möglich — "
168 + "alle Spielerkombinationen sind ausgeschöpft. "
169 + "Möglicherweise müssen Wiederholungen in den Regeln zugelassen werden.");
170 }
171 }
172 }
173
174 // neue Runde ist gefunden und Rückgabewert kann instanziert werden.
175 this.finalizeGrid(grid);
176 MeleeSpielRunde retval = new MeleeSpielRunde(rndNr);
177 retval.addTeamsWennNichtVorhanden(grid);
178 return retval;
179 }

273 private List<Team> tryEncounter(Team teamA, Team teamB)
274 {
275 if((null == teamA) || (null == teamB))
276 { // da ein Team nicht erzeugt wurde, gibt es keine Lösung
277 if(null != teamA)
278 this.decreaseUniqueTeamId();
279 if(null != teamB)
280 this.decreaseUniqueTeamId();
281 return null;
282 } else {
283 // in diesem Fall ist zu prüfen, ob die Konstellation passt
284 if(this.ruleset.isMatchupValid(teamA, teamB))
285 { // Spielpaarung ist möglich
286 List<Team> retval = new ArrayList<>();
287 retval.add(teamA);
288 retval.add(teamB);
289 return retval;
290 } else {
291 // in diesem Fall ist die Spielpaarung zu verwerfen
292 this.uniqueTeamId -=2;
293 return null;
294 }
295 }

423 private void finalizeGrid(List<Team> runde)
424 {
425 for(int i=0; i < runde.size(); ++i)
426 {
427 Team team = runde.get(i);
428
429 for(Spieler it: team.spieler())
430 {
431 // optional muss ein Einsatz im Doublette erfasst werden
432 if(2 == team.size())
433 it.incAnzMalKleinesTeam();
434
435 // alle Mitspieler sind zu erfassen (Spieler prüft auf "selbst")
436 for(Spieler teammate: team.spieler())
437 it.addWarImSpielMit(teammate);
438
439 // Gegner sind zu erfassen, falls zweite Mannschaft einer Partie
440 if(1 == i % 2)
441 { // ungerader Index steht für 2. Mannschaft einer Partie
442 // Somit gibt es immer das Team runde.get(i-1)
443 for(Spieler opps: runde.get(i-1).spieler())
444 it.addGegner(opps); // addGegner sorgt für opps.addGegner(it)
445 }
446 }

487 public void updateDoubletteValue()
488 {
489 if(null == this.participants || this.participants.isEmpty())
490 { // In diesem Fall kann kein vernünftiger Wert ermittelt werden.
491 this.maximumForDoublette = -1;
492 }
493 else
494 {
495 @SuppressWarnings("null")
496 IntSummaryStatistics stats = participants.stream()
497 .mapToInt(Spieler::getAnzMalKleinesTeam)
498 .summaryStatistics();
499 this.maximumForDoublette = stats.getMax();
500 int smaller = 0;
501 for(Spieler p: this.participants)
502 smaller += (this.maximumForDoublette < p.getAnzMalKleinesTeam())?1:0;
503
504 if((stats.getMin() == this.maximumForDoublette) || (smaller < 10))
505 ++this.maximumForDoublette;
506 }
507 }
