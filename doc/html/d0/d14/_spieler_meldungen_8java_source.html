52 @Override
53 public SpielerMeldungen addNewWennNichtVorhanden(RowData meldungZeile) {
54 int spielerNr = meldungZeile.get(0).getIntVal(-1);
55 if (spielerNr > 0) {
56 Spieler spieler = Spieler.from(spielerNr);
57 if (setzPostionAktiv && setzPosIndex < meldungZeile.size()) {
58 int nichtzusammen = meldungZeile.get(setzPosIndex).getIntVal(-1);
59 if (nichtzusammen > 0) {
60 spieler.setSetzPos(nichtzusammen);
61 }
62 }
63 this.addSpielerWennNichtVorhanden(spieler);
64 }
65 return this;
66 }
