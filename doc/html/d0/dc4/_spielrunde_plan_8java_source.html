99 public void generate() throws GenerateException {
100
101 MeldeListeSheet_Update meldeListe = new MeldeListeSheet_Update(getWorkingSpreadsheet());
102 meldeListe.setSpielTag(getKonfigurationSheet().getAktiveSpieltag());
103 SpielerMeldungen meldungen = meldeListe.getAktiveMeldungen();
104
105 // Spielrunde sheet ?
106 processBoxinfo("processbox.neuer.spielrundeplan", getSpielRundeNr().getNr(), getSpielTag().getNr());
107
108 var neuesSheet = NewSheet
109 .from(this, getSheetName(getSpielTag(), getSpielRundeNr()),
110 SheetMetadataHelper.schluesselSupermeleeSpielrundePlan(getSpielTag().getNr(),
111 getSpielRundeNr().getNr()))
112 .pos(DefaultSheetPos.SUPERMELEE_WORK).spielTagPageStyle(getSpielTag()).setForceCreate(true).setActiv()
113 .tabColor(konfigurationSheet.getSpielrundeTabFarbe()).create();
114 if (!neuesSheet.isDidCreate()) {
115 ProcessBox.from().info("Abbruch vom Benutzer, Spielrundeplan wurde nicht erstellt");
116 return;
117 }
118
119 NumberCellValue spierNrVal = NumberCellValue
120 .from(getXSpreadSheet(), Position.from(SPIELER_NR_SPALTE, ERSTE_DATEN_ZEILE))
121 .setBorder(BorderFactory.from().allThin().toBorder())
122 .setCharColor(ColorHelper.CHAR_COLOR_GRAY_SPIELER_NR).setCharHeight(10).centerJustify();
123
124 // Team A/B und Bahn Nr
125 StringCellValue strVal = StringCellValue.from(spierNrVal).removeCharColor().setCharHeight(12).centerJustify();
126
127 int maxAnzSpielerInSpalte = 0;
128 int spielerCntr = 1;
129
130 StringCellValue nameFormula = StringCellValue
131 .from(getXSpreadSheet(), Position.from(SPIELER_NR_SPALTE, ERSTE_DATEN_ZEILE))
132 .setBorder(BorderFactory.from().allThin().toBorder()).setShrinkToFit(true).setCharHeight(12);
133
134 spalteFormat(spierNrVal.getPos());
135 blockHeader(spierNrVal.getPos());
136 boolean insertTrenner = false;
137
138 Integer maxAnzSpielerInSpalteFromKonfig = getKonfigurationSheet().getMaxAnzSpielerInSpalte();
139
140 for (Spieler spieler : meldungen.getSpielerList()) {
141
142 if (insertTrenner) {
143 spierNrVal.spalte((spielerCntr / maxAnzSpielerInSpalteFromKonfig) * (ANZ_SPALTEN_IN_BLOCK + 1))
144 .zeile(ERSTE_DATEN_ZEILE);
145 spalteFormat(spierNrVal.getPos());
146 blockHeader(spierNrVal.getPos());
147 insertTrenner = false;
148
149 }
150
151 int spielerNrSpalte = spierNrVal.getPos().getSpalte();
152 spierNrVal.setValue((double) spieler.getNr());
153 getSheetHelper().setNumberValueInCell(spierNrVal);
154
155 // spieler name
156 nameFormula.setValue(meldeliste.formulaSverweisSpielernamen(spierNrVal.getPos().getAddress()));
157 nameFormula.spalte(spielerNrSpalte).spaltePlus(SPIELER_NAME_OFFS_SPALTE);
158 nameFormula.zeile(spierNrVal.getPos().getZeile());
159 getSheetHelper().setFormulaInCell(nameFormula);
160
161 strVal.spalte(spielerNrSpalte).spaltePlus(SPIELER_TEAM_OFFS_SPALTE).zeile(spierNrVal.getPos().getZeile());
162 getSheetHelper().setFormulaInCell(strVal.setValue(sVerweisSpielrundeSpalte(spierNrVal.getPos(),
163 SpielrundeSheetKonstanten.SPALTE_VERTIKALE_ERGEBNISSE_AB, 4)));
164
165 getSheetHelper().setFormulaInCell(strVal.spalte(spielerNrSpalte).spaltePlus(SPIELER_BAHN_NR_OFFS_SPALTE)
166 .setValue(sVerweisSpielrundeSpalte(spierNrVal.getPos(),
167 SpielrundeSheetKonstanten.SPALTE_VERTIKALE_ERGEBNISSE_BA_NR, 5)));
168
169 spierNrVal.zeilePlusEins();
170
171 // Block Trenner
172 if ((spielerCntr / maxAnzSpielerInSpalteFromKonfig) * maxAnzSpielerInSpalteFromKonfig == spielerCntr) {
173 insertTrenner = true;
174 }
175
176 spielerCntr++;
177 if (maxAnzSpielerInSpalte < maxAnzSpielerInSpalteFromKonfig) {
178 maxAnzSpielerInSpalte++;
179 }
180 }
181
182 Position rechtsUnten = Position.from(spierNrVal.getPos().getSpalte() + 3,
183 ERSTE_DATEN_ZEILE + (maxAnzSpielerInSpalte - 1));
184 // Header Spielrunde
185 StringCellValue spielrunde = StringCellValue.from(getXSpreadSheet(), SPIELER_NR_SPALTE, HEADER_ZEILE)
186 .setEndPosMergeSpalte(rechtsUnten.getSpalte()).setValue("Spielrunde " + getSpielRundeNr().getNr())
187 .centerHoriJustify().centerVertJustify().setCharHeight(14);
188 getSheetHelper().setStringValueInCell(spielrunde);
189 printBereichDefinieren(rechtsUnten);
190 }

199 private void blockHeader(final Position ersteSpielerNr) throws GenerateException {
200 Position ersteHeader = Position.from(ersteSpielerNr).zeile(HEADER_ZEILE_2);
201 int spielerSpalte = ersteSpielerNr.getSpalte();
202 StringCellValue header = StringCellValue.from(getXSpreadSheet(), ersteHeader).setCharWeight(FontWeight.BOLD)
203 .setCharHeight(8).setShrinkToFit(true);
204 getSheetHelper().setStringValueInCell(header.setValue("Nr."));
205 getSheetHelper().setStringValueInCell(
206 header.spalte(spielerSpalte).spaltePlus(SPIELER_NAME_OFFS_SPALTE).setValue("Name"));
207 getSheetHelper().setStringValueInCell(
208 header.spalte(spielerSpalte).spaltePlus(SPIELER_TEAM_OFFS_SPALTE).setValue("Team"));
209 getSheetHelper().setStringValueInCell(
210 header.spalte(spielerSpalte).spaltePlus(SPIELER_BAHN_NR_OFFS_SPALTE).setValue(I18n.get("column.header.bahn")));
211 }

213 private String sVerweisSpielrundeSpalte(Position spielerNr, int letzteSpalte, int idx) throws GenerateException {
214 String spielRundeSheetName = "$'" + aktuelleSpielrundeSheet.getSheetName(getSpielTag(), getSpielRundeNr())
215 + "'.";
216 String verweisAufSpalteSpielerNr = "INDIRECT(ADDRESS(ROW();" + (spielerNr.getSpalte() + 1) + ";4))";
217
218 int ersteSpalteVertikaleErgebnisse = SpielrundeSheetKonstanten.ERSTE_SPALTE_VERTIKALE_ERGEBNISSE;
219 int spielrundeSheetErsteDatenzeile = SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE;
220 Position erstePos = Position.from(ersteSpalteVertikaleErgebnisse, spielrundeSheetErsteDatenzeile);
221 Position letztePos = Position.from(letzteSpalte, 1000 + spielrundeSheetErsteDatenzeile);
222 String suchMatrixTeam = erstePos.getAddressWith$() + ":" + letztePos.getAddressWith$();
223 String formula = "VLOOKUP(" + verweisAufSpalteSpielerNr + ";" + spielRundeSheetName + suchMatrixTeam + ";" + idx
224 + ";0)";
225 return formula;
226 }
