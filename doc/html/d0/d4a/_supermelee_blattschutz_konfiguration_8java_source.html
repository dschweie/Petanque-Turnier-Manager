59 @Override
60 public void zelleStylesAktualisieren(WorkingSpreadsheet ws) {
61 var doc = ws.getWorkingSpreadsheetDocument();
62 // Styles direkt auf Dokumentebene erzeugen/aktualisieren – unabhängig davon,
63 // ob die Meldeliste im Dokument gerade existiert.
64 CellStyleHelper.from(doc,
65 new EditierbareZelleHintergrundFarbeGeradeStyle(
66 EditierbaresZelleFormatHelper.EDITIERBAR_GERADE_FARBE)).apply();
67 CellStyleHelper.from(doc,
68 new EditierbareZelleHintergrundFarbeUnGeradeStyle(
69 EditierbaresZelleFormatHelper.EDITIERBAR_UNGERADE_FARBE)).apply();
70 }

72 @Override
73 public List<SheetSchutzInfo> berechneSchutzInfos(WorkingSpreadsheet ws) {
74 var xDoc = ws.getWorkingSpreadsheetDocument();
75 var infos = new ArrayList<SheetSchutzInfo>();
76
77 sammleMeldelisteSchutzInfo(xDoc, infos);
78 sammleSpielrundenSchutzInfos(xDoc, infos);
79 sammleVollGesperrteSheets(xDoc, infos);
80
81 return infos;
82 }

87 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc,
88 List<SheetSchutzInfo> infos) {
89 SheetMetadataHelper.findeSheet(xDoc,
90 SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_MELDELISTE).ifPresent(sheet -> {
91 var bereiche = berechneMeldelisteBereiche(xDoc);
92 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, bereiche));
93 });
94 }
95
97 private void sammleSpielrundenSchutzInfos(XSpreadsheetDocument xDoc,
98 List<SheetSchutzInfo> infos) {
99 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
100 SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_SPIELRUNDE_PREFIX);
101 for (var key : schluessel) {
102 SheetMetadataHelper.findeSheet(xDoc, key).ifPresent(sheet ->
103 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet,
104 List.of(berechneSpielrundeErgebnisBereich(sheet)))));
105 }
106 }
107
109 private void sammleVollGesperrteSheets(XSpreadsheetDocument xDoc,
110 List<SheetSchutzInfo> infos) {
111 fuegeVollgesperrteAusPrefix(xDoc, SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_SPIELRUNDE_PLAN_PREFIX, infos);
112 fuegeVollgesperrteAusPrefix(xDoc, SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_ANMELDUNGEN_PREFIX, infos);
113 fuegeVollgesperrteAusPrefix(xDoc, SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_SPIELTAG_TEILNEHMER_PREFIX, infos);
114 fuegeVollgesperrteAusPrefix(xDoc, SheetMetadataHelper.SCHLUESSEL_SPIELTAG_RANGLISTE_PREFIX, infos);
115
116 SheetMetadataHelper.findeSheet(xDoc,
117 SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_ENDRANGLISTE)
118 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
119 SheetMetadataHelper.findeSheet(xDoc,
120 SheetMetadataHelper.SCHLUESSEL_SUPERMELEE_TEAMS)
121 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
122 }
123
124 private void fuegeVollgesperrteAusPrefix(XSpreadsheetDocument xDoc, String prefix,
125 List<SheetSchutzInfo> infos) {
126 for (var key : SheetMetadataHelper.getSchluesselMitPrefix(xDoc, prefix)) {
127 SheetMetadataHelper.findeSheet(xDoc, key)
128 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
129 }
130 }

146 private List<RangePosition> berechneMeldelisteBereiche(XSpreadsheetDocument xDoc) {
147 int ersteDatenZeile = MeldeListeKonstanten.ERSTE_DATEN_ZEILE;
148 int maxZeile = MeldungenSpalte.MAX_ANZ_MELDUNGEN;
149
150 var bereiche = new ArrayList<RangePosition>();
151 bereiche.add(RangePosition.from(1, ersteDatenZeile, 1, maxZeile)); // Name
152 bereiche.add(RangePosition.from(2, ersteDatenZeile, 2, maxZeile)); // SP
153
154 int anzSpieltage = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
155 SheetMetadataHelper.SCHLUESSEL_SPIELTAG_RANGLISTE_PREFIX).length;
156 if (anzSpieltage > 0) {
157 // Spieltage: Spalte 3 bis 2+anzSpieltage (inklusiv)
158 bereiche.add(RangePosition.from(3, ersteDatenZeile, 2 + anzSpieltage, maxZeile));
159 }
160 return bereiche;
161 }
162
172 private RangePosition berechneSpielrundeErgebnisBereich(XSpreadsheet sheet) {
173 return RangePosition.from(
174 SpielrundeSheetKonstanten.ERSTE_SPALTE_ERGEBNISSE,
175 SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE,
176 SpielrundeSheetKonstanten.ERSTE_SPALTE_ERGEBNISSE + 1,
177 ermittleLetzteSpielrundeZeile(sheet));
178 }
179
185 private int ermittleLetzteSpielrundeZeile(XSpreadsheet sheet) {
186 int letzteZeile = SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE;
187 try {
188 for (int zeile = SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE;
189 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
190 XCell xCell = sheet.getCellByPosition(SpielrundeSheetKonstanten.PAARUNG_CNTR_SPALTE, zeile);
191 if (CellContentType.EMPTY.equals(xCell.getType())) {
192 break;
193 }
194 letzteZeile = zeile;
195 }
196 } catch (Exception e) {
197 logger.warn("Letzte Spielrunde-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
198 }
199 return letzteZeile;
200 }
