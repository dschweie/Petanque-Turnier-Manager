114 private Map<String, List<Integer>> teamPositionenBerechnen(int naechsteRundeNr, KaskadenKoRundenPlan plan)
115 throws GenerateException {
116 Map<String, List<Integer>> aktuelleMap = new HashMap<>();
117 Set<Integer> bereitsFreilosTeams = new HashSet<>();
118
119 for (int rundenNr = 1; rundenNr < naechsteRundeNr; rundenNr++) {
120 var rundenInfo = plan.kaskadeRunden().get(rundenNr - 1);
121 var sheetDaten = leseRundenSheetDaten(rundenNr);
122 var naechsteMap = new HashMap<String, List<Integer>>();
123 int zeilenOffset = 0;
124
125 for (var gruppe : rundenInfo.gruppenRunden()) {
126 var pfad = gruppe.pfad();
127 var teamsInGruppe = (rundenNr == 1)
128 ? leseTeamNrsAusGruppe(sheetDaten, zeilenOffset, gruppe)
129 : aktuelleMap.getOrDefault(pfad, Collections.emptyList());
130
131 List<Integer> sieger = new ArrayList<>();
132 List<Integer> verlierer = new ArrayList<>();
133
134 for (int i = 0; i < gruppe.spielPaare().size(); i++) {
135 var spiel = gruppe.spielPaare().get(i);
136 int teamA = spiel.positionA() <= teamsInGruppe.size()
137 ? teamsInGruppe.get(spiel.positionA() - 1) : 0;
138 int teamB = spiel.positionB() <= teamsInGruppe.size()
139 ? teamsInGruppe.get(spiel.positionB() - 1) : 0;
140
141 var row = sheetDaten.get(zeilenOffset + i);
142 int ergA = row.size() > 2 ? row.get(2).getIntVal(0) : 0;
143 int ergB = row.size() > 3 ? row.get(3).getIntVal(0) : 0;
144
145 if (ergA >= ergB) {
146 sieger.add(teamA);
147 verlierer.add(teamB);
148 } else {
149 sieger.add(teamB);
150 verlierer.add(teamA);
151 }
152 }
153
154 if (gruppe.anzFreilose() > 0) {
155 int freilosPos = gruppe.anzTeams() - 1;
156 int freilosTeam = freilosPos < teamsInGruppe.size()
157 ? teamsInGruppe.get(freilosPos) : 0;
158
159 var freilosRow = sheetDaten.get(zeilenOffset + gruppe.spielPaare().size());
160 int freilosErgA = freilosRow.size() > 2 ? freilosRow.get(2).getIntVal(0) : 0;
161 int freilosErgB = freilosRow.size() > 3 ? freilosRow.get(3).getIntVal(0) : 0;
162
163 if (freilosTeam > 0) {
164 bereitsFreilosTeams.add(freilosTeam);
165 }
166 if (freilosErgA > freilosErgB) {
167 sieger.add(freilosTeam);
168 } else {
169 verlierer.add(freilosTeam);
170 }
171 }
172
173 sieger = verhindereFreilosDuplikat(sieger, bereitsFreilosTeams);
174 verlierer = verhindereFreilosDuplikat(verlierer, bereitsFreilosTeams);
175
176 naechsteMap.put(pfad + "S", Collections.unmodifiableList(sieger));
177 naechsteMap.put(pfad + "V", Collections.unmodifiableList(verlierer));
178 zeilenOffset += gruppe.spielPaare().size() + gruppe.anzFreilose();
179 }
180
181 aktuelleMap = naechsteMap;
182 }
183
184 return aktuelleMap;
185 }

213 private static List<Integer> leseTeamNrsAusGruppe(RangeData sheetDaten, int zeilenOffset,
214 KaskadenKoGruppenRunde gruppe) {
215 var teams = new ArrayList<Integer>(gruppe.anzTeams());
216 for (int i = 0; i < gruppe.spielPaare().size(); i++) {
217 var row = sheetDaten.get(zeilenOffset + i);
218 teams.add(row.size() > 0 ? row.get(0).getIntVal(0) : 0); // Team A
219 teams.add(row.size() > 1 ? row.get(1).getIntVal(0) : 0); // Team B
220 }
221 if (gruppe.anzFreilose() > 0) {
222 var freilosRow = sheetDaten.get(zeilenOffset + gruppe.spielPaare().size());
223 teams.add(freilosRow.size() > 0 ? freilosRow.get(0).getIntVal(0) : 0);
224 }
225 return teams;
226 }

239 private static List<Integer> verhindereFreilosDuplikat(
240 List<Integer> teams, Set<Integer> bereitsFreilosTeams) {
241 if (teams.size() % 2 == 0) {
242 return teams;
243 }
244 int letzteIdx = teams.size() - 1;
245 if (!bereitsFreilosTeams.contains(teams.get(letzteIdx))) {
246 return teams;
247 }
248 for (int i = letzteIdx - 1; i >= 0; i--) {
249 if (!bereitsFreilosTeams.contains(teams.get(i))) {
250 var ergebnis = new ArrayList<>(teams);
251 Collections.swap(ergebnis, i, letzteIdx);
252 return ergebnis;
253 }
254 }
255 LOGGER.warn("Alle Teams in Gruppe hatten bereits ein Freilos. Kein Tausch möglich.");
256 return teams;
257 }
