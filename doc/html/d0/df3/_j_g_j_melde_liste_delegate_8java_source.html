274 void formatDatenSpalten() throws GenerateException {
275 Formation formation = konfigurationSheet.getMeldeListeFormation();
276 int anzSpieler = formation.getAnzSpieler();
277 int letzteDatenZeile = getLetzteDatenZeileUseMin();
278
279 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
280 getTeamNrSpalte(), letzteDatenZeile);
281 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), nrRange,
282 CellProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
283
284 String kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
285 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
286 ConditionalFormatHelper.from(sheet, nrRange).clear()
287 .formulaIsText().styleIsFehler().applyAndDoReset()
288 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
289 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
290 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset();
291
292 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
293 RangePosition teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, 1, letzteDatenZeile);
294 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), teamnameRange,
295 CellProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()).setShrinkToFit(true));
296 }
297
298 for (int s = 0; s < anzSpieler; s++) {
299 int ersteSpielSpalte = getVornameSpalte(s);
300 int letzteSpielSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
301 ? getVereinsnameSpalte(s)
302 : getNachnameSpalte(s);
303 RangePosition spielerRange = RangePosition.from(ersteSpielSpalte, ERSTE_DATEN_ZEILE,
304 letzteSpielSpalte, letzteDatenZeile);
305 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), spielerRange,
306 CellProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()).setShrinkToFit(true));
307 }
308
309 RangePosition spRange = RangePosition.from(getSetzPositionSpalte(), ERSTE_DATEN_ZEILE,
310 getSetzPositionSpalte(), letzteDatenZeile);
311 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), spRange,
312 CellProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
313
314 RangePosition aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE,
315 getAktivSpalte(), letzteDatenZeile);
316 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), aktivRange,
317 CellProperties.from().centerJustify().setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
318
319 RangePosition editierbareRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, getAktivSpalte(), letzteDatenZeile);
320 EditierbaresZelleFormatHelper.anwenden(sheet, editierbareRange);
321 }

401 Map<Integer, String> leseTeamNamen() throws GenerateException {
402 Map<Integer, String> result = new HashMap<>();
403 XSpreadsheet mlSheet = sheet.getXSpreadSheet();
404 if (mlSheet == null) {
405 return result;
406 }
407 boolean zeigeTeamname = konfigurationSheet.isMeldeListeTeamnameAnzeigen();
408 boolean zeigeVerein = konfigurationSheet.isMeldeListeVereinsnameAnzeigen();
409 Formation formation = konfigurationSheet.getMeldeListeFormation();
410 int anzSpieler = formation.getAnzSpieler();
411 int ersterSpielerOffset = zeigeTeamname ? 2 : 1;
412 int spaltenProSpieler = zeigeVerein ? 3 : 2;
413 int maxSpalte = ersterSpielerOffset + anzSpieler * spaltenProSpieler - 1;
414
415 var xDoc = sheet.getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
416 RangeData data = RangeHelper.from(mlSheet, xDoc,
417 RangePosition.from(0, ERSTE_DATEN_ZEILE, maxSpalte, ERSTE_DATEN_ZEILE + 999)).getDataFromRange();
418
419 for (RowData row : data) {
420 if (row.isEmpty()) {
421 break;
422 }
423 int nr = row.get(0).getIntVal(0);
424 if (nr <= 0) {
425 break;
426 }
427 String name = zeigeTeamname
428 ? (row.size() > 1 ? row.get(1).getStringVal() : "")
429 : bauspielerNamenZusammen(row, anzSpieler, ersterSpielerOffset, spaltenProSpieler);
430 result.put(nr, name != null ? name : "");
431 }
432 return result;
433 }

435 private String bauspielerNamenZusammen(RowData row, int anzSpieler, int ersterSpielerOffset, int spaltenProSpieler) {
436 var sb = new StringBuilder();
437 for (int s = 0; s < anzSpieler; s++) {
438 int vorSpalte = ersterSpielerOffset + s * spaltenProSpieler;
439 int nachSpalte = vorSpalte + 1;
440 String vorname = vorSpalte < row.size() ? row.get(vorSpalte).getStringVal() : "";
441 String nachname = nachSpalte < row.size() ? row.get(nachSpalte).getStringVal() : "";
442 String spielerName = baueSpielerName(vorname, nachname);
443 if (!spielerName.isEmpty()) {
444 if (sb.length() > 0) {
445 sb.append(" / ");
446 }
447 sb.append(spielerName);
448 }
449 }
450 return sb.toString();
451 }

453 private static String baueSpielerName(String vorname, String nachname) {
454 String vn = vorname != null ? vorname.trim() : "";
455 String nn = nachname != null ? nachname.trim() : "";
456 if (vn.isEmpty() && nn.isEmpty()) {
457 return "";
458 }
459 if (vn.isEmpty()) {
460 return nn;
461 }
462 if (nn.isEmpty()) {
463 return vn;
464 }
465 return vn + " " + nn;
466 }

479 letzte = zeile;
480 }
481 }
482 return letzte;

604 private void bereinigeSpalte(XSpreadsheet xSheet, int spalte, int zeile) throws GenerateException {
605 if (spalte < 0) {
606 return;
607 }
608 String original = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(spalte, zeile));
609 if (original == null || original.isEmpty()) {
610 return;
611 }
612 String bereinigt = original.replaceAll("[\\p{Cntrl}]", "").strip();
613 if (!bereinigt.equals(original)) {
614 sheet.getSheetHelper().setStringValueInCell(StringCellValue.from(xSheet, Position.from(spalte, zeile), bereinigt));
