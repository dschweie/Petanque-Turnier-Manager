124 private void handleStatischOderRoot(HttpExchange exchange) throws IOException {
125 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
126 exchange.sendResponseHeaders(405, -1);
127 return;
128 }
129 String path = exchange.getRequestURI().getPath();
130 if ("/".equals(path) || path.isEmpty()) {
131 serviereRessource(exchange, "/index.html", CONTENT_TYPE_HTML);
132 } else if (path.startsWith("/assets/")) {
133 serviereRessource(exchange, "/assets/" + path.substring("/assets/".length()),
134 ermittleContentType(path));
135 } else if (path.startsWith("/images/")) {
136 serviereRessource(exchange, "/images/" + path.substring("/images/".length()),
137 ermittleContentType(path));
138 } else {
139 exchange.sendResponseHeaders(404, -1);
140 }
141 }

143 private void handleEvents(HttpExchange exchange) throws IOException {
144 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
145 exchange.sendResponseHeaders(405, -1);
146 return;
147 }
148 exchange.getResponseHeaders().set("Content-Type", CONTENT_TYPE_SSE);
149 exchange.getResponseHeaders().set("Cache-Control", "no-cache");
150 exchange.getResponseHeaders().set("Connection", "keep-alive");
151 exchange.getResponseHeaders().set("X-Accel-Buffering", "no");
152 exchange.getResponseHeaders().set("Access-Control-Allow-Origin", "*");
153 exchange.sendResponseHeaders(200, 0);
154
155 OutputStream os = exchange.getResponseBody();
156 var verbindung = new SseVerbindung(os, this);
157 sseVerbindungen.add(verbindung);
158 try {
159 os.write("retry: 3000\n\n".getBytes(StandardCharsets.UTF_8));
160 os.flush();
161 } catch (IOException e) {
162 verbindungEntfernen(verbindung);
163 return;
164 }
165 verbindung.sendeInitNachricht();
166 }

168 private void handleTurnierlogo(HttpExchange exchange) throws IOException {
169 if (!"GET".equalsIgnoreCase(exchange.getRequestMethod())) {
170 exchange.sendResponseHeaders(405, -1);
171 return;
172 }
173 String quelle = logoQuelle;
174 if (quelle.isEmpty()) {
175 exchange.sendResponseHeaders(404, -1);
176 return;
177 }
178 // Externe http(s)-URL: Weiterleiten an den Browser.
179 if (quelle.startsWith("http://") || quelle.startsWith("https://")) {
180 exchange.getResponseHeaders().set("Location", quelle);
181 exchange.sendResponseHeaders(302, -1);
182 return;
183 }
184 // Lokale Datei: file:// oder absoluter Pfad.
185 Path datei;
186 try {
187 datei = quelle.startsWith("file:")
188 ? Paths.get(URI.create(quelle))
189 : Paths.get(quelle);
190 } catch (IllegalArgumentException e) {
191 logger.warn("Ungültige Logo-Quelle: {}", quelle, e);
192 exchange.sendResponseHeaders(404, -1);
193 return;
194 }
195 if (!Files.isRegularFile(datei) || !Files.isReadable(datei)) {
196 logger.warn("Turnierlogo-Datei nicht lesbar: {}", datei);
197 exchange.sendResponseHeaders(404, -1);
198 return;
199 }
200 byte[] body = Files.readAllBytes(datei);
201 var headers = exchange.getResponseHeaders();
202 headers.set("Content-Type", ermittleContentType(datei.getFileName().toString()));
203 headers.set("Cache-Control", "no-cache");
204 headers.set("Access-Control-Allow-Origin", "*");
205 exchange.sendResponseHeaders(200, body.length);
206 try (OutputStream os = exchange.getResponseBody()) {
207 os.write(body);
208 }
209 }

211 private void serviereRessource(HttpExchange exchange, String relativerPfad, String contentType)
212 throws IOException {
213 String ressourcePfad = (STATIC_RESOURCE_PREFIX + relativerPfad).replaceFirst("^/", "");
214 InputStream gefunden = getClass().getClassLoader().getResourceAsStream(ressourcePfad);
215 if (gefunden == null) {
216 gefunden = getClass().getResourceAsStream("/" + ressourcePfad);
217 }
218 if (gefunden == null) {
219 logger.warn("Classpath-Ressource nicht gefunden: {}", ressourcePfad);
220 exchange.sendResponseHeaders(404, -1);
221 return;
222 }
223 try (InputStream in = gefunden) {
224 byte[] body = in.readAllBytes();
225 var headers = exchange.getResponseHeaders();
226 headers.set("Content-Type", contentType);
227 headers.set("Cache-Control", "no-cache");
228 headers.set("Access-Control-Allow-Origin", "*");
229 exchange.sendResponseHeaders(200, body.length);
230 try (OutputStream os = exchange.getResponseBody()) {
231 os.write(body);
232 }
233 }

236 private static String ermittleContentType(String dateiname) {
237 if (dateiname.endsWith(".js")) return "text/javascript; charset=UTF-8";
238 if (dateiname.endsWith(".css")) return "text/css; charset=UTF-8";
239 if (dateiname.endsWith(".html")) return CONTENT_TYPE_HTML;
240 if (dateiname.endsWith(".svg")) return "image/svg+xml";
241 if (dateiname.endsWith(".png")) return "image/png";
242 if (dateiname.endsWith(".ico")) return "image/x-icon";
243 return "application/octet-stream";
244 }
