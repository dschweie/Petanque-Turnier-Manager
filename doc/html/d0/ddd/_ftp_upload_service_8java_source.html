29 client.setControlEncoding("UTF-8");

31 if (!FTPReply.isPositiveCompletion(client.getReplyCode())) {
32 throw new IOException("FTP-Verbindung abgelehnt, Antwortcode: " + client.getReplyCode());
33 }
34 if (!client.login(konfiguration.benutzer(), passwort)) {
35 throw new IOException("FTP-Anmeldung fehlgeschlagen für Benutzer: " + konfiguration.benutzer());
36 }
37 client.enterLocalPassiveMode();
38 client.setFileType(FTP.BINARY_FILE_TYPE);
39
40 String remotePfad = normalisiereRemotePfad(konfiguration.remotePfad());
41 client.makeDirectory(remotePfad);
42
43 int anzahl = 0;
44 for (Path datei : dateien) {
45 Path dateiName = datei.getFileName();
46 if (dateiName == null) {
47 logger.warn("FTP: Datei ohne Namen übersprungen: {}", datei);
48 continue;
49 }
50 String ziel = remotePfad + "/" + dateiName;
51 try (InputStream is = Files.newInputStream(datei)) {
52 if (client.storeFile(ziel, is)) {
53 anzahl++;
54 logger.info("FTP hochgeladen: {}", ziel);
55 } else {
56 throw new IOException("FTP Upload nicht bestätigt: " + ziel
57 + " (Antwortcode: " + client.getReplyCode() + ")");
58 }
59 }
60 }
61 return anzahl;
62 } finally {
63 if (client.isConnected()) {
64 try {
65 client.logout();
66 } catch (IOException e) {
67 logger.debug("FTP-Logout-Fehler ignoriert", e);
68 }
