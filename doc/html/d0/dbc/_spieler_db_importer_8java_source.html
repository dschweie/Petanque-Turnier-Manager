55 public ImportErgebnis importiere(ValidierteDaten daten, ImportRequest request)
56 throws SpielerDbException {
57 Connection con = dbConn.getConnection();
58 ImportIdMapping mapping = new ImportIdMapping();
59 Counter counter = new Counter();
60 List<ImportWarnung> warnungen = new ArrayList<>(daten.warnungen());
61
62 boolean vorigerAutoCommit = true;
63 try {
64 vorigerAutoCommit = con.getAutoCommit();
65 con.setAutoCommit(false);
66
67 verarbeiteVereine(con, daten.vereine(), mapping, counter, request.modus());
68 verarbeiteLabels(con, daten.labels(), mapping, counter, request.modus());
69 verarbeiteSpieler(con, daten.spieler(), mapping, counter, warnungen, request.modus());
70 verarbeiteJunction(con, daten.spielerLabels(), mapping, counter, request.modus());
71
72 if (request.dryRun()) {
73 con.rollback();
74 } else {
75 con.commit();
76 }
77 return counter.zuErgebnis(request.dryRun(), warnungen);
78 } catch (SQLException e) {
79 try {
80 con.rollback();
81 } catch (SQLException rb) {
82 logger.warn("Rollback fehlgeschlagen", rb);
83 }
84 throw new SpielerDbException("Import fehlgeschlagen", e);
85 } finally {
86 try {
87 con.setAutoCommit(vorigerAutoCommit);
88 } catch (SQLException e) {
89 logger.warn("AutoCommit-Wiederherstellung fehlgeschlagen", e);
90 }
91 }
92 }

213 private static void verarbeiteSpieler(Connection con, List<ValSpieler> spieler,
214 ImportIdMapping mapping, Counter counter, List<ImportWarnung> warnungen,
215 ImportModus modus) throws SQLException {
216 for (ValSpieler s : spieler) {
217 Integer neueVereinNr = mapping.neueVereinNr(s.altVereinNr());
218 Optional<DbSpieler> bestehend = findeSpieler(con, s, neueVereinNr);
219
220 if (bestehend.isEmpty()) {
221 int neueNr = insertSpieler(con, s.vorname(), s.nachname(),
222 neueVereinNr, s.lizenznr());
223 counter.spielerEingefuegt++;
224 mapping.merkeSpieler(s.altNr(), neueNr);
225 continue;
226 }
227
228 DbSpieler db = bestehend.get();
229 switch (modus) {
230 case NUR_NEUE -> {
231 counter.spielerUebersprungen++;
232 mapping.merkeSpieler(s.altNr(), db.nr());
233 }
234 case AKTUALISIEREN -> {
235 aktualisiereSpielerMerge(con, db, s, neueVereinNr, warnungen);
236 counter.spielerAktualisiert++;
237 mapping.merkeSpieler(s.altNr(), db.nr());
238 }
239 case DUPLIKATE_SEPARAT -> {
240 Optional<Integer> neu = versucheInsertSpieler(con, s.vorname(), s.nachname(),
241 neueVereinNr, s.lizenznr(), s.quellZeile(), warnungen);
242 if (neu.isPresent()) {
243 counter.spielerEingefuegt++;
244 mapping.merkeSpieler(s.altNr(), neu.get());
245 } else {
246 counter.spielerUebersprungen++;
247 }
248 }
249 }
250 }

275 private static Optional<DbSpieler> findeSpielerByNameUndVerein(Connection con,
276 String vorname, String nachname, @Nullable Integer vereinNr) throws SQLException {
277 // Suche per normalisiertem Schlüssel; SQL-LOWER + Java-Whitespace-Collapse
278 // auf Datenbankseite ist nicht trivial — daher load-and-compare.
279 String sql = vereinNr == null
280 ? "SELECT NR, VORNAME, NACHNAME, VEREIN_NR, LIZENZNR FROM SPIELER "
281 + "WHERE VEREIN_NR IS NULL"
282 : "SELECT NR, VORNAME, NACHNAME, VEREIN_NR, LIZENZNR FROM SPIELER "
283 + "WHERE VEREIN_NR = ?";
284 String suchKey = SpielerMatchKeyNormalizer.spielerSchluesselMitVereinNr(
285 vorname, nachname, vereinNr);
286 try (PreparedStatement ps = con.prepareStatement(sql)) {
287 if (vereinNr != null) {
288 ps.setInt(1, vereinNr);
289 }
290 try (ResultSet rs = ps.executeQuery()) {
291 while (rs.next()) {
292 DbSpieler db = mapDb(rs);
293 String dbKey = SpielerMatchKeyNormalizer.spielerSchluesselMitVereinNr(
294 db.vorname(), db.nachname(), db.vereinNr());
295 if (dbKey.equals(suchKey)) {
296 return Optional.of(db);
297 }
298 }
299 }
300 }
301 return Optional.empty();
302 }

362 private static void aktualisiereSpielerMerge(Connection con, DbSpieler db, ValSpieler s,
363 @Nullable Integer neueVereinNr, List<ImportWarnung> warnungen) throws SQLException {
364 // Feldweise:
365 // - Vorname/Nachname/Verein: überschreiben
366 // - Lizenznr: nur setzen, wenn Ziel leer
367 @Nullable String neueLizenz = db.lizenznr();
368 if ((neueLizenz == null || neueLizenz.isEmpty())
369 && s.lizenznr() != null && !s.lizenznr().isEmpty()) {
370 neueLizenz = s.lizenznr();
371 } else if (s.lizenznr() != null && !s.lizenznr().isEmpty()
372 && !s.lizenznr().equals(db.lizenznr())) {
373 String praefix = s.quellZeile() != null ? "Zeile " + s.quellZeile() + ": " : "";
374 warnungen.add(new ImportWarnung(praefix
375 + "Lizenznummer für '" + s.vorname() + " " + s.nachname()
376 + "' bleibt erhalten (DB: '" + db.lizenznr()
377 + "', Import: '" + s.lizenznr() + "')"));
378 }
379
380 try (PreparedStatement ps = con.prepareStatement(
381 "UPDATE SPIELER SET VORNAME = ?, NACHNAME = ?, VEREIN_NR = ?, LIZENZNR = ? "
382 + "WHERE NR = ?")) {
383 ps.setString(1, s.vorname());
384 ps.setString(2, s.nachname());
385 if (neueVereinNr == null) {
386 ps.setNull(3, Types.INTEGER);
387 } else {
388 ps.setInt(3, neueVereinNr);
389 }
390 if (neueLizenz == null || neueLizenz.isEmpty()) {
391 ps.setNull(4, Types.VARCHAR);
392 } else {
393 ps.setString(4, neueLizenz);
394 }
395 ps.setInt(5, db.nr());
396 ps.executeUpdate();
397 }

402 private static void verarbeiteJunction(Connection con, List<ValSpielerLabel> junction,
403 ImportIdMapping mapping, Counter counter, ImportModus modus) throws SQLException {
404 // Bei NUR_NEUE und DUPLIKATE_SEPARAT könnte man Junction-Inserts überspringen
405 // für übersprungene Spieler — aber das mapping liefert dann eh null.
406 // INSERT OR IGNORE macht doppelte Einträge zum No-Op (Merge-Semantik).
407 Map<Integer, Integer> inserted = new HashMap<>();
408 try (PreparedStatement ps = con.prepareStatement(
409 "INSERT OR IGNORE INTO SPIELER_LABEL (SPIELER_NR, LABEL_NR) VALUES (?, ?)")) {
410 for (ValSpielerLabel j : junction) {
411 Integer sNr = mapping.neueSpielerNr(j.altSpielerNr());
412 Integer lNr = mapping.neueLabelNr(j.altLabelNr());
413 if (sNr == null || lNr == null) {
414 continue;
415 }
416 ps.setInt(1, sNr);
417 ps.setInt(2, lNr);
418 int n = ps.executeUpdate();
419 if (n > 0) {
420 counter.junctionEingefuegt++;
421 inserted.merge(sNr, 1, Integer::sum);
422 }
423 }
424 }
425 // Stille Verwendung von modus — vermeidet "unused parameter"-Warnungen
426 // und dokumentiert, dass das aktuelle Verhalten modus-unabhängig ist.
427 logger.debug("Junction-Phase abgeschlossen, Modus={}, neue Einträge={}",
428 modus, inserted.size());
429 }
