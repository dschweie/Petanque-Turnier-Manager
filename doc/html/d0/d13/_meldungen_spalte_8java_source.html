76 MeldungenSpalte(int ersteDatenZiele, int spielerNrSpalte, ISheet iSheet, Formation formation, int anzZeilenInHeader,
77 int spalteMeldungNameWidth, int minAnzZeilen, int ersteMeldungNameSpalteOffset,
78 Integer anzNamenSpaltenOverride, List<String> headerKeysProSpielerOverride) {
79 checkNotNull(iSheet);
80 checkArgument(ersteDatenZiele > -1);
81 checkArgument(spielerNrSpalte > -1);
82 checkArgument(anzZeilenInHeader > 0);
83 checkNotNull(formation);
84 checkArgument(spalteMeldungNameWidth > 1);
85
86 this.formation = checkNotNull(formation);
87 this.ersteDatenZiele = ersteDatenZiele;
88 this.anzZeilenInHeader = anzZeilenInHeader;
89 this.spalteMeldungNameWidth = spalteMeldungNameWidth;
90 this.meldungNrSpalte = spielerNrSpalte;
91 this.ersteMeldungNameSpalte = spielerNrSpalte + ersteMeldungNameSpalteOffset;
92 this.anzNamenSpalten = (anzNamenSpaltenOverride != null && anzNamenSpaltenOverride > 0)
93 ? anzNamenSpaltenOverride
94 : this.formation.getAnzSpieler();
95 this.letzteMeldungNameSpalte = this.ersteMeldungNameSpalte + this.anzNamenSpalten - 1;
96 if (headerKeysProSpielerOverride != null && !headerKeysProSpielerOverride.isEmpty()) {
97 checkArgument(headerKeysProSpielerOverride.size() == this.anzNamenSpalten,
98 "headerKeysProSpieler.size() muss anzNamenSpalten entsprechen");
99 this.headerKeysProSpieler = List.copyOf(headerKeysProSpielerOverride);
100 } else {
101 this.headerKeysProSpieler = Collections.nCopies(this.anzNamenSpalten, HEADER_SPIELER_NAME_KEY);
102 }
103 this.iSheet = iSheet;
104 this.minAnzZeilen = minAnzZeilen;
105 }

161 getISheet().processBoxinfo("processbox.meldeliste.spalten.header");

163 ColumnProperties columnProperties = ColumnProperties.from().setWidth(DEFAULT_SPALTE_NUMBER_WIDTH)
164 .setHoriJustify(CellHoriJustify.CENTER).setVertJustify(CellVertJustify2.CENTER)
165 .margin(MeldeListeKonstanten.CELL_MARGIN);

169 .setComment(I18n.get("meldeliste.comment.meldenummer")).addColumnProperties(columnProperties)
170 .setBorder(BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()).setCellBackColor(headerColor)
171 .setVertJustify(CellVertJustify2.CENTER).setShrinkToFit(true);
172
173 if (anzZeilenInHeader > 1) {
174 celVal.setEndPosMergeZeilePlus(1);
175 }
176 getSheetHelper().setStringValueInCell(celVal); // spieler nr
177 // --------------------------------------------------------------------------------------------
178
179 celVal.addColumnProperties(columnProperties.setWidth(spalteMeldungNameWidth)).setComment(null)
180 .spalte(ersteMeldungNameSpalte)
181 .setBorder(BorderFactory.from().allThin().boldLn().forTop().toBorder()).setCellBackColor(headerColor);
182
183 if (anzZeilenInHeader > 1) {
184 // weil spalte sich geändert hat
185 celVal.setEndPosMergeZeilePlus(1);
186 }
187
188 for (int anzSpieler = 0; anzSpieler < getAnzahlSpielerNamenSpalten(); anzSpieler++) {
189 celVal.setValue(I18n.get(headerKeysProSpieler.get(anzSpieler)));
190 getSheetHelper().setStringValueInCell(celVal);
191 celVal.spaltePlusEins();
192 }

285 public int getSpielerZeileNr(int spielerNr) throws GenerateException {
286 checkArgument(spielerNr > 0);
287 // muss in komplette spalte wert stehen. Deswegen mit ^ und $
288 Position result = RangeSearchHelper
289 .from(getISheet(),
290 RangePosition.from(meldungNrSpalte, getErsteDatenZiele(), meldungNrSpalte, MAX_ANZ_MELDUNGEN))
291 .searchNachRegExprInSpalte("^" + spielerNr + "$");
292 if (result != null) {
293 // Validieren !
294 Integer intFromCell = getSheetHelper().getIntFromCell(getXSpreadsheet(), result);
295 if (intFromCell != spielerNr) {
296 throw new GenerateException(I18n.get("error.spieler.suchen", spielerNr, result));
297 }
298 return result.getZeile();
299 }
300 return -1;

315 public void alleSpielerNrEinfuegen(Collection<Integer> spielerNummerList,
316 IMeldeliste<MLD_LIST_TYPE, MLDTYPE> meldeliste) throws GenerateException {
317 checkNotNull(meldeliste);
318 checkNotNull(spielerNummerList);
319
320 if (spielerNummerList.isEmpty()) {
321 return; // nichts zu tun
322 }
323
324 RangeData spielrNrData = new RangeData();
325 spielerNummerList.stream().forEachOrdered(nr -> {
326 spielrNrData.addNewRow().newInt(nr);
327 });
328
329 Position startPosNr = Position.from(meldungNrSpalte, getErsteDatenZiele());
330 RangeHelper.from(getISheet(), spielrNrData.getRangePosition(startPosNr)).setDataInRange(spielrNrData);
331
332 // filldown formula fuer name
333 String verweisAufMeldeListeFormula = meldeliste.formulaSverweisSpielernamen("INDIRECT(ADDRESS(ROW();1;4))");
334 StringCellValue strCelValSpielerName = StringCellValue.from(getXSpreadsheet(),
335 Position.from(meldungNrSpalte, getErsteDatenZiele()));
336 getSheetHelper().setFormulaInCell(strCelValSpielerName.spaltePlusEins().setValue(verweisAufMeldeListeFormula)
337 .setFillAutoDown(getLetzteMitDatenZeileInSpielerNrSpalte()));

356 public List<Integer> getSpielerNrList() throws GenerateException {
357 List<Integer> spielerNrList = new ArrayList<>();
358 int letzteZeile = getLetzteMitDatenZeileInSpielerNrSpalte();
359
360 if (letzteZeile >= ersteDatenZiele) {
361 RangePosition spielNrRange = RangePosition.from(meldungNrSpalte, ersteDatenZiele, meldungNrSpalte,
362 letzteZeile);
363 RangeData dataFromRange = RangeHelper.from(getISheet(), spielNrRange).getDataFromRange();
364 for (RowData zeile : dataFromRange) {
365 int spielerNr = zeile.get(0).getIntVal(-1);
366 if (spielerNr < 1) {
367 break; // fertig
368 }
369 spielerNrList.add(spielerNr);
370 }
371 }
372 java.util.Collections.sort(spielerNrList);
373 return spielerNrList;

381 public List<String> getSpielerNamenList() throws GenerateException {
382 List<String> spielerNamen = new ArrayList<>();
383 int letzteZeile = getLetzteMitDatenZeileInSpielerNrSpalte();
384 XSpreadsheet sheet = getXSpreadsheet();
385
386 if (letzteZeile >= ersteDatenZiele) {
387 for (int spielerZeile = ersteDatenZiele; spielerZeile <= letzteZeile; spielerZeile++) {
388 String spielerName = leseSpielerNameZeile(sheet, spielerZeile);
389 if (StringUtils.isNotBlank(spielerName)) {
390 spielerNamen.add(spielerName);
391 }
392 }
393 }
394
395 java.util.Collections.sort(spielerNamen);
396
397 return spielerNamen;

404 public String leseSpielerNameZeile(XSpreadsheet sheet, int zeile) throws GenerateException {
405 if (anzNamenSpalten == 1) {
406 return StringUtils.trimToEmpty(
407 getSheetHelper().getTextFromCell(sheet, Position.from(ersteMeldungNameSpalte, zeile)));
408 }
409 if (anzNamenSpalten == 2) {
410 String vorname = StringUtils.trimToEmpty(
411 getSheetHelper().getTextFromCell(sheet, Position.from(ersteMeldungNameSpalte, zeile)));
412 String nachname = StringUtils.trimToEmpty(
413 getSheetHelper().getTextFromCell(sheet, Position.from(letzteMeldungNameSpalte, zeile)));
414 if (nachname.isEmpty()) {
415 return vorname;
416 }
417 if (vorname.isEmpty()) {
418 return nachname;
419 }
420 return nachname + ", " + vorname;
421 }
422 StringBuilder sb = new StringBuilder();
423 for (int sp = ersteMeldungNameSpalte; sp <= letzteMeldungNameSpalte; sp++) {
424 String wert = StringUtils.trimToEmpty(
425 getSheetHelper().getTextFromCell(sheet, Position.from(sp, zeile)));
426 if (!wert.isEmpty()) {
427 if (sb.length() > 0) {
428 sb.append(' ');
429 }
430 sb.append(wert);
431 }
432 }
433 return sb.toString();
434 }
