185 public boolean vollstaendigAktualisieren() throws GenerateException {
186 XSpreadsheet xSheet = getXSpreadSheet();
187 if (xSheet == null) {
188 logger.warn("Schweizer Meldeliste nicht gefunden");
189 return false;
190 }
191 stringsBesinigen(xSheet);
192 teamnummernVergeben(xSheet);
193 nachTeamNrSortieren(xSheet);
194 pruefeAufDoppelteTeamNr(xSheet);
195 upDateSheet();
196 return true;
197 }

199private void stringsBesinigen(XSpreadsheet xSheet) throws GenerateException {
200 Formation formation = getKonfigurationSheet().getMeldeListeFormation();
201 boolean teamnameAktiv = getKonfigurationSheet().isMeldeListeTeamnameAnzeigen();
202 boolean vereinsnameAktiv = getKonfigurationSheet().isMeldeListeVereinsnameAnzeigen();
203 int letzteZeile = letzteZeileMitDaten(xSheet) + MIN_ANZAHL_MELDUNGEN_ZEILEN;
204 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
205 if (teamnameAktiv) {
206 bereinigeSpalte(xSheet, getTeamnameSpalte(), zeile);
207 }
208 for (int s = 0; s < formation.getAnzSpieler(); s++) {
209 bereinigeSpalte(xSheet, getVornameSpalte(s), zeile);
210 bereinigeSpalte(xSheet, getNachnameSpalte(s), zeile);
211 if (vereinsnameAktiv) {
212 bereinigeSpalte(xSheet, getVereinsnameSpalte(s), zeile);
213 }
214 }
215 }
216 }

218 private void bereinigeSpalte(XSpreadsheet xSheet, int spalte, int zeile) throws GenerateException {
219 if (spalte < 0) {
220 return;
221 }
222 String original = getSheetHelper().getTextFromCell(xSheet, Position.from(spalte, zeile));
223 if (original == null || original.isEmpty()) {
224 return;
225 }
226 String bereinigt = original.replaceAll("[\\p{Cntrl}]", "").strip();
227 if (!bereinigt.equals(original)) {
228 getSheetHelper().setStringValueInCell(StringCellValue.from(xSheet, Position.from(spalte, zeile), bereinigt));
229 }
230 }
231
232 private void teamnummernVergeben(XSpreadsheet xSheet) throws GenerateException {
233 int letzteZeile = letzteZeileMitDaten(xSheet);
234 if (letzteZeile < ERSTE_DATEN_ZEILE) {
235 return;
236 }
237 int vornameSpalte = getVornameSpalte(0);
238 // Nr-Spalte absteigend sortieren (höchste Nr zuerst, leere ans Ende)
239 RangePosition range = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
240 getSetzPositionSpalte(), letzteZeile);
241 SortHelper.from(this, range).spalteToSort(getTeamNrSpalte()).abSteigendSortieren().doSort();
242
243 // Höchste vorhandene Nr lesen
244 int letztNr = Math.max(0,
245 getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE)));
246
247 // Fehlende Nummern vergeben (nr <= 0 bedeutet: leer oder kein Wert vergeben)
248 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
249 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
250 if (vorname == null || vorname.isEmpty()) {
251 continue;
252 }
253 int nr = getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
254 if (nr <= 0) {
255 letztNr++;
256 getSheetHelper().setNumberValueInCell(
257 NumberCellValue.from(xSheet, Position.from(getTeamNrSpalte(), zeile)).setValue(letztNr));
258 }
259 }
260 }

262 private void nachTeamNrSortieren(XSpreadsheet xSheet) throws GenerateException {

264 if (letzteZeile < ERSTE_DATEN_ZEILE) {
265 return;
266 }
267 RangePosition range = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
268 getSetzPositionSpalte(), letzteZeile);
269 SortHelper.from(this, range).spalteToSort(getTeamNrSpalte()).aufSteigendSortieren(true).doSort();
270 }
