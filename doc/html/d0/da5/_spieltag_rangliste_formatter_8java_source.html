47 public void updateHeader() throws GenerateException {
48
49 rangliste.processBoxinfo("processbox.formatiere.header");
50
51 int anzRunden = rangliste.getAnzahlRunden();
52 if (anzRunden < 1) {
53 return;
54 }
55 XSpreadsheet sheet = rangliste.getXSpreadSheet();
56 // -------------------------
57 // spielrunden spalten
58 // -------------------------
59 boolean rundensummenAnzeigen = getPropertiesSpalte().getSpieltagRanglisteRundensummenAnzeigen();
60 ColumnProperties columnProperties = ColumnProperties.from()
61 .setWidth(MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH).setHoriJustify(CellHoriJustify.CENTER)
62 .isVisible(rundensummenAnzeigen);
63 StringCellValue headerPlus = StringCellValue
64 .from(sheet, Position.from(ersteSpielRundeSpalte, DRITTE_KOPFDATEN_ZEILE), "+")
65 .addColumnProperties(columnProperties).setCellBackColor(getHeaderFarbe()).setShrinkToFit(true);
66 StringCellValue headerMinus = StringCellValue.from(headerPlus).setValue("-");
67
68 headerPlus.setBorder(borderThinLeftBold());
69
70 for (int spielRunde = 1; spielRunde <= anzRunden; spielRunde++) {
71 int plusSpalte = ersteSpielRundeSpalte + ((spielRunde - 1) * 2);
72 getSheetHelper().setStringValueInCell(
73 headerPlus.spalte(plusSpalte).setComment(I18n.get("supermelee.spieltagrangliste.comment.runde.punkte.plus", spielRunde)));
74 getSheetHelper().setStringValueInCell(
75 headerMinus.spalte(plusSpalte + 1).setComment(I18n.get("supermelee.spieltagrangliste.comment.runde.punkte.minus", spielRunde)));
76
77 // Runden Counter
78 StringCellValue headerRndCounter = StringCellValue.from(headerPlus).setValue(spielRunde)
79 .zeile(ZWEITE_KOPFDATEN_ZEILE).setEndPosMergeSpaltePlus(1).setComment(null);
80 getSheetHelper().setStringValueInCell(headerRndCounter);
81 }
82 // Erste Zeile Runde
83 TableBorder2 border = BorderFactory.from().allBold().thinLn().forBottom().toBorder();
84 StringCellValue rundeHeaderBez = StringCellValue.from(sheet).zeile(ERSTE_KOPFDATEN_ZEILE)
85 .spalte(ersteSpielRundeSpalte).setEndPosMergeSpaltePlus((anzRunden * 2) - 1).setValue(I18n.get("column.header.runde"))
86 .setCellBackColor(getHeaderFarbe()).setBorder(border).setShrinkToFit(true);
87 getSheetHelper().setStringValueInCell(rundeHeaderBez);
88
89 // -------------------------
90 // summen spalten
91 // -------------------------
92 formatEndSummen(rangliste.getErsteSummeSpalte());
93 }

102 public int countAnzahlRunden() throws GenerateException {
103 int anzRunden = 0;
104 Position rundeNrPos = Position.from(ersteSpielRundeSpalte, ZWEITE_KOPFDATEN_ZEILE);
105 for (int spielRunde = 1; spielRunde <= 1000; spielRunde++) {
106 Integer rundeNrFromSheet = getSheetHelper().getIntFromCell(getSheet(), rundeNrPos);
107 if (rundeNrFromSheet != spielRunde) {
108 break;
109 }
110 rundeNrPos.spaltePlus(2);
111 anzRunden++;
112 }
113 return anzRunden;
114 }

116 public void formatDaten() throws GenerateException {
117
118 rangliste.processBoxinfo("processbox.formatiere.daten");
119
120 MeldungenSpalte<SpielerMeldungen, Spieler> spielerSpalte = getSpielerSpalte();
121
122 int anzRunden = rangliste.getAnzahlRunden();
123 if (anzRunden < 1) {
124 return;
125 }
126
127 int ersteDatenZeile = spielerSpalte.getErsteDatenZiele();
128 int letzteDatenZeile = spielerSpalte.getLetzteMitDatenZeileInSpielerNrSpalte();
129 TableBorder2 border = BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder();
130
131 for (int spielRunde = 1; spielRunde <= anzRunden; spielRunde++) {
132 Position posPunktePlusStart = Position
133 .from(ersteSpielRundeSpalte + ((spielRunde - 1) * anzSpaltenInSpielrunde), ersteDatenZeile);
134 Position posPunkteMinusEnd = Position.from(posPunktePlusStart).spaltePlusEins().zeile(letzteDatenZeile);
135 RangePosition datenRange = RangePosition.from(posPunktePlusStart, posPunkteMinusEnd);
136 getSheetHelper().setPropertyInRange(rangliste.getXSpreadSheet(), datenRange, TABLE_BORDER2, border);
137 }
138
139 formatDatenSpielTagSpalten(rangliste.getErsteSummeSpalte());
140
141 RangePosition gesamtDatenRange = RangePosition.from(
142 ersteSpielRundeSpalte, ersteDatenZeile, rangliste.getLetzteSpalte(), letzteDatenZeile);
143 getSheetHelper().setPropertiesInRange(rangliste.getXSpreadSheet(), gesamtDatenRange,
144 CellProperties.from().margin(MeldeListeKonstanten.CELL_MARGIN));
145 }

159 .setEndPosMergeSpaltePlus(getLetzteSpalte())
160 .addCellProperty(IS_TEXT_WRAPPED, Boolean.TRUE)
161 .addRowProperty("OptimalHeight", Boolean.TRUE)
162 .setValue("Nicht gespielten Runden werden mit " + nichtgespieltPlus + ":" + nichtgespieltMinus + " gewertet"));
163
164 return stringVal;
