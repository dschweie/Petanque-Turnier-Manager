188 private void renderKonfigurationsZuZellen(XSpreadsheet xKonfigSheet) throws GenerateException {
189 int headerFarbe = Integer.parseInt(BasePropertiesSpalte.HEADER_BACK_COLOR.replace("#", ""), 16);
190
191 sheet.getSheetHelper().setStringValueInCell(
192 StringCellValue.from(xKonfigSheet, Position.from(0, 0), "Eigenschaft")
193 .setCharWeight(FontWeight.BOLD)
194 .setCellBackColor(headerFarbe));
195 sheet.getSheetHelper().setStringValueInCell(
196 StringCellValue.from(xKonfigSheet, Position.from(1, 0), "Wert")
197 .setCharWeight(FontWeight.BOLD)
198 .setCellBackColor(headerFarbe));
199
200 List<ConfigProperty<?>> props = FormuleXPropertiesSpalte.KONFIG_PROPERTIES;
201 int zeile = 1;
202 for (ConfigProperty<?> prop : props) {
203 String beschreibung = prop.getDescription() != null ? prop.getDescription() : prop.getKey();
204 beschreibung = beschreibung.split("[\r\n]")[0];
205
206 String wert;
207 if (prop.getType() == ConfigPropertyType.COLOR) {
208 int farbwert = konfigurationSheet.getPropertiesSpalte().readIntProperty(prop.getKey());
209 wert = String.format("#%06X", farbwert & 0xFFFFFF);
210 } else if (prop.getType() == ConfigPropertyType.INTEGER) {
211 wert = String.valueOf(konfigurationSheet.getPropertiesSpalte().readIntProperty(prop.getKey()));
212 } else {
213 wert = konfigurationSheet.getPropertiesSpalte().readStringProperty(prop.getKey());
214 }
215
216 sheet.getSheetHelper().setStringValueInCell(
217 StringCellValue.from(xKonfigSheet, Position.from(0, zeile), beschreibung));
218 sheet.getSheetHelper().setStringValueInCell(
219 StringCellValue.from(xKonfigSheet, Position.from(1, zeile), wert));
220 zeile++;
221 }

361 private void formatDatenSpalten() throws GenerateException {
362 sheet.processBoxinfo("processbox.ko.meldeliste.formatieren");
363 Formation formation = konfigurationSheet.getMeldeListeFormation();
364 int anzSpieler = formation.getAnzSpieler();
365 int letzteDatenZeile = getLetzteDatenZeileUseMin();
366
367 MeldungenHintergrundFarbeGeradeStyle farbeGerade = konfigurationSheet.getMeldeListeHintergrundFarbeGeradeStyle();
368 MeldungenHintergrundFarbeUnGeradeStyle farbeUngerade = konfigurationSheet
369 .getMeldeListeHintergrundFarbeUnGeradeStyle();
370
371 RangePosition nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE,
372 getTeamNrSpalte(), letzteDatenZeile);
373 RangeHelper.from(sheet, nrRange).setRangeProperties(
374 RangeProperties.from().setBorder(
375 BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
376
377 String kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
378 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
379 ConditionalFormatHelper.from(sheet, nrRange).clear()
380 .formulaIsText().styleIsFehler().applyAndDoReset()
381 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
382 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
383 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset()
384 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
385 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
386
387 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
388 RangePosition teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, 1, letzteDatenZeile);
389 RangeHelper.from(sheet, teamnameRange).setRangeProperties(
390 RangeProperties.from().setBorder(
391 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
392 }
393
394 for (int s = 0; s < anzSpieler; s++) {
395 int ersteSpielerSpalte = getVornameSpalte(s);
396 int letzteSpielerSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
397 ? getVereinsnameSpalte(s)
398 : getNachnameSpalte(s);
399 RangePosition spielerRange = RangePosition.from(ersteSpielerSpalte, ERSTE_DATEN_ZEILE,
400 letzteSpielerSpalte, letzteDatenZeile);
401 RangeHelper.from(sheet, spielerRange).setRangeProperties(
402 RangeProperties.from().setBorder(
403 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
404 }
405
406 RangePosition spRange = RangePosition.from(getSetzPositionSpalte(), ERSTE_DATEN_ZEILE,
407 getSetzPositionSpalte(), letzteDatenZeile);
408 RangeHelper.from(sheet, spRange).setRangeProperties(
409 RangeProperties.from().setBorder(
410 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
411
412 ConditionalFormatHelper.from(sheet, spRange).clear()
413 .formulaIsText().styleIsFehler().applyAndDoReset()
414 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
415 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
416
417 RangePosition aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE,
418 getAktivSpalte(), letzteDatenZeile);
419 RangeHelper.from(sheet, aktivRange).setRangeProperties(
420 RangeProperties.from().centerJustify()
421 .setBorder(BorderFactory.from().allThin().toBorder()));
422
423 String kondAktivUngueltig = "AND(NOT(ISBLANK(" + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "));"
424 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>1;"
425 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + "<>2)";
426 ConditionalFormatHelper.from(sheet, aktivRange).clear()
427 .formula1(kondAktivUngueltig).operator(ConditionOperator.FORMULA)
428 .styleIsFehler().applyAndDoReset()
429 .formulaIsEvenRow().style(farbeGerade).applyAndDoReset()
430 .formulaIsOddRow().style(farbeUngerade).applyAndDoReset();
431
432 EditierbaresZelleFormatHelper.anwenden(sheet, RangePosition.from(1, ERSTE_DATEN_ZEILE, getAktivSpalte(), letzteDatenZeile));
433 }

452 TeamMeldungen getAktiveMeldungen() throws GenerateException {
453 sheet.processBoxinfo("processbox.ko.meldeliste.einlesen");
454 XSpreadsheet xSheet = sheet.getXSpreadSheet();
455 int vornameSpalte = getVornameSpalte(0);
456 int spSpalte = getSetzPositionSpalte();
457 int aktivSpalte = getAktivSpalte();
458 int letzteZeile = letzteZeileMitDaten(xSheet);
459
460 TeamMeldungen meldungen = new TeamMeldungen();
461 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
462 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
463 if (vorname == null || vorname.isEmpty()) {
464 continue;
465 }
466 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
467 if (nr <= 0) {
468 continue;
469 }
470 int aktiv = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
471 if (aktiv == AKTIV_WERT_NIMMT_TEIL) {
472 int setzPos = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(spSpalte, zeile));
473 meldungen.addTeamWennNichtVorhanden(Team.from(nr).setSetzPos(setzPos));
474 }
475 }
476 return meldungen;

483 TeamMeldungen getMeldungenSortiertNachSetzposition() throws GenerateException {
484 sheet.processBoxinfo("processbox.ko.meldeliste.sortieren");
485 XSpreadsheet xSheet = sheet.getXSpreadSheet();
486 int vornameSpalte = getVornameSpalte(0);
487 int spSpalte = getSetzPositionSpalte();
488 int aktivSpalte = getAktivSpalte();
489 int letzteZeile = letzteZeileMitDaten(xSheet);
490
491 record TeamZeile(int nr, int sp) {}
492 List<TeamZeile> alleTeams = new ArrayList<>();
493
494 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
495 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
496 if (vorname == null || vorname.isEmpty()) {
497 continue;
498 }
499 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
500 if (nr <= 0) {
501 continue;
502 }
503 int aktiv = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
504 if (aktiv != AKTIV_WERT_NIMMT_TEIL) {
505 continue;
506 }
507 int sp = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(spSpalte, zeile));
508 alleTeams.add(new TeamZeile(nr, sp));
509 }
510
511 Map<Integer, List<TeamZeile>> setzGruppen = new LinkedHashMap<>();
512 List<TeamZeile> ungesetzte = new ArrayList<>();
513 for (TeamZeile t : alleTeams) {
514 if (t.sp() > 0) {
515 setzGruppen.computeIfAbsent(t.sp(), k -> new ArrayList<>()).add(t);
516 } else {
517 ungesetzte.add(t);
518 }
519 }
520 setzGruppen.values().forEach(gruppe -> gruppe.sort(Comparator.comparingInt(TeamZeile::nr)));

611 void pruefeAufDoppelteTeamNr(XSpreadsheet xSheet) throws GenerateException {
612 int letzteZeile = letzteZeileMitDaten(xSheet);
613 if (letzteZeile < ERSTE_DATEN_ZEILE) {
614 return;
615 }
616 int vornameSpalte = getVornameSpalte(0);
617 Map<Integer, List<Integer>> alleNrn = new LinkedHashMap<>();
618 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
619 String vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
620 if (vorname == null || vorname.isEmpty()) {
621 continue;
622 }
623 int nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
624 if (nr <= 0) {
625 continue;
626 }
627 alleNrn.computeIfAbsent(nr, k -> new ArrayList<>()).add(zeile);
628 }
629 Map<Integer, List<Integer>> duplikate = new LinkedHashMap<>();
630 for (Map.Entry<Integer, List<Integer>> entry : alleNrn.entrySet()) {
631 if (entry.getValue().size() > 1) {
632 duplikate.put(entry.getKey(), entry.getValue());
633 }
634 }
635 if (duplikate.isEmpty()) {
636 return;
637 }
638 var sb = new StringBuilder("Meldeliste wurde nicht aktualisiert.\nDoppelte Startnummern:");
639 for (Map.Entry<Integer, List<Integer>> entry : duplikate.entrySet()) {
640 sb.append("\nNr. ").append(entry.getKey()).append(": Zeilen ");
641 List<Integer> zeilen = entry.getValue();
642 for (int i = 0; i < zeilen.size(); i++) {
643 if (i > 0) {
644 sb.append(", ");
645 }
646 sb.append(zeilen.get(i) + 1);
647 }
648 }
649 throw new GenerateException(sb.toString());
650 }
