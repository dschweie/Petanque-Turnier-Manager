322 public static XControl createControl(GuiFactoryCreateParam guiFactoryCreateParam, String type,
323 Map<String, Object> props, Rectangle rectangle) {
324
325 try {
326 XControl control = Lo.qi(XControl.class, guiFactoryCreateParam.getxMCF().createInstanceWithContext(type,
327 guiFactoryCreateParam.getContext()));
328 XControlModel controlModel = Lo.qi(XControlModel.class, guiFactoryCreateParam.getxMCF()
329 .createInstanceWithContext(type + "Model", guiFactoryCreateParam.getContext()));
330 control.setModel(controlModel);
331 XMultiPropertySet properties = Lo.qi(XMultiPropertySet.class, control.getModel());
332 if (props != null && props.size() > 0) {
333 properties.setPropertyValues(props.keySet().toArray(new String[props.size()]),
334 props.values().toArray(new Object[props.size()]));
335 }
336 control.createPeer(guiFactoryCreateParam.getToolkit(), guiFactoryCreateParam.getWindowPeer());
337 XWindow controlWindow = Lo.qi(XWindow.class, control);
338 if (rectangle != null) {
339 setWindowPosSize(controlWindow, rectangle);
340 }
341 return control;
342 } catch (Exception ex) {
343 logger.error(ex.getMessage(), ex);
344 return null;
345 }
346 }
