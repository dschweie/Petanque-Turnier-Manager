54 public static Map<Integer, String> lesePreservedGruppen(SchweizerRanglisteSheet rangliste) {
55 Map<Integer, String> result = new HashMap<>();
56 try {
57 XSpreadsheet sheet = rangliste.getXSpreadSheet();
58 if (sheet == null) {
59 return result;
60 }
61 RangePosition leseRange = RangePosition.from(
62 SchweizerRanglisteSheet.TEAM_NR_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE,
63 GRUPPE_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE + 999);
64 RangeData rowsData = RangeHelper
65 .from(sheet, rangliste.getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), leseRange)
66 .getDataFromRange();
67 for (RowData row : rowsData) {
68 if (row.size() <= GRUPPE_SPALTE) {
69 break;
70 }
71 int teamNr = row.get(SchweizerRanglisteSheet.TEAM_NR_SPALTE).getIntVal(0);
72 if (teamNr <= 0) {
73 break;
74 }
75 String gruppe = row.get(GRUPPE_SPALTE).getStringVal();
76 if (gruppe != null && !gruppe.isEmpty()) {
77 result.put(teamNr, gruppe.trim());
78 }
79 }
80 } catch (GenerateException e) {
81 logger.debug("lesePreservedGruppen: Sheet nicht lesbar, leere Map verwendet", e);
82 }
83 return result;
84 }

116 public static void schreibeGruppenZuweisungen(SchweizerRanglisteSheet rangliste, XSpreadsheet sheet,
117 Map<Integer, String> teamNrZuGruppe) throws GenerateException {
118 RangePosition leseRange = RangePosition.from(
119 SchweizerRanglisteSheet.TEAM_NR_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE,
120 SchweizerRanglisteSheet.TEAM_NR_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE + 999);
121 RangeData rowsData = RangeHelper
122 .from(sheet, rangliste.getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), leseRange)
123 .getDataFromRange();
124
125 RangeData schreibBlock = new RangeData();
126 int anzZeilen = 0;
127 for (RowData row : rowsData) {
128 if (row.size() < 1) {
129 break;
130 }
131 int teamNr = row.get(0).getIntVal(0);
132 if (teamNr <= 0) {
133 break;
134 }
135 RowData neueRow = schreibBlock.addNewRow();
136 neueRow.newString(teamNrZuGruppe.getOrDefault(teamNr, ""));
137 anzZeilen++;
138 }
139 if (anzZeilen == 0) {
140 return;
141 }
142 RangeHelper.from(rangliste,
143 schreibBlock.getRangePosition(Position.from(GRUPPE_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE)))
144 .setDataInRange(schreibBlock);
145 }

151 public static void schreibeDaten(SchweizerRanglisteSheet rangliste, XSpreadsheet sheet,
152 List<SchweizerTeamErgebnis> sortiert, int letzteZeile,
153 Map<Integer, String> teamNrZuGruppe) throws GenerateException {
154 if (sortiert.isEmpty()) {
155 return;
156 }
157 RangeData block = new RangeData();
158 for (SchweizerTeamErgebnis erg : sortiert) {
159 RowData row = block.addNewRow();
160 row.newString(teamNrZuGruppe.getOrDefault(erg.teamNr(), ""));
161 }
162 RangeHelper.from(rangliste,
163 block.getRangePosition(Position.from(GRUPPE_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE)))
164 .setDataInRange(block);
165
166 rangliste.getSheetHelper().setPropertiesInRange(sheet,
167 RangePosition.from(GRUPPE_SPALTE, SchweizerRanglisteSheet.ERSTE_DATEN_ZEILE,
168 GRUPPE_SPALTE, letzteZeile),
169 CellProperties.from()
170 .setAllThinBorder()
171 .centerJustify()
172 .setCharWeight(FontWeight.BOLD));
173 }
