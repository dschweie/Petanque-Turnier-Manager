86 XNameContainer xCellStylesNA = Lo.qi(XNameContainer.class, pageStylesObj);
87
88 Object pageStyle = null;

90 pageStyle = xCellStylesNA.getByName(styleName);
91 } catch (NoSuchElementException e) {
92 // create a new Page style
93 XMultiServiceFactory xDocServiceManager = Lo.qi(XMultiServiceFactory.class, currentSpreadsheetDocument);
94 // TablePageStyle

96 xCellStylesNA.insertByName(styleName, pageStyle);
97 }

99 // modify properties of the (new) style
