106 private void upDateSheet() throws GenerateException {
107 meldeListe.upDateSheet();
108 TeamMeldungen alleMeldungen = meldeListe.getAlleMeldungen();
109 this.spielPlan = new LigaSpielPlan(alleMeldungen);
110
111 getxCalculatable().enableAutomaticCalculation(false); // speed up
112 if (!alleMeldungen.isValid()) {
113 processBoxinfo("processbox.abbruch");
114 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK).caption(I18n.get("msg.caption.jgj.spielplan"))
115 .message(I18n.get("msg.text.ungueltige.anzahl.meldungen")).show();
116 return;
117 }
118
119 if (!NewSheet.from(this, SheetNamen.direktvergleich(), METADATA_SCHLUESSEL)
120 .pos(DefaultSheetPos.JGJ_DIREKTEVERGLEICH).setForceCreate(true).setActiv()
121 .hideGrid().tabColor(konfigurationSheet.getDirektvergleichTabFarbe()).create().isDidCreate()) {
122 ProcessBox.from().info("Abbruch vom Benutzer, JGJ Direktvergleich wurde nicht erstellt");
123 return;
124 }
125 meldungenSpalte.alleAktiveUndAusgesetzteMeldungenAusmeldelisteEinfuegen(meldeListe);
126 int headerBackColor = getKonfigurationSheet().getRanglisteHeaderFarbe();
127 meldungenSpalte.insertHeaderInSheet(headerBackColor);
128
129 dateneinfuegen(alleMeldungen);
130 formatData();
131 addConditionalFormuleForDirektVergleichReturnCode();
132 StringCellValue lastPos = addFooter();
133 printBereichDefinieren(lastPos.getPos());
134 }

167 private void dateneinfuegen(TeamMeldungen alleMeldungen) throws GenerateException {
168 processBoxinfo("processbox.direktvergleich.einfuegen");
169
170 Position startTeamNrPos = Position.from(ERSTE_SPALTE_DIREKTVERGLEICH, TEAM_NR_HEADER_ZEILE);
171 int headerBackColor = getKonfigurationSheet().getRanglisteHeaderFarbe();
172 StringCellValue htmNr = StringCellValue.from(getXSpreadSheet()).setPos(startTeamNrPos)
173 .setCellBackColor(headerBackColor).centerJustify().setAllThinBorder();
174
175 for (IMeldung<Team> mld : alleMeldungen.getMeldungenSortedByNr()) {
176 htmNr.setValue(mld.getNr());
177 getSheetHelper().setStringValueInCell(htmNr);
178 htmNr.spaltePlusEins();
179 }
180
181 // formula einfuegen
182 StringCellValue formula = StringCellValue.from(getXSpreadSheet())
183 .setCharWeight(com.sun.star.awt.FontWeight.BOLD);
184 StringCellValue xStr = StringCellValue.from(getXSpreadSheet()).setValue("x");
185 String spielplanBegegnungenVerweis = ligaSpielPlanVerweis(JGJSpielPlanSheet.TEAM_A_NR_SPALTE);
186 String spielplanSpieleVerweis = ligaSpielPlanVerweis(JGJSpielPlanSheet.SPIELE_A_SPALTE);
187 String spielplanSpielPunkteVerweis = ligaSpielPlanVerweis(JGJSpielPlanSheet.SPIELPNKT_A_SPALTE);
188 for (IMeldung<Team> mldA : alleMeldungen.getMeldungenSortedByNr()) {
189 for (IMeldung<Team> mldB : alleMeldungen.getMeldungenSortedByNr()) {
190 if (mldA.getNr() != mldB.getNr()) {
191 String formuleStr = direktVergleichFormula(mldA.getNr(), mldB.getNr(), spielplanBegegnungenVerweis,
192 spielplanSpieleVerweis, spielplanSpielPunkteVerweis);
193 formula.setPos(startTeamNrPos).zeilePlus(mldA.getNr()).spaltePlus(mldB.getNr() - 1)
194 .setComment(mldA.getNr() + ":" + mldB.getNr()).setValue(formuleStr);
195 getSheetHelper().setFormulaInCell(formula);
196 } else {
197 xStr.setPos(startTeamNrPos).zeilePlus(mldA.getNr()).spaltePlus(mldB.getNr() - 1);
198 getSheetHelper().setStringValueInCell(xStr);
199 }
200 }
201 }

204 private String ligaSpielPlanVerweis(int startSpalte) throws GenerateException {
205 int anzZeilen = (spielPlan.anzRunden() * 2) * spielPlan.anzBegnungenProRunde();
206 Position startBegegnungenPos = Position.from(startSpalte, JGJSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE);
207 return "$'" + JGJSpielPlanSheet.sheetName() + "'." + startBegegnungenPos.getAddressWith$() + ":"
208 + startBegegnungenPos.spaltePlusEins().zeilePlus(anzZeilen - 1).getAddressWith$();
209 }

215 + spielplanSpieleVerweis + ";" + spielplanSpielPunkteVerweis + ")";

246 private StringCellValue addFooter() throws GenerateException {
247 processBoxinfo("processbox.fusszeile.einfuegen");
248 Position startPos = Position.from(TEAM_NR_SPALTE, footerStartLinePos() + 1);
249 StringCellValue direktvergleichResultCode = StringCellValue.from(getXSpreadSheet()).setPos(startPos)
250 .setCharHeight(8);
251 StringCellValue direktvergleichResultVal = StringCellValue.from(direktvergleichResultCode).spaltePlusEins()
252 .setHoriJustify(com.sun.star.table.CellHoriJustify.LEFT);
253
254 List<DirektvergleichResult> sortedList = DirektvergleichResult.stream()
255 .sorted(Comparator.comparingInt(DirektvergleichResult::getCode)).collect(Collectors.toList());
256
257 for (DirektvergleichResult drslt : sortedList) {
258 direktvergleichResultCode.setValue(drslt.getCode());
259 direktvergleichResultVal.setValue(drslt.getAnzeigeText());
260 getSheetHelper().setStringValueInCell(direktvergleichResultCode);
261 getSheetHelper().setStringValueInCell(direktvergleichResultVal);
262 direktvergleichResultCode.zeilePlusEins();
263 direktvergleichResultVal.zeilePlusEins();
264 }
265 return direktvergleichResultCode.zeilePlus(-1); // letzte zeile
266 }
