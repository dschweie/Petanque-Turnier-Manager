57 public List<List<TeamPaarung>> generate() {
58 List<List<TeamPaarung>> result = new ArrayList<>();
59
60 int anzRndn = anzRunden();
61 int letzteMeldungNr = letzteMeldungNr();
62 int anzPaarungen = anzPaarungen();
63
64 for (int rundenCntr = 1; rundenCntr <= anzRndn; rundenCntr++) {
65
66 List<TeamPaarung> runde = new ArrayList<>();
67 result.add(runde);
68
69 // letzte Teampaarung ist einfach nach schema F
70 // wegen heim und gast je nach runde umdrehen, nur wenn kein freispiel
71 // bei freispiel nicht drehen sondern reihenfolge bleibt
72 if (IsEvenOrOdd.isOdd(rundenCntr) || freiSpiel) {
73 runde.add(newTeamPaarung(rundenCntr - 1, letzteMeldungNr - 1));
74 } else {
75 runde.add(newTeamPaarung(letzteMeldungNr - 1, rundenCntr - 1));
76 }
77
78 // restliche paarungen
79 for (int teamPaarungcntr = 1; teamPaarungcntr < anzPaarungen; teamPaarungcntr++) {
80 int moduloA = (rundenCntr + teamPaarungcntr) % anzRndn;
81 int idxMeldungA = (moduloA < 1) ? anzRndn + moduloA : moduloA;
82 int moduloB = (rundenCntr - teamPaarungcntr) % anzRndn;
83 int idxMeldungB = (moduloB < 1) ? anzRndn + moduloB : moduloB;
84
85 runde.add(newTeamPaarung(idxMeldungA - 1, idxMeldungB - 1));
86 }
87 }
88 return result;
89 }

91 private TeamPaarung newTeamPaarung(int idxMeldungA, int idxMeldungB) {
92 Team teamA = meldungen.get(idxMeldungA);
93 Optional<Team> teamB;

96 teamB = Optional.empty();
97 } else {
98 teamB = Optional.of(meldungen.get(idxMeldungB));
99 }
100 TeamPaarung ret = new TeamPaarung(teamA, teamB);
101 return ret;
102 }
