104 public void beginCommandScope(IBlattschutzKonfiguration konfig, WorkingSpreadsheet ws) {
105 ScopeState state = SCOPE.get();
106 if (state == null) {
107 SCOPE.set(new ScopeState(konfig, ws));
108 return;
109 }
110 if (!state.konfig.equals(konfig)) {
111 throw new IllegalStateException(
112 "BlattschutzScope: Konfigurationswechsel in nested scope nicht erlaubt – "
113 + "äußerer Scope: " + state.konfig.getClass().getSimpleName()
114 + ", innerer Scope: " + konfig.getClass().getSimpleName());
115 }
116 state.refCount++;
117 }
118
129 public void endCommandScope() {
130 ScopeState state = SCOPE.get();
131 if (state == null) {
132 return;
133 }
134 if (--state.refCount > 0) {
135 return;
136 }
137 try {
138 // doSchuetzen() toggelt Protect/Unprotect und aktualisiert CellStyles → das setzt in
139 // LibreOffice das Modified-Flag. Hat der Lauf selbst nichts geändert (Dokument vor dem
140 // Schützen unverändert – z.B. ein idempotenter Formatierer-Lauf beim Tab-Wechsel), darf
141 // das bloße Wiederherstellen des Schutzes das Dokument NICHT als geändert markieren –
142 // sonst Phantom-Speicherung/„Speichern?"-Abfrage. Bei echten Änderungen war das Flag
143 // bereits vor doSchuetzen true und bleibt erhalten.
144 var xDoc = state.ws.getWorkingSpreadsheetDocument();
145 if (xDoc == null) {
146 doSchuetzen(state.konfig, state.ws);
147 } else {
148 new DocumentPropertiesHelper(xDoc).ohneModifiedFlag(() -> doSchuetzen(state.konfig, state.ws));
149 }
150 } finally {
151 SCOPE.remove();
152 }
153 }

172 public void ensureUnprotectedInScope() {
173 ScopeState state = SCOPE.get();
174 if (state == null) {
175 if (TurnierModus.get().istAktiv()) {
176 throw new IllegalStateException(
177 "Style/CF-Operation außerhalb eines aktiven BlattschutzScopes – "
178 + "diese Operation muss innerhalb eines SheetRunner.run() laufen.");
179 }
180 return;
181 }
182 if (!state.wurdeEntsperrt) {
183 doEntsperren(state.konfig, state.ws);
184 state.wurdeEntsperrt = true;
185 }
186 }

206 public BlattschutzScope scopeFuer(TurnierSystem ts, WorkingSpreadsheet ws) {

267 private void doSchuetzen(IBlattschutzKonfiguration konfiguration, WorkingSpreadsheet ws) {
268 protectCount.incrementAndGet();
269 // Schritt 1: CellStyles sichern BEVOR irgendein Sheet geschützt wird
270 konfiguration.zelleStylesAktualisieren(ws);
271
272 // Schritt 2: Schutz-Infos einmalig berechnen, ergänzt um system-übergreifende Sheets
273 var infos = mitGlobalenSchutzInfos(konfiguration.berechneSchutzInfos(ws), ws);
274
275 // Schritt 3: Pro Sheet entsperren → Zellschutz → sperren
276 for (var info : infos) {
277 try {
278 entsperreSheetFallsNoetig(info.sheet());
279 for (var range : info.editierbareBereich()) {
280 setzeZellSchutzFreigegeben(info.sheet(), range);
281 }
282 schuetzeSheet(info.sheet());
283 } catch (Exception e) {
284 logger.warn("Sheet konnte nicht gesperrt werden: {}", e.getMessage(), e);
285 }
286 }
287 }
288
289 private void doEntsperren(IBlattschutzKonfiguration konfiguration, WorkingSpreadsheet ws) {
290 unprotectCount.incrementAndGet();
291 var infos = mitGlobalenSchutzInfos(konfiguration.berechneSchutzInfos(ws), ws);
292 for (var info : infos) {
293 try {
294 entsperreSheet(info.sheet());
295 } catch (Exception e) {
296 logger.warn("Sheet konnte nicht entsperrt werden: {}", e.getMessage(), e);
297 }
298 }
299 }

346 private void setzeZellSchutzFreigegeben(XSpreadsheet sheet, RangePosition range) {
347 try {
348 var xRange = sheet.getCellRangeByPosition(
349 range.getStartSpalte(), range.getStartZeile(),
350 range.getEndeSpalte(), range.getEndeZeile());
351 var props = Lo.qi(XPropertySet.class, xRange);
352
353 var cp = new CellProtection();
354 cp.IsLocked = false; // Zelle bleibt editierbar trotz Sheet-Schutz
355 props.setPropertyValue("CellProtection", cp);
356
357 // Verifikation: LO kann setPropertyValue still ignorieren wenn nLockCount > 0
358 var result = (CellProtection) props.getPropertyValue("CellProtection");
359 if (result.IsLocked) {
360 logger.warn("IsLocked=false konnte nicht gesetzt werden für Bereich {} " +
361 "(LO nLockCount aktiv?) – wird beim nächsten formatDaten() korrigiert.", range);
362 }
363 } catch (Exception e) {
364 logger.error("Zellschutz für Bereich {} konnte nicht gesetzt werden: {}",
365 range, e.getMessage(), e);
366 }
