120 public static KaskadenKoRundenPlan berechne(
121 int gesamtTeams, int kaskadenStufen, KaskadenKoPaarungsStrategie strategie, boolean freispielGewonnen) {
122
123 checkArgument(gesamtTeams >= 2, "gesamtTeams muss mindestens 2 sein, war: %s", gesamtTeams);
124 checkArgument(kaskadenStufen >= 1, "kaskadenStufen muss mindestens 1 sein, war: %s", kaskadenStufen);
125
126 List<GruppeKnoten> aktuelleGruppen = new ArrayList<>();
127 aktuelleGruppen.add(new GruppeKnoten("", gesamtTeams));
128
129 var runden = new ArrayList<KaskadenKoRunde>(kaskadenStufen);
130
131 for (int rundenNr = 1; rundenNr <= kaskadenStufen; rundenNr++) {
132 var gruppenRunden = new ArrayList<KaskadenKoGruppenRunde>(aktuelleGruppen.size());
133 var naechsteGruppen = new ArrayList<GruppeKnoten>(aktuelleGruppen.size() * 2);
134
135 for (var gruppe : aktuelleGruppen) {
136 var gruppenRunde = erstelleGruppenRunde(gruppe, rundenNr, strategie);
137 gruppenRunden.add(gruppenRunde);
138
139 int m = gruppe.anzTeams();
140 int siegerTeams = freispielGewonnen ? (m + 1) / 2 : m / 2;
141 int verliererTeams = m - siegerTeams;
142 naechsteGruppen.add(new GruppeKnoten(gruppe.pfad() + "S", siegerTeams));
143 naechsteGruppen.add(new GruppeKnoten(gruppe.pfad() + "V", verliererTeams));
144 }
145
146 runden.add(new KaskadenKoRunde(rundenNr, Collections.unmodifiableList(gruppenRunden)));
147 aktuelleGruppen = naechsteGruppen;
148 }
149
150 var felder = KaskadenKoFeldRechner.berechne(gesamtTeams, kaskadenStufen, freispielGewonnen);
151 return new KaskadenKoRundenPlan(
152 gesamtTeams,
153 kaskadenStufen,
154 Collections.unmodifiableList(runden),
155 felder);
156 }

158 private static KaskadenKoGruppenRunde erstelleGruppenRunde(
159 GruppeKnoten gruppe, int rundenNr, KaskadenKoPaarungsStrategie strategie) {
160
161 int m = gruppe.anzTeams();
162 int erwarteteAnzPaare = m / 2;
163 int freilose = m % 2;
164
165 var spielPaare = strategie.generiere(gruppe.pfad(), rundenNr, m);
166
167 checkState(spielPaare.size() == erwarteteAnzPaare,
168 "Strategie liefert falsche Paaranzahl: erwartet %s, erhalten %s (Gruppe pfad='%s', anzTeams=%s)",
169 erwarteteAnzPaare, spielPaare.size(), gruppe.pfad(), m);
170
171 return new KaskadenKoGruppenRunde(
172 gruppe.pfad(),
173 m,
174 spielPaare,
175 freilose);
176 }
