84 public static XSpreadsheetDocument getCurrentSpreadsheetDocument(XComponentContext xContext) {
85 checkNotNull(xContext, "xContext = null");
86
87 // Normalfall: aktive Komponente (funktioniert mit UI-Fokus)
88 XSpreadsheetDocument doc = Lo.qi(XSpreadsheetDocument.class, getCurrentComponent(xContext));
89 if (doc != null) {
90 return doc;
91 }
92
93 // Fallback für Headless-Modus: alle offenen Frames durchsuchen
94 try {
95 XDesktop desktop = getCurrentDesktop(xContext);
96 XFramesSupplier framesSupplier = Lo.qi(XFramesSupplier.class, desktop);
97 if (framesSupplier == null) return null;
98
99 com.sun.star.frame.XFrames frames = framesSupplier.getFrames();
100 if (frames == null) return null;
101
102 for (int i = 0; i < frames.getCount(); i++) {
103 Object frameObj = frames.getByIndex(i);
104 XFrame frame = Lo.qi(XFrame.class, frameObj);
105 if (frame == null) continue;
106 XController controller = frame.getController();
107 if (controller == null) continue;
108 XModel model = controller.getModel();
109 XSpreadsheetDocument spreadsheet = Lo.qi(XSpreadsheetDocument.class, model);
110 if (spreadsheet != null) return spreadsheet;
111 }
112 } catch (Exception e) {
113 logger.error("Fehler beim Durchsuchen der Frames", e);
114 }
115 return null;
116 }
