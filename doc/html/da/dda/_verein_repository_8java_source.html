154 private static boolean istUniqueVerletzung(SQLException e) {
155 String raw = e.getMessage();
156 if (raw == null) {
157 return false;
158 }
159 String msg = raw.toLowerCase(java.util.Locale.ROOT);
160 // SQLite: "[SQLITE_CONSTRAINT_UNIQUE] ... UNIQUE constraint failed: ..."
161 if (msg.contains("unique constraint failed") || msg.contains("sqlite_constraint_unique")) {
162 return true;
163 }
164 // Generischer 23xxx-Constraint-State (HSQLDB / sonstige) als Fallback.
165 String state = e.getSQLState();
166 return state != null && state.startsWith("23");
167 }

169 private static boolean istFkVerletzung(SQLException e) {
170 String raw = e.getMessage();
171 if (raw == null) {
172 return false;
173 }
174 String msg = raw.toLowerCase(java.util.Locale.ROOT);
175 if (msg.contains("foreign key constraint failed") || msg.contains("sqlite_constraint_foreignkey")) {
176 return true;
177 }
178 String state = e.getSQLState();
179 return state != null && state.startsWith("23")
180 && (msg.contains("foreign") || msg.contains("integrity"));
181 }
