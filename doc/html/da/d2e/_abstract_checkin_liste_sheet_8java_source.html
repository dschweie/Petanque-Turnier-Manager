162 private void sortiereUndFuelle() throws GenerateException {
163 List<Integer> nummern = ladeNummern();
164 if (!nummern.isEmpty()) {
165 TeilnehmerListeSortModus modus = getKonfigurationSheet().getCheckinListeSortModus();
166 if (modus == TeilnehmerListeSortModus.TEAMNAME && !teamnameVerfuegbar()) {
167 modus = TeilnehmerListeSortModus.NAME;
168 }
169 nummern = sortiereNummern(nummern, ladeSortDaten(), modus);
170 }
171 fuelleBereich(nummern);
172 }

178 private static List<Integer> sortiereNummern(List<Integer> nummern, Map<Integer, SortSchluessel> sortDaten,
179 TeilnehmerListeSortModus modus) {
180 List<TeilnehmerEintrag> eintraege = new ArrayList<>(nummern.size());
181 for (int nr : nummern) {
182 SortSchluessel sk = sortDaten.getOrDefault(nr, SortSchluessel.LEER);
183 eintraege.add(new TeilnehmerEintrag(nr, sk.teamname(), "", sk.sortNachname()));
184 }
185 eintraege.sort(modus.comparator());
186 return eintraege.stream().map(TeilnehmerEintrag::nr).toList();
187 }

197 private void fuelleBereich(List<Integer> nummern) throws GenerateException {
198 final BlockLayout layout = new BlockLayout(teamSpalteAktiv());
199
200 if (nummern.isEmpty()) {
201 // Leere Meldeliste: dennoch Kopfzeile + Fußzeile (Anzahl 0) anzeigen und Druckbereich setzen.
202 kopfzeileSchreiben(0, layout);
203 int letzteSpalte = layout.spielerSpalte(0);
204 int footerZeile = footerSchreiben(0, KOPF_ZEILE, letzteSpalte);
205 printBereichDefinieren(footerZeile, letzteSpalte);
206 return;
207 }
208
209 final int maxProSpalte = getMaxProSpalte();
210 final int anzBloecke = (int) Math.ceil((double) nummern.size() / maxProSpalte);
211 final Map<Integer, String> spielerNamen = namenNachNummer();
212 final Map<Integer, String> teamnamen = layout.teamSpalte() ? teamnamenNachNummer() : Map.of();
213 final Map<Integer, Boolean> checkboxStatus = checkboxStatusNachNummer();
214 final String haken = I18n.get("checkinliste.haken");
215
216 RangeData data = new RangeData();
217 for (int zeileImBlock = 0; zeileImBlock < maxProSpalte; zeileImBlock++) {
218 RowData zeileData = data.addNewRow();
219 for (int blkCntr = 1; blkCntr <= anzBloecke; blkCntr++) {
220 int idx = zeileImBlock + (blkCntr - 1) * maxProSpalte;
221 if (idx < nummern.size()) {
222 int nr = nummern.get(idx);
223 zeileData.newInt(nr);
224 if (layout.teamSpalte()) {
225 zeileData.newString(teamnamen.getOrDefault(nr, ""));
226 }
227 zeileData.newString(spielerNamen.getOrDefault(nr, ""));
228 // Checkbox-Spalte: Haken, wenn der Aktiv-Status in der Meldeliste gesetzt ist.
229 if (checkboxStatus.getOrDefault(nr, false)) {
230 zeileData.newString(haken);
231 } else {
232 zeileData.newEmpty();
233 }
234 if (blkCntr != anzBloecke) {
235 zeileData.newEmpty(); // Trennspalte
236 }
237 }
238 }
239 }
240 RangePosition rangePosition = data.getRangePosition(Position.from(NR_SPALTE, ERSTE_DATEN_ZEILE));
241 RangeHelper.from(this, rangePosition).setDataInRange(data);
242
243 FormatErgebnis fmt = spaltenFormatieren(nummern, anzBloecke, maxProSpalte, layout);
244 int footerZeile = footerSchreiben(nummern.size(), fmt.letzteDatenZeile(), fmt.letzteSpalte());
245 printBereichDefinieren(footerZeile, fmt.letzteSpalte());
246 }

252 private FormatErgebnis spaltenFormatieren(List<Integer> nummern, int anzBloecke, int maxProSpalte,
253 BlockLayout layout) throws GenerateException {
254 RangeProperties rangePropNr = RangeProperties.from().setHoriJustify(CellHoriJustify.CENTER)
255 .setCharColor(ColorHelper.CHAR_COLOR_GRAY_SPIELER_NR);
256 ColumnProperties columnPropNr = ColumnProperties.from().setWidth(MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH);
257 ColumnProperties columnPropTeam = ColumnProperties.from().setHoriJustify(CellHoriJustify.LEFT);
258 ColumnProperties columnPropName = ColumnProperties.from().setHoriJustify(CellHoriJustify.CENTER);
259 ColumnProperties columnPropChkBox = ColumnProperties.from().setHoriJustify(CellHoriJustify.CENTER)
260 .setWidth(MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH);
261 RangeProperties rangePropBorderOnly = RangeProperties.from().setBorder(BorderFactory.from().allThin().toBorder())
262 .setShrinkToFit(true);
263
264 int maxMeldungZeile = 0;
265 int letzteSpalte = 0;
266
267 for (int blkCntr = 0; blkCntr < anzBloecke; blkCntr++) {
268 int letzteZeile = ERSTE_DATEN_ZEILE + maxProSpalte - 1;
269 if (blkCntr + 1 == anzBloecke) {
270 letzteZeile = ERSTE_DATEN_ZEILE + (nummern.size() - ((anzBloecke - 1) * maxProSpalte)) - 1;
271 }
272 maxMeldungZeile = Math.max(maxMeldungZeile, letzteZeile);
273
274 // Kopfzeile (Überschrift) je Block
275 kopfzeileSchreiben(blkCntr, layout);
276
277 int nrSpalte = layout.nrSpalte(blkCntr);
278 RangePosition nrRange = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, nrSpalte, letzteZeile);
279 RangeHelper.from(this, nrRange).setRangeProperties(rangePropNr);
280 getSheetHelper().setColumnProperties(getXSpreadSheet(), nrSpalte, columnPropNr);
281
282 // Namen sind bereits als Werte geschrieben; nur Spaltenbreite/Ausrichtung setzen.
283 if (layout.teamSpalte()) {
284 getSheetHelper().setColumnProperties(getXSpreadSheet(), layout.teamSpalte(blkCntr), columnPropTeam);
285 getSheetHelper().setOptimaleBreitePlusMarge(getXSpreadSheet(), layout.teamSpalte(blkCntr),
286 SheetHelper.OPTIMALE_BREITE_MARGE);
287 }
288 getSheetHelper().setColumnProperties(getXSpreadSheet(), layout.spielerSpalte(blkCntr), columnPropName);
289 getSheetHelper().setOptimaleBreitePlusMarge(getXSpreadSheet(), layout.spielerSpalte(blkCntr),
290 SheetHelper.OPTIMALE_BREITE_MARGE);
291 getSheetHelper().setColumnProperties(getXSpreadSheet(), layout.checkboxSpalte(blkCntr), columnPropChkBox);
292
293 int blockEnde = layout.letzteBlockSpalte(blkCntr);
294 RangePosition blockAll = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, blockEnde, letzteZeile);
295 RangeHelper.from(this, blockAll).setRangeProperties(rangePropBorderOnly);
296
297 // Trennspalte hinter dem Block auf Nr-Breite zurücksetzen
298 getSheetHelper().setColumnProperties(getXSpreadSheet(), blockEnde + 1, columnPropNr);
299 letzteSpalte = blockEnde;
300 }
301
302 return new FormatErgebnis(maxMeldungZeile, letzteSpalte);
303 }

311 private int footerSchreiben(int anzahl, int letzteDatenZeile, int letzteSpalte) throws GenerateException {

329 private void kopfzeileSchreiben(int blkCntr, BlockLayout layout) throws GenerateException {
330 int headerColor = getKonfigurationSheet().getMeldeListeHeaderFarbe();
331 // Header-Stil wie Teilnehmerliste: untere fette Trennlinie, nicht fett, ShrinkToFit.
332 var border = BorderFactory.from().allThin().boldLn().forBottom().toBorder();
333
334 headerZelleSchreiben(layout.nrSpalte(blkCntr), "column.header.nr",
335 MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH, headerColor, border);
336 if (layout.teamSpalte()) {
337 headerZelleSchreiben(layout.teamSpalte(blkCntr), "column.header.teamname",
338 TEAMNAME_SPALTE_WIDTH, headerColor, border);
339 headerZelleSchreiben(layout.spielerSpalte(blkCntr), "column.header.spieler",
340 getNameSpalteWidth(), headerColor, border);
341 } else {
342 headerZelleSchreiben(layout.spielerSpalte(blkCntr), "column.header.name",
343 getNameSpalteWidth(), headerColor, border);
344 }
