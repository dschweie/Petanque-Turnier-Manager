127 private void ergebnisseEinfuegen() throws GenerateException {
128 var meldungen = meldeliste.getAktiveMeldungen();
129 if (meldungen.size() < 3) {
130 return;
131 }
132
133 var gruppenGroessen = PouleGruppenRechner.berechneGruppenGroessen(meldungen.size());
134 var poules = PouleSeedingService.verteileTeams(meldungen, gruppenGroessen);
135
136 var xSheet = SheetMetadataHelper.findeSheetUndHeile(
137 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
138 SheetMetadataHelper.SCHLUESSEL_POULE_VORRUNDE,
139 SheetNamen.pouleVorrunde());
140
141 if (xSheet == null) {
142 return;
143 }
144
145 int aktuelleZeile = PouleVorrundeSheet.ERSTE_DATEN_ZEILE;
146
147 for (var poule : poules) {
148 SheetRunner.testDoCancelTask();
149 if (poule.teams().size() == 4) {
150 schreibeViererPouleErgebnisse(xSheet, aktuelleZeile);
151 aktuelleZeile += PouleVorrundeSheet.VIERER_POULE_ZEILEN;
152 } else {
153 schreibeDreierPouleErgebnisse(xSheet, aktuelleZeile);
154 aktuelleZeile += PouleVorrundeSheet.DREIER_POULE_ZEILEN;
155 }
156 }
157 }

187 private void schreibeErgebnis(XSpreadsheet xSheet, int zeile) throws GenerateException {
188 int loserPts = RandomSource.nextInt(0, 13);
189 int winner = RandomSource.nextInt(2);
190 int ergA = (winner == 0) ? 13 : loserPts;
191 int ergB = (winner == 0) ? loserPts : 13;
192
193 getSheetHelper().setNumberValueInCell(
194 NumberCellValue.from(xSheet, Position.from(PouleVorrundeSheet.SPALTE_ERG_A, zeile)).setValue(ergA));
195 getSheetHelper().setNumberValueInCell(
196 NumberCellValue.from(xSheet, Position.from(PouleVorrundeSheet.SPALTE_ERG_B, zeile)).setValue(ergB));
197 }
