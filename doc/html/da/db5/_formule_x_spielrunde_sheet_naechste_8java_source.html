43 public boolean naechsteSpielrundeEinfuegen() throws GenerateException {
44 SpielRundeNr aktuelleSpielrunde = getKonfigurationSheet().getAktiveSpielRunde();
45 setSpielRundeNrInSheet(aktuelleSpielrunde);
46 getMeldeListe().vollstaendigAktualisieren();
47 TeamMeldungen aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
48
49 if (aktiveMeldungen.size() == 0) {
50 TeamMeldungen alleMeldungen = getMeldeListe().getAlleMeldungen();
51 if (alleMeldungen.size() > 0) {
52 MessageBoxResult result = MessageBox.from(getxContext(), MessageBoxTypeEnum.WARN_YES_NO)
53 .caption(I18n.get("msg.caption.keine.aktiven.meldungen"))
54 .message(I18n.get("msg.text.keine.aktiven.teams.aktivieren", alleMeldungen.size()))
55 .show();
56 if (result == MessageBoxResult.YES) {
57 getMeldeListe().alleTeamsAktivieren();
58 aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
59 }
60 }
61 }
62
63 if (!canStart(aktiveMeldungen)) {
64 return false;
65 }
66
67 int neueSpielrundeNr = aktuelleSpielrunde.getNr();
68 if (getXSpreadSheet() != null) {
69 if (!alleErgebnisseEingetragen(aktuelleSpielrunde)) {
70 getSheetHelper().setActiveSheet(getXSpreadSheet());
71 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
72 .caption(I18n.get("msg.caption.naechste.runde.nicht.moeglich"))
73 .message(I18n.get("msg.text.naechste.runde.ergebnisse.fehlen", aktuelleSpielrunde.getNr()))
74 .show();
75 return false;
76 }
77 neueSpielrundeNr++;
78 }
79
80 int maxRunden = getKonfigurationSheet().getAnzahlRunden();
81 if (neueSpielrundeNr > maxRunden) {
82 MessageBox.from(getxContext(), MessageBoxTypeEnum.INFO_OK)
83 .caption(I18n.get("msg.caption.formulex.max.runden"))
84 .message(I18n.get("msg.text.formulex.max.runden.erreicht", maxRunden))
85 .show();
86 return false;
87 }
88
89 // Bestehende Rangliste aktualisieren, damit die Sortierung der Paarungen
90 // auf dem aktuellen Stand basiert. Nur ab Runde >= 2 sinnvoll.
91 if (neueSpielrundeNr >= 2) {
92 vorNaechsterRunde();
93 }
94
95 getKonfigurationSheet().setAktiveSpielRunde(SpielRundeNr.from(neueSpielrundeNr));
96
97 SpielrundenAkkumulation akkumulierung = gespieltenRundenEinlesen(aktiveMeldungen, 1, neueSpielrundeNr - 1);
98
99 return neueSpielrunde(aktiveMeldungen, SpielRundeNr.from(neueSpielrundeNr), akkumulierung);
100 }

118 private boolean alleErgebnisseEingetragen(SpielRundeNr spielRundeNr) throws GenerateException {
119 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
120 XSpreadsheet sheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
121 getSpielrundeSchluessel(spielRundeNr.getNr()), getSheetName(spielRundeNr));
122 if (sheet == null) {
123 return true;
124 }
125 RangePosition readRange = RangePosition.from(TEAM_A_SPALTE, ERSTE_DATEN_ZEILE, ERG_TEAM_B_SPALTE,
126 ERSTE_DATEN_ZEILE + 999);
127 RangeData rowsData = RangeHelper
128 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange).getDataFromRange();
129
130 for (RowData row : rowsData) {
131 if (row.size() < 2) {
132 break;
133 }
134 int nrA = row.get(0).getIntVal(0);
135 if (nrA <= 0) {
136 break;
137 }
138 int nrB = row.get(1).getIntVal(0);
139 if (nrB <= 0) {
140 continue; // Freilos-Zeile, kein Ergebnis nötig
141 }
142 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0;
143 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0;
144 if (ergA == 0 && ergB == 0) {
145 return false;
146 }
147 }
148 return true;
149 }
