124 public void generate(TeamMeldungen meldungen) throws GenerateException {
125 if (!meldungen.isValid()) {
126 processBoxinfo("processbox.abbruch");
127 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
128 .caption(I18n.get("msg.caption.triptete.spielplan"))
129 .message(I18n.get("msg.text.ungueltige.anzahl.meldungen")).show();
130 return;
131 }
132
133 if (!NewSheet.from(this, sheetName(), METADATA_SCHLUESSEL)
134 .pos(DefaultSheetPos.TRIPTETE_WORK).setActiv().hideGrid()
135 .tabColor(konfigurationSheet.getSpielrundeTabFarbe()).create().isDidCreate()) {
136 ProcessBox.from().info("Abbruch vom Benutzer, Trip-Tête-Spielplan wurde nicht erstellt");
137 return;
138 }
139
140 List<List<TeamPaarung>> spielPlan = TripTetePaarungen.jederGegenJeden(meldungen);
141
142 getxCalculatable().enableAutomaticCalculation(false);
143 try {
144 insertDatenHeaderUndSpalteBreite();
145 insertArbeitsspalten(spielPlan);
146 insertSpielNrSpalte(spielPlan);
147 insertBahnenSpalten(spielPlan);
148 insertFormulaTeamNamen();
149 insertFormulaWertSpalten();
150 formatieren(spielPlan);
151 printBereichDefinieren();
152 SheetFreeze.from(getTurnierSheet()).anzZeilen(2).doFreeze();
153 } finally {
154 getxCalculatable().enableAutomaticCalculation(true);
155 }
156 }

265 private void insertBahnenSpalten(List<List<TeamPaarung>> spielPlan) throws GenerateException {
266 // Pro Runde werden 2 physische Bahnen pro Begegnung vergeben:
267 // Bahn-Tri (ungerade): Triplette läuft hier, danach Doublette (sequenziell)
268 // Bahn-Dou (ungerade): dieselbe Bahn wie Tri – Doublette folgt Triplette
269 // Bahn-Tête (gerade): parallel zur Doublette, separate Bahn
270 RangeData rangeData = new RangeData();
271 for (List<TeamPaarung> runde : spielPlan) {
272 int bahnInRunde = 1;
273 for (TeamPaarung ignored : runde) {
274 int bahnTri = 2 * bahnInRunde - 1;
275 int bahnTete = 2 * bahnInRunde;
276 RowData row = rangeData.addNewRow();
277 row.newInt(bahnTri);
278 row.newInt(bahnTri); // Doublette auf gleicher Bahn wie Triplette (sequenziell)
279 row.newInt(bahnTete); // Tête auf separater Parallelbahn
280 bahnInRunde++;
281 }
282 }
283 Position startPos = Position.from(BAHN_TRI_SPALTE, ERSTE_DATEN_ZEILE);
284 RangeHelper.from(this, rangeData.getRangePosition(startPos)).setDataInRange(rangeData);
285 }

365 private String siegeFormel(boolean fuerA) {
366 Position triA = Position.from(TRI_A_SPALTE, ERSTE_DATEN_ZEILE);
367 Position triB = Position.from(TRI_B_SPALTE, ERSTE_DATEN_ZEILE);
368 Position douA = Position.from(DOU_A_SPALTE, ERSTE_DATEN_ZEILE);
369 Position douB = Position.from(DOU_B_SPALTE, ERSTE_DATEN_ZEILE);
370 Position teteA = Position.from(TETE_A_SPALTE, ERSTE_DATEN_ZEILE);
371 Position teteB = Position.from(TETE_B_SPALTE, ERSTE_DATEN_ZEILE);
372 String aAdr = fuerA ? triA.getAddress() : triB.getAddress();
373 String bAdr = fuerA ? triB.getAddress() : triA.getAddress();
374 String aDouAdr = fuerA ? douA.getAddress() : douB.getAddress();
375 String bDouAdr = fuerA ? douB.getAddress() : douA.getAddress();
376 String aTeteAdr = fuerA ? teteA.getAddress() : teteB.getAddress();
377 String bTeteAdr = fuerA ? teteB.getAddress() : teteA.getAddress();
378 return "WENN(" + aAdr + ">" + bAdr + ";1;0)"
379 + "+WENN(" + aDouAdr + ">" + bDouAdr + ";1;0)"
380 + "+WENN(" + aTeteAdr + ">" + bTeteAdr + ";1;0)";
381 }

397 private void formatieren(List<List<TeamPaarung>> spielPlan) throws GenerateException {
398 int letzteSpielZeile = letzteSpielZeile();
399 RangePosition allDataMitHeader = RangePosition.from(SPIEL_NR_SPALTE, ERSTE_HEADER_ZEILE,
400 SP_PUNKTE_B, letzteSpielZeile);
401 RangeProperties rangeProp = RangeProperties.from().setBorder(BorderFactory.from().allThin().toBorder())
402 .centerJustify().setShrinkToFit(true).topMargin(110).bottomMargin(110).setCharHeight(10);
403 RangeHelper.from(this, allDataMitHeader).setRangeProperties(rangeProp);
404
405 // gerade/ungerade
406 RangePosition runden = RangePosition.from(SPIEL_NR_SPALTE, ERSTE_DATEN_ZEILE, SP_PUNKTE_B, letzteSpielZeile);
407 Integer farbeGerade = getKonfigurationSheet().getSpielPlanHintergrundFarbeGerade();
408 Integer farbeUngerade = getKonfigurationSheet().getSpielPlanHintergrundFarbeUnGerade();
409 RanglisteGeradeUngeradeFormatHelper.from(this, runden).geradeFarbe(farbeGerade)
410 .ungeradeFarbe(farbeUngerade).apply();
411
412 // Editierbare Zellen: Bahnen + Ergebnisse aller drei Partien
413 EditierbaresZelleFormatHelper.anwenden(this, RangePosition.from(
414 BAHN_TRI_SPALTE, ERSTE_DATEN_ZEILE, BAHN_TETE_SPALTE, letzteSpielZeile));
415 EditierbaresZelleFormatHelper.anwenden(this, RangePosition.from(
416 TRI_A_SPALTE, ERSTE_DATEN_ZEILE, TETE_B_SPALTE, letzteSpielZeile));
417
418 // Trenner: zwischen Runden (= je anzPaarungen Zeilen) ein dicker Strich
419 RangeProperties trenner = RangeProperties.from()
420 .setBorder(BorderFactory.from().doubleLn().forBottom().toBorder());
421 int anzRunden = spielPlan.size();
422 int anzPaarungen = spielPlan.get(0).size();
423 Position trennerPos = Position.from(SPIEL_NR_SPALTE, ERSTE_DATEN_ZEILE);
424 for (int i = 1; i < anzRunden; i++) {
425 trennerPos.zeilePlus(anzPaarungen - 1);
426 RangePosition trennerRange = RangePosition.from(SPIEL_NR_SPALTE, trennerPos.getZeile(),
427 SP_PUNKTE_B, trennerPos.getZeile());
428 RangeHelper.from(this, trennerRange).setRangeProperties(trenner);
429 trennerPos.zeilePlusEins();
430 }
431
