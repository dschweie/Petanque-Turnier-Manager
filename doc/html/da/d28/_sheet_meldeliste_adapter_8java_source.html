105 static Optional<MeldelisteZiel> fuer(WorkingSpreadsheet ws, String sheetName,
106 TurnierSystem ts, Formation formation, boolean teamnameAktiv, boolean vereinsnameAktiv) {
107 try {
108 if (ts == null || ts == TurnierSystem.KEIN) {
109 return Optional.empty();
110 }
111 SheetHelper sh = new SheetHelper(ws);
112 XSpreadsheet sheet = sh.findByName(sheetName);
113 if (sheet == null) {
114 return Optional.empty();
115 }
116 int datenZeile = ermittleErsteDatenZeile(sh, sheet);
117 return Optional.of(new SheetMeldelisteAdapter(
118 ws.getWorkingSpreadsheetDocument(), sheet, sh, ts, formation, datenZeile,
119 teamnameAktiv, vereinsnameAktiv));
120 } catch (Exception e) {
121 logger.warn("Adapter-Erkennung fehlgeschlagen", e);
122 return Optional.empty();
123 }
124 }

126 private static int ermittleErsteDatenZeile(SheetHelper sh, XSpreadsheet sheet) {
127 for (int zeile = 0; zeile <= HEADER_ZEILE_MAX_SCAN; zeile++) {
128 String inhalt = sicherText(sh, sheet, SPALTE_NR, zeile).strip();
129 if (inhalt.matches("\\d+")) {
130 return zeile;
131 }
132 }
133 // Fallback: erste Zeile direkt nach erkanntem Header — Spalte A leer.
134 for (int zeile = 0; zeile <= HEADER_ZEILE_MAX_SCAN; zeile++) {
135 if (sicherText(sh, sheet, SPALTE_NR, zeile).strip().isEmpty()) {
136 return zeile;
137 }
138 }
139 return 2;
140 }

178 namen.add(voll);

190 @Override

192 List<MeldelisteSpielerDaten> result = new ArrayList<>();

197 break;
198 }
199 for (int s = 0; s < anzSpieler; s++) {
200 String vor = sicherText(sheetHelper, sheet, vornameSpalte(s), zeile).strip();
201 String nach = sicherText(sheetHelper, sheet, nachnameSpalte(s), zeile).strip();
202 if (vor.isEmpty() && nach.isEmpty()) {
203 continue;
204 }
205 String verein = vereinsnameAktiv
206 ? sicherText(sheetHelper, sheet, vereinSpalte(s), zeile).strip()
207 : null;
208 result.add(new MeldelisteSpielerDaten(vor, nach, verein, zeile + 1));
209 }

225 String nach = sicherText(sheetHelper, sheet, nachnameSpalte(s), zeile).strip();

243 @Override
244 public int schreibeBlock(List<SpielerMitVerein> spieler) throws MeldelisteSchreibException {
245 if (spieler.isEmpty()) {
246 return 0;
247 }
248 if (spieler.size() > anzSpieler) {
249 throw new MeldelisteSchreibException(
250 "Mehr Spieler als Slots: " + spieler.size() + " > " + anzSpieler);
251 }
252 int zeile = naechsteFreieZeile();
253 if (zeile < 0) {
254 throw new MeldelisteSchreibException("Keine freie Zeile in der Meldeliste");
255 }
256
257 try {
258 // Team-Nr (Spalte 0) wird bewusst nicht hier gesetzt — der
259 // anschließende „Meldeliste Aktualisieren"-Lauf vergibt fortlaufend
260 // konsistente Nummern für alle aktiven Teams.
261 RowData row = new RowData();
262 // Teamname-Slot bleibt leer — User füllt das ggf. manuell.
263 if (teamnameAktiv) {
264 row.newEmpty();
265 }
266 for (SpielerMitVerein s : spieler) {
267 row.newString(s.vorname());
268 row.newString(s.nachname());
269 if (vereinsnameAktiv) {
270 row.newString(s.vereinName() == null ? "" : s.vereinName());
271 }
272 }
273 // Padding für nicht gefüllte Slots (Schutz, eigentlich blockiert der
274 // Dialog die Übernahme bevor das Team unvollständig ist):
275 int luecke = (anzSpieler - spieler.size()) * spaltenProSpieler;
276 for (int i = 0; i < luecke; i++) {
277 row.newEmpty();
278 }
279
280 RangeData rangeData = new RangeData();
281 rangeData.add(row);
282
283 // Range startet immer in Spalte 1 (rechts neben Spalte 0 = Nr).
284 RangePosition pos = RangePosition.from(1, zeile, letzteSchreibSpalte, zeile);
285 RangeHelper.from(sheet, doc, pos).setDataInRange(rangeData);
286
287 // Aktiv-Spalte (= letzteDatenSpalte + 2) auf „nimmt teil" setzen,
288 // sonst kommt „Meldeliste Aktualisieren" mit der Frage „Es sind
289 // keine Teams aktiv. Sollen alle aktiviert werden?".
290 int aktivSpalte = letzteSchreibSpalte + 2;
291 sheetHelper.setNumberValueInCell(NumberCellValue
292 .from(sheet, Position.from(aktivSpalte, zeile)).setValue(AKTIV_WERT_NIMMT_TEIL));
293 return spieler.size();
294 } catch (Exception e) {
295 throw new MeldelisteSchreibException("Schreibvorgang fehlgeschlagen", e);
296 }
297 }

305 return zeile;
306 }
307 }
308 return -1;
