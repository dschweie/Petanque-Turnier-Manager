91 @Override
92 public void doRun() throws GenerateException {
93 processBoxinfo("processbox.maastrichter.finalrunde.erstellen");
94
95 if (!pruefeUndAktualisiereVorrundenRangliste()) {
96 return;
97 }
98
99 var meldeliste = new MaastrichterMeldeListeSheetUpdate(getWorkingSpreadsheet());
100 TeamMeldungen aktiveMeldungen = meldeliste.getAktiveMeldungen();
101 if (aktiveMeldungen == null || aktiveMeldungen.size() < 2) {
102 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
103 .caption(I18n.get("maastrichter.finalrunde.caption"))
104 .message(I18n.get("maastrichter.finalrunde.fehler.zu.wenige.teams"))
105 .show();

108
109 MaastrichterKonfigurationSheet konfigSheet = getKonfigurationSheet();
110 int anzVorrunden = konfigSheet.getAnzVorrunden();
111 SchweizerRankingModus modus = konfigSheet.getRankingModus();
112
113 // Alle Vorrunden-Ergebnisse einlesen
114 List<SchweizerTeamErgebnis> ergebnisse = leseVorrundenErgebnisse(
115 aktiveMeldungen, anzVorrunden, meldeliste);
116
117 // Nach Schweizer Kriterien sortieren (für korrekte Setzliste pro Gruppe)
118 List<SchweizerTeamErgebnis> sortiert = new SchweizerSystem()
119 .sortiereNachAuswertungskriterien(ergebnisse, modus);
120
121 if (sortiert.isEmpty()) {
122 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
123 .caption(I18n.get("maastrichter.finalrunde.caption"))
124 .message(I18n.get("maastrichter.finalrunde.fehler.keine.ergebnisse"))
125 .show();
126 return;

128
129 // Gruppen gemäß konfiguriertem Modus bilden

131 List<List<SchweizerTeamErgebnis>> gruppen = switch (gruppenModus) {
132 case NACH_SIEGEN -> teileNachSiegen(sortiert, anzVorrunden);
133 case NACH_GROESSE -> teileNachGroesse(sortiert, konfigSheet.getGruppenGroesse(),
134 konfigSheet.getMinLetzteGruppeGroesse());
135 };
136
137 // Alte Finale-Blätter löschen
138 alleFinaleSheetNamenLoeschen();

142 KoTurnierbaumSheet koSheet = new KoTurnierbaumSheet(getWorkingSpreadsheet());
143 short sheetPos = DefaultSheetPos.MAASTRICHTER_FINALE;
144 char naechsterBuchstabe = 'A';
145 Map<Integer, String> teamNrZuGruppe = new HashMap<>();
146
147 for (List<SchweizerTeamErgebnis> gruppeErg : gruppen) {
148 SheetRunner.testDoCancelTask();
149 TeamMeldungen gruppeTeams = erstelleGruppeTeams(gruppeErg, aktiveMeldungen);
150 if (gruppeTeams.size() >= 2) {
151 String gruppenBuchstabe = String.valueOf(naechsterBuchstabe++);

153 processBoxinfo("processbox.erstelle.sheet.teams", sheetName, gruppeErg.size());

155 SheetMetadataHelper.schluesselMaastrichterFinalrunde(gruppenBuchstabe), gruppenBuchstabe);

157 for (SchweizerTeamErgebnis erg : gruppeErg) {
158 teamNrZuGruppe.put(erg.teamNr(), gruppenBuchstabe);
159 }
160 }
161 }
162
163 // Gruppen-Buchstaben einmalig in die Vorrunden-Rangliste schreiben
164 // (Spalte "Gruppe"). Nachfolgende Rangliste-Refreshs erhalten die Werte.
165 if (!teamNrZuGruppe.isEmpty()) {
166 new MaastrichterVorrundenRanglisteSheetUpdate(getWorkingSpreadsheet())
167 .schreibeGruppenZuweisungen(teamNrZuGruppe);
168 }

175 private List<List<SchweizerTeamErgebnis>> teileNachSiegen(
176 List<SchweizerTeamErgebnis> sortiert, int anzVorrunden) {
177
178 Map<Integer, List<SchweizerTeamErgebnis>> nachSiegen = new HashMap<>();
179 for (SchweizerTeamErgebnis erg : sortiert) {
180 nachSiegen.computeIfAbsent(erg.siege(), k -> new ArrayList<>()).add(erg);
181 }
182
183 List<List<SchweizerTeamErgebnis>> gruppen = new ArrayList<>();
184 for (int siege = anzVorrunden; siege >= 0; siege--) {
185 List<SchweizerTeamErgebnis> gruppe = nachSiegen.get(siege);
186 if (gruppe != null && !gruppe.isEmpty()) {
187 gruppen.add(gruppe);
188 }
189 }
190 return gruppen;
191 }

197 private List<List<SchweizerTeamErgebnis>> teileNachGroesse(
198 List<SchweizerTeamErgebnis> sortiert, int gruppenGroesse, int minLetzteGruppe) {
199
200 List<Integer> gruppenGroessen = GruppenAufteilungRechner.berechne(
201 sortiert.size(), gruppenGroesse, minLetzteGruppe);
202 List<List<SchweizerTeamErgebnis>> gruppen = new ArrayList<>();
203 int startIndex = 0;
204 for (int groesse : gruppenGroessen) {
205 gruppen.add(new ArrayList<>(sortiert.subList(startIndex, startIndex + groesse)));
206 startIndex += groesse;
207 }
208 return gruppen;
209 }

214 private List<SchweizerTeamErgebnis> leseVorrundenErgebnisse(
215 TeamMeldungen aktiveMeldungen, int anzVorrunden,
216 MaastrichterMeldeListeSheetUpdate meldeliste) throws GenerateException {
217
218 Map<Integer, int[]> statsMap = new HashMap<>(); // teamNr → [0]=siege, [1]=punkte+, [2]=punkte-
219 Map<Integer, List<Integer>> gegnerMap = new HashMap<>();
220 for (Team team : aktiveMeldungen.teams()) {
221 statsMap.put(team.getNr(), new int[3]);
222 gegnerMap.put(team.getNr(), new ArrayList<>());
223 }
224
225 for (int runde = 1; runde <= anzVorrunden; runde++) {
226 SheetRunner.testDoCancelTask();
227 String legacyName = runde + ". " + SheetNamen.LEGACY_MAASTRICHTER_VORRUNDE_PRAEFIX;
228 XSpreadsheet rundeSheet = SheetMetadataHelper.findeSheetUndHeile(
229 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
230 SheetMetadataHelper.schluesselMaastrichterVorrunde(runde), legacyName);
231 if (rundeSheet == null) {
232 logger.warn("Vorrunden-Sheet '{}' nicht gefunden, übersprungen.", legacyName);
233 continue;
234 }
235 leseRundeEin(rundeSheet, aktiveMeldungen, statsMap, gegnerMap, meldeliste);
236 }
237
238 List<SchweizerTeamErgebnis> ergebnisse = new ArrayList<>();
239 for (Team team : aktiveMeldungen.teams()) {
240 int[] stats = statsMap.getOrDefault(team.getNr(), new int[3]);
241 List<Integer> gegnerNrn = gegnerMap.getOrDefault(team.getNr(), new ArrayList<>());
242 int punkteDiff = stats[1] - stats[2];
243 ergebnisse.add(new SchweizerTeamErgebnis(team.getNr(), stats[0], punkteDiff, stats[1], gegnerNrn));
244 }
245 return ergebnisse;
246 }

251 private void leseRundeEin(XSpreadsheet rundeSheet, TeamMeldungen aktiveMeldungen,
252 Map<Integer, int[]> statsMap, Map<Integer, List<Integer>> gegnerMap,
253 MaastrichterMeldeListeSheetUpdate meldeliste) throws GenerateException {
254
255 RangePosition readRange = RangePosition.from(
256 SchweizerAbstractSpielrundeSheet.TEAM_A_SPALTE,
257 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE,
258 SchweizerAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE,
259 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE + 999);
260
261 RangeData rowsData = RangeHelper
262 .from(rundeSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
263 .getDataFromRange();
264
265 for (RowData row : rowsData) {
266 if (row.size() < 2) break;
267
268 int nrA = resolveTeamNr(row.get(0), meldeliste);
269 if (nrA <= 0) break;
270 Team teamA = aktiveMeldungen.getTeam(nrA);
271 if (teamA == null) continue;
272
273 int nrB = resolveTeamNr(row.get(1), meldeliste);
274 if (nrB <= 0) {
275 // Freilos für Team A
276 statsMap.computeIfAbsent(nrA, k -> new int[3])[0]++;
277 continue;
278 }
279 Team teamB = aktiveMeldungen.getTeam(nrB);
280 if (teamB == null) continue;
281
282 gegnerMap.computeIfAbsent(nrA, k -> new ArrayList<>()).add(nrB);
283 gegnerMap.computeIfAbsent(nrB, k -> new ArrayList<>()).add(nrA);
284
285 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0;
286 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0;
287
288 if (ergA > 0 || ergB > 0) {
289 statsMap.computeIfAbsent(nrA, k -> new int[3])[1] += ergA;
290 statsMap.computeIfAbsent(nrA, k -> new int[3])[2] += ergB;
291 statsMap.computeIfAbsent(nrB, k -> new int[3])[1] += ergB;
292 statsMap.computeIfAbsent(nrB, k -> new int[3])[2] += ergA;
293 if (ergA > ergB) {
294 statsMap.computeIfAbsent(nrA, k -> new int[3])[0]++;
295 } else if (ergB > ergA) {
296 statsMap.computeIfAbsent(nrB, k -> new int[3])[0]++;
297 }
298 }
299 }
300 }

302 private int resolveTeamNr(CellData cell, MaastrichterMeldeListeSheetUpdate meldeliste) throws GenerateException {
303 int nr = cell.getIntVal(0);

305 String name = cell.getStringVal();
