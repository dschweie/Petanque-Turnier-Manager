75 processBoxinfo("processbox.poule.ko.erstellen");
76
77 if (!pruefeUndAktualisiereVorrundenRangliste()) {
78 return;
79 }
80
81 var vorrundeSheet = SheetMetadataHelper.findeSheetUndHeile(
82 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
83 SheetMetadataHelper.SCHLUESSEL_POULE_VORRUNDE,
84 SheetNamen.pouleVorrunde());

86 if (vorrundeSheet == null) {

88 .caption(I18n.get("poule.ko.caption"))

98 .caption(I18n.get("poule.ko.caption"))
99 .message(I18n.get("poule.ko.fehler.keine.ergebnisse"))
100 .show();
101 return;
102 }
103
104 var meldeliste = new PouleMeldeListeSheetUpdate(getWorkingSpreadsheet());
105 TeamMeldungen aktiveMeldungen = meldeliste.getAktiveMeldungen();
106
107 List<PouleTeamErgebnis> aTeamsList = bestimmeATeams(gruppenErgebnisse);
108 List<PouleTeamErgebnis> bTeamsList = bestimmeBTeams(gruppenErgebnisse);
109

111 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
112 .caption(I18n.get("poule.ko.caption"))
113 .message(I18n.get("poule.ko.fehler.keine.ergebnisse"))
114 .show();

117
118 // Alte KO-Sheets löschen
119 alleKoSheetsLoeschen();
120
121 var konfigAdapter = new PouleKoConfigAdapter(getKonfigurationSheet());
122 var koSheet = new KoTurnierbaumSheet(getWorkingSpreadsheet());
123
124 // A-Turnier erstellen

136 processBoxinfo("processbox.erstelle.sheet.teams", bSheetName, bTeams.size());

139 konfigAdapter, SheetMetadataHelper.schluesselPouleKo("B"));
140 }

148 private List<List<PouleTeamErgebnis>> leseUndSortiereGruppen(XSpreadsheet vorrundeSheet)
149 throws GenerateException {
150
151 var readRange = RangePosition.from(
152 AbstractPouleVorrundeSheet.SPALTE_POULE_NR,
153 AbstractPouleVorrundeSheet.ERSTE_DATEN_ZEILE,
154 AbstractPouleVorrundeSheet.SPALTE_ERG_B,
155 AbstractPouleVorrundeSheet.ERSTE_DATEN_ZEILE + 9999);
156
157 var rowsData = RangeHelper
158 .from(vorrundeSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
159 .getDataFromRange();
160
161 List<Map<Integer, int[]>> gruppenRoh = new ArrayList<>();
162 Map<Integer, int[]> aktuelleGruppe = null;
163
164 for (RowData row : rowsData) {
165 if (row.size() < 7) {
166 break;
167 }
168
169 // Neue Gruppe: SPALTE_POULE_NR ist nur in der ersten Zeile des Merge-Blocks nicht leer.
170 String pouleNrStr = row.get(0).getStringVal();
171 if (pouleNrStr != null && !pouleNrStr.isEmpty()) {
172 aktuelleGruppe = new HashMap<>();
173 gruppenRoh.add(aktuelleGruppe);
174 }
175
176 if (aktuelleGruppe == null) {
177 continue;
178 }
179
180 int teamANr = row.get(1).getIntVal(0);
181 int teamBNr = row.get(3).getIntVal(0);
182
183 int ergA = row.get(5).getIntVal(0);
184 int ergB = row.get(6).getIntVal(0);
185
186 if (teamANr > 0 && teamBNr == 0) {
187 aktuelleGruppe.computeIfAbsent(teamANr, k -> new int[4])[0]++;
188 continue;
189 }
190
191 if (teamANr <= 0) {
192 continue;
193 }
194
195 aktuelleGruppe.computeIfAbsent(teamANr, k -> new int[4]);
196 aktuelleGruppe.computeIfAbsent(teamBNr, k -> new int[4]);
197
198 if (ergA > 0 || ergB > 0) {
199 aktuelleGruppe.get(teamANr)[2] += ergA;
200 aktuelleGruppe.get(teamANr)[3] += ergB;
201 aktuelleGruppe.get(teamBNr)[2] += ergB;
202 aktuelleGruppe.get(teamBNr)[3] += ergA;
203
204 if (ergA > ergB) {
205 aktuelleGruppe.get(teamANr)[0]++;
206 aktuelleGruppe.get(teamBNr)[1]++;
207 } else if (ergB > ergA) {
208 aktuelleGruppe.get(teamBNr)[0]++;
209 aktuelleGruppe.get(teamANr)[1]++;
210 }
211 }
212 }
213
214 var rechner = new PouleRanglisteRechner();
215 var sortiertGruppen = new ArrayList<List<PouleTeamErgebnis>>();
216
217 for (var gruppeRoh : gruppenRoh) {
218 var ergebnisse = new ArrayList<PouleTeamErgebnis>();
219 for (var entry : gruppeRoh.entrySet()) {
220 int teamNr = entry.getKey();
221 int[] stats = entry.getValue();
222 ergebnisse.add(new PouleTeamErgebnis(teamNr, stats[0], stats[1],
223 stats[2] - stats[3], stats[2], List.of()));
224 }
225 sortiertGruppen.add(rechner.sortiere(ergebnisse));
226 }
227
228 return sortiertGruppen;
229 }

239 private List<PouleTeamErgebnis> bestimmeATeams(List<List<PouleTeamErgebnis>> gruppen) {
240 var aTeams = new ArrayList<PouleTeamErgebnis>();
241
242 // Erst alle Platz-1-Teams in Paarfolge
243 for (int i = 0; i < gruppen.size(); i += 2) {
244 if (!gruppen.get(i).isEmpty()) {
245 aTeams.add(gruppen.get(i).get(0));
246 }
247 if (i + 1 < gruppen.size() && !gruppen.get(i + 1).isEmpty()) {
248 aTeams.add(gruppen.get(i + 1).get(0));
249 }
250 }
251
252 // Dann alle Platz-2-Teams in umgekehrter Reihenfolge
253 for (int i = gruppen.size() - 1; i >= 0; i--) {
254 if (gruppen.get(i).size() >= 2) {
255 aTeams.add(gruppen.get(i).get(1));
256 }
257 }
258
259 return aTeams;
260 }

265 private List<PouleTeamErgebnis> bestimmeBTeams(List<List<PouleTeamErgebnis>> gruppen) {
266 var bTeams = new ArrayList<PouleTeamErgebnis>();
267
268 // Platz-3-Teams in Paarfolge
269 for (int i = 0; i < gruppen.size(); i += 2) {
270 if (gruppen.get(i).size() >= 3) {
271 bTeams.add(gruppen.get(i).get(2));
272 }
273 if (i + 1 < gruppen.size() && gruppen.get(i + 1).size() >= 3) {
274 bTeams.add(gruppen.get(i + 1).get(2));
275 }
276 }
277
278 // Platz-4-Teams in umgekehrter Reihenfolge
279 for (int i = gruppen.size() - 1; i >= 0; i--) {
280 if (gruppen.get(i).size() >= 4) {
281 bTeams.add(gruppen.get(i).get(3));
282 }
283 }
284
285 return bTeams;
286 }
