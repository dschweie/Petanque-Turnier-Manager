120 public void createMeldelisteWithParams() throws GenerateException {
121 var neuesSheet = NewSheet.from(this, SheetNamen.meldeliste(), METADATA_SCHLUESSEL)
122 .pos(DefaultSheetPos.MELDELISTE).hideGrid().tabColor(getKonfigurationSheet().getMeldelisteTabFarbe()).setDocVersionWhenNew().create();

124 upDateSheet();

128 @Override
129 protected void doRun() throws GenerateException {
130 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.KO)
131 .prefix(getLogPrefix()).validate()) {
132 return;
133 }
134
135 // KonfigSheet anlegen, damit Defaults lesbar sind
136 getKonfigurationSheet().update();
137
138 // Dialog anzeigen
139 KoKonfigurationSheet konfig = getKonfigurationSheet();
140 Optional<KoTurnierParameterDialog.TurnierParameter> result;
141 try {
142 result = KoTurnierParameterDialog.from(getWorkingSpreadsheet())
143 .show(konfig.getMeldeListeFormation(),
144 konfig.isMeldeListeTeamnameAnzeigen(),
145 konfig.isMeldeListeVereinsnameAnzeigen(),
146 konfig.getSpielbaumTeamAnzeige(),
147 konfig.getSpielbaumSpielbahn(),
148 konfig.isSpielbaumSpielUmPlatz3(),
149 konfig.getGruppenGroesse(),
150 konfig.getMinLetzteGruppeGroesse());
151 } catch (com.sun.star.uno.Exception e) {
152 String errMsg = I18n.get("error.dialog.parameterdialog", e.getMessage());
153 logger.error(errMsg, e);
154 throw new GenerateException(errMsg);
155 }
156
157 if (result.isEmpty()) {
158 return;
159 }
160
161 // Werte in KonfigSheet speichern
162 var params = result.get();
163 konfig.setMeldeListeFormation(params.formation);
164 konfig.setMeldeListeTeamnameAnzeigen(params.teamnameAnzeigen);
165 konfig.setMeldeListeVereinsnameAnzeigen(params.vereinsnameAnzeigen);
166 konfig.setSpielbaumTeamAnzeige(params.spielbaumTeamAnzeige);
167 konfig.setSpielbaumSpielbahn(params.spielbaumSpielbahn);
168 konfig.setSpielbaumSpielUmPlatz3(params.spielUmPlatz3);
169 konfig.setGruppenGroesse(params.gruppenGroesse);
170 konfig.setMinLetzteGruppeGroesse(params.minLetzteGruppeGroesse);
171
172 // KonfigSheet mit neuen Werten neu rendern
173 getKonfigurationSheet().update();
174
175 // Alle anderen Blätter entfernen, dann Meldeliste erstellen
176 getSheetHelper().removeAllSheetsExclude();
177 createMeldelisteWithParams();
178 }
