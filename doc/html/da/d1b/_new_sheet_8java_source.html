143 public NewSheet create() throws GenerateException {
144 XSpreadsheetDocument xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
145 sheet = sheetHelper.findByName(sheetName);
146 didCreate = false;
147 didCreateRun = true;
148
149 TurnierSheet turnierSheet = null;
150
151 // i18n-aware Suche: Sheet könnte nach Sprachenwechsel anders heißen
152 if (sheet == null && hatMetadaten()) {
153 Optional<XSpreadsheet> byKey = SheetMetadataHelper.findeSheet(xDoc, metadatenSchluessel);
154 if (byKey.isPresent()) {
155 XSpreadsheet gefunden = byKey.get();
156 String alterName = TurnierSheet.from(gefunden, getWorkingSpreadsheet()).getName();
157 if (alterName.equals(sheetName)) {
158 logger.error("Inkonsistenz: Sheet '{}' per Metadaten gefunden, aber nicht per findByName (key='{}')",
159 sheetName, metadatenSchluessel);
160 } else {
161 logger.warn("Sheet '{}' → '{}' (i18n-Umbenennung, Risiko: String-Referenzen in Formeln)",
162 alterName, sheetName);
163 sheetHelper.reNameSheet(gefunden, sheetName);
164 sheet = gefunden;
165 }
166 }
167 }
168
169 if (sheet != null) {
170 turnierSheet = TurnierSheet.from(sheet, getWorkingSpreadsheet());
171 if (createNewIfExist) {
172 // setActiv() nur wenn der User einen Dialog sieht – bei forceCreate() (programmatisch)
173 // würde setActiv() unnötig den SheetSyncListener auslösen (isRunning=false im Test)
174 if (!forceOkCreateNewWhenExist) {
175 turnierSheet.setActiv();
176 }
177 MessageBoxResult result = MessageBox
178 .from(getWorkingSpreadsheet().getxContext(), MessageBoxTypeEnum.WARN_YES_NO)
179 .caption(I18n.get("msg.caption.tabelle.erstellen", sheetName))
180 .message(I18n.get("msg.text.tabelle.erstellen.frage", sheetName))
181 .forceOk(forceOkCreateNewWhenExist).show();
182 if (MessageBoxResult.YES != result) {
183 return this;
184 }
185 // removeSheet() bereinigt intern bereits verwaiste Named-Range-Einträge
186 sheetHelper.removeSheet(sheetName);
187 sheet = null;
188 turnierSheet = null;
189 }
190 }
191
192 if (sheet == null) {
193 try {
194 sheetHelper.newIfNotExist(sheetName, pos);
195
196 if (setDocVersionWhenNew) {
197 DocumentHelper.setDocErstelltMitVersion(getWorkingSpreadsheet());
198 }
199
200 sheet = sheetHelper.findByName(sheetName);
201 turnierSheet = TurnierSheet.from(sheet, getWorkingSpreadsheet());
202 if (!showGrid) {
203 // nur bei Neu, einmal abschalten
204 turnierSheet.toggleSheetGrid();
205 }
206
207 if (pageStyleDef != null) {
208 // Info: alle PageStyles werden in KonfigurationSheet initialisiert, (Header etc)
209 // @see KonfigurationSheet#initPageStyles
210 // @see SheetRunner#updateKonfigurationSheet
211 // sheet direkt übergeben – Metadaten sind zu diesem Zeitpunkt noch nicht geschrieben,
212 // daher darf applytoSheet() nicht über iSheet.getXSpreadSheet() suchen.
213 PageStyleHelper.from(getISheet(), pageStyleDef).initDefaultFooter().create().applytoSheet(sheet);
214 } else {
215 // dann nur der default, mit copyright footer
216 PageStyleHelper.from(getISheet(), new PageStyleDef(PageStyle.PETTURNMNGR)).initDefaultFooter()
217 .create().applytoSheet(sheet);
218 }
219 didCreate = true;
220
221 } catch (IllegalArgumentException e) {
222 logger.error(e.getMessage(), e);
223 }
224 }
225
226 if (turnierSheet != null) {
227 turnierSheet.protect(protect).tabColor(tabColor).setActiv(setActiv);
228 }
229
230 if (hatMetadaten() && sheet != null) {
231 if (didCreate || !SheetMetadataHelper.istRegistriertesSheet(xDoc, sheet, metadatenSchluessel)) {
232 SheetMetadataHelper.schreibeSheetMetadaten(xDoc, sheet, metadatenSchluessel);
233 }
234 }
235
236 return this;
237 }
