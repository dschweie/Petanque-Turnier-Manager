119 public RanglisteGeradeUngeradeFormatHelper apply() throws GenerateException {
120 // Direkte Zeilenfärbung als Basis; bedingte Formatierungen überschreiben bei Bedarf
121 SheetHelper.faerbeZeilenAbwechselnd(sheet, rangePos, geradeColor, ungeradeColor);
122
123 ConditionalFormatHelper conditionalFormatHelper = ConditionalFormatHelper.from(sheet, rangePos).clear();
124
125 // formula when in validate spalte ein error
126 // Formula fuer sort error, komplette zeile rot einfärben wenn fehler meldung
127 // Achtung spalte plus 1 weil A ist nicht 0 sondern 1
128 if (validateSpalteNr > -1) {
129 String formulaSortError = "LEN(TRIM(INDIRECT(ADDRESS(ROW();" + (validateSpalteNr + 1) + "))))>0";
130 // -----------------------------
131 // Formula fuer sort error, komplette zeile rot einfärben wenn fehler meldung
132 conditionalFormatHelper.formula1(formulaSortError).operator(ConditionOperator.FORMULA)
133 .style(new FehlerStyle()).applyAndDoReset();
134 }
135
136 if (redCharEqualToValue.isPresent()) {
137 conditionalFormatHelper.formulaIsEvenAndEqualToInt(redCharEqualToValue.get())
138 .style(new RanglisteHintergrundFarbeGeradeCharRedStyle(geradeColor)).applyAndDoReset();
139 conditionalFormatHelper.formulaIsOddAndEqualToInt(redCharEqualToValue.get())
140 .style(new RanglisteHintergrundFarbeUnGeradeCharRedStyle(ungeradeColor)).applyAndDoReset();
141 }
142
143 if (greenCharEqualToValue.isPresent()) {
144 conditionalFormatHelper.formulaIsEvenAndEqualToInt(greenCharEqualToValue.get())
145 .style(new RanglisteHintergrundFarbeGeradeCharGreenStyle(geradeColor)).applyAndDoReset();
146 conditionalFormatHelper.formulaIsOddAndEqualToInt(greenCharEqualToValue.get())
147 .style(new RanglisteHintergrundFarbeUnGeradeCharGreenStyle(ungeradeColor)).applyAndDoReset();
148 }
149
150 if (orangeCharEvenFormula.isPresent()) {
151 conditionalFormatHelper.formula1(orangeCharEvenFormula.get()).isFormula()
152 .style(new RanglisteHintergrundFarbeGeradeCharOrangeStyle(geradeColor)).applyAndDoReset();
153 }
154 if (orangeCharOddFormula.isPresent()) {
155 conditionalFormatHelper.formula1(orangeCharOddFormula.get()).isFormula()
156 .style(new RanglisteHintergrundFarbeUnGeradeCharOrangeStyle(ungeradeColor)).applyAndDoReset();
157 }
158
159 if (streichSpieltagFormulaGerade.isPresent()) {
160 conditionalFormatHelper.formula1(streichSpieltagFormulaGerade.get()).isFormula()
161 .style(new StreichSpieltagHintergrundFarbeGeradeStyle(streichSpieltagFarbeGerade.get()))
162 .applyAndDoReset();
163 }
164 if (streichSpieltagFormulaUnGerade.isPresent()) {
165 conditionalFormatHelper.formula1(streichSpieltagFormulaUnGerade.get()).isFormula()
166 .style(new StreichSpieltagHintergrundFarbeUnGeradeStyle(streichSpieltagFarbeUnGerade.get()))
167 .applyAndDoReset();
168 }
169
170 if (nichtGespieltSpalteNr > -1) {
171 String nichtGespieltBasis = "LEN(TRIM(INDIRECT(ADDRESS(ROW();" + (nichtGespieltSpalteNr + 1) + "))))>0";
172 conditionalFormatHelper.formula1("AND(ISEVEN(ROW());" + nichtGespieltBasis + ")").isFormula()
173 .style(new NichtGespieltHintergrundFarbeGeradeStyle(nichtGespieltGeradeFarbe)).applyAndDoReset();
174 conditionalFormatHelper.formula1("AND(ISODD(ROW());" + nichtGespieltBasis + ")").isFormula()
175 .style(new NichtGespieltHintergrundFarbeUnGeradeStyle(nichtGespieltUnGeradeFarbe)).applyAndDoReset();
176 }
177
178 return this;
179 }
