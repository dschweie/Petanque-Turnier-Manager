88 public SignaturErgebnis berechne(XSpreadsheetDocument xDoc, int versuch) {
89 checkNotNull(xDoc);
90 List<SignaturQuelle> quellen;
91 try {
92 quellen = quellenLieferant.apply(xDoc);
93 } catch (DisposedException e) {
94 return new SignaturErgebnis.TransientFehler("Dokument verworfen", e, versuch);
95 } catch (RuntimeException e) {
96 logger.warn("Quellen-Lieferant warf RuntimeException", e);
97 return new SignaturErgebnis.PermanenterFehler("Quellen-Lieferant fehlerhaft", e);
98 }
99 if (quellen == null || quellen.isEmpty()) {
100 return new SignaturErgebnis.PermanenterFehler("Quellen-Lieferant lieferte keine Quellen", null);
101 }
102
103 List<SignaturQuelle> sortiert = new ArrayList<>(quellen);
104 sortiert.sort(Comparator.comparing(SignaturQuelle::stabileId));
105
106 MessageDigest digest;
107 try {
108 digest = MessageDigest.getInstance("SHA-256");
109 } catch (NoSuchAlgorithmException e) {
110 return new SignaturErgebnis.PermanenterFehler("SHA-256 nicht verfügbar", e);
111 }
112
113 for (SignaturQuelle quelle : sortiert) {
114 SignaturErgebnis teilErgebnis = serialisiereQuelle(xDoc, quelle, digest, versuch);
115 if (teilErgebnis != null) {
116 return teilErgebnis;
117 }
118 }
119
120 String zusatzKontext;
121 try {
122 zusatzKontext = zusatzKontextLieferant.apply(xDoc);
123 } catch (DisposedException e) {
124 return new SignaturErgebnis.TransientFehler("Zusatz-Kontext verworfen", e, versuch);
125 } catch (RuntimeException e) {
126 logger.warn("Zusatz-Kontext-Lieferant warf RuntimeException", e);
127 return new SignaturErgebnis.PermanenterFehler("Zusatz-Kontext-Lieferant fehlerhaft", e);
128 }
129 if (zusatzKontext != null && !zusatzKontext.isEmpty()) {
130 digest.update(("CTX=" + zusatzKontext + "\n").getBytes(StandardCharsets.UTF_8));
131 }
132
133 String hex = HexFormat.of().formatHex(digest.digest());
134 return new SignaturErgebnis.Ok(hex);
135 }

190 private void kanonisiereInDigest(SignaturQuelle quelle, RangeData daten, MessageDigest digest) {
191 int ersteAbsoluteSpalte = quelle.ersteRelevanteSpalte();
192 int anzNichtleere = 0;
193 StringBuilder zeilen = new StringBuilder(256);
194
195 for (int zeilenIdx = 0; zeilenIdx < daten.size(); zeilenIdx++) {
196 RowData zeile = daten.get(zeilenIdx);
197 StringBuilder zellen = new StringBuilder(64);
198 boolean zeileHatInhalt = false;
199 for (int cellIdx = 0; cellIdx < zeile.size(); cellIdx++) {
200 int absSpalte = ersteAbsoluteSpalte + cellIdx;
201 if (!quelle.relevanteSpalten().contains(absSpalte)) {
202 continue;
203 }
204 Object rohwert = zeile.get(cellIdx).getData();
205 String normiert = normalisiereWert(rohwert);
206 if (normiert == null) {
207 continue;
208 }
209 zellen.append("C=").append(absSpalte).append(";V=").append(normiert).append('\n');
210 zeileHatInhalt = true;
211 }
212 if (zeileHatInhalt) {
213 int absoluteZeile = quelle.ersteZeile() + zeilenIdx;
214 zeilen.append("R=").append(absoluteZeile).append('\n').append(zellen);
215 anzNichtleere++;
216 }
217 }
218
219 digest.update(("Q=" + quelle.stabileId() + ";N=" + anzNichtleere + "\n")
220 .getBytes(StandardCharsets.UTF_8));
221 digest.update(zeilen.toString().getBytes(StandardCharsets.UTF_8));
222 }

231 static String normalisiereWert(Object rohwert) {
232 if (rohwert == null) {
233 return null;
234 }
235 if (rohwert instanceof Number num) {
236 double d = num.doubleValue();
237 if (d == 0.0d && !(rohwert instanceof Double)) {
238 // Integer 0 → bleibt "0"
239 return Long.toString(num.longValue());
240 }
241 if (Double.isFinite(d) && d == Math.floor(d) && !Double.isInfinite(d)
242 && Math.abs(d) < 1e15) {
243 return Long.toString((long) d);
244 }
245 return Double.toString(d);
246 }
247 String s = rohwert.toString().trim();
248 return s.isEmpty() ? null : s;
249 }
