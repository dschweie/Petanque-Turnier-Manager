82 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, WorkingSpreadsheet ws,
83 List<SheetSchutzInfo> infos) {
84 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_KASKADE_MELDELISTE).ifPresent(sheet -> {
85 try {
86 var konfigSheet = new KaskadeKonfigurationSheet(ws);
87 int aktivSpalte = berechneAktivSpalte(konfigSheet);
88 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
89 RangePosition.from(1, MELDELISTE_ERSTE_DATEN_ZEILE,
90 aktivSpalte, MeldungenSpalte.MAX_ANZ_MELDUNGEN))));
91 } catch (GenerateException e) {
92 logger.warn("Editierbare Meldeliste-Spalten konnten nicht berechnet werden: {}", e.getMessage(), e);
93 }
94 });
95 }
96
97 private void sammleKaskadenRundenSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
98 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
99 SheetMetadataHelper.SCHLUESSEL_KASKADE_RUNDE_PREFIX);
100 for (var key : schluessel) {
101 SheetMetadataHelper.findeSheet(xDoc, key).ifPresent(sheet ->
102 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet,
103 List.of(berechneRundeErgebnisBereich(sheet)))));
104 }
105 }
106
107 private RangePosition berechneRundeErgebnisBereich(XSpreadsheet sheet) {
108 return RangePosition.from(
109 KaskadeSpielrundeSheet.ERG_TEAM_A_SPALTE,
110 KaskadeSpielrundeSheet.ERSTE_DATEN_ZEILE,
111 KaskadeSpielrundeSheet.ERG_TEAM_B_SPALTE,
112 ermittleLetzteRundeZeile(sheet));
113 }
114
115 private int ermittleLetzteRundeZeile(XSpreadsheet sheet) {
116 int letzteZeile = KaskadeSpielrundeSheet.ERSTE_DATEN_ZEILE;
117 try {
118 for (int zeile = KaskadeSpielrundeSheet.ERSTE_DATEN_ZEILE;
119 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
120 var xCell = sheet.getCellByPosition(KaskadeSpielrundeSheet.TEAM_A_SPALTE, zeile);
121 if (CellContentType.EMPTY.equals(xCell.getType())) {
122 break;
123 }
124 letzteZeile = zeile;
125 }
126 } catch (Exception e) {
127 logger.warn("Letzte Kaskadenrunden-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
128 }
129 return letzteZeile;
130 }
131
132 private void sammleKaskadenFelderSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
133 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
134 SheetMetadataHelper.SCHLUESSEL_KASKADE_FELD_PREFIX);
135 for (var feldKey : schluessel) {
136 SheetMetadataHelper.findeSheet(xDoc, feldKey).ifPresent(sheet -> {
137 var scoreKey = SheetMetadataHelper.scoreSchluessel(feldKey);
138 var encoded = SheetMetadataHelper.leseScoreText(xDoc, scoreKey);
139 var editierbareBereiche = (encoded != null)
140 ? KoTurnierbaumSheet.decodeScoreBereiche(encoded)
141 : List.<RangePosition>of();
142 if (editierbareBereiche.isEmpty()) {
143 infos.add(SheetSchutzInfo.vollGesperrt(sheet));
144 } else {
145 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, editierbareBereiche));
146 }
147 });
148 }
149 }
150
151 private void sammleGruppenRanglisteSchutzInfo(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
152 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_KASKADE_GRUPPENRANGLISTE)
153 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
154 }

160 private int berechneAktivSpalte(KaskadeKonfigurationSheet konfigSheet) throws GenerateException {
161 int spaltenProSpieler = konfigSheet.isMeldeListeVereinsnameAnzeigen() ? 3 : 2;
162 int ersterOffset = konfigSheet.isMeldeListeTeamnameAnzeigen() ? 2 : 1;
163 int letzteDataSpalte = ersterOffset + konfigSheet.getMeldeListeFormation().getAnzSpieler() * spaltenProSpieler - 1;
164 return letzteDataSpalte + 2; // +1=SP, +2=Aktiv
165 }
