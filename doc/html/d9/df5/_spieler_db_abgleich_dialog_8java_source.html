143 private void ladeFehlende() {
144 UnoControlsHelper c = this.controls;
145 if (c == null) {
146 return;
147 }
148 List<MeldelisteSpielerDaten> ausMeldeliste = quelle.leseAlleSpielerRoh();
149 this.quellSnapshot = List.copyOf(ausMeldeliste);
150 Set<String> dbIndex;
151 try {
152 dbIndex = baueDbIndex();
153 } catch (SpielerDbException e) {
154 logger.error("Spieler-DB lesen fehlgeschlagen", e);
155 zeigeFehler(I18n.get("spielerdb.fehler.dbinit",
156 e.getMessage() == null ? "" : e.getMessage()));
157 return;
158 }
159 if (this.dbKeysAnfang == null) {
160 this.dbKeysAnfang = Set.copyOf(dbIndex);
161 }
162
163 Map<String, FehlendeSpielerDaten> fehlendDedup = new LinkedHashMap<>();
164 int bereitsInDb = 0;
165 for (MeldelisteSpielerDaten m : ausMeldeliste) {
166 String schluessel = matchSchluessel(m.vorname(), m.nachname(), m.vereinName());
167 if (dbIndex.contains(schluessel)) {
168 bereitsInDb++;
169 continue;
170 }
171 fehlendDedup.putIfAbsent(schluessel, new FehlendeSpielerDaten(
172 m.vorname(), m.nachname(), m.vereinName(), m.zeile1Basiert()));
173 }
174 this.fehlend = List.copyOf(fehlendDedup.values());
175
176 c.label("lblKopf", I18n.get("spielerdb.abgleich.kopf",
177 quelle.getSystemBezeichnung(), ausMeldeliste.size(), bereitsInDb, fehlend.size()));
178 c.setzeListItems("lstFehlend",
179 fehlend.stream().map(SpielerDbAbgleichDialog::formatZeile).toArray(String[]::new));
180 boolean nichtLeer = !fehlend.isEmpty();
181 c.enabled("btnImport", nichtLeer);
182 c.enabled("btnAlle", nichtLeer);
183 c.enabled("btnKeine", nichtLeer);

253 private void berichteStatusAnSenke() {
254 if (senke == null || quellSnapshot.isEmpty()) {
255 return;
256 }
257 Set<String> dbAnfang = dbKeysAnfang == null ? Set.of() : dbKeysAnfang;
258 List<ZeilenStatus> liste = new ArrayList<>(quellSnapshot.size());
259 for (MeldelisteSpielerDaten m : quellSnapshot) {
260 String key = matchSchluessel(m.vorname(), m.nachname(), m.vereinName());
261 AbgleichStatus st;
262 String reason = null;
263 if (dbAnfang.contains(key)) {
264 st = AbgleichStatus.IN_DB;
265 } else if (importErfolgKeys.contains(key)) {
266 st = AbgleichStatus.NEU;
267 } else if (importFehlerKeys.containsKey(key)) {
268 st = AbgleichStatus.FEHLER;
269 reason = importFehlerKeys.get(key);
270 } else {
271 st = AbgleichStatus.FEHLT;
272 }
273 liste.add(new ZeilenStatus(m.zeile1Basiert(), st, reason));
274 }
275 try {
276 senke.schreibeStatus(liste);
277 } catch (RuntimeException e) {
278 logger.warn("Status-Rückschreiben in Quelle fehlgeschlagen", e);
279 }

282 private ImportErgebnis importiere(short[] indizes) {
283 // Cache: Vereinsname (normiert) -> Verein-Nr; vermeidet wiederholte
284 // findByName-Lookups innerhalb desselben Import-Vorgangs.
285 Map<String, Integer> vereinNrCache = new HashMap<>();
286 ImportErgebnis erg = new ImportErgebnis();
287 for (short idx : indizes) {
288 if (idx < 0 || idx >= fehlend.size()) {
289 continue;
290 }
291 FehlendeSpielerDaten d = fehlend.get(idx);
292 String key = matchSchluessel(d.vorname(), d.nachname(), d.vereinName());
293 try {
294 Integer vereinNr = vereinNrFuer(d.vereinName(), vereinNrCache, erg);
295 spielerRepo.insert(SpielerDatensatz.neu(
296 d.vorname(), d.nachname(), vereinNr, List.of(), null));
297 erg.importiert++;
298 importErfolgKeys.add(key);
299 importFehlerKeys.remove(key);
300 } catch (SpielerDbException e) {
301 logger.error("Import von '{} {}' fehlgeschlagen", d.vorname(), d.nachname(), e);
302 erg.fehler++;
303 importFehlerKeys.put(key,
304 e.getMessage() == null ? e.getClass().getSimpleName() : e.getMessage());
305 }
306 }
307 return erg;
308 }

310 @Nullable
311 private Integer vereinNrFuer(@Nullable String vereinName,
312 Map<String, Integer> cache, ImportErgebnis erg) throws SpielerDbException {
313 if (vereinName == null || vereinName.isEmpty()) {
314 return null;
315 }
316 String key = SpielerMatchKeyNormalizer.vereinSchluessel(vereinName);
317 Integer cached = cache.get(key);
318 if (cached != null) {
319 return cached;
320 }
321 var vorhanden = vereinRepo.findByName(vereinName);
322 VereinDatensatz v;
323 if (vorhanden.isPresent()) {
324 v = vorhanden.get();
325 } else {
326 try {
327 v = vereinRepo.insert(vereinName);
328 erg.neueVereine++;
329 } catch (VereinRepository.DuplikatException dup) {
330 // Race-Sicherung: Wenn ein paralleler Insert oder ein
331 // case-Variant-Treffer dazwischen kam, beim zweiten Versuch
332 // erneut lesen statt Importschritt abbrechen.
333 v = vereinRepo.findByName(vereinName).orElseThrow(() -> dup);
334 }
335 }
336 Integer nr = v.nr();
337 if (nr != null) {
338 cache.put(key, nr);
339 }
340 return nr;
341 }
