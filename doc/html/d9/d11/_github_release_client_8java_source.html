76 public Optional<ReleaseInfo> ladeLetztesRelease() {
77 var url = baseUri.resolve("/repos/" + repository + "/releases/latest");
78 var request = HttpRequest.newBuilder(url)
79 .timeout(readTimeout)
80 .header("Accept", "application/vnd.github+json")
81 .header("X-GitHub-Api-Version", "2022-11-28")
82 .header("User-Agent", USER_AGENT)
83 .GET()
84 .build();
85 try {
86 var response = httpClient.send(request, HttpResponse.BodyHandlers.ofString());
87 if (response.statusCode() != 200) {
88 logger.warn("GitHub-Release-Abruf für {} lieferte HTTP {}", repository, response.statusCode());
89 return Optional.empty();
90 }
91 return parseAntwort(response.body());
92 } catch (IOException e) {
93 logger.warn("GitHub-Release-Abruf für {} schlug fehl: {}", repository, e.getMessage());
94 return Optional.empty();
95 } catch (InterruptedException e) {
96 Thread.currentThread().interrupt();
97 logger.debug("GitHub-Release-Abruf wurde unterbrochen");
98 return Optional.empty();
99 }
100 }

102 private Optional<ReleaseInfo> parseAntwort(String json) {
103 try {
104 var dto = gson.fromJson(json, ReleaseDto.class);
105 if (dto == null || dto.tagName == null) {
106 logger.warn("GitHub-Antwort enthielt kein tag_name-Feld");
107 return Optional.empty();
108 }
109 var name = (dto.name != null && !dto.name.isBlank()) ? dto.name : dto.tagName;
110 Instant publishedAt = parseInstant(dto.publishedAt);
111 List<AssetInfo> assets = new ArrayList<>();
112 if (dto.assets != null) {
113 for (var asset : dto.assets) {
114 if (asset != null && asset.name != null && asset.browserDownloadUrl != null) {
115 assets.add(new AssetInfo(asset.name, asset.browserDownloadUrl));
116 }
117 }
118 }
119 return Optional.of(new ReleaseInfo(dto.tagName, name, publishedAt, dto.prerelease, dto.body, assets));
120 } catch (JsonSyntaxException e) {
121 logger.warn("GitHub-Antwort war kein gültiges JSON", e);
122 return Optional.empty();
123 }
124 }

126 private static @Nullable Instant parseInstant(@Nullable String iso) {
127 if (iso == null || iso.isBlank()) {
128 return null;

131 return Instant.parse(iso);

133 return null;
