74 public static void speichereNachRebuild(XSpreadsheetDocument xDoc, String schluessel,
75 String hash, String reason) {
76 checkNotNull(xDoc);
77 checkNotNull(schluessel);
78 checkNotNull(hash);
79 checkNotNull(reason);
80 try {
81 DocumentPropertiesHelper props = new DocumentPropertiesHelper(xDoc);
82 String jetzt = Instant.now().toString();
83 props.ohneModifiedFlag(() -> {
84 props.setStringPropertyOhneEvent(propName(schluessel, SUFFIX_HASH), hash);
85 props.setStringPropertyOhneEvent(propName(schluessel, SUFFIX_REBUILD_TS), jetzt);
86 props.setStringPropertyOhneEvent(propName(schluessel, SUFFIX_REASON), reason);
87 props.setStringPropertyOhneEvent(propName(schluessel, SUFFIX_VERIFY_TS), jetzt);
88 props.setBooleanPropertyOhneEvent(propName(schluessel, SUFFIX_RECOVERY), false);
89 });
90 } catch (RuntimeException e) {
