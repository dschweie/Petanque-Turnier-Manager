38 private static File logFile = null;
39
40 static {
41 // logfile aus configuration lesen
42 final LoggerContext ctx = (LoggerContext) LogManager.getContext(false);
43 final Configuration config = ctx.getConfiguration();
44 Map<String, Appender> appenders = config.getAppenders();
45
46 RollingFileAppender fileAppender = (RollingFileAppender) appenders.entrySet().stream()
47 .filter(e -> e.getValue() instanceof RollingFileAppender).map(Map.Entry::getValue).findFirst()
48 .orElse(null);
49
50 if (fileAppender != null) {
51 logFile = new File(fileAppender.getFileName());
52 }
53 }

55 public static final void openLogFile() {
56
57 if (logFile != null) {
58 try {
59 ProcessBox.from().prefix("Log4J").info("Open:" + logFile.getCanonicalPath());
60 String[] cmd = null;
61 // AWT Desktop => coredump, not working
62 // https://stackoverflow.com/questions/526037/how-to-open-user-system-preferred-editor-for-given-file
63 String osName = System.getProperty("os.name");
64 String filePath = logFile.getCanonicalPath();
65 if (StringTools.containsIgnoreCase(osName, "win")) {
66 cmd = new String[] { "rundll32", "url.dll,FileProtocolHandler", filePath };
67 } else if (StringTools.containsIgnoreCase(osName, "linux")) {
68 cmd = new String[] { "xdg-open", filePath };
69 } else if (StringTools.containsIgnoreCase(osName, "mac")) {
70 cmd = new String[] { "open", filePath };
71 }
72 if (cmd != null) {
73 logger.info(String.join(" ", cmd));
74 Runtime.getRuntime().exec(cmd);
75 }
76 } catch (IOException e) {
77 logger.error(e);
78 }
79 } else {
80 ProcessBox.from().prefix("Log4J").fehler("Kann Logdatei nicht ermitteln");
81 }
82 }
