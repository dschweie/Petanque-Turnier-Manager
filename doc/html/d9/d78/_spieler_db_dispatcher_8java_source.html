84 public static void uebernehmenInMeldeliste(WorkingSpreadsheet ws) {
85 XComponentContext ctx = ws.getxContext();
86 Optional<MeldelisteZiel> ziel = MeldelisteZielFactory.fuerAktivesSheet(ws);
87 if (ziel.isEmpty()) {
88 zeigeFehlerKeinZiel(ctx, ws);
89 return;
90 }
91 Optional<SpielerDbConnection> conn = oeffneOderMelde(ctx);
92 if (conn.isEmpty()) {
93 return;
94 }
95 aktiviereMeldelisteSheet(ws);
96 ProcessBox pb = ProcessBox.from();
97 boolean warSichtbar = pb.istSichtbar();
98 if (warSichtbar) {
99 pb.hide();
100 }
101 SpielerSucheDialog dialog = new SpielerSucheDialog(ctx,
102 new SpielerRepository(conn.get()),
103 new VereinRepository(conn.get()),
104 new LabelRepository(conn.get()),
105 ziel.get());
106 try (var _ = BlattschutzManager.get().scopeFuer(aktivesTurnierSystem(ws), ws)) {
107 dialog.zeigen();
108 } catch (com.sun.star.uno.Exception e) {
109 logger.error("Spieler-Suche-Dialog fehlgeschlagen", e);
110 } finally {
111 if (warSichtbar) {
112 pb.visibleWennAutomatisch();
113 }
114 }
115 // Nach Dialog-Schließen (Scope geschlossen, ProcessBox wiederhergestellt):
116 // wurden Spieler übernommen, vergibt der Aktualisieren-Lauf die Team-Nr.
117 if (dialog.wurdenSpielerUebernommen()) {
118 starteMeldelisteUpdate(ws, aktivesTurnierSystem(ws));
119 }

129 private static void starteMeldelisteUpdate(WorkingSpreadsheet ws, TurnierSystem ts) {
130 try {
131 SheetRunner runner = switch (ts) {
132 case SUPERMELEE -> new MeldeListeSheet_Update(ws).testTurnierSystem(TurnierSystem.SUPERMELEE);
133 case SCHWEIZER -> new SchweizerMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.SCHWEIZER);
134 case FORMULEX -> new FormuleXMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.FORMULEX);
135 case KO -> new KoMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.KO).backUpDocument();
136 case JGJ -> new JGJMeldeListeSheet_Update(ws).testTurnierSystem(TurnierSystem.JGJ).backUpDocument();
137 case MAASTRICHTER -> new MaastrichterMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.MAASTRICHTER).backUpDocument();
138 case KASKADE -> new KaskadeMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.KASKADE).backUpDocument();
139 case POULE -> new PouleMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.POULE).backUpDocument();
140 case TRIPTETE -> new de.petanqueturniermanager.triptete.meldeliste.TripTeteMeldeListeSheetUpdate(ws).testTurnierSystem(TurnierSystem.TRIPTETE).backUpDocument();
141 case LIGA, KEIN -> null;
142 };
143 if (runner != null) {
144 runner.start();
145 }
146 } catch (GenerateException e) {
147 logger.error("Meldeliste-Aktualisieren nach Spieler-Übernahme fehlgeschlagen", e);

369 private static Optional<SpielerDbConnection> oeffneOderMelde(XComponentContext ctx) {
370 try {
371 return Optional.of(SpielerDbConnection.getInstance());
372 } catch (SpielerDbException e) {
373 logger.error("Spieler-DB konnte nicht geöffnet werden", e);
374 String text = e.istLockFehler()
375 ? I18n.get("spielerdb.fehler.db_locked")
376 : I18n.get("spielerdb.fehler.dbinit",
377 e.getMessage() == null ? "" : e.getMessage());
378 MessageBox.from(ctx, MessageBoxTypeEnum.ERROR_OK)
379 .caption(I18n.get("spielerdb.fehler.titel"))
380 .message(text)
381 .show();
382 return Optional.empty();
383 } catch (RuntimeException | LinkageError e) {
384 // LinkageError: schützt vor stillem JVM-Tod bei künftigen
385 // Classloader-Konflikten (vgl. HSQLDB-Shading, build.gradle).
386 logger.error("Spieler-DB: unerwarteter Fehler beim Initialisieren", e);
387 MessageBox.from(ctx, MessageBoxTypeEnum.ERROR_OK)
388 .caption(I18n.get("spielerdb.fehler.titel"))
389 .message(I18n.get("spielerdb.fehler.dbinit",
390 e.getClass().getSimpleName() + ": " + e.getMessage()))
391 .show();
392 return Optional.empty();
393 }
