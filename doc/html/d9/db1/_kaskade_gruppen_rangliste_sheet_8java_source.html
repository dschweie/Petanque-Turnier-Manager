117 @Override
118 protected void doRun() throws GenerateException {
119 processBoxinfo("processbox.kaskade.gruppenrangliste.erstellen");
120 getxCalculatable().enableAutomaticCalculation(false);
121
122 var plan = ermittlePlan();
123 var belegungen = ermittleBelegungen(plan);
124 if (belegungen.isEmpty()) {
125 LOGGER.info("Keine Feldbelegungen vorhanden – Gruppenrangliste wird nicht erstellt.");
126 return;
127 }
128
129 var sheetName = SheetNamen.kaskadeGruppenrangliste();
130 var schluessel = SheetMetadataHelper.SCHLUESSEL_KASKADE_GRUPPENRANGLISTE;
131 var tabFarbe = konfigurationSheet.getRanglisteTabFarbe();
132
133 var erstellt = NewSheet.from(this, sheetName, schluessel)
134 .pos(DefaultSheetPos.KASKADE_GRUPPENRANGLISTE)
135 .setForceCreate(isForceOk())
136 .tabColor(tabFarbe)
137 .setActiv()
138 .hideGrid()
139 .create()
140 .isDidCreate();
141
142 if (!erstellt) {
143 return;
144 }
145
146 hauptHeaderSchreiben(plan, belegungen);
147 gruppeHeadersSchreiben(belegungen);
148 datenSchreiben(belegungen);
149 formatieren(belegungen);
150 druckBereichSetzen(plan, belegungen);
151
152 SheetSyncSignaturStore.commitVollaufbau(
153 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
154 SheetMetadataHelper.SCHLUESSEL_KASKADE_GRUPPENRANGLISTE,
155 new EingabeSignatur(SignaturQuellen::fuerKaskade));
156 }

216 private void gruppeHeadersSchreiben(List<KaskadenFeldBelegung> belegungen) throws GenerateException {
217 var sheet = getXSpreadSheet();
218 var headerFarbe = konfigurationSheet.getMeldeListeHeaderFarbe();
219 var posLabel = I18n.get("kaskade.gruppenrangliste.spalte.pos");
220 var nrLabel = I18n.get("kaskade.gruppenrangliste.spalte.team");
221
222 for (int i = 0; i < belegungen.size(); i++) {
223 var belegung = belegungen.get(i);
224 int basisSpalte = i * BLOCK_BREITE;
225 boolean istLetzte = (i == belegungen.size() - 1);
226
227 var gruppeHeaderBorder = istLetzte
228 ? BorderFactory.from().allThin().toBorder()
229 : BorderFactory.from().allThin().doubleLn().forRight().toBorder();
230
231 // Spalten-Header-Border: fette Unterlinie als Trenner zum Datenbereich + ggf. doppelte rechte Trennlinie
232 var spaltenHeaderBorderNr = istLetzte
233 ? BorderFactory.from().allThin().boldLn().forBottom().toBorder()
234 : BorderFactory.from().allThin().boldLn().forBottom().doubleLn().forRight().toBorder();
235
236 // Gruppen-Header (über beide Spalten des Blocks)
237 var gruppeHeader = StringCellValue.from(sheet,
238 Position.from(basisSpalte + BLOCK_POS_OFFSET, GRUPPE_HEADER_ZEILE))
239 .setValue(I18n.get("kaskade.feld.info.header",
240 belegung.bezeichner(), belegung.feld().gesamtTeams()))
241 .setVertJustify(CellVertJustify2.CENTER)
242 .centerHoriJustify()
243 .setCharHeight(CHARHEIGHT_HEADER)
244 .setCharWeight(FontWeight.BOLD)
245 .setShrinkToFit(true)
246 .setCellBackColor(headerFarbe)
247 .setEndPosMergeSpaltePlus(BLOCK_BREITE - 1)
248 .setBorder(gruppeHeaderBorder);
249 getSheetHelper().setStringValueInCell(gruppeHeader);
250
251 // Spalten-Header: Pos – fette Unterlinie als Trenner zum Datenbereich
252 var posHeader = StringCellValue.from(sheet,
253 Position.from(basisSpalte + BLOCK_POS_OFFSET, SPALTEN_HEADER_ZEILE))
254 .setValue(posLabel)
255 .setVertJustify(CellVertJustify2.CENTER)
256 .centerHoriJustify()
257 .setCharHeight(CHARHEIGHT_HEADER)
258 .setCellBackColor(headerFarbe)
259 .setBorder(BorderFactory.from().allThin().boldLn().forBottom().toBorder());
260 getSheetHelper().setStringValueInCell(posHeader);
261
262 // Spalten-Header: Nr – fette Unterlinie + bei nicht-letzter Gruppe doppelte rechte Trennlinie
263 var nrHeader = StringCellValue.from(sheet,
264 Position.from(basisSpalte + BLOCK_NR_OFFSET, SPALTEN_HEADER_ZEILE))
265 .setValue(nrLabel)
266 .setVertJustify(CellVertJustify2.CENTER)
267 .centerHoriJustify()
268 .setCharHeight(CHARHEIGHT_HEADER)
269 .setCellBackColor(headerFarbe)
270 .setBorder(spaltenHeaderBorderNr);
271 getSheetHelper().setStringValueInCell(nrHeader);
272 }

278 private void datenSchreiben(List<KaskadenFeldBelegung> belegungen) throws GenerateException {
279 int maxZeilen = belegungen.stream()
280 .mapToInt(b -> b.teamNrs().size())
281 .max()
282 .orElse(0);
283
284 if (maxZeilen == 0) {
285 return;
286 }
287
288 // Pro Gruppe einen separaten Datenblock schreiben
289 for (int i = 0; i < belegungen.size(); i++) {
290 var belegung = belegungen.get(i);
291 var teamNrs = belegung.teamNrs();
292 int basisSpalte = i * BLOCK_BREITE;
293
294 var datenBlock = new RangeData();
295 for (int pos = 1; pos <= teamNrs.size(); pos++) {
296 var zeile = datenBlock.addNewRow();
297 zeile.add(new CellData(pos)); // Pos
298 zeile.add(new CellData(teamNrs.get(pos - 1))); // Team-Nr
299 }
300
301 var datenStart = Position.from(basisSpalte, ERSTE_DATEN_ZEILE);
302 RangeHelper.from(this, datenBlock.getRangePosition(datenStart)).setDataInRange(datenBlock);
303 }
304 }

309 private void formatieren(List<KaskadenFeldBelegung> belegungen) throws GenerateException {
310 var sheet = getXSpreadSheet();
311 int maxZeilen = belegungen.stream()
312 .mapToInt(b -> b.teamNrs().size())
313 .max()
314 .orElse(0);
315
316 if (maxZeilen == 0) {
317 return;
318 }
319
320 int letzteZeile = ERSTE_DATEN_ZEILE + maxZeilen - 1;
321 int letzteSpalte = belegungen.size() * BLOCK_BREITE - 1;
322 var datenRange = RangePosition.from(0, ERSTE_DATEN_ZEILE, letzteSpalte, letzteZeile);
323
324 TableBorder2 border = BorderFactory.from().allThin().toBorder();
325 getSheetHelper().setPropertyInRange(sheet, datenRange, TABLE_BORDER2, border);
326 getSheetHelper().setPropertyInRange(sheet, datenRange, HORI_JUSTIFY, CellHoriJustify.CENTER);
327 getSheetHelper().setPropertiesInRange(sheet, datenRange, CellProperties.from().margin(MeldeListeKonstanten.CELL_MARGIN));
328 getSheetHelper().setPropertyInRange(sheet, datenRange, VERT_JUSTIFY, CellVertJustify2.CENTER);
329 getSheetHelper().setPropertyInRange(sheet, datenRange, CHAR_HEIGHT, CHARHEIGHT_DATEN);
330
331 // Zebra-Formatierung für Datenbereich
332 RanglisteGeradeUngeradeFormatHelper.from(this, datenRange)
333 .geradeFarbe(konfigurationSheet.getRanglisteHintergrundFarbeGerade())
334 .ungeradeFarbe(konfigurationSheet.getRanglisteHintergrundFarbeUnGerade())
335 .apply();
336
337 // Team-Nr-Spalten fett
338 for (int i = 0; i < belegungen.size(); i++) {
339 int nrSpalte = i * BLOCK_BREITE + BLOCK_NR_OFFSET;
340 var nrRange = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, nrSpalte, letzteZeile);
341 getSheetHelper().setPropertyInRange(sheet, nrRange, CHAR_WEIGHT, FontWeight.BOLD);
342 }
343
344 // Doppelte vertikale Trennlinie rechts an jeder NR-Spalte (außer der letzten Gruppe)
345 var trennBorder = BorderFactory.from().allThin().doubleLn().forRight().toBorder();
346 for (int i = 0; i < belegungen.size() - 1; i++) {
347 int nrSpalte = i * BLOCK_BREITE + BLOCK_NR_OFFSET;
348 var trennRange = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, nrSpalte, letzteZeile);
349 getSheetHelper().setPropertyInRange(sheet, trennRange, TABLE_BORDER2, trennBorder);
350 }
351
352 // Alle Spalten und Zeilen auf optimale Breite/Höhe setzen
353 getSheetHelper().setOptimaleBreiteUndHoeheAlles(sheet, HAUPTHEADER_ZEILE, letzteZeile, 0, letzteSpalte);
354 }

356 private void druckBereichSetzen(KaskadenKoRundenPlan plan, List<KaskadenFeldBelegung> belegungen)
357 throws GenerateException {
358 processBoxinfo("processbox.print.bereich");
359 var xSheet = getXSpreadSheet();
360 if (xSheet == null) {
361 return;
362 }
363 int maxZeilen = belegungen.stream().mapToInt(b -> b.teamNrs().size()).max().orElse(0);
364 if (maxZeilen == 0) {
365 return;
366 }
367 int letzteZeile = ERSTE_DATEN_ZEILE + maxZeilen - 1;
368 int letzteSpalte = belegungen.size() * BLOCK_BREITE - 1;
369 var druckBereich = RangePosition.from(0, HAUPTHEADER_ZEILE, letzteSpalte, letzteZeile);
370 PrintArea.from(xSheet, getWorkingSpreadsheet()).setPrintArea(druckBereich);
371 }
