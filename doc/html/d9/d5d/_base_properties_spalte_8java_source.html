243 public int readIntProperty(String key) {
244 int hardcodedDefault = getHardcodedDefault(key);
245 int effectiveDefault = istTabFarbProp(key)
246 ? GlobalProperties.get().getTabFarbe(key, hardcodedDefault)
247 : hardcodedDefault;
248 return docPropHelper.getIntProperty(key, effectiveDefault);
249 }
250
251 private int getHardcodedDefault(String key) {
252 Object defaultVal = getDefaultProp(key);
253 return defaultVal == null ? 0 : (Integer) defaultVal;
254 }
255
256 private boolean istTabFarbProp(String key) {
257 return getKonfigProperties().stream()
258 .anyMatch(p -> p.getKey().equals(key) && p.isTabFarbe());
259 }

283 public String readStringProperty(String key) {
284 // value aus Document properties lesen
285 Object defaultVal = getDefaultProp(key);
286 String defaultStr = (defaultVal == null) ? "" : defaultVal.toString();
287 // Fehlende Property einmalig mit dem Konfig-Default in die UserDefinedProperties
288 // schreiben — sonst bleibt der Default nur Code-seitig sichtbar, und externe
289 // Leser (z.B. MeldelisteZielFactory) bekämen weiter leeres Ergebnis.
290 // initStringPropertyIfAbsent ist no-op, sobald die Property gesetzt wurde,
291 // es entstehen also keine wiederholten Doc-Mutationen.
292 if (defaultVal != null) {
293 docPropHelper.initStringPropertyIfAbsent(key, defaultStr);
294 }
295 return docPropHelper.getStringProperty(key, defaultStr);
296 }

316 protected <E extends Enum<E>> E readEnumProperty(String key, Class<E> enumClass, E defaultValue) {
317 String val = readStringProperty(key);
318 if (val == null || val.isEmpty()) {
319 return defaultValue;
320 }
321 try {
322 return Enum.valueOf(enumClass, val);
323 } catch (IllegalArgumentException e) {
324 return defaultValue;
325 }
326 }

333 for (ConfigProperty<?> konfigProp : getKonfigProperties()) {
