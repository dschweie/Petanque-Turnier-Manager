80 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, WorkingSpreadsheet ws,
81 List<SheetSchutzInfo> infos) {
82 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_MAASTRICHTER_MELDELISTE).ifPresent(sheet -> {
83 try {
84 var konfigSheet = new MaastrichterKonfigurationSheet(ws);
85 int aktivSpalte = berechneAktivSpalte(konfigSheet);
86 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
87 RangePosition.from(1, MELDELISTE_ERSTE_DATEN_ZEILE,
88 aktivSpalte, MeldungenSpalte.MAX_ANZ_MELDUNGEN))));
89 } catch (GenerateException e) {
90 logger.warn("Editierbare Meldeliste-Spalten konnten nicht berechnet werden: {}", e.getMessage(), e);
91 }
92 });
93 }

100 private void sammleVorrundenUndRanglisteSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
101 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
102 SheetMetadataHelper.SCHLUESSEL_MAASTRICHTER_VORRUNDE_PREFIX);
103 for (var key : schluessel) {
104 SheetMetadataHelper.findeSheet(xDoc, key).ifPresent(sheet -> {
105 if (key.equals(SheetMetadataHelper.SCHLUESSEL_MAASTRICHTER_VORRUNDE_PREFIX)) {
106 // Vorrunden-Rangliste: kein Schlüssel-Suffix → vollständig gesperrt
107 infos.add(SheetSchutzInfo.vollGesperrt(sheet));
108 } else {
109 // Vorrunde: Ergebnis-Spalten editierbar
110 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet,
111 List.of(berechneVorrundeErgebnisBereich(sheet))));
112 }
113 });
114 }
115 }
116
117 private void sammleFinalrundenSchutzInfos(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
118 var schluessel = SheetMetadataHelper.getSchluesselMitPrefix(xDoc,
119 SheetMetadataHelper.SCHLUESSEL_MAASTRICHTER_FINALRUNDE_PREFIX);
120 for (var finalrundeKey : schluessel) {
121 SheetMetadataHelper.findeSheet(xDoc, finalrundeKey).ifPresent(sheet -> {
122 var scoreKey = SheetMetadataHelper.scoreSchluessel(finalrundeKey);
123 var encoded = SheetMetadataHelper.leseScoreText(xDoc, scoreKey);
124 var editierbareBereiche = (encoded != null)
125 ? KoTurnierbaumSheet.decodeScoreBereiche(encoded)
126 : List.<RangePosition>of();
127 if (editierbareBereiche.isEmpty()) {
128 infos.add(SheetSchutzInfo.vollGesperrt(sheet));
129 } else {
130 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, editierbareBereiche));
131 }
132 });
133 }
134 }

140 private RangePosition berechneVorrundeErgebnisBereich(XSpreadsheet sheet) {
141 return RangePosition.from(
142 SchweizerAbstractSpielrundeSheet.ERG_TEAM_A_SPALTE,
143 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE,
144 SchweizerAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE,
145 ermittleLetzteVorrundeZeile(sheet));
146 }
147
148 private int ermittleLetzteVorrundeZeile(XSpreadsheet sheet) {
149 int letzteZeile = SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE;
150 try {
151 for (int zeile = SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE;
152 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
153 var xCell = sheet.getCellByPosition(SchweizerAbstractSpielrundeSheet.TEAM_A_SPALTE, zeile);
154 if (com.sun.star.table.CellContentType.EMPTY.equals(xCell.getType())) {
155 break;
156 }
157 letzteZeile = zeile;
158 }
159 } catch (Exception e) {
160 logger.warn("Letzte Vorrunde-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
161 }
162 return letzteZeile;
163 }

169 private int berechneAktivSpalte(MaastrichterKonfigurationSheet konfigSheet) throws GenerateException {
170 int spaltenProSpieler = konfigSheet.isMeldeListeVereinsnameAnzeigen() ? 3 : 2;
171 int ersterOffset = konfigSheet.isMeldeListeTeamnameAnzeigen() ? 2 : 1;
172 int letzteDataSpalte = ersterOffset + konfigSheet.getMeldeListeFormation().getAnzSpieler() * spaltenProSpieler - 1;
173 return letzteDataSpalte + 2; // +1=SP, +2=Aktiv
174 }
