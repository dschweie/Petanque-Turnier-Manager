32 @Override
33 protected final void doRun() throws GenerateException {
34 var konfigurierbar = getUploadKonfigurierbar();
35 var konfig = new UploadKonfiguration(
36 konfigurierbar.getUploadProtokoll(),
37 konfigurierbar.getUploadHost(),
38 konfigurierbar.getUploadPort(),
39 konfigurierbar.getUploadBenutzer(),
40 konfigurierbar.getUploadVerzeichnis());
41
42 if (!konfig.istVollstaendig()) {
43 throw new GenerateException(I18n.get("ftp.upload.fehler.konfiguration"));
44 }
45
46 Path tempVerzeichnis = erstelleTempVerzeichnis();
47 try {
48 var ergebnis = exportFactory.apply(tempVerzeichnis).exportiere();
49 String passwort = GlobalProperties.get().getUploadPasswort(konfig.host());
50 if (passwort.isEmpty()) {
51 passwort = fragePasswort(konfig);
52 }
53
54 try {
55 int anzahl = ladeHoch(konfig, passwort, ergebnis);
56 processBox().info(I18n.get("ftp.upload.erfolg", anzahl));
57 } catch (IOException e) {
58 if (istAnmeldefehler(e)) {
59 try {
60 GlobalProperties.get().setUploadPasswort(konfig.host(), "");
61 String neuesPasswort = fragePasswort(konfig);
62 int anzahl = ladeHoch(konfig, neuesPasswort, ergebnis);
63 processBox().info(I18n.get("ftp.upload.erfolg", anzahl));
64 return;
65 } catch (IOException retryFehler) {
66 logger.error("FTP/SFTP-Upload nach erneuter Passwort-Eingabe fehlgeschlagen", retryFehler);
67 throw new GenerateException(retryFehler.getMessage());
68 }
69 }
70 logger.error("FTP/SFTP-Upload fehlgeschlagen", e);
71 throw new GenerateException(e.getMessage());
72 }
73 } finally {
74 try {
75 FileUtils.deleteDirectory(tempVerzeichnis.toFile());
76 } catch (IOException e) {
77 logger.warn("Temporäres Exportverzeichnis konnte nicht gelöscht werden: {}", tempVerzeichnis, e);
78 }
79 }
