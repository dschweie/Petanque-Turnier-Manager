78 protected void doRun() throws GenerateException {
79 if (!NewTestDatenValidator.from(getWorkingSpreadsheet(), getSheetHelper(), TurnierSystem.KASKADE)
80 .prefix(getLogPrefix()).validate()) {
81 return;
82 }
83
84 // 1. Meldeliste mit 97 Teams anlegen (setzt intern anzahlKaskaden=2)
85 meldelisteTestDaten.erstelleMeldelisteWithTestdaten();
86
87 // 2. Kaskadenrunden-Anzahl auf 3 überschreiben (→ Endfelder A–H)
88 konfigurationSheet.setAnzahlKaskaden(ANZ_KASKADEN);
89
90 // 3. Kaskadenrunden 1–3 erstellen und mit Zufallsergebnissen füllen
91 spielrundeSheet.setForceOk(true);
92 for (int rundeNr = 1; rundeNr <= ANZ_KASKADEN; rundeNr++) {
93 SheetRunner.testDoCancelTask();
94 processBoxinfo("processbox.erstelle.spielrunde", rundeNr, ANZ_KASKADEN);
95 spielrundeSheet.naechsteRunde();
96 ergebnisseEinfuegen(rundeNr);
97 }
98
99 // 4. Gruppenrangliste + KO-Felder A–H erstellen
100 // (aktiveRunde=3 >= anzahlKaskaden=3 → KaskadeSpielrundeSheet erkennt Abschluss)
101 SheetRunner.testDoCancelTask();
102 spielrundeSheet.naechsteRunde();
103
104 // 5. Kopfzeile setzen
105 konfigurationSheet.setKopfZeileMitte("Kaskaden-KO 97 Teams (A\u2013H)");
106 konfigurationSheet.seitenstileAktualisieren();
107 }

115 private void ergebnisseEinfuegen(int rundeNr) throws GenerateException {
116 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
117 var sheet = SheetMetadataHelper.findeSheetUndHeile(
118 xDoc, SheetMetadataHelper.schluesselKaskadenRunde(rundeNr),
119 SheetNamen.kaskadenRunde(rundeNr));
120 if (sheet == null) {
121 return;
122 }
123
124 // Lese Spalten SPIEL_NR..ERG_TEAM_B; Offsets im Ergebnis:
125 // 0=SPIEL_NR, 1=TEAM_A, 2=TEAM_B, 3=ERG_A, 4=ERG_B
126 var readRange = RangePosition.from(
127 KaskadeSpielrundeSheet.SPIEL_NR_SPALTE, KaskadeSpielrundeSheet.ERSTE_DATEN_ZEILE,
128 KaskadeSpielrundeSheet.ERG_TEAM_B_SPALTE, KaskadeSpielrundeSheet.ERSTE_DATEN_ZEILE + 200);
129 var daten = RangeHelper.from(sheet, xDoc, readRange).getDataFromRange();
130
131 for (int i = 0; i < daten.size(); i++) {
132 var zeile = daten.get(i);
133 if (zeile.isEmpty() || zeile.get(0).getIntVal(0) <= 0) {
134 break; // Ende der Spieldaten
135 }
136 // Team B = 0/leer → Freilos, Ergebnis ist vorbelegt
137 if (zeile.size() < 3 || zeile.get(2).getIntVal(0) <= 0) {
138 continue;
139 }
140
141 int datenZeile = KaskadeSpielrundeSheet.ERSTE_DATEN_ZEILE + i;
142 int gewinner = RandomSource.nextInt(2);
143 int verliererP = RandomSource.nextInt(0, 13);
144 int ergA = (gewinner == 0) ? 13 : verliererP;
145 int ergB = (gewinner == 0) ? verliererP : 13;
146
147 getSheetHelper().setNumberValueInCell(
148 NumberCellValue.from(sheet,
149 Position.from(KaskadeSpielrundeSheet.ERG_TEAM_A_SPALTE, datenZeile))
150 .setValue(ergA));
151 getSheetHelper().setNumberValueInCell(
152 NumberCellValue.from(sheet,
153 Position.from(KaskadeSpielrundeSheet.ERG_TEAM_B_SPALTE, datenZeile))
154 .setValue(ergB));
155 }
156 }
