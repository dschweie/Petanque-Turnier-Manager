162 public List<TeamPaarung> ersteRunde(List<Team> teams) {
163 boolean freiSpiel = teams.size() % 2 != 0;
164 int letzteMeldungNr = freiSpiel ? teams.size() + 1 : teams.size();
165 int anzTeamPaarungen = letzteMeldungNr / 2;
166
167 // shuffle für zufällige Losung
168 List<Team> shuffled = new ArrayList<>(teams);
169 Collections.shuffle(shuffled, RandomSource.asJavaRandom());
170
171 // split into 2 lists
172 List<List<Team>> partition = ListUtils.partition(shuffled, anzTeamPaarungen);
173
174 List<Team> listATeams = partition.get(0);
175 List<Team> listBTeams = partition.get(1);
176 int listBSize = listBTeams.size();
177
178 return IntStream.range(0, anzTeamPaarungen).mapToObj(i -> {
179 Team teamA = listATeams.get(i);
180 Team teamB = (((anzTeamPaarungen - i) <= listBSize) ? listBTeams.get(anzTeamPaarungen - i - 1) : null);
181 if (teamB == null) {
182 teamA.setHatteFreilos(true);
183 }
184 return new TeamPaarung(teamA, teamB).addGegner();
185 }).sorted((tp1, tp2) -> {
186 // Freilos an letzte Stelle
187 if (tp1.getB() == null) {
188 return 1;
189 } else if (tp2.getB() == null) {
190 return -1;
191 }
192 return Integer.compare(tp1.getA().getNr(), tp2.getA().getNr());
193 }).toList();
194 }

210 public List<TeamPaarung> weitereRunde(List<FormuleXErgebnis> rangliste) {
211 List<Team> teamsInRangfolge = rangliste.stream()
212 .map(e -> Team.from(e.teamNr()))
213 .toList();
214
215 List<Team> zuPaarende;
216 TeamPaarung freilosPaarung = null;
217
218 if (teamsInRangfolge.size() % 2 != 0) {
219 Team byeTeam = findeByeTeam(rangliste, teamsInRangfolge);
220 byeTeam.setHatteFreilos(true);
221 freilosPaarung = new TeamPaarung(byeTeam);
222 zuPaarende = teamsInRangfolge.stream()
223 .filter(t -> t.getNr() != byeTeam.getNr())
224 .collect(Collectors.toList());
225 } else {
226 zuPaarende = new ArrayList<>(teamsInRangfolge);
227 }
228
229 List<TeamPaarung> paarungen = new ArrayList<>(paareLinearMitSwap(zuPaarende, rangliste));
230 if (freilosPaarung != null) {
231 paarungen.add(freilosPaarung);
232 }
233 return paarungen;

278 @VisibleForTesting
279 List<TeamPaarung> paareLinearMitSwap(List<Team> rangliste, List<FormuleXErgebnis> ergebnisse) {
280 List<TeamPaarung> paarungen = new ArrayList<>();
281 Map<Integer, FormuleXErgebnis> ergebnisMap = ergebnisse.stream()
282 .collect(Collectors.toMap(FormuleXErgebnis::teamNr, e -> e));
283 Set<Integer> verarbeitet = new HashSet<>();
284
285 for (int i = 0; i < rangliste.size() - 1; i += 2) {
286 if (verarbeitet.contains(i)) {
287 continue;
288 }
289
290 Team teamA = rangliste.get(i);
291 Team teamB = rangliste.get(i + 1);
292
293 if (!hatGegeneinanderGespielt(teamA, teamB, ergebnisMap)) {
294 paarungen.add(new TeamPaarung(teamA, teamB));
295 } else {
296 boolean swapErfolgreich = false;
297
298 for (int j = i + 2; j + 1 < rangliste.size(); j += 2) {
299 if (verarbeitet.contains(j)) {
300 continue;
301 }
302 Team teamC = rangliste.get(j);
303 Team teamD = rangliste.get(j + 1);
304
305 // Teste Swap B↔D: (A,C) und (B,D)
306 if (!hatGegeneinanderGespielt(teamA, teamC, ergebnisMap)
307 && !hatGegeneinanderGespielt(teamB, teamD, ergebnisMap)) {
308 paarungen.add(new TeamPaarung(teamA, teamC));
309 paarungen.add(new TeamPaarung(teamB, teamD));
310 verarbeitet.add(j);
311 swapErfolgreich = true;
312 break;
313 }
314
315 // Teste Swap A↔C: (A,D) und (C,B)
316 if (!hatGegeneinanderGespielt(teamA, teamD, ergebnisMap)
317 && !hatGegeneinanderGespielt(teamC, teamB, ergebnisMap)) {
318 paarungen.add(new TeamPaarung(teamA, teamD));
319 paarungen.add(new TeamPaarung(teamC, teamB));
320 verarbeitet.add(j);
321 swapErfolgreich = true;
322 break;
323 }
324 }
325
326 if (!swapErfolgreich) {
327 // Fail-Safe: Rematch akzeptieren
328 paarungen.add(new TeamPaarung(teamA, teamB));
329 }
330 }
331 }
332
333 return paarungen;
334 }
