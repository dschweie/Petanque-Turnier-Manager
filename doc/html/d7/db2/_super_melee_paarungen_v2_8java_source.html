197 public MeleeSpielRunde neueSpielrundeDoubletteMode(int rndNr, SpielerMeldungen meldungen, boolean nurTriplette)
198 throws AlgorithmenException {
199 checkNotNull(meldungen, "Meldungen = null");
200 SuperMeleeTeamRechner teamRechner = new SuperMeleeTeamRechner(meldungen.spieler().size(), SuperMeleeMode.Doublette);
201
202 if (!teamRechner.valideAnzahlSpieler()) {
203 throw new AlgorithmenException(I18n.get("error.algorithmus.spieleranzahl.doublette", meldungen.spieler().size()));
204 }
205 if (nurTriplette && !teamRechner.isNurTripletteMoeglich()) {
206 throw new AlgorithmenException(I18n.get("error.algorithmus.keine.triplette"));
207 }
208
209 // Doublette-Hauptmodus: das Dummy-Team wird nach Cleanup zum Doublette = Default-Größe.
210 // Die "Ausnahme" sind hier die dummylosen Triplettes. Spieler im Dummy-Team auf
211 // anzMalKleinesTeam zu sperren wäre kontraproduktiv → Fairness-Constraint deaktiviert.
212 MeleeSpielRunde spielRunde = nurTriplette
213 ? generiereRundeMitFesteTeamGroese(rndNr, 3, meldungen)
214 : generiereRundeMitDummies(rndNr, 3, teamRechner.getAnzDoublette(), meldungen, false);
215 return finalizeRunde(spielRunde);

262 public MeleeSpielRunde erzeugeZufallsRunde(int rndNr, SpielerMeldungen meldungen, boolean doubletteRunde)
263 throws AlgorithmenException {
264 checkNotNull(meldungen, "Meldungen = null");
265 SuperMeleeMode mode = doubletteRunde ? SuperMeleeMode.Doublette : SuperMeleeMode.Triplette;
266 SuperMeleeTeamRechner rechner = new SuperMeleeTeamRechner(meldungen.spieler().size(), mode);
267 if (!rechner.valideAnzahlSpieler()) {
268 throw new AlgorithmenException(I18n.get(
269 doubletteRunde ? "error.algorithmus.spieleranzahl.doublette"
270 : "error.algorithmus.spieleranzahl.triplette",
271 meldungen.spieler().size()));
272 }

276 Collections.shuffle(shuffled, RandomSource.asJavaRandom());

283 team.addSpielerWennNichtVorhanden(shuffled.get(idx++));

290 }
291 }

349 private void optimiereGegnerPaarung(MeleeSpielRunde spielRunde) throws AlgorithmenException {
350 List<Team> doublettes = new ArrayList<>();
351 List<Team> triplettes = new ArrayList<>();
352 for (Team team : spielRunde.teams()) {
353 (team.size() == 2 ? doublettes : triplettes).add(team);
354 }
355
356 // Bei ungerader Parität in beiden Gruppen: genau eine 5er-Partie bilden.
357 // Laut SuperMeleeTeamRechner fallen D- und T-Parität immer zusammen, daher
358 // entsteht maximal eine gemischte Paarung pro Runde.
359 Team[] mixedPaar = null;
360 if (doublettes.size() % 2 != 0 && triplettes.size() % 2 != 0) {
361 mixedPaar = besteMischpaarung(doublettes, triplettes);
362 doublettes.remove(mixedPaar[0]);
363 triplettes.remove(mixedPaar[1]);
364 }
365
366 // Reihenfolge: erst alle D-Paare, dann ggf. 5er-Partie, dann alle T-Paare —
367 // bleibt damit aufsteigend nach Teamgröße sortiert.
368 List<Team> ergebnis = new ArrayList<>(spielRunde.teams().size());
369 paareInnerhalbGruppe(doublettes, ergebnis);
370 if (mixedPaar != null) {
371 ergebnis.add(mixedPaar[0]);
372 ergebnis.add(mixedPaar[1]);
373 }
374 paareInnerhalbGruppe(triplettes, ergebnis);
375
376 spielRunde.setzeTeamReihenfolge(ergebnis);
377
378 if (ergebnis.size() % 2 != 0) {
379 throw new AlgorithmenException(
380 "Ungerade Teamanzahl (" + ergebnis.size() + ") in optimiereGegnerPaarung – " +
381 "SuperMeleeTeamRechner muss immer eine gerade Anzahl Teams liefern.");
382 }
383
384 // Gegner paarweise eintragen: ergebnis[2i] vs ergebnis[2i+1]
385 for (int i = 0; i < ergebnis.size() - 1; i += 2) {
386 Team teamA = ergebnis.get(i);
387 Team teamB = ergebnis.get(i + 1);
388 for (Spieler sA : teamA.spieler()) {
389 for (Spieler sB : teamB.spieler()) {
390 sA.addGegner(sB);
391 }
392 }
393 }

423 private void suchePaarungenRekursiv(List<Team> teams, List<Team> aktuelle, int aktScore,
424 List<Team> besteReihenfolge, int[] bestScore) {
425 if (teams.isEmpty()) {
426 if (aktScore < bestScore[0]) {
427 bestScore[0] = aktScore;
428 besteReihenfolge.clear();
429 besteReihenfolge.addAll(aktuelle);
430 }
431 return;
432 }
433 if (aktScore >= bestScore[0]) {
434 return; // Branch-Cut: aktueller Pfad kann nicht besser werden
435 }
436 Team team1 = teams.getFirst();
437 for (int i = 1; i < teams.size(); i++) {
438 Team partner = teams.get(i);
439 int score = berechneGegnerScore(team1, partner);
440 List<Team> remaining = new ArrayList<>(teams.size() - 2);
441 for (int j = 1; j < teams.size(); j++) {
442 if (j != i) {
443 remaining.add(teams.get(j));
444 }
445 }
446 aktuelle.add(team1);
447 aktuelle.add(partner);
448 suchePaarungenRekursiv(remaining, aktuelle, aktScore + score, besteReihenfolge, bestScore);
449 aktuelle.removeLast();
450 aktuelle.removeLast();
451 }

510 List<List<Integer>> doublettes = new ArrayList<>();
511 List<List<Integer>> triplettes = new ArrayList<>();
512 for (List<Integer> team : teams) {
513 List<Integer> effektivesTeam = effektivesTeamOhneDummies(team, spieler);
514 if (effektivesTeam.isEmpty()) {
515 continue;

517 (effektivesTeam.size() == 2 ? doublettes : triplettes).add(effektivesTeam);

522 if (doublettes.size() % 2 != 0 && triplettes.size() % 2 != 0) {

524 score += mixed[2];
525 doublettes.remove(mixed[0]);
526 triplettes.remove(mixed[1]);
527 }
528 score += berechneBesteGruppenPaarungScore(doublettes, spieler);
529 score += berechneBesteGruppenPaarungScore(triplettes, spieler);
530 return score;

589 private int berechneGreedyGruppenPaarungScore(List<List<Integer>> gruppe, List<Spieler> spieler) {
590 List<List<Integer>> offen = new ArrayList<>(gruppe);
591 int score = 0;
592 while (offen.size() > 1) {
593 int teamIndex = findeTeamMitHoechstemKonfliktpotenzial(offen, spieler);
594 List<Integer> team = offen.remove(teamIndex);
595 int besterPartnerIndex = 0;
596 int besterPartnerScore = Integer.MAX_VALUE;
597 for (int i = 0; i < offen.size(); i++) {
598 int partnerScore = berechneIndexTeamScore(team, offen.get(i), spieler);
599 if (partnerScore < besterPartnerScore) {
600 besterPartnerScore = partnerScore;
601 besterPartnerIndex = i;
602 }
603 }
604 score += besterPartnerScore;
605 offen.remove(besterPartnerIndex);
606 }
607 return score;

629 List<Spieler> spieler) {
630 if (teams.isEmpty()) {
631 bestScore[0] = Math.min(bestScore[0], aktScore);
632 return;
633 }
634 if (teams.size() == 1) {
635 bestScore[0] = Math.min(bestScore[0], aktScore);
636 return;
637 }
638 if (aktScore >= bestScore[0]) {
639 return;
640 }
641 if (teams.size() % 2 != 0) {
642 for (int i = 0; i < teams.size(); i++) {
643 List<List<Integer>> remaining = new ArrayList<>(teams.size() - 1);
644 for (int j = 0; j < teams.size(); j++) {
645 if (j != i) {
646 remaining.add(teams.get(j));
647 }
648 }
649 sucheIndexPaarungenRekursiv(remaining, aktScore, bestScore, spieler);
650 }
651 return;
652 }
653 List<Integer> team1 = teams.getFirst();
654 for (int i = 1; i < teams.size(); i++) {
655 List<Integer> partner = teams.get(i);
656 int score = berechneIndexTeamScore(team1, partner, spieler);
657 List<List<Integer>> remaining = new ArrayList<>(teams.size() - 2);
658 for (int j = 1; j < teams.size(); j++) {
659 if (j != i) {
660 remaining.add(teams.get(j));
661 }
662 }
663 sucheIndexPaarungenRekursiv(remaining, aktScore + score, bestScore, spieler);
664 }

695 private MeleeSpielRunde generiereRundeMitDummies(int rndNr, int teamSize, int anzDummies,
696 SpielerMeldungen meldungen, boolean dummyTeamIstAusnahme) throws AlgorithmenException {
697 // Snapshot der realen Spieler VOR dem Hinzufügen der Dummies — die Fairness-Schwellen
698 // dürfen sich nicht auf Dummies stützen.
699 List<Spieler> realeSpieler = new ArrayList<>(meldungen.spieler());
700
701 Spieler[] dummies = new Spieler[anzDummies];
702 for (int i = 0; i < anzDummies; i++) {
703 dummies[i] = Spieler.from(DUMMY_SPIELER_START_NR + i).setSetzPos(DUMMY_SPIELER_SETZPOS);
704 meldungen.addSpielerWennNichtVorhanden(dummies[i]);
705 }
706 try {
707 MeleeSpielRunde spielRunde = generiereRundeMitFairnessConstraint(rndNr, teamSize, meldungen, dummies,
708 realeSpieler, dummyTeamIstAusnahme);
709 for (Spieler dummy : dummies) {
710 spielRunde.removeSpieler(dummy);
711 // Dummy-Einträge aus warImTeamMit der echten Spieler entfernen:
712 // Dummies sind künstliche Platzhalter — ihre Teamzuordnung soll keine
713 // Paarungsconstraints für echte Spieler in späteren Runden erzeugen.
714 for (Spieler spieler : meldungen.spieler()) {
715 spieler.deleteWarImTeam(dummy);
716 }
717 }
718 if (dummyTeamIstAusnahme) {
719 // Buchhaltung: jeder reale Spieler in einem Team kleiner als teamSize hat in
720 // dieser Runde im Ausnahme-Team (Doublette) gespielt. Konsistent zum Eintrag
721 // beim Wieder-Einlesen der Sheets ({@code SpielrundeDelegate.gespieltenRundenEinlesen}).
722 for (Team team : spielRunde.teams()) {
723 if (team.size() > 0 && team.size() < teamSize) {
724 for (Spieler s : team.spieler()) {
725 s.incAnzMalKleinesTeam();
726 }
727 }
728 }
729 }
730 return spielRunde;
731 } finally {
732 // Garantiertes Cleanup — auch bei Exception
733 for (Spieler dummy : dummies) {
734 meldungen.removeSpieler(dummy);
735 }
736 }
737 }

747 private MeleeSpielRunde generiereRundeMitFairnessConstraint(int rndNr, int teamSize,
748 SpielerMeldungen meldungen, Spieler[] dummies, List<Spieler> realeSpieler,
749 boolean dummyTeamIstAusnahme) throws AlgorithmenException {
750 if (!dummyTeamIstAusnahme || dummies.length == 0 || realeSpieler.isEmpty()) {
751 return generiereRundeMitFesteTeamGroese(rndNr, teamSize, meldungen);
752 }
753
754 int maxKlein = realeSpieler.stream().mapToInt(Spieler::getAnzMalKleinesTeam).max().orElse(0);
755 int minKlein = realeSpieler.stream().mapToInt(Spieler::getAnzMalKleinesTeam).min().orElse(0);
756 // Reduziertes Budget fuer Fairness-Versuche: loesbare Faelle werden in <<1000 Knoten gefunden,
757 // unloesbare Faelle (dichte Teamhistorie + Fairness-Constraints) werden damit schnell erkannt.
758 // Das volle Budget wird nur fuer den uneingeschraenkten Fallback-Call danach verwendet.
759 int fairnessMaxKnoten = maxKnotenFairnessVersuch(meldungen.spieler().size());
760
761 AlgorithmenException letzteException = null;
762 // Von strikt nach locker, in Richtung „minimaler Counter bevorzugt":
763 // Schwelle = minKlein + 1 sperrt alle Spieler, die mehr als minKlein mal im
764 // Ausnahme-Team waren — übrig bleiben nur die Spieler mit dem niedrigsten Zähler.
765 // Findet das Backtracking keine Lösung, wird die Schwelle schrittweise angehoben
766 // (mehr Spieler werden zugelassen). So landen Doublette-Slots immer zuerst bei
767 // den am wenigsten belasteten Spielern.
768 for (int schwelle = minKlein + 1; schwelle <= maxKlein; schwelle++) {
769 try {
770 for (Spieler dummy : dummies) {
771 for (Spieler s : realeSpieler) {
772 if (s.getAnzMalKleinesTeam() >= schwelle) {
773 dummy.addWarImTeamMitWennNichtVorhanden(s);
774 }
775 }
776 }
777 return generiereRunde(rndNr, teamSize, meldungen, fairnessMaxKnoten, "Fairness-Schwelle " + schwelle);
778 } catch (AlgorithmenException ex) {
779 letzteException = ex;
780 logger.debug("Spielrunde {}: Fairness-Schwelle {} nicht lösbar — lockere.", rndNr, schwelle);
781 // Dummy-Konflikte vor nächstem Versuch zurücksetzen — beide Seiten, weil
782 // addWarImTeamMitWennNichtVorhanden symmetrisch ist, deleteWarImTeam aber nicht.
783 for (Spieler dummy : dummies) {
784 for (Spieler s : realeSpieler) {
785 dummy.deleteWarImTeam(s);
786 s.deleteWarImTeam(dummy);
787 }
788 }
789 }
790 }
791 // Schwelle vollständig zurückgenommen → identisches Verhalten zum Algorithmus
792 // vor der Fairness-Erweiterung. Vorhandene Dummies haben hier keine zusätzlichen
793 // Constraints mehr.
794 try {
795 return generiereRundeMitFesteTeamGroese(rndNr, teamSize, meldungen);
796 } catch (AlgorithmenException ex) {
797 // Wenn auch das nicht klappt: die ursprüngliche Exception ist informativer.
798 if (letzteException != null) {
799 throw letzteException;
800 }
801 throw ex;
802 }

836 private MeleeSpielRunde generiereRunde(int rndNr, int teamSize, SpielerMeldungen meldungen, int maxKnoten,
837 String kontext) throws AlgorithmenException {
838
839 // SpielerMeldungen.spieler() liefert eine Kopie — sicher zum Sortieren
840 List<Spieler> spieler = new ArrayList<>(meldungen.spieler());
841 int n = spieler.size();
842 int numTeams = n / teamSize;
843
844 // Mehrere Shuffle-Versuche mit budget-begrenztem Backtracking pro Pass. Existiert
845 // eine Lösung, findet MCV+Forward-Checking sie typischerweise in <<1 000 Knoten im
846 // ersten Versuch. Ist die Aufgabe unlösbar, scheitert jeder Versuch schnell.
847 // Das Budget skaliert mit n: kleine Turniere erkennen Unlösbarkeit schneller,
848 // grosse Turniere erhalten mehr Suchraum.
849 boolean irgendeinLimitErreicht = false;
850 RundenKandidat besterKandidat = new RundenKandidat();
851
852 // Pass-1-Tauglichkeit: warImSpielMit-Beziehungen aendern sich zwischen Shuffles nicht.
853 // Daher reicht eine einmalige Pruefung im ersten Versuch.
854 // Pass 1 ist strukturell unloesbar, wenn ein Spieler so viele Soft-Konflikte hat,
855 // dass er keine (teamSize-1) kompatiblen Mitspieler mehr finden kann.
856 boolean pass1NichtLoesbar = false;
857
858 for (int versuch = 0; versuch < MAX_SHUFFLE_VERSUCHE; versuch++) {
859 // Schritt 1: Zufaellig mischen — jeder Versuch startet mit anderer Spieler-Reihenfolge
860 Collections.shuffle(spieler, RandomSource.asJavaRandom());
861
862 // Schritt 2: Adjazenz-Matrix aufbauen + MCV-Grade in einem einzigen O(n^2)-Durchlauf.
863 // Parallel: Soft-Matrix fuer "war schon im selben Spiel" (Value-Ordering-Heuristik)
864 // und Soft-Grade fuer die Pass-1-Tauglichkeitspruefung.
865 boolean[][] matrix = new boolean[n][n];
866 boolean[][] softMatrix = new boolean[n][n];
867 int[] degrees = new int[n];
868 int[] softDegrees = new int[n];
869 for (int i = 0; i < n; i++) {
870 for (int j = i + 1; j < n; j++) {
871 if (spieler.get(i).warImTeamMit(spieler.get(j))) {
872 matrix[i][j] = matrix[j][i] = true;
873 degrees[i]++;
874 degrees[j]++;
875 }
876 if (spieler.get(i).warImSpielMit(spieler.get(j))) {
877 softMatrix[i][j] = softMatrix[j][i] = true;
878 softDegrees[i]++;
879 softDegrees[j]++;
880 }
881 }
882 }
883
884 // Einmalig im ersten Versuch pruefen: kann Pass 1 ueberhaupt eine Loesung finden?
885 // Bedingung: jeder Spieler braucht mindestens (teamSize-1) Soft-kompatible Partner.
886 // softDegrees[i] > n - teamSize bedeutet: weniger als (teamSize-1) Partner verfuegbar.
887 if (versuch == 0) {
888 for (int i = 0; i < n; i++) {
889 if (softDegrees[i] > n - teamSize) {
890 pass1NichtLoesbar = true;
891 logger.debug("Spielrunde {}: Pass 1 uebersprungen — Spieler-Index {} hat {} Soft-Konflikte"
892 + " (Schwelle: {}), kein Zero-Crossover-Team moeglich.",
893 rndNr, i, softDegrees[i], n - teamSize);
894 break;
895 }
896 }
897 }
898
899 // Schritt 3: MCV-Sortierung — Index-Array nach Constraint-Grad absteigend sortieren
900 Integer[] order = new Integer[n];
901 for (int i = 0; i < n; i++) {
902 order[i] = i;
903 }
904 Arrays.sort(order, (a, b) -> Integer.compare(degrees[b], degrees[a]));
905
906 List<List<Integer>> teams = new ArrayList<>(numTeams);
907 for (int i = 0; i < numTeams; i++) {
908 teams.add(new ArrayList<>(teamSize));
909 }
910
911 // Pass 1: Soft-Constraint als zusaetzlichen harten Constraint behandeln —
912 // Adjazenz wird zur Union (matrix OR softMatrix). Findet das Backtracking
913 // hier eine Loesung, hat sie 0 Mitspieler-Crossover (keine zwei Spieler im
914 // selben Team waren bereits gemeinsam in einer frueheren Partie als Team-
915 // Partner oder Gegner). Damit ist das vom Anwender als stoerend empfundene
916 // "Team-dann-Gegner"-Muster vermieden.
917 // Uebersprungen wenn strukturell unloesbar (zu dichte Soft-Matrix).
918 if (!pass1NichtLoesbar) {
919 boolean[][] unionMatrix = new boolean[n][n];
920 for (int i = 0; i < n; i++) {
921 for (int j = 0; j < n; j++) {
922 unionMatrix[i][j] = matrix[i][j] || softMatrix[i][j];
923 }
924 }
925 int[] zaehler1 = {0};
926 BacktrackingErgebnis ergebnis1 = new BacktrackingErgebnis();
927 backtrackBeste(order, 0, teams, teamSize, unionMatrix, softMatrix, zaehler1, maxKnoten,
928 spieler, ergebnis1);
929 if (ergebnis1.hatLoesung()) {
930 aktualisiereBestenKandidaten(besterKandidat, ergebnis1, spieler, 1);
931 if (ergebnis1.perfekteLoesungGefunden) {
932 return buildSpielRunde(rndNr, besterKandidat.teams, besterKandidat.spieler, meldungen);
933 }
934 }
935 if (ergebnis1.limitErreicht || ergebnis1.bewertungslimitErreicht) {
936 irgendeinLimitErreicht = true;
937 }
938 for (List<Integer> team : teams) {
939 team.clear();
940 }
941 }
942
943 // Pass 2: Soft-Constraint nur noch als Value-Ordering-Tie-Breaker (nicht zum Pruning).
944 // Bekommt ein eigenes Budget — unabhaengig davon, ob Pass 1 ausgefuehrt wurde.
945 int[] zaehler2 = {0};
946 BacktrackingErgebnis ergebnis2 = new BacktrackingErgebnis();
947 backtrackBeste(order, 0, teams, teamSize, matrix, softMatrix, zaehler2, maxKnoten,
948 spieler, ergebnis2);
949 if (ergebnis2.hatLoesung()) {
950 aktualisiereBestenKandidaten(besterKandidat, ergebnis2, spieler, 2);
951 if (pass1NichtLoesbar && ergebnis2.perfekteLoesungGefunden) {
952 return buildSpielRunde(rndNr, besterKandidat.teams, besterKandidat.spieler, meldungen);
953 }
954 }
955 if (ergebnis2.limitErreicht || ergebnis2.bewertungslimitErreicht) {
956 irgendeinLimitErreicht = true;
957 }
958 }
959
960 if (besterKandidat.hatLoesung()) {
961 return buildSpielRunde(rndNr, besterKandidat.teams, besterKandidat.spieler, meldungen);
962 }
963
964 if (irgendeinLimitErreicht) {
965 var pass1Info = pass1NichtLoesbar ? " [Pass 1 wegen Soft-Dichte uebersprungen]" : "";
966 var spieltagPraefix = spieltagNrFuerLog > 0 ? "Spieltag " + spieltagNrFuerLog + ", " : "";
967 var kontextInfo = kontext.isEmpty() ? "" : " [" + kontext + "]";
968 logger.warn("{}Spielrunde {}: Knotenlimit ({} Knoten/Pass x {} Paesse x {} Versuche) erreicht ({} Spieler).{}{}",
969 spieltagPraefix, rndNr, maxKnoten, pass1NichtLoesbar ? 1 : 2, MAX_SHUFFLE_VERSUCHE, n, pass1Info, kontextInfo);
970 throw new AlgorithmenException(
971 "Keine gueltige Spielrunde fuer Runde " + rndNr + " moeglich — "
972 + "Knotenlimit (" + maxKnoten + " x " + (pass1NichtLoesbar ? 1 : 2) + " Paesse x " + MAX_SHUFFLE_VERSUCHE + " Versuche) erreicht. "
973 + "Moeglicherweise muessen Wiederholungen in den Regeln zugelassen werden.");
974 }
975
976 var spieltagPraefix = spieltagNrFuerLog > 0 ? "Spieltag " + spieltagNrFuerLog + ", " : "";
977 logger.warn("{}Spielrunde {}: Alle möglichen Spielerkombinationen ausgeschöpft ({} Spieler).",
978 spieltagPraefix, rndNr, n);
979 throw new AlgorithmenException(
980 "Keine gültige Spielrunde für Runde " + rndNr + " möglich — "
981 + "alle Spielerkombinationen sind ausgeschöpft. "
982 + "Möglicherweise müssen Wiederholungen in den Regeln zugelassen werden.");
983 }

1026 private void backtrackBeste(Integer[] order, int idx, List<List<Integer>> teams,
1027 int teamSize, boolean[][] matrix, boolean[][] softMatrix, int[] knotenZaehler, int knotenLimit,
1028 List<Spieler> spieler, BacktrackingErgebnis ergebnis) {
1029 if (ergebnis.perfekteLoesungGefunden || ergebnis.limitErreicht || ergebnis.bewertungslimitErreicht) {
1030 return;
1031 }
1032 if (idx == order.length) {
1033 int score = berechneLayoutGegnerScore(teams, spieler);
1034 ergebnis.bewerteteLayouts++;
1035 if (score < ergebnis.besterScore) {
1036 ergebnis.besterScore = score;
1037 ergebnis.besteTeams = kopiereTeams(teams);
1038 ergebnis.perfekteLoesungGefunden = (score == 0);
1039 }
1040 if (ergebnis.bewerteteLayouts >= MAX_BEWERTETE_LAYOUTS_PRO_PASS) {
1041 ergebnis.bewertungslimitErreicht = true;
1042 }
1043 return;
1044 }
1045
1046 if (++knotenZaehler[0] >= knotenLimit) {
1047 ergebnis.limitErreicht = true;
1048 return; // Sicherheitsnetz: Knotenlimit pro Pass erreicht
1049 }
1050
1051 int currentOrigIdx = order[idx];
1052
1053 // Value-Ordering: Teams nach Crossover-Score zum aktuellen Spieler aufsteigend
1054 // sortieren (Soft-Constraint). Leere Teams haben Score 0 und liegen damit vorn
1055 // — die Symmetriebrechung filtert davon nur das erste leere Team.
1056 List<List<Integer>> sortierteTeams = new ArrayList<>(teams);
1057 sortierteTeams.sort(Comparator.comparingInt(t -> crossoverScoreTeam(currentOrigIdx, t, softMatrix)));
1058
1059 boolean triedEmptyTeam = false;
1060 for (List<Integer> team : sortierteTeams) {
1061 if (team.size() >= teamSize) {
1062 continue; // Team bereits voll
1063 }
1064
1065 // Symmetriebrechung: leere Teams sind äquivalent — nur das erste versuchen
1066 if (team.isEmpty()) {
1067 if (triedEmptyTeam) {
1068 continue;
1069 }
1070 triedEmptyTeam = true;
1071 }
1072
1073 if (kannTeamBeitreten(currentOrigIdx, team, matrix)) {
1074 team.add(currentOrigIdx);
1075 boolean teamVoll = (team.size() >= teamSize);
1076 if (vorwaertsCheckInkrementell(currentOrigIdx, order, idx + 1, teams, teamSize, matrix, teamVoll)) {
1077 backtrackBeste(order, idx + 1, teams, teamSize, matrix, softMatrix, knotenZaehler, knotenLimit,
1078 spieler, ergebnis);
1079 }
1080 team.removeLast(); // Backtrack: Zuweisung rückgängig machen
1081 if (ergebnis.perfekteLoesungGefunden || ergebnis.limitErreicht || ergebnis.bewertungslimitErreicht) {
1082 return;
1083 }
1084 }
1085 }
