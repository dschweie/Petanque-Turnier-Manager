33 public List<String> listeMitTestNamen(int anzahl, int skip) {
34 List<String> testNamen = new ArrayList<>();
35
36 InputStream csvFile = TestnamenLoader.class.getResourceAsStream(TESTNAMEN_1000_CSV);
37
38 if (csvFile == null) {
39 String errmsg = TESTNAMEN_1000_CSV + " File not found";
40 logger.error(errmsg);
41 throw new NullPointerException(errmsg);
42 }
43
44 CSVParser parser = new CSVParserBuilder().withSeparator(';').withIgnoreQuotations(true).build();
45
46 try (CSVReader reader = new CSVReaderBuilder(new InputStreamReader(csvFile, StandardCharsets.UTF_8))
47 .withSkipLines(1 + skip)
48 .withCSVParser(parser).build()) {
49 String[] nextLine;
50 int anzRead = 1;
51 while ((nextLine = reader.readNext()) != null && anzRead <= anzahl) {
52 testNamen.add(nextLine[2] + ", " + nextLine[1]);
53 anzRead++;
54 }
55
56 } catch (IOException | CsvException e) {
57 logger.error(e.getMessage(), e);
58 }
59 return testNamen;
60 }
