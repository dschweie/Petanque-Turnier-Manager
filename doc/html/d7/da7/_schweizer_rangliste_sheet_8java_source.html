336 private List<TeamRanglisteData> leseAlleSpielergebnisse(TeamMeldungen aktiveMeldungen, int bisSpielrunde,
337 SchweizerMeldeListeSheetUpdate meldeliste) throws GenerateException {
338
339 Map<Integer, int[]> statsMap = new HashMap<>(); // teamNr → [0]=siege, [1]=punkte+, [2]=punkte-
340 Map<Integer, List<Integer>> gegnerMap = new HashMap<>();
341 for (Team team : aktiveMeldungen.teams()) {
342 statsMap.put(team.getNr(), new int[3]);
343 gegnerMap.put(team.getNr(), new ArrayList<>());
344 }
345
346 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
347 for (int runde = 1; runde <= bisSpielrunde; runde++) {
348 SheetRunner.testDoCancelTask();
349 // Iterations-Lookup: Metadaten-first (überlebt Umbenennung), Fallback auf Namen

352 if (rundeSheet == null) {

355 }

358 }

361 for (Team team : aktiveMeldungen.teams()) {

365 }

369 private void leseRundeEin(XSpreadsheet rundeSheet, TeamMeldungen aktiveMeldungen,
370 Map<Integer, int[]> statsMap, Map<Integer, List<Integer>> gegnerMap,
371 SchweizerMeldeListeSheetUpdate meldeliste) throws GenerateException {
372 int dbgSiegeVorher = statsMap.values().stream().mapToInt(a -> a[0]).sum();
373
374 // Lese ab TEAM_A_SPALTE(1) bis ERG_TEAM_B_SPALTE(4), ab ERSTE_DATEN_ZEILE(2)
375 RangePosition readRange = RangePosition.from(
376 SchweizerAbstractSpielrundeSheet.TEAM_A_SPALTE,
377 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE,
378 SchweizerAbstractSpielrundeSheet.ERG_TEAM_B_SPALTE,
379 SchweizerAbstractSpielrundeSheet.ERSTE_DATEN_ZEILE + 999);
380 RangeData rowsData = RangeHelper
381 .from(rundeSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
382 .getDataFromRange();
383
384 int dbgZeilen = 0;
385 for (RowData row : rowsData) {
386 if (row.size() < 2) break;
387
388 int nrA = resolveTeamNr(row.get(0), meldeliste);
389 if (nrA <= 0) break; // Ende der Daten
390 dbgZeilen++;
391 Team teamA = aktiveMeldungen.getTeam(nrA);
392 if (teamA == null) continue;
393
394 int nrB = resolveTeamNr(row.get(1), meldeliste);
395 if (nrB <= 0) {
396 // Freilos für Team A
397 statsMap.computeIfAbsent(nrA, k -> new int[3])[0]++;
398 continue;
399 }
400 Team teamB = aktiveMeldungen.getTeam(nrB);
401 if (teamB == null) continue;
402
403 gegnerMap.computeIfAbsent(nrA, k -> new ArrayList<>()).add(nrB);
404 gegnerMap.computeIfAbsent(nrB, k -> new ArrayList<>()).add(nrA);
405
406 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0;
407 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0;
408
409 if (ergA > 0 || ergB > 0) {
410 statsMap.computeIfAbsent(nrA, k -> new int[3])[1] += ergA; // punkte+
411 statsMap.computeIfAbsent(nrA, k -> new int[3])[2] += ergB; // punkte-
412 statsMap.computeIfAbsent(nrB, k -> new int[3])[1] += ergB; // punkte+
413 statsMap.computeIfAbsent(nrB, k -> new int[3])[2] += ergA; // punkte-
414 if (ergA > ergB) {
415 statsMap.computeIfAbsent(nrA, k -> new int[3])[0]++; // siege für A
416 } else if (ergB > ergA) {
417 statsMap.computeIfAbsent(nrB, k -> new int[3])[0]++; // siege für B
418 }
419 }
420 }
421 int dbgSiegeNachher = statsMap.values().stream().mapToInt(a -> a[0]).sum();
422 logger.debug("leseRundeEin: {} Paarungszeilen verarbeitet, neue Siege in dieser Runde: {}",
423 dbgZeilen, dbgSiegeNachher - dbgSiegeVorher);
424 }

589 }
