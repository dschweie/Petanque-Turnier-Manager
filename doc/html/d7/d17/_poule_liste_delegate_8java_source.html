296 void formatDatenSpalten() throws GenerateException {
297 var formation = konfigurationSheet.getMeldeListeFormation();
298 var anzSpieler = formation.getAnzSpieler();
299 var letzteDatenZeile = getLetzteDatenZeileUseMin();
300
301 // Team-Nr-Spalte
302 var nrRange = RangePosition.from(getTeamNrSpalte(), ERSTE_DATEN_ZEILE, getTeamNrSpalte(), letzteDatenZeile);
303 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), nrRange,
304 CellProperties.from().centerJustify().setBorder(
305 BorderFactory.from().allThin().boldLn().forTop().forLeft().doubleLn().forRight().toBorder()));
306
307 // Bedingte Formatierung Team-Nr (Duplikate + Bereichsprüfung)
308 var kondDoppeltNr = "COUNTIF(" + Position.from(getTeamNrSpalte(), 0).getSpalteAddressWith$() + ";"
309 + ConditionalFormatHelper.FORMULA_CURRENT_CELL + ")>1";
310 ConditionalFormatHelper.from(sheet, nrRange).clear()
311 .formulaIsText().styleIsFehler().applyAndDoReset()
312 .formula1(kondDoppeltNr).operator(ConditionOperator.FORMULA).styleIsFehler().applyAndDoReset()
313 .formula1("0").formula2("" + MeldungenSpalte.MAX_ANZ_MELDUNGEN)
314 .operator(ConditionOperator.NOT_BETWEEN).styleIsFehler().applyAndDoReset();
315
316 // Teamname-Spalte (optional)
317 if (konfigurationSheet.isMeldeListeTeamnameAnzeigen()) {
318 var teamnameRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, 1, letzteDatenZeile);
319 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), teamnameRange,
320 CellProperties.from().setBorder(
321 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()).setShrinkToFit(true));
322 }
323
324 // Spieler-Spalten (Vorname + Nachname [+ Vereinsname])
325 for (int s = 0; s < anzSpieler; s++) {
326 var ersteSpielSpalte = getVornameSpalte(s);
327 var letzteSpielSpalte = konfigurationSheet.isMeldeListeVereinsnameAnzeigen()
328 ? getVereinsnameSpalte(s)
329 : getNachnameSpalte(s);
330 var spielerRange = RangePosition.from(ersteSpielSpalte, ERSTE_DATEN_ZEILE, letzteSpielSpalte, letzteDatenZeile);
331 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), spielerRange,
332 CellProperties.from().setBorder(
333 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()).setShrinkToFit(true));
334 }
335
336 // Setzposition-Spalte
337 var spRange = RangePosition.from(getSetzPositionSpalte(), ERSTE_DATEN_ZEILE, getSetzPositionSpalte(), letzteDatenZeile);
338 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), spRange,
339 CellProperties.from().centerJustify().setBorder(
340 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
341
342 // Aktiv-Spalte
343 var aktivRange = RangePosition.from(getAktivSpalte(), ERSTE_DATEN_ZEILE, getAktivSpalte(), letzteDatenZeile);
344 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), aktivRange,
345 CellProperties.from().centerJustify().setBorder(
346 BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder()));
347
348 // Editierbare Felder hervorheben: Spalte 1 bis Aktiv
349 var editierbareRange = RangePosition.from(1, ERSTE_DATEN_ZEILE, getAktivSpalte(), letzteDatenZeile);
350 EditierbaresZelleFormatHelper.anwenden(sheet, editierbareRange);
351 var editierbar = new CellProtection();
352 editierbar.IsLocked = false;
353 sheet.getSheetHelper().setPropertiesInRange(sheet.getXSpreadSheet(), editierbareRange,
354 CellProperties.from().setCellProtection(editierbar));
355 }

386 TeamMeldungen getAktiveMeldungen() throws GenerateException {
387 var xSheet = sheet.getXSpreadSheet();
388 var letzteZeile = letzteZeileMitDaten(xSheet);
389
390 record TeamZeile(int nr, int setzPos, int aktivWert) {}
391 List<TeamZeile> alleTeams = new ArrayList<>();
392 var hatAktivEintrag = false;
393
394 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
395 var vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(getVornameSpalte(0), zeile));
396 if (vorname == null || vorname.isEmpty()) {
397 continue;
398 }
399 var nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
400 if (nr <= 0) {
401 continue;
402 }
403 var setzPos = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getSetzPositionSpalte(), zeile));
404 var aktivWert = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getAktivSpalte(), zeile));
405 if (aktivWert > 0) {
406 hatAktivEintrag = true;
407 }
408 alleTeams.add(new TeamZeile(nr, setzPos, aktivWert));
409 }
410
411 var meldungen = new TeamMeldungen();
412 for (var tz : alleTeams) {
413 if (!hatAktivEintrag || tz.aktivWert() == AKTIV_WERT_NIMMT_TEIL) {
414 meldungen.addTeamWennNichtVorhanden(Team.from(tz.nr()).setSetzPos(tz.setzPos()));
415 }
416 }
417 return meldungen;
418 }

478 void pruefeAufDoppelteTeamNr(XSpreadsheet xSheet) throws GenerateException {
479 var letzteZeile = letzteZeileMitDaten(xSheet);
480 if (letzteZeile < ERSTE_DATEN_ZEILE) {
481 return;
482 }
483 var vornameSpalte = getVornameSpalte(0);
484 Map<Integer, List<Integer>> alleNrn = new LinkedHashMap<>();
485 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
486 var vorname = sheet.getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
487 if (vorname == null || vorname.isEmpty()) {
488 continue;
489 }
490 var nr = sheet.getSheetHelper().getIntFromCell(xSheet, Position.from(getTeamNrSpalte(), zeile));
491 if (nr <= 0) {
492 continue;
493 }
494 alleNrn.computeIfAbsent(nr, k -> new ArrayList<>()).add(zeile);
495 }
496 Map<Integer, List<Integer>> duplikate = new LinkedHashMap<>();
497 for (var entry : alleNrn.entrySet()) {
498 if (entry.getValue().size() > 1) {
499 duplikate.put(entry.getKey(), entry.getValue());
500 }
501 }
502 if (duplikate.isEmpty()) {
503 return;
504 }
505 var sb = new StringBuilder(I18n.get("error.meldeliste.doppelte.startnummern"));
506 for (var entry : duplikate.entrySet()) {
507 sb.append("\nNr. ").append(entry.getKey()).append(": Zeilen ");
508 var zeilen = entry.getValue();
509 for (int i = 0; i < zeilen.size(); i++) {
510 if (i > 0) {
511 sb.append(", ");
512 }
513 sb.append(zeilen.get(i) + 1);
514 }
515 }
516 throw new GenerateException(sb.toString());
517 }
