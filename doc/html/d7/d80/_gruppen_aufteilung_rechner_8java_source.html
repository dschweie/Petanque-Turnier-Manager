58 public static List<Integer> berechne(int anzTeams, int maxGruppenGroesse, int minLetzteGruppeGroesse) {
59 checkArgument(anzTeams > 0, "anzTeams muss groesser als 0 sein");
60 checkArgument(maxGruppenGroesse >= 2, "maxGruppenGroesse muss >= 2 sein, war: %s", maxGruppenGroesse);
61 checkArgument(minLetzteGruppeGroesse >= 2, "minLetzteGruppeGroesse muss >= 2 sein, war: %s", minLetzteGruppeGroesse);
62
63 List<Integer> ergebnis = new ArrayList<>();
64 for (int start = 0; start < anzTeams; start += maxGruppenGroesse) {
65 ergebnis.add(Math.min(maxGruppenGroesse, anzTeams - start));
66 }
67 if (ergebnis.size() >= 2 && ergebnis.getLast() < minLetzteGruppeGroesse) {
68 int rest = ergebnis.removeLast();
69 ergebnis.set(ergebnis.size() - 1, ergebnis.getLast() + rest);
70 }
71 return ergebnis;
72 }
