77 public RangeHelper clearRange() {
78 checkNotNull(rangePos);
79
80 XCellRange xRangetoClear;
81 xRangetoClear = getCellRange();
82 if (xRangetoClear != null) {
83 // --- Sheet operation. ---
84 XSheetOperation xSheetOp = Lo.qi(com.sun.star.sheet.XSheetOperation.class, xRangetoClear);
85 // clearContents mit CellFlags.STYLES löscht Cell-Styles – im Turnier-Modus
86 // muss das Sheet dafür entsperrt sein (Lazy-Trigger im offenen Scope).
87 BlattschutzManager.get().ensureUnprotectedInScope();
88 xSheetOp.clearContents(CellFlags.ANNOTATION | CellFlags.DATETIME | CellFlags.EDITATTR | CellFlags.FORMATTED
89 | CellFlags.FORMULA | CellFlags.HARDATTR | CellFlags.OBJECTS | CellFlags.STRING | CellFlags.STYLES
90 | CellFlags.VALUE);
91 }
92 return this;
93 }

143 public RangeHelper setDataInRange(RangeData rangeData, boolean adjustEndPosToDataRange) throws GenerateException {
144
145 // adjust endpos to dataArray size
146 if (adjustEndPosToDataRange) {
147 this.rangePos = rangeData.getRangePosition(rangePos.getStart());
148 }
149
150 XCellRangeData xCellRangeData = getXCellRangeData();
151 if (xCellRangeData != null) {
152 Object[][] dataArray = rangeData.toDataArray();
153
154 if (dataArray.length == 0 || dataArray[0].length == 0) {
155 // keine Daten
156 return this;
157 }
158
159 // array muss genau mit rangepos übereinstimmen
160 if (rangePos.getAnzahlZeilen() != dataArray.length) {
161 throw new GenerateException(I18n.get("error.rangehelper.anzahl.zeilen", rangePos.getAnzahlZeilen(), dataArray.length));
162 }
163
164 if (rangePos.getAnzahlSpalten() != dataArray[0].length) {
165 throw new GenerateException(I18n.get("error.rangehelper.anzahl.spalten", rangePos.getAnzahlSpalten(), dataArray[0].length));
166 }
167
168 // setDataArray() wirft RuntimeException auf voll gesperrten Sheets (z.B. Rangliste-Sheets im
169 // Turnier-Modus). Lazy-Entsperren im aktiven Scope; außerhalb eines Scopes No-Op.
170 BlattschutzManager.get().ensureUnprotectedInScope();
171 xCellRangeData.setDataArray(dataArray);
172 }
173 return this;
174 }
