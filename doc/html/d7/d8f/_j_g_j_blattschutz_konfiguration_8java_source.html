76 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
77 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_JGJ_MELDELISTE).ifPresent(sheet ->
78 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
79 RangePosition.from(SPIELER_NR_SPALTE + 1, ERSTE_DATEN_ZEILE,
80 SPIELER_NR_SPALTE + 1, MeldungenSpalte.MAX_ANZ_MELDUNGEN)))));
81 }
82
83 private void sammleSpielplanSchutzInfo(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
84 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_JGJ_SPIELPLAN).ifPresent(sheet -> {
85 int ersteDatenZeile = JGJSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE;
86 int letzteZeile = ermittleLetzteSpielplanZeile(sheet);
87 var bereiche = List.of(
88 RangePosition.from(JGJSpielPlanSheet.SPIELPNKT_A_SPALTE, ersteDatenZeile,
89 JGJSpielPlanSheet.SPIELPNKT_B_SPALTE, letzteZeile));
90 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, bereiche));
91 });
92 }
93
94 private int ermittleLetzteSpielplanZeile(XSpreadsheet sheet) {
95 int letzteZeile = JGJSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE;
96 try {
97 for (int zeile = JGJSpielPlanSheet.ERSTE_SPIELTAG_DATEN_ZEILE;
98 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
99 XCell xCell = sheet.getCellByPosition(JGJSpielPlanSheet.SPIEL_NR_SPALTE, zeile);
100 if (CellContentType.EMPTY.equals(xCell.getType())) {
101 break;
102 }
103 letzteZeile = zeile;
104 }
105 } catch (Exception e) {
106 logger.warn("Letzte Spielplan-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
107 }
108 return letzteZeile;
109 }
110
111 private void sammleVollGesperrteSheets(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
112 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_JGJ_RANGLISTE)
113 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
114 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_JGJ_DIREKTVERGLEICH)
115 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
116 fuegeVollgesperrteAusPrefix(xDoc, SheetMetadataHelper.SCHLUESSEL_JGJ_GRUPPE_SPIELPLAN_PREFIX, infos);
117 }
118
119 private void fuegeVollgesperrteAusPrefix(XSpreadsheetDocument xDoc, String prefix,
120 List<SheetSchutzInfo> infos) {
121 for (var key : SheetMetadataHelper.getSchluesselMitPrefix(xDoc, prefix)) {
122 SheetMetadataHelper.findeSheet(xDoc, key)
123 .ifPresent(sheet -> infos.add(SheetSchutzInfo.vollGesperrt(sheet)));
124 }
