68 public String erstelle() throws GenerateException {
69 XSpreadsheetDocument doc = workingSpreadsheet.getWorkingSpreadsheetDocument();
70 var tabellenHtml = new ArrayList<String>();

85 private String renderSheet(String sheetName, XSpreadsheetDocument doc) throws GenerateException {
86 var sheet = sheetHelper.findByName(sheetName);
87 if (sheet == null) {
88 return fehlendeTabelleHtml(sheetName);
89 }
90 TabelleModel model = tabellenMapper.map(sheet, doc);
91 return tabelleHtmlRenderer.render(model);
92 }

96 + "</em></p>";

99 private String assembliere(List<String> tabellenHtml) {
100 var sb = new StringBuilder(16384);
101 String seitentitel = StringUtils.defaultIfBlank(titel, "Export");
102
103 sb.append("<!DOCTYPE html>\n<html lang=\"de\">\n<head>\n");
104 sb.append("<meta charset=\"UTF-8\">\n");
105 sb.append("<meta name=\"viewport\" content=\"width=device-width, initial-scale=1.0\">\n");
106 sb.append("<title>").append(StringEscapeUtils.escapeHtml4(seitentitel)).append("</title>\n");
107 sb.append("<style>\n").append(css()).append("</style>\n");
108 sb.append("</head>\n<body>\n");
109
110 sb.append("<header class=\"page-header\">\n");
111 sb.append("<div class=\"header-text\">");
112 sb.append("<h1>").append(StringEscapeUtils.escapeHtml4(seitentitel)).append("</h1>");
113 sb.append("<div class=\"ptm-subline\">Pétanque Turnier Manager</div>");
114 sb.append("</div>");
115 if (StringUtils.isNotBlank(logoUrl)) {
116 sb.append("<img class=\"turnier-logo\" src=\"").append(StringEscapeUtils.escapeHtml4(logoUrl))
117 .append("\" alt=\"Logo\">");
118 }
119 sb.append("\n</header>\n");
120
121 sb.append("<nav class=\"section-nav\" aria-label=\"Abschnitte\">\n");
122 for (var section : sections) {
123 sb.append(navLink(section.id(), section.titel()));
124 }
125 sb.append("\n</nav>\n");
126
127 sb.append("<main>\n");
128 for (int i = 0; i < sections.size(); i++) {
129 var section = sections.get(i);
130 appendSection(sb, section.id(), section.titel(), section.pdfUrl(), tabellenHtml.get(i));
131 }
132 sb.append("</main>\n");
133
134 sb.append(ExportFooterHtml.html(true));
135
136 sb.append("</body>\n</html>");
137 return sb.toString();
138 }

140 private void appendSection(StringBuilder sb, String id, String titel, String pdfUrl, String tabelleHtml) {
141 sb.append("<section class=\"export-section\" id=\"").append(StringEscapeUtils.escapeHtml4(id)).append("\">\n");
142 sb.append("<div class=\"section-heading\">\n");
143 sb.append("<h2>").append(StringEscapeUtils.escapeHtml4(titel)).append("</h2>\n");
144 if (StringUtils.isNotBlank(pdfUrl)) {
145 sb.append("<a class=\"pdf-btn\" href=\"").append(StringEscapeUtils.escapeHtml4(pdfUrl))
146 .append("\" download>&#128462; ")
147 .append(StringEscapeUtils.escapeHtml4(I18n.get("export.liga.pdf.herunterladen")))
148 .append("</a>\n");
149 }
150 sb.append("</div>\n");
151 sb.append("<div class=\"tbl-scroll\">\n").append(tabelleHtml).append("\n</div>\n");
152 sb.append("</section>\n");
153 }
