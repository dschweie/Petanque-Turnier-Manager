48 @Override
49 public void notifyEvent(com.sun.star.document.EventObject docEvent) {
50 // Der try-catch-Block verhindert, daß die Funktion und damit der
51 // ganze Listener ohne Fehlermeldung abstürzt.
52 try {
53 if (docEvent.Source == null) {
54 return;
55 }
56 String event = docEvent.EventName;
57 if (logger.isDebugEnabled()) {
58 logger.debug("{} [controller={}]", event, ermittleControllerTyp(docEvent.Source));
59 }
60 // Erstes Vorkommen jedes Event-Typs zur Startup-Vermessung loggen
61 StartupClock.logErstesVorkommen("event:" + event, "GlobalEvent " + event);
62
63 // https://wiki.openoffice.org/wiki/Documentation/DevGuide/WritingUNO/Jobs/List_of_Supported_Events
64 // https://api.libreoffice.org/docs/idl/ref/servicecom_1_1sun_1_1star_1_1document_1_1Events.html
65 if (ON_CREATE.equals(event)) {
66 onCreate(docEvent.Source);
67 } else if (ON_VIEW_CREATED.equals(event)) {
68 onViewCreated(docEvent.Source);
69 } else if (ON_VIEW_CLOSED.equals(event)) {
70 onViewClosed(docEvent.Source);
71 } else if (ON_UNLOAD.equals(event)) {
72 onUnload(docEvent.Source);
73 } else if (ON_SAVE.equals(event)) {
74 onSaveOrSaveAs(docEvent.Source);
75 } else if (ON_SAVE_AS.equals(event)) {
76 onSaveOrSaveAs(docEvent.Source);
77 } else if (ON_NEW.equals(event)) {
78 onNew(docEvent.Source);
79 } else if (ON_LOAD_FINISHED.equals(event)) { // 1
80 onLoadFinished(docEvent.Source);
81 } else if (ON_LOAD.equals(event)) { // 5
82 onLoad(docEvent.Source);
83 } else if (ON_UNFOCUS.equals(event)) {
84 onUnfocus(docEvent.Source);
85 } else if (ON_FOCUS.equals(event)) {
86 onFocus(docEvent.Source);
87 }
88
89 } catch (Exception e) {
90 LogUtil.error(logger, "Globaler Event-Dispatch fehlgeschlagen", e);
91 } catch (Error e) {
92 throw e;
93 }
94 }
