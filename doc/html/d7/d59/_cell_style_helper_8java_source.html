88 private void applyAufDokument(XSpreadsheetDocument currentSpreadsheetDocument) {
89 var styleName = cellStyleDef.getName();
90
91 // LO-Einschränkung (sc/source/ui/unoobj/styleuno.cxx): Zellstile können nicht
92 // geändert werden, solange irgendein Sheet im Doc tab-geschützt ist. Daher
93 // alle Sheets mit leerem Passwort temporär entsperren, Style schreiben,
94 // danach Schutz wiederherstellen. Passwortgeschützte Sheets (durch User)
95 // bleiben gesperrt – dort logged der Catch-Zweig wie zuvor.
96 var temporaerEntsperrt = entsperreAlleSheetsMitLeeremPasswort(currentSpreadsheetDocument);
97 try {
98 var xFamiliesSupplier = Lo.qi(XStyleFamiliesSupplier.class, currentSpreadsheetDocument);
99 var xFamiliesNA = xFamiliesSupplier.getStyleFamilies();
100 var aCellStylesObj = xFamiliesNA.getByName("CellStyles");
101 var xCellStylesNA = Lo.qi(XNameContainer.class, aCellStylesObj);
102
103 Object aCellStyle;
104 try {
105 aCellStyle = xCellStylesNA.getByName(styleName);
106 } catch (NoSuchElementException e) {
107 // create a new cell style
108 var xDocServiceManager = Lo.qi(XMultiServiceFactory.class, currentSpreadsheetDocument);
109 aCellStyle = xDocServiceManager.createInstance("com.sun.star.style.CellStyle");
110 xCellStylesNA.insertByName(styleName, aCellStyle);
111 }
112
113 // modify properties of the (new) style
114 var xPropSet = Lo.qi(XPropertySet.class, aCellStyle);
115 for (var propKey : cellStyleDef.getCellProperties().keySet()) {
116 if (xPropSet.getPropertySetInfo().hasPropertyByName(propKey)) {
117 var neuerWert = cellStyleDef.getCellProperties().get(propKey);
118 var aktuellerWert = xPropSet.getPropertyValue(propKey);
119 if (!Objects.equals(neuerWert, aktuellerWert)) {
120 xPropSet.setPropertyValue(propKey, neuerWert);
121 }
122 }
123 }
124 } catch (RuntimeException e) {
125 if (!temporaerEntsperrt.isEmpty()) {
126 getLogger().warn(
127 "Zellstil '{}' konnte nicht gesetzt werden – LO-Einschränkung: " +
128 "Zellstile können nicht geändert werden, solange ein Sheet im Dokument " +
129 "tab-geschützt ist. (sc/source/ui/unoobj/styleuno.cxx)",
130 styleName);
131 } else {
132 getLogger().error(e.getMessage(), e);
133 }
134 } catch (Exception e) {
135 getLogger().error(e.getMessage(), e);
136 } finally {
137 schuetzeWiederMitLeeremPasswort(temporaerEntsperrt);
138 }
139 }

146 private List<XProtectable> entsperreAlleSheetsMitLeeremPasswort(XSpreadsheetDocument doc) {
147 List<XProtectable> entsperrt = new ArrayList<>();
148 var sheets = doc.getSheets();
149 for (var name : sheets.getElementNames()) {
150 try {
151 var xSheet = Lo.qi(XSpreadsheet.class, sheets.getByName(name));
152 var xProt = Lo.qi(XProtectable.class, xSheet);
153 if (xProt != null && xProt.isProtected()) {
154 try {
155 xProt.unprotect("");
156 // Verifizieren: bei Passwortschutz schlägt unprotect("") still fehl
157 if (!xProt.isProtected()) {
158 entsperrt.add(xProt);
159 }
160 } catch (RuntimeException e) {
161 // Sheet mit Passwort – überspringen, bleibt geschützt
162 }
163 }
164 } catch (NoSuchElementException | WrappedTargetException e) {
165 // Sheet nicht zugreifbar – überspringen
166 }
167 }
168 return entsperrt;
169 }
170
171 private void schuetzeWiederMitLeeremPasswort(List<XProtectable> sheets) {
172 for (var xProt : sheets) {
173 try {
174 xProt.protect("");
175 } catch (RuntimeException e) {
176 getLogger().warn("Sheet-Schutz konnte nicht wiederhergestellt werden: {}", e.getMessage());
177 }
178 }
