170 private Formation leseFormation(XControlContainer xcc) {
171 return switch (leseListBoxAuswahl(xcc, "lstFormation")) {
172 case 1 -> Formation.DOUBLETTE;
173 case 2 -> Formation.TRIPLETTE;
174 default -> Formation.TETE;
175 };
176 }

178 private static short formationIndex(Formation formation) {
179 return switch (formation) {
180 case DOUBLETTE -> 1;
181 case TRIPLETTE -> 2;
182 default -> 0;
183 };
184 }

186 private short leseListBoxAuswahl(XControlContainer xcc, String name) {
187 XControl ctrl = xcc.getControl(name);
188 if (ctrl == null) {
189 return 0;
190 }
191 XListBox lb = Lo.qi(XListBox.class, ctrl);
192 return lb != null ? lb.getSelectedItemPos() : 0;
193 }
194
195 private boolean leseCheckBoxZustand(XControlContainer xcc, String name) {
196 XControl ctrl = xcc.getControl(name);
197 if (ctrl == null) {
198 return false;
199 }
200 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
201 return cb != null && cb.getState() == 1;
202 }

206 if (ctrl != null) {
207 XButton btn = Lo.qi(XButton.class, ctrl);
208 if (btn != null) {
209 btn.addActionListener(listener);
210 }
211 }
