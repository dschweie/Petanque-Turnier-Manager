147 public void generate(TeamMeldungen meldungen) throws GenerateException {
148 int gruppengroesse = getKonfigurationSheet().getGruppengroesse();
149 boolean gruppenModus = gruppengroesse > 0 && meldungen.size() > gruppengroesse;
150
151 if (gruppenModus) {
152 if (meldungen.size() < 2) {
153 zeigeUngueltigeMeldungenFehler();
154 return;
155 }
156 } else {
157 if (!meldungen.isValid()) {
158 zeigeUngueltigeMeldungenFehler();
159 return;
160 }
161 }
162
163 if (!NewSheet.from(this, sheetName(), METADATA_SCHLUESSEL)
164 .pos(DefaultSheetPos.JGJ_WORK).setActiv().hideGrid()
165 .tabColor(getKonfigurationSheet().getSpielrundeTabFarbe()).create().isDidCreate()) {
166 ProcessBox.from().info("Abbruch vom Benutzer, Jeder gegen Jeden SpielPlan wurde nicht erstellt");
167 return;
168 }
169
170 insertDatenHeaderUndSpalteBreite();
171
172 boolean mitRueckrunde = getKonfigurationSheet().isRueckrunde();
173
174 if (gruppenModus) {
175 generiereGruppenSpielplan(meldungen, gruppengroesse, mitRueckrunde);
176 } else {
177 var ligaSpielPlan = new LigaSpielPlan(meldungen);
178 var spielPlanHRunde = ligaSpielPlan.schufflePlan().getSpielPlanClone();
179 var spielPlanRRunde = mitRueckrunde
180 ? ligaSpielPlan.flipTeams().getSpielPlanClone()
181 : List.<List<TeamPaarung>>of();
182 insertSpieltageDaten(spielPlanHRunde, spielPlanRRunde, ERSTE_SPIELTAG_DATEN_ZEILE);
183 insertArbeitsspalten(spielPlanHRunde, spielPlanRRunde, ERSTE_SPIELTAG_DATEN_ZEILE);
184 insertFormulaPunkte();
185 insertFormulaTeamNamen();
186 insertFormulaValidierung();
187 verbergeArbeitsspalten();
188 formatieren(spielPlanHRunde, spielPlanRRunde, ERSTE_SPIELTAG_DATEN_ZEILE);
189 }
190
191 printBereichDefinieren();
192 SheetFreeze.from(getTurnierSheet()).anzZeilen(2).doFreeze();
193 }

202 private void generiereGruppenSpielplan(TeamMeldungen meldungen, int gruppengroesse, boolean mitRueckrunde) throws GenerateException {
203 List<TeamMeldungen> gruppen = JGJGruppenAufteiler.teileInGruppen(meldungen, gruppengroesse);
204 List<Integer> gruppenHeaderZeilen = new ArrayList<>();
205 List<Integer> gruppenStartZeilen = new ArrayList<>();
206 List<List<List<TeamPaarung>>> gruppenSpielplaeneH = new ArrayList<>();
207 List<List<List<TeamPaarung>>> gruppenSpielplaeneR = new ArrayList<>();
208 int aktuelleZeile = ERSTE_SPIELTAG_DATEN_ZEILE;
209
210 for (int g = 0; g < gruppen.size(); g++) {
211 TeamMeldungen gruppe = gruppen.get(g);
212 gruppenHeaderZeilen.add(aktuelleZeile);
213 schreibeGruppenNrZeile(aktuelleZeile);
214 int gruppeStartZeile = aktuelleZeile + 1;
215 gruppenStartZeilen.add(gruppeStartZeile);
216
217 var ligaSpielPlan = new LigaSpielPlan(gruppe);
218 var spielPlanHRunde = ligaSpielPlan.schufflePlan().getSpielPlanClone();
219 var spielPlanRRunde = mitRueckrunde
220 ? ligaSpielPlan.flipTeams().getSpielPlanClone()
221 : List.<List<TeamPaarung>>of();
222
223 gruppenSpielplaeneH.add(spielPlanHRunde);
224 gruppenSpielplaeneR.add(spielPlanRRunde);
225
226 insertSpieltageDaten(spielPlanHRunde, spielPlanRRunde, gruppeStartZeile);
227 insertArbeitsspalten(spielPlanHRunde, spielPlanRRunde, gruppeStartZeile);
228
229 int anzSpiele = anzahlSpiele(spielPlanHRunde, spielPlanRRunde);
230 aktuelleZeile = gruppeStartZeile + anzSpiele;
231 }
232
233 // Formeln erst nach allen Dateneintragungen einfügen, damit fillAuto die
234 // danach folgende Zebra-Formatierung nicht überschreibt
235 insertFormulaPunkte();
236 insertFormulaTeamNamen();
237 insertFormulaValidierung();
238 verbergeArbeitsspalten();
239
240 for (int g = 0; g < gruppen.size(); g++) {
241 formatieren(gruppenSpielplaeneH.get(g), gruppenSpielplaeneR.get(g), gruppenStartZeilen.get(g));
242 }
243
244 for (int g = 0; g < gruppen.size(); g++) {
245 schreibeGruppenHeaderBeschriftung(gruppenHeaderZeilen.get(g), gruppenBuchstabe(g));
246 }
247
248 erstelleAushangSheets(gruppenStartZeilen, gruppenSpielplaeneH, gruppenSpielplaeneR);
249 }
