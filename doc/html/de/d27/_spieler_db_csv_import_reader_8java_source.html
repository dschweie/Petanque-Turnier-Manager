60 @Override
61 public ImportRohdaten read(ImportRequest request) throws SpielerDbException {
62 Path datei = request.source();
63 if (!Files.isRegularFile(datei)) {
64 throw new SpielerDbException("CSV-Quelle ist keine Datei: " + datei);
65 }
66
67 try (Reader r = Files.newBufferedReader(datei, StandardCharsets.UTF_8);
68 BufferedReader br = new BufferedReader(r)) {
69 int aktuelleZeile = 0;
70 String erste = br.readLine();
71 aktuelleZeile++;
72 if (erste == null) {
73 throw new SpielerDbException("CSV-Datei ist leer: " + datei);
74 }
75 erste = entferneBom(erste);
76
77 String headerZeile;
78 if (SpielerDbCsvFormat.istMarkerZeile(erste)) {
79 pruefeVersion(datei, erste);
80 headerZeile = br.readLine();
81 aktuelleZeile++;
82 } else {
83 // Marker fehlt — tolerant weiterlesen.
84 headerZeile = erste;
85 }
86 if (headerZeile == null) {
87 throw new SpielerDbException("CSV-Datei enthält keinen Header: " + datei);
88 }
89
90 // Header und Daten-Zeilen über opencsv parsen, damit Quoting konsistent
91 // mit dem Exporter behandelt wird (auch im Header — falls jemand mal
92 // einen Header mit Sonderzeichen erfindet).
93 try (CSVReader csv = new CSVReaderBuilder(new java.io.StringReader(headerZeile))
94 .withCSVParser(new CSVParserBuilder()
95 .withSeparator(SpielerDbCsvFormat.SEPARATOR).build())
96 .build()) {
97 String[] header = csv.readNext();
98 if (header == null) {
99 throw new SpielerDbException("CSV-Datei enthält keinen Header: " + datei);
100 }
101 validiereHeader(datei, header);
102 } catch (CsvException e) {
103 throw new SpielerDbException("CSV-Header nicht parsebar: " + datei, e);
104 }
105
106 List<RohSpieler> spieler = leseSpielerZeilen(datei, br, aktuelleZeile);
107 return new ImportRohdaten(spieler, List.of(), List.of(), List.of());
108 } catch (IOException e) {
109 throw new SpielerDbException("CSV-Lesen fehlgeschlagen: " + datei, e);
110 }
111 }

113 private static List<RohSpieler> leseSpielerZeilen(Path datei, BufferedReader br,
114 int bereitsGeleseneZeilen) throws SpielerDbException {
115 try (CSVReader csv = new CSVReaderBuilder(br)
116 .withCSVParser(new CSVParserBuilder()
117 .withSeparator(SpielerDbCsvFormat.SEPARATOR).build())
118 .build()) {
119 List<RohSpieler> erg = new ArrayList<>();
120 int datenZeileInDatei = bereitsGeleseneZeilen;
121 String[] zeile;
122 while ((zeile = csv.readNext()) != null) {
123 datenZeileInDatei++;
124 if (zeile.length == 0 || (zeile.length == 1 && zeile[0].isEmpty())) {
125 continue; // Leerzeile überspringen
126 }
127 String vorname = normalisiere(spalte(zeile, IDX_VORNAME));
128 String nachname = normalisiere(spalte(zeile, IDX_NACHNAME));
129 String verein = leerAlsNull(normalisiere(spalte(zeile, IDX_VEREIN)));
130 String lizenz = leerAlsNull(normalisiere(spalte(zeile, IDX_LIZENZ)));
131 erg.add(new RohSpieler(null, vorname, nachname, null, verein, lizenz,
132 datenZeileInDatei));
133 }
134 return erg;
135 } catch (IOException | CsvException e) {
136 throw new SpielerDbException("CSV-Daten nicht parsebar: " + datei, e);
137 }
138 }

140 private static void pruefeVersion(Path datei, String markerZeile) throws SpielerDbException {
141 Integer version = SpielerDbCsvFormat.leseVersion(markerZeile);
142 if (version == null) {
143 throw new SpielerDbException("Format-Marker in " + datei
144 + " enthält keine gültige Versionsangabe: " + markerZeile);
145 }
146 if (version > SpielerDbCsvFormat.VERSION) {
147 throw new SpielerDbException("CSV-Datei " + datei + " ist Version " + version
148 + ", dieses Plugin kennt höchstens Version " + SpielerDbCsvFormat.VERSION
149 + " — Plugin aktualisieren?");
150 }

153 private static void validiereHeader(Path datei, String[] header) throws SpielerDbException {
154 String[] erwartet = SpielerDbCsvFormat.HEADER;
155 if (header.length < erwartet.length) {
156 throw new SpielerDbException("Header in " + datei + " hat zu wenige Spalten: "
157 + header.length + " < " + erwartet.length);
158 }
159 for (int i = 0; i < erwartet.length; i++) {
160 String gefunden = header[i] == null ? "" : header[i].strip();
161 if (i == 0) {
162 gefunden = entferneBom(gefunden);
163 }
164 if (!erwartet[i].equalsIgnoreCase(gefunden)) {
165 throw new SpielerDbException("Header-Spalte " + (i + 1) + " in " + datei
166 + " erwartet '" + erwartet[i] + "', gefunden '" + gefunden + "'");
167 }
168 }

178 private static String spalte(String[] zeile, int idx) {

186 private static String normalisiere(String s) {
187 if (s.isEmpty()) {
188 return "";
189 }
190 String nfc = Normalizer.normalize(s, Normalizer.Form.NFC);
191 return nfc.replaceAll("\\s+", " ").strip();
192 }
