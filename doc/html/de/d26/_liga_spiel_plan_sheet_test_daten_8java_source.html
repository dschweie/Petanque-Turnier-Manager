71 private List<List<List<SpielErgebnis>>> getTestDaten(int anzRunden, int anzPaarungen, int anzSpielInBegnung) {
72
73 // ein begegnung sind:
74 // Runde1 = 1x Tete + 2 x Doublette
75 // Runde2 = 1 x Doublette + 1 x Triplette
76
77 ArrayList<List<List<SpielErgebnis>>> result = new ArrayList<>();
78
79 for (int runde = 0; runde < anzRunden; runde++) {
80 ArrayList<List<SpielErgebnis>> paarungen = new ArrayList<>();
81 for (int i = 0; i < anzPaarungen; i++) {
82 ArrayList<SpielErgebnis> ergebnisse = new ArrayList<>();
83 for (int sp = 0; sp < anzSpielInBegnung; sp++) {
84 int welchenTeamHatGewonnen = RandomSource.nextInt(0, 2); // 0,1
85 int verliererPunkte = RandomSource.nextInt(0, 13); // 0 - 12
86 SpielErgebnis ergebnis;
87 if (welchenTeamHatGewonnen == 0) {
88 ergebnis = new SpielErgebnis(13, verliererPunkte);
89 } else {
90 ergebnis = new SpielErgebnis(verliererPunkte, 13);
91 }
92 ergebnisse.add(ergebnis);
93 }
94 paarungen.add(ergebnisse);
95 }
96 result.add(paarungen);
97 }
98
99 return result;
100 }
