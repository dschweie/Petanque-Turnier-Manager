71 private void validiereHeader(XSpreadsheet sheet) {
72 String erwarteterVorname = I18n.get("spielerdb.vorlage.spalte.vorname");
73 String erwarteterNachname = I18n.get("spielerdb.vorlage.spalte.nachname");
74 String erwarteterVerein = I18n.get("spielerdb.vorlage.spalte.verein");
75
76 XSpreadsheetDocument doc = ws.getWorkingSpreadsheetDocument();
77 RangePosition headerPos = RangePosition.from(
78 SpielerDbVorlageSheet.SPALTE_VORNAME, SpielerDbVorlageSheet.HEADER_ZEILE,
79 SpielerDbVorlageSheet.SPALTE_VEREIN, SpielerDbVorlageSheet.HEADER_ZEILE);
80 RangeData header = RangeHelper.from(sheet, doc, headerPos).getDataFromRange();
81
82 String gefundenVor = headerWert(header, SpielerDbVorlageSheet.SPALTE_VORNAME);
83 String gefundenNach = headerWert(header, SpielerDbVorlageSheet.SPALTE_NACHNAME);
84 String gefundenVer = headerWert(header, SpielerDbVorlageSheet.SPALTE_VEREIN);
85
86 if (!gefundenVor.equalsIgnoreCase(erwarteterVorname.strip())
87 || !gefundenNach.equalsIgnoreCase(erwarteterNachname.strip())
88 || !gefundenVer.equalsIgnoreCase(erwarteterVerein.strip())) {
89 String gefunden = gefundenVor + " | " + gefundenNach + " | " + gefundenVer;
90 throw new VorlageNichtVerfuegbarException(
91 I18n.get("spielerdb.vorlage.fehler.header",
92 erwarteterVorname, erwarteterNachname, erwarteterVerein, gefunden));
93 }
94 }

96 private static String headerWert(RangeData header, int spalte) {
97 if (header.isEmpty()) {
98 return "";
99 }
100 RowData row = header.get(0);
101 if (spalte >= row.size()) {
102 return "";

104 CellData cell = row.get(spalte);
105 String s = cell == null ? null : cell.getStringVal();
106 return s == null ? "" : s.strip();

109 private List<MeldelisteSpielerDaten> leseDatenzeilen(XSpreadsheet sheet) {
110 XSpreadsheetDocument doc = ws.getWorkingSpreadsheetDocument();
111 RangePosition datenPos = RangePosition.from(
112 SpielerDbVorlageSheet.SPALTE_VORNAME, SpielerDbVorlageSheet.HEADER_ZEILE + 1,
113 SpielerDbVorlageSheet.SPALTE_VEREIN, SpielerDbVorlageSheet.MAX_DATEN_ZEILE);
114 RangeData daten = RangeHelper.from(sheet, doc, datenPos).getDataFromRange();
115
116 List<MeldelisteSpielerDaten> result = new ArrayList<>();
117 int zeileOffset = SpielerDbVorlageSheet.HEADER_ZEILE + 1;
118 for (int i = 0; i < daten.size(); i++) {
119 RowData row = daten.get(i);
120 String vor = zellWert(row, SpielerDbVorlageSheet.SPALTE_VORNAME);
121 String nach = zellWert(row, SpielerDbVorlageSheet.SPALTE_NACHNAME);
122 String ver = zellWert(row, SpielerDbVorlageSheet.SPALTE_VEREIN);
123 if (vor.isEmpty() && nach.isEmpty() && ver.isEmpty()) {
124 break;
125 }
126 if (vor.isEmpty() && nach.isEmpty()) {
127 continue;
128 }
129 String vereinOderNull = ver.isEmpty() ? null : ver;
130 int zeile1Basiert = zeileOffset + i + 1;
131 result.add(new MeldelisteSpielerDaten(vor, nach, vereinOderNull, zeile1Basiert));
132 }
133 return result;
134 }

145 @Override
146 public void schreibeStatus(List<ZeilenStatus> eintraege) {
147 if (eintraege.isEmpty()) {
148 return;
149 }
150 SheetHelper sh = new SheetHelper(ws);
151 XSpreadsheet sheet = sh.findByName(SheetNamen.spielerDbVorlage());
152 if (sheet == null) {
153 return;
154 }
155
156 Map<Integer, ZeilenStatus> byZeile1 = new HashMap<>();
157 int minZeile1 = Integer.MAX_VALUE;
158 int maxZeile1 = Integer.MIN_VALUE;
159 for (ZeilenStatus zs : eintraege) {
160 byZeile1.put(zs.zeile1Basiert(), zs);
161 minZeile1 = Math.min(minZeile1, zs.zeile1Basiert());
162 maxZeile1 = Math.max(maxZeile1, zs.zeile1Basiert());
163 }
164 // Schreibblock: vom kleinsten gemeldeten Zeilenwert bis zum größten
165 // Reservebereich. Zeilen, für die kein Status vorliegt (Lücken zwischen
166 // gefüllten Zeilen oder hinter den aktuellen Daten), werden geleert –
167 // so verschwinden Stand-Reste eines früheren Laufs zuverlässig.
168 int startZeile0 = minZeile1 - 1;
169 int endZeile0 = SpielerDbVorlageSheet.MAX_DATEN_ZEILE;
170
171 RangeData rd = new RangeData();
172 for (int zeile0 = startZeile0; zeile0 <= endZeile0; zeile0++) {
173 int zeile1 = zeile0 + 1;
174 ZeilenStatus zs = byZeile1.get(zeile1);
175 RowData row = new RowData();
176 if (zs == null) {
177 row.newString("");
178 row.newString("");
179 } else {
180 row.newString(I18n.get(statusKey(zs.status())));
181 row.newString(zs.fehlerursache() == null ? "" : zs.fehlerursache());
182 }
183 rd.add(row);
184 }
185
186 try {
187 RangePosition pos = RangePosition.from(
188 SpielerDbVorlageSheet.SPALTE_STATUS, startZeile0,
189 SpielerDbVorlageSheet.SPALTE_FEHLERURSACHE, endZeile0);
190 RangeHelper.from(sheet, ws.getWorkingSpreadsheetDocument(), pos)
191 .setDataInRange(rd);
192 } catch (GenerateException e) {
193 logger.warn("Status-Block-Schreiben in Vorlage fehlgeschlagen", e);
194 }
195 }
