61 public void generate() throws GenerateException {
62 // 1. Meldeliste erstellen
63 meldelisteTestDaten.erstelleMeldelisteWithTestdaten();
64 naechsteSpielrunde.getKonfigurationSheet().setSpielrundeSpielbahn(SpielrundeSpielbahn.R);
65
66 naechsteSpielrunde.getKonfigurationSheet().setAnzahlRunden(ANZ_RUNDEN);
67
68 // 2. Spielrunden erstellen und mit Zufallsergebnissen füllen
69 for (int runde = 1; runde <= ANZ_RUNDEN; runde++) {
70 SheetRunner.testDoCancelTask();
71 processBoxinfo("processbox.erstelle.spielrunde", runde, ANZ_RUNDEN);
72 naechsteSpielrunde.doRun();
73
74 XSpreadsheet sheet = SheetMetadataHelper.findeSheetUndHeile(
75 getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
76 getSpielrundeSchluessel(runde),
77 getSheetName(SpielRundeNr.from(runde)));
78 if (sheet != null) {
79 ergebnisseEinfuegen(sheet);
80 }
81 }
82
83 // 3. Rangliste erstellen
84 SheetRunner.testDoCancelTask();
85 ranglisteSheet.doRun();
86
87 // 4. Teilnehmer erstellen
88 SheetRunner.testDoCancelTask();
89 teilnehmerSheet.generate();

91 // 5. Seitenstile aktualisieren
92 naechsteSpielrunde.getKonfigurationSheet().seitenstileAktualisieren();
93 }

95 private void ergebnisseEinfuegen(XSpreadsheet sheet) throws GenerateException {
96 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
97 RangePosition readRange = RangePosition.from(
98 TEAM_A_SPALTE, ERSTE_DATEN_ZEILE, ERG_TEAM_B_SPALTE, ERSTE_DATEN_ZEILE + 100);
99 RangeData data = RangeHelper.from(sheet, xDoc, readRange).getDataFromRange();
100
101 for (int i = 0; i < data.size(); i++) {
102 RowData row = data.get(i);
103 if (row.size() < 2) {
104 break;
105 }
106 int nrA = row.get(0).getIntVal(0);
107 if (nrA <= 0) {
108 break;
109 }
110 int nrB = row.get(1).getIntVal(0);
111 if (nrB <= 0) {
112 continue; // Freilos – Ergebnis ist bereits per Freispiel-Properties vorbelegt
113 }
114
115 int zeile = ERSTE_DATEN_ZEILE + i;
116 int winner = RandomSource.nextInt(2);
117 int loserPts = RandomSource.nextInt(0, 13);
118 int ergA = (winner == 0) ? 13 : loserPts;
119 int ergB = (winner == 0) ? loserPts : 13;
120
121 getSheetHelper().setNumberValueInCell(
122 NumberCellValue.from(sheet, Position.from(ERG_TEAM_A_SPALTE, zeile)).setValue(ergA));
123 getSheetHelper().setNumberValueInCell(
124 NumberCellValue.from(sheet, Position.from(ERG_TEAM_B_SPALTE, zeile)).setValue(ergB));
125 }
126 }
