205 private String getFormulastreichSpieltag(boolean iseven) throws GenerateException {
206 // 13 = streichspieltag spalte
207 // -1 = offset spalten links
208 // /3 = anzahl spalten in spieltag
209 // UND(INDIREKT(ADRESSE(ZEILE();13;4;))=AUFRUNDEN((SPALTE()-1)/3;0);ISTGERADE(ZEILE()))
210 int schlechtesteSpielTageSpalte = getSchlechtesteSpielTageSpalte() + 1; // In Formula erste spalte ab 1
211 int anzSpaltenInSpieltag = getAnzSpaltenInSpieltag();
212 int anzahlSpieltage = getAnzahlSpieltage();
213
214 String isCondition = "";
215 if (iseven) {
216 isCondition = "ISEVEN";
217 } else {
218 isCondition = "ISODD";
219 }
220
221 String verweisAufStreichSpalte = "INDIRECT(ADDRESS(ROW();" + schlechtesteSpielTageSpalte + ";4))";
222 // 1. prüfen ob wert in steichspieltag spalte > 0
223 String erstePruefung = verweisAufStreichSpalte + ">0";
224 // 2. prüfen ob wert in steichspieltag spalte < anzahlSpieltage
225 String zweitePruefung = verweisAufStreichSpalte + "<" + (anzahlSpieltage + 1);
226 // 3. prüfen ob wert in steichspieltag spalte == aktuelle spalte block
227 String drittePruefung = verweisAufStreichSpalte + "=ROUNDUP((COLUMN()-" + ERSTE_SPIELTAG_SPALTE + ")/"
228 + anzSpaltenInSpieltag + ";0)";
229 // 4. prüfen ob gerade oder ungerade zeile
230 String viertePruefung = isCondition + "(ROW())";
231 return "AND(" + erstePruefung + ";" + zweitePruefung + ";" + drittePruefung + ";" + viertePruefung + ")";
232 }

327 protected void berechnungUndSchreiben(XSpreadsheet sheet) throws GenerateException {
328 processBoxinfo("processbox.endrangliste.spieltage");
329
330 int anzSpieltage = getAnzahlSpieltage();
331 if (anzSpieltage < 2) {
332 return;
333 }
334
335 List<Integer> spielerNrList = leseSpielerNrInSheetOrdnung();
336 if (spielerNrList.isEmpty()) {
337 return;
338 }
339
340 Map<Integer, Map<Integer, SpielerSpieltagErgebnis>> cache = leseSpieltagRanglisteCache(anzSpieltage);
341
342 int letzteDatenzeile = ERSTE_DATEN_ZEILE + spielerNrList.size() - 1;
343 int ersteSpalteEndsumme = getErsteSummeSpalte();
344 int anzSpielTageSpalte = anzSpielTageSpalte();
345 int schlechtesterSpielTageSpalte = getSchlechtesteSpielTageSpalte();
346
347 RangeData block = new RangeData();
348 for (Integer spielerNr : spielerNrList) {
349 SheetRunner.testDoCancelTask();
350 List<SpielerSpieltagErgebnis> ergebnisseDesSpielers = ergebnisseDesSpielers(cache, anzSpieltage, spielerNr);
351 SpielTagNr streich = ermittleStreichSpieltag(ergebnisseDesSpielers);
352
353 RowData row = block.addNewRow();
354 int endSpielePlus = 0;
355 int endSpieleMinus = 0;
356 int endSpieleDiff = 0;
357 int endPunktePlus = 0;
358 int endPunkteMinus = 0;
359 int endPunkteDiff = 0;
360 int anzGespielt = 0;
361
362 for (SpielerSpieltagErgebnis erg : ergebnisseDesSpielers) {
363 if (erg.getSpielPlus() < 0) {
364 // nuller Spieltag – komplette Zellgruppe leer lassen
365 for (int i = 0; i < ANZAHL_SPALTEN_IN_SUMME; i++) {
366 row.newEmpty();
367 }
368 } else {
369 int sPlus = erg.getSpielPlus();
370 int sMinus = erg.getSpielMinus();
371 int pPlus = erg.getPunktePlus();
372 int pMinus = erg.getPunkteMinus();
373 row.newInt(sPlus);
374 row.newInt(sMinus);
375 row.newInt(sPlus - sMinus);
376 row.newInt(pPlus);
377 row.newInt(pMinus);
378 row.newInt(pPlus - pMinus);
379 anzGespielt++;
380 if (streich == null || streich.getNr() != erg.getSpielTag().getNr()) {
381 endSpielePlus += sPlus;
382 endSpieleMinus += sMinus;
383 endSpieleDiff += sPlus - sMinus;
384 endPunktePlus += pPlus;
385 endPunkteMinus += pMinus;
386 endPunkteDiff += pPlus - pMinus;
387 }
388 }
389 }
390 row.newInt(endSpielePlus);
391 row.newInt(endSpieleMinus);
392 row.newInt(endSpieleDiff);
393 row.newInt(endPunktePlus);
394 row.newInt(endPunkteMinus);
395 row.newInt(endPunkteDiff);
396
397 row.newInt(anzGespielt);
398 if (streich != null) {
399 row.newInt(streich.getNr());
400 } else {
401 row.newEmpty();
402 }
403 }
404
405 // Sanity check: Layout muss zusammenhängend sein.
406 int endBlockSpalte = ersteSpalteEndsumme + ANZAHL_SPALTEN_IN_SUMME - 1;
407 int expectedAnzTageSpalte = endBlockSpalte + 1;
408 int expectedSchlechtSpalte = expectedAnzTageSpalte + 1;
409 if (anzSpielTageSpalte != expectedAnzTageSpalte || schlechtesterSpielTageSpalte != expectedSchlechtSpalte) {
410 throw new GenerateException("EndranglisteSheet: unerwartetes Spaltenlayout für Block-Write");
411 }
412
413 RangeHelper.from(this,
414 RangePosition.from(ERSTE_SPIELTAG_SPALTE, ERSTE_DATEN_ZEILE,
415 schlechtesterSpielTageSpalte, letzteDatenzeile))
416 .setDataInRange(block);
417 }

443 private Map<Integer, Map<Integer, SpielerSpieltagErgebnis>> leseSpieltagRanglisteCache(int anzSpieltage)
444 throws GenerateException {
445 Map<Integer, Map<Integer, SpielerSpieltagErgebnis>> cache = new HashMap<>();
446 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
447 for (int spieltagCntr = 1; spieltagCntr <= anzSpieltage; spieltagCntr++) {
448 SheetRunner.testDoCancelTask();
449 SpielTagNr spielTag = SpielTagNr.from(spieltagCntr);
450 Map<Integer, SpielerSpieltagErgebnis> proSpieler = new HashMap<>();
451 cache.put(spieltagCntr, proSpieler);
452
453 XSpreadsheet spieltagSheet = spieltagRanglisteSheet.getSheet(spielTag);
454 if (spieltagSheet == null) {
455 continue;
456 }
457 int ersteSummeSpalte = spieltagRanglisteSheet.getErsteSummeSpalte(spielTag);
458 int letzteSummeSpalte = ersteSummeSpalte + ANZAHL_SPALTEN_IN_SUMME - 1;
459 int spielerNrSpalteRangliste = SpieltagRanglisteSheet.SPIELER_NR_SPALTE;
460 int ersteDatenZeile = SpieltagRanglisteSheet.ERSTE_DATEN_ZEILE;
461 int letzteDatenZeile = ersteDatenZeile + 999;
462
463 // Block 1: Spielernr-Spalte
464 RangeData spielerNrBlock = RangeHelper
465 .from(spieltagSheet, xDoc,
466 RangePosition.from(spielerNrSpalteRangliste, ersteDatenZeile,
467 spielerNrSpalteRangliste, letzteDatenZeile))
468 .getDataFromRange();
469 // Block 2: Summenspalten
470 RangeData summenBlock = RangeHelper
471 .from(spieltagSheet, xDoc,
472 RangePosition.from(ersteSummeSpalte, ersteDatenZeile,
473 letzteSummeSpalte, letzteDatenZeile))
474 .getDataFromRange();
475
476 int rowCount = Math.min(spielerNrBlock.size(), summenBlock.size());
477 for (int r = 0; r < rowCount; r++) {
