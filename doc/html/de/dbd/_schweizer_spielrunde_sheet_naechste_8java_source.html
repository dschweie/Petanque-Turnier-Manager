47 public boolean naechsteSpielrundeEinfuegen() throws GenerateException {
48 SpielRundeNr aktuelleSpielrunde = getKonfigurationSheet().getAktiveSpielRunde();
49 setSpielRundeNrInSheet(aktuelleSpielrunde);
50 getMeldeListe().vollstaendigAktualisieren();
51 TeamMeldungen aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
52
53 if (aktiveMeldungen.size() == 0) {
54 TeamMeldungen alleMeldungen = getMeldeListe().getAlleMeldungen();
55 if (alleMeldungen.size() > 0) {
56 MessageBoxResult result = MessageBox.from(getxContext(), MessageBoxTypeEnum.WARN_YES_NO)
57 .caption(I18n.get("msg.caption.keine.aktiven.meldungen"))
58 .message(I18n.get("msg.text.keine.aktiven.teams.aktivieren", alleMeldungen.size()))
59 .show();
60 if (result == MessageBoxResult.YES) {
61 getMeldeListe().alleTeamsAktivieren();
62 aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
63 }
64 }
65 }
66
67 if (!canStart(aktiveMeldungen)) {
68 return false;
69 }
70
71 // Ermitteln ob es eine aktuelle Runde gibt (= Sheet vorhanden); Metadaten-first via getXSpreadSheet()
72 int neueSpielrunde = aktuelleSpielrunde.getNr();
73 if (getXSpreadSheet() != null) {
74 // Aktuelle Runde vorhanden → prüfen ob alle Ergebnisse eingetragen sind
75 if (!alleErgebnisseEingetragen(aktuelleSpielrunde)) {
76 getSheetHelper().setActiveSheet(getXSpreadSheet());
77 MessageBox.from(getxContext(), MessageBoxTypeEnum.ERROR_OK)
78 .caption(I18n.get("msg.caption.naechste.runde.nicht.moeglich"))
79 .message(I18n.get("msg.text.naechste.runde.ergebnisse.fehlen", aktuelleSpielrunde.getNr()))
80 .show();
81 return false;
82 }
83 neueSpielrunde++;
84 }
85
86 if (!pruefeKannNeueRundeErstellen(neueSpielrunde)) {
87 return false;
88 }
89
90 // Bestehende Rangliste aktualisieren, damit die Auslosung der neuen Runde auf
91 // einem konsistenten Stand basiert. Nur ab Runde >= 2 sinnvoll.
92 if (neueSpielrunde >= 2) {
93 vorNaechsterRunde();
94 }
95
96 // Konfiguration auf neue Rundennummer setzen (muss VOR neueSpielrunde() passieren,
97 // da getSpielRundeNr() den Sheet-Namen daraus ableitet)
98 getKonfigurationSheet().setAktiveSpielRunde(SpielRundeNr.from(neueSpielrunde));
99
100 List<SchweizerTeamErgebnis> ergebnisse = gespieltenRundenEinlesen(aktiveMeldungen, 1, neueSpielrunde - 1);
101
102 // Teams nach Rangliste sortieren (ab Runde 2)
103 TeamMeldungen meldungenFuerAuslosung = (neueSpielrunde > 1)
104 ? sortierteTeamMeldungen(aktiveMeldungen, ergebnisse)
105 : aktiveMeldungen;
106
107 return neueSpielrunde(meldungenFuerAuslosung, SpielRundeNr.from(neueSpielrunde), ergebnisse);
108 }

142 private boolean alleErgebnisseEingetragen(SpielRundeNr spielRundeNr) throws GenerateException {
143 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
144 XSpreadsheet sheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
145 getSpielrundeSchluessel(spielRundeNr.getNr()), getSheetName(spielRundeNr));
146 if (sheet == null) {
147 return true;
148 }
149 RangePosition readRange = RangePosition.from(TEAM_A_SPALTE, ERSTE_DATEN_ZEILE, ERG_TEAM_B_SPALTE,
150 ERSTE_DATEN_ZEILE + 999);
151 RangeData rowsData = RangeHelper
152 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange).getDataFromRange();
153
154 for (RowData row : rowsData) {
155 if (row.size() < 2) {
156 break;
157 }
158 int nrA = row.get(0).getIntVal(0);
159 if (nrA <= 0) {
160 // Name statt Zahl → keine Ergebnisprüfung möglich per int, String-Inhalt prüfen
161 String valA = row.get(0).getStringVal();
162 if (valA == null || valA.isEmpty()) {
163 break; // Ende der Daten
164 }
165 }
166 // Team B: leer = Freilos, Freilos braucht kein Ergebnis
167 int nrB = row.get(1).getIntVal(0);
168 if (nrB <= 0) {
169 String valB = row.get(1).getStringVal();
170 if (valB == null || valB.isEmpty()) {
171 continue; // Freilos-Zeile
172 }
173 }
174 // Ergebnisse prüfen
175 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0;
176 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0;
177 if (ergA == 0 && ergB == 0) {
178 return false; // Ergebnis fehlt
179 }
180 }
181 return true;
182 }
