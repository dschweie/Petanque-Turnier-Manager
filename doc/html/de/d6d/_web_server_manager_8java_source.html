719 private void sseRefreshSendenIntern(WorkingSpreadsheet ws) {
720 // Serialisiert: niemals zwei parallele Render-/Push-Wellen. Konkurrierende Aufrufer
721 // signalisieren stattdessen "nochmal" über den Listener-Dirty-Flag.
722 if (!refreshLock.tryLock()) {
723 modifyListener.markDirty();
724 logger.debug("Refresh läuft bereits – markDirty()");
725 return;
726 }
727 try {
728 registriereModifyListenerFallsNoetig(ws);
729 // Genau ein calculate() pro Refresh-Welle, vor der Composite-Schleife –
730 // spart bei N Compositen N-1 Recalc-Durchläufe und stellt sicher, dass
731 // abhängige Formelzellen (z.B. Rangliste) den aktuellen Stand spiegeln.
732 forceRecalc(ws);
733 for (var instanz : compositeInstanzen) {
734 renderUndPushenComposite(instanz, ws);

737 // keine Diff-Engine, nur zwei Integer + Diff-Cache.
738 pushStartseiteFallsAktiv(ws);

740 } finally {
741 refreshLock.unlock();
742 // Livelock-Schutz: konkurrierende Aufrufer haben evtl. nur markDirty() gemacht
743 // und sind an tryLock() abgeprallt. Falls jetzt dirty noch true ist, garantiert
744 // der erfolgreiche Owner eine Folge-Welle.
745 if (modifyListener.isDirty()) {
746 modifyListener.scheduleIfNeededExternal();
747 }

877 var panelModelle = letzteCompositeModelle.computeIfAbsent(port, p -> new ConcurrentHashMap<>());
878 var panelTitel = letzteCompositeTitel.computeIfAbsent(port, p -> new ConcurrentHashMap<>());
879 var fehlendCache = fehlendePanels.computeIfAbsent(port, p -> ConcurrentHashMap.newKeySet());
880 try {
881 var panelNachrichten = new ArrayList<CompositePanelNachricht>();
882 boolean irgendeineAenderung = false;
883 boolean erstesRendering = false;
884
885 var urlCache = letzteCompositeUrls.computeIfAbsent(port, p -> new ConcurrentHashMap<>());
886 for (int i = 0; i < konfig.panels().size(); i++) {
887 var panelKonfig = konfig.panels().get(i);

890 if (panelKonfig.typ() == PanelTyp.TIMER) {

907 urlCache.put(i, panelKonfig.externeUrl());

910 }
911 continue;

915 if (sheetOpt.isEmpty()) {
916 // Nicht aufgelöstes Sheet: nur dieses Panel zeigt einen Hinweis,
917 // alle anderen Panels rendern weiterhin normal. Init-Cache enthält
918 // weiter unten die fehlend-Nachricht für reconnectende Clients.
919 panelModelle.remove(i);
920 panelTitel.remove(i);

924 I18n.get("webserver.hinweis.sheet.nicht.gefunden.titel"),

934 String alterTitel = panelTitel.get(i);
935 boolean warVorherFehlend = fehlendCache.remove(i);
936
937 if (altesModell != null && layoutGeaendert(altesModell, neuesModell)) {

944 || kopfFusszeileGeaendert(altesModell, neuesModell)
945 || !java.util.Objects.equals(alterTitel, neuerTitel)
946 || warVorherFehlend;
947
948 if (altesModell == null) {
949 erstesRendering = true;
950 }
951
952 panelModelle.put(i, neuesModell);
953 panelTitel.put(i, neuerTitel);

956 irgendeineAenderung = true;
957 }

959 // vollständiges Modell senden, sonst nur Diff wenn nötig

961 panelNachrichten.add(CompositePanelNachricht.init(i, zuSendendesModell, neuerTitel, panelKonfig.zoom(), panelKonfig.horizontalAusrichtung(), panelKonfig.vertikalAusrichtung(), panelKonfig.blattnameAnzeigen()));
962 }
963

965 return;
966 }
967

969 .computeIfAbsent(port, p -> new AtomicInteger(0))
970 .incrementAndGet();

973 var alleInitPanels = new ArrayList<CompositePanelNachricht>();
974 for (int i = 0; i < konfig.panels().size(); i++) {
975 var panelKonfig = konfig.panels().get(i);
976 if (panelKonfig.typ() == PanelTyp.TIMER) {
977 alleInitPanels.add(CompositePanelNachricht.timer(i, panelKonfig.zoom(), panelKonfig.horizontalAusrichtung(), panelKonfig.vertikalAusrichtung(), letzterTimerZustand));
978 continue;
979 }
980 if (panelKonfig.typ() == PanelTyp.URL) {
981 String cachedUrl = urlCache.get(i);
982 if (cachedUrl != null) {
983 alleInitPanels.add(CompositePanelNachricht.url(i, cachedUrl));
984 }
985 continue;
986 }
987 var vollModell = panelModelle.get(i);
988 if (vollModell != null) {
989 alleInitPanels.add(CompositePanelNachricht.init(i, vollModell, panelTitel.getOrDefault(i, ""), panelKonfig.zoom(), panelKonfig.horizontalAusrichtung(), panelKonfig.vertikalAusrichtung(), panelKonfig.blattnameAnzeigen()));
990 } else if (fehlendCache.contains(i)) {
991 alleInitPanels.add(CompositePanelNachricht.fehlend(i,
992 I18n.get("webserver.hinweis.sheet.nicht.gefunden.titel"),
993 I18n.get("webserver.hinweis.sheet.nicht.gefunden.text",
994 panelKonfig.resolver().getAnzeigeName())));
995 }
996 }
997 CompositeSseNachricht initNachricht = CompositeSseNachricht.init(version, alleInitPanels, konfig.wurzel(), konfig.zoom(), konfig.mitHeaderFooter());
998 instanz.setCachedInitJson(GSON.toJson(initNachricht));
999
1000 // Push: beim ersten Rendering init, sonst diff mit geänderten Panels
1001 CompositeSseNachricht push = erstesRendering
1002 ? initNachricht
1003 : CompositeSseNachricht.diff(version, panelNachrichten, konfig.wurzel(), konfig.zoom(), konfig.mitHeaderFooter());
1004 instanz.sseNachrichtPushen(GSON.toJson(push));
1005
1006 } catch (Exception e) {
1007 logger.error("Fehler beim Rendern des Composite Views für Port {}: {}", port, e.getMessage(), e);
1008 safeProcessBoxFehler(I18n.get("webserver.prozessbox.render.fehler", port, e.getMessage()));
1009 }
