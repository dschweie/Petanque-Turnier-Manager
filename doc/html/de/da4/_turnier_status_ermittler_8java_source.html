28 public static String ermitteln(WorkingSpreadsheet ws) {
29 if (ws == null) {
30 return "";
31 }
32 var docPropHelper = new DocumentPropertiesHelper(ws);
33 TurnierSystem system = docPropHelper.getTurnierSystemAusDocument();
34 if (system == null) {
35 return "";
36 }
37 return switch (system) {
38 case SUPERMELEE -> {
39 int spieltag = docPropHelper.getIntProperty(SuperMeleePropertiesSpalte.KONFIG_PROP_NAME_SPIELTAG, 1);
40 var xDoc = ws.getWorkingSpreadsheetDocument();
41 String spielrundeSheetName = I18n.get("sheet.name.supermelee.spielrunde.muster", spieltag, 1);
42 var spielrundeSheet = SheetMetadataHelper.findeSheetUndHeile(xDoc,
43 SheetMetadataHelper.schluesselSupermeleeSpielrunde(spieltag, 1),
44 spielrundeSheetName);
45 if (spielrundeSheet == null) {
46 yield I18n.get("sidebar.info.meldungen.erfassen");
47 }
48 int runde = docPropHelper.getIntProperty(SuperMeleePropertiesSpalte.KONFIG_PROP_NAME_SPIELRUNDE, 1);
49 yield I18n.get("sidebar.info.supermelee.schritt", spieltag, runde);
50 }
51 case SCHWEIZER -> {
52 int runde = docPropHelper.getIntProperty(SchweizerPropertiesSpalte.KONFIG_PROP_NAME_SPIELRUNDE, 1);
53 yield I18n.get("sidebar.info.spielrunde", runde);
54 }
55 case LIGA -> {
56 var status = LigaStatusLeser.von(ws).liesStatus();
57 if (!status.spielplanVorhanden()) {
58 yield I18n.get("sidebar.info.meldungen.erfassen");
59 }
60 if (status.alleGespielt()) {
61 yield I18n.get("sidebar.info.turnier.beendet");
62 }
63 yield I18n.get("sidebar.info.liga.schritt",
64 status.hrGespielt(), status.hrGesamt(),
65 status.rrGespielt(), status.rrGesamt());
66 }
67 case JGJ -> {
68 var status = JGJStatusLeser.von(ws).liesStatus();
69 if (!status.spielplanVorhanden()) {
70 yield I18n.get("sidebar.info.meldungen.erfassen");
71 }
72 if (status.alleGespielt()) {
73 yield I18n.get("sidebar.info.turnier.beendet");
74 }
75 yield I18n.get("sidebar.info.jgj.schritt",
76 status.hrGespielt(), status.hrGesamt(),
77 status.rrGespielt(), status.rrGesamt());
78 }
79 case POULE -> {
80 var status = PouleStatusLeser.von(ws).liesStatus();
81 if (!status.vorrundeVorhanden()) {
82 yield I18n.get("sidebar.info.meldungen.erfassen");
83 }
84 if (status.beendet()) {
85 yield I18n.get("sidebar.info.turnier.beendet");
86 }
87 if (status.koVorhanden()) {
88 yield I18n.get("sidebar.poule.ko");
89 }
90 yield I18n.get("sidebar.info.poule.vorrunde",
91 status.vorrundeGespielt(), status.vorrundeGesamt());
92 }
93 case MAASTRICHTER -> {
94 var status = MaastrichterStatusLeser.von(ws).liesStatus();
95 if (!status.vorrundeVorhanden()) {
96 yield I18n.get("sidebar.info.meldungen.erfassen");
97 }
98 if (status.beendet()) {
99 yield I18n.get("sidebar.info.turnier.beendet");
100 }
101 if (status.finalrundeVorhanden()) {
102 yield I18n.get("sidebar.maastrichter.finalrunde");
103 }
104 yield I18n.get("sidebar.info.maastrichter.vorrunde",
105 status.aktuelleVorrundeNr(),
106 status.vorrundeGespielt(), status.vorrundeGesamt());
107 }
108 case KASKADE -> {
109 var status = KaskadeStatusLeser.von(ws).liesStatus();
110 if (!status.rundeVorhanden()) {
111 yield I18n.get("sidebar.info.meldungen.erfassen");
112 }
113 if (status.beendet()) {
114 yield I18n.get("sidebar.info.turnier.beendet");
115 }
116 if (status.koPhaseVorhanden()) {
117 yield I18n.get("sidebar.kaskade.ko.felder");
118 }
119 yield I18n.get("sidebar.info.kaskade.runde",
120 status.aktuelleRundeNr(),
121 status.rundeGespielt(), status.rundeGesamt());
122 }
123 case KO -> {
124 var status = KoStatusLeser.von(ws).liesStatus();
125 if (!status.turnierbaumVorhanden()) {
126 yield I18n.get("sidebar.info.meldungen.erfassen");
127 }
128 if (status.beendet()) {
129 yield I18n.get("sidebar.info.turnier.beendet");
130 }
131 yield I18n.get("sidebar.info.ko.laeuft");
132 }
133 case FORMULEX -> {
134 var status = FormuleXStatusLeser.von(ws).liesStatus();
135 if (!status.spielrundeVorhanden()) {
136 yield I18n.get("sidebar.info.meldungen.erfassen");
137 }
138 if (status.beendet()) {
139 yield I18n.get("sidebar.info.turnier.beendet");
140 }
141 yield I18n.get("sidebar.info.formulex.schritt",
142 status.aktuelleRundeNr(), status.anzahlRunden(),
143 status.rundeGespielt(), status.rundeGesamt());
144 }
145 default -> "";
146 };
147 }
