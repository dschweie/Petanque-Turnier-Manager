44 public void updateHeader() throws GenerateException {
45 int anzSpieltagen = rangliste.getAnzahlSpieltage();
46 if (anzSpieltagen < 1) {
47 return;
48 }
49 // -------------------------
50 // spieltag spalten
51 // -------------------------
52 StringCellValue spieltagheader = StringCellValue.from(getSheet(), Position.from(ERSTE_KOPFDATEN_ZEILE, 0))
53 .setCellBackColor(getHeaderFarbe())
54 .setBorder(BorderFactory.from().allThin().boldLn().forLeft().forTop().forRight().toBorder())
55 .setHoriJustify(CellHoriJustify.CENTER)
56 .setShrinkToFit(true);
57 for (int spielTag = 1; spielTag <= anzSpieltagen; spielTag++) {
58 int ersteSpalteSpieltagBlock = ersteSpielTagSpalte + ((spielTag - 1) * anzSpaltenInSpieltag);
59 // ERSTE_KOPFDATEN_ZEILE
60 spieltagheader.spalte(ersteSpalteSpieltagBlock).setValue(spielTag + ". Spieltag")
61 .setEndPosMergeSpaltePlus(anzSpaltenInSpieltag - 1);
62 getSheetHelper().setStringValueInCell(spieltagheader);
63 formatZweiteZeileSpielTagSpalten(ersteSpalteSpieltagBlock); // ZWEITE_KOPFDATEN_ZEILE
64 formatDritteZeileSpielTagSpalten(ersteSpalteSpieltagBlock, MeldungenSpalte.DEFAULT_SPALTE_NUMBER_WIDTH);
65 }
66
67 // -------------------------
68 // endsumme spalten
69 // -------------------------
70 formatEndSummen(rangliste.getErsteSummeSpalte());
71 // -------------------------
72 }
73
74 public void formatDaten() throws GenerateException {
75 int anzSpielTage = rangliste.getAnzahlSpieltage();
76 if (anzSpielTage < 1) {
77 return;
78 }
79 int ersteDatenZeile = getSpielerSpalte().getErsteDatenZiele();
80 int letzteDatenZeile = getSpielerSpalte().getLetzteMitDatenZeileInSpielerNrSpalte();
81 if (letzteDatenZeile < ersteDatenZeile) {
82 return;
83 }
84
85 // Statt 2 Range-Calls pro Spieltag: alle gleichartigen Sub-Blöcke je Border-Typ
86 // in einem einzigen Multi-Range-Call setzen.
87 List<RangePosition> spieleBloecke = new ArrayList<>();
88 List<RangePosition> punkteBloecke = new ArrayList<>();
89 for (int spielTag = 1; spielTag <= anzSpielTage; spielTag++) {
90 int ersteSpalteSpieltagBlock = ersteSpielTagSpalte + ((spielTag - 1) * anzSpaltenInSpieltag);
91 sammelSpieleUndPunkteBloecke(ersteSpalteSpieltagBlock, ersteDatenZeile, letzteDatenZeile,
92 spieleBloecke, punkteBloecke);
93 }
94 // Endsumme-Block (gleiches Layout)
95 sammelSpieleUndPunkteBloecke(rangliste.getErsteSummeSpalte(), ersteDatenZeile, letzteDatenZeile,
96 spieleBloecke, punkteBloecke);
97
98 TableBorder2 spieleBorder = BorderFactory.from().allThin().boldLn().forTop().forLeft().toBorder();
99 TableBorder2 punkteBorder = BorderFactory.from().allThin().boldLn().forTop().forRight()
100 .doubleLn().forLeft().toBorder();
101 getSheetHelper().setPropertyInMultipleRanges(rangliste, spieleBloecke, ICommonProperties.TABLE_BORDER2,
102 spieleBorder);
103 getSheetHelper().setPropertyInMultipleRanges(rangliste, punkteBloecke, ICommonProperties.TABLE_BORDER2,
104 punkteBorder);
105 }
106
107 private void sammelSpieleUndPunkteBloecke(int ersteSummeSpalte, int ersteDatenZeile, int letzteDatenZeile,
108 List<RangePosition> spieleBloecke, List<RangePosition> punkteBloecke) {
109 // Spiele-Block: +,-,Δ (3 Spalten ab ersteSummeSpalte)
110 spieleBloecke.add(RangePosition.from(ersteSummeSpalte, ersteDatenZeile,
111 ersteSummeSpalte + 2, letzteDatenZeile));
112 // Punkte-Block: +,-,Δ (3 Spalten direkt anschließend)
113 punkteBloecke.add(RangePosition.from(ersteSummeSpalte + 3, ersteDatenZeile,
114 ersteSummeSpalte + 5, letzteDatenZeile));
115 }

117 @Override
118 public StringCellValue addFooter() throws GenerateException {
119 StringCellValue stringVal = super.addFooter();
120
121 getSheetHelper().setStringValueInCell(stringVal.zeilePlusEins()
122 .setEndPosMergeSpaltePlus(getLetzteSpalte())
123 .addCellProperty(IS_TEXT_WRAPPED, Boolean.TRUE)
124 .addRowProperty("OptimalHeight", Boolean.TRUE)
125 .setValue("Aus der Endranglistenwertung entfallen eine einmalige Nichtteilnahme bzw. das"));
126 getSheetHelper().setStringValueInCell(stringVal.zeilePlusEins()
127 .setEndPosMergeSpaltePlus(getLetzteSpalte())
128 .addCellProperty(IS_TEXT_WRAPPED, Boolean.TRUE)
129 .addRowProperty("OptimalHeight", Boolean.TRUE)
130 .setValue("schlechteste Tagesergebnis wenn an allen Spieltagen teilgenommen wurde"));
131
132 return stringVal;
133 }
