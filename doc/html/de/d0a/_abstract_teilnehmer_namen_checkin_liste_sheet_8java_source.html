63 @Override
64 protected List<Integer> ladeNummern() throws GenerateException {
65 List<Integer> nummern = new ArrayList<>();
66 ISheet meldelisteSheet = getMeldelisteSheet();
67 XSpreadsheet xSheet = meldelisteSheet.getXSpreadSheet();
68 if (xSheet == null) {
69 return nummern;
70 }
71 int ersteZeile = getMeldelisteErsteDatenZeile();
72 RangeData data = RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
73 RangePosition.from(0, ersteZeile, 0, ersteZeile + LESE_BIS_ZEILE_OFFSET)).getDataFromRange();
74 for (RowData row : data) {
75 if (row.isEmpty()) {
76 break;
77 }
78 int nr = row.get(0).getIntVal(0);
79 if (nr <= 0) {
80 break;
81 }
82 nummern.add(nr);
83 }
84 return nummern;

87 @Override
88 protected Map<Integer, Boolean> checkboxStatusNachNummer() throws GenerateException {

90 ISheet meldelisteSheet = getMeldelisteSheet();
91 XSpreadsheet xSheet = meldelisteSheet.getXSpreadSheet();
92 if (xSheet == null) {
93 return ergebnis;

95 int aktivSpalte = getMeldelisteAktivSpalte();
96 int ersteZeile = getMeldelisteErsteDatenZeile();
97 RangeData data = RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
98 RangePosition.from(0, ersteZeile, aktivSpalte, ersteZeile + LESE_BIS_ZEILE_OFFSET)).getDataFromRange();
99 for (RowData row : data) {
100 if (row.isEmpty()) {
101 break;
102 }
103 int nr = row.get(0).getIntVal(0);
104 if (nr <= 0) {
105 break;
106 }
107 boolean aktivGesetzt = StringUtils.isNotBlank(row.get(aktivSpalte).getStringVal());
108 ergebnis.put(nr, aktivGesetzt);
109 }
110 return ergebnis;
