432 @Override
433 public String[][] ptmdbspielersuche(String suche, boolean wildcard) {
434 String normalized = (suche == null ? "" : suche).trim().toLowerCase(Locale.ROOT);
435 if (normalized.length() > SUCHE_MAX_LEN) {
436 normalized = normalized.substring(0, SUCHE_MAX_LEN);
437 }
438 String key = "suche:" + wildcard + ":" + normalized;
439 String suchTerm = normalized;
440 try {
441 synchronized (spielerCache) {
442 return spielerCache.holeOderBerechne(key, () -> {
443 try {
444 List<SpielerMitVerein> treffer = spielerRepository()
445 .findeMitWildcard(suchTerm, wildcard, SUCHE_LIMIT);
446 String[][] tabelle = new String[treffer.size()][3];
447 for (int i = 0; i < treffer.size(); i++) {
448 SpielerMitVerein s = treffer.get(i);
449 tabelle[i][0] = s.vorname();
450 tabelle[i][1] = s.nachname();
451 tabelle[i][2] = s.vereinName() == null ? "" : s.vereinName();
452 }
453 return tabelle;
454 } catch (SpielerDbException e) {
455 logger.error("PTM.DB.SPIELER.SUCHE fehlgeschlagen: '{}'", suchTerm, e);
456 return LEERES_SUCH_ERGEBNIS;
457 }
458 });
459 }
460 } catch (RuntimeException | LinkageError e) {
461 logger.error("PTM.DB.SPIELER.SUCHE: harter Init-Fehler", e);
462 return LEERES_SUCH_ERGEBNIS;
463 }
