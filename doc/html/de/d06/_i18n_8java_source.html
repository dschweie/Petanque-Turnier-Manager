121 private static Locale ermittleLocale(XComponentContext xContext) {
122 if (xContext == null) {
123 return Locale.GERMAN;
124 }
125 try {
126 XMultiServiceFactory msf = Lo.qi(XMultiServiceFactory.class,
127 xContext.getServiceManager().createInstanceWithContext(
128 "com.sun.star.configuration.ConfigurationProvider", xContext));
129 var pv = new com.sun.star.beans.PropertyValue(
130 "nodepath", 0, "/org.openoffice.Setup/L10N",
131 com.sun.star.beans.PropertyState.DIRECT_VALUE);
132 Object acc = msf.createInstanceWithArguments(
133 "com.sun.star.configuration.ConfigurationAccess", new Object[] { pv });
134 String ooLocale = (String) Lo.qi(XHierarchicalNameAccess.class, acc)
135 .getByHierarchicalName("ooLocale");
136 return Locale.forLanguageTag(ooLocale.replace('_', '-'));
137 } catch (Exception e) {
138 logger.error("Locale-Ermittlung fehlgeschlagen: {}", e.getMessage(), e);
139 return Locale.GERMAN;
