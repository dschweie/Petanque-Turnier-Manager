76 private void rangListeNachCadrageErstellen() throws GenerateException {
77 NewSheet ranglisteNachCadrage = NewSheet.temporary(this, RANGLISTE_NACH_CADRAGE_SHEET).pos(DefaultSheetPos.MELEE_WORK).forceCreate().setActiv().create();
78 for (int grpCntr = 0; grpCntr < 10; grpCntr++) {
79 TeamRangliste gruppeAusRanglist = ranglisten.gruppeAusRanglisteEinlesen(grpCntr, RANGLISTE_AUS_VORUNDE_SHEET);
80 if (gruppeAusRanglist.isEmpty()) {
81 break; // fertig keine weitere gruppen
82 }
83 int anzTeamsinGruppeAusRangliste = gruppeAusRanglist.size();
84 int anzCadrageTeams = new CadrageRechner(anzTeamsinGruppeAusRangliste).anzTeams();
85 TeamRangliste ranglisteOhneCadrage = gruppeAusRanglist.ranglisteVonOben(anzTeamsinGruppeAusRangliste - anzCadrageTeams);
86 Position posTeamNr = Position.from(grpCntr, 0);
87 getSheetHelper().setActiveSheet(ranglisteNachCadrage.getSheet());
88 for (Team team : ranglisteOhneCadrage.getTeamListe()) {
89 NumberCellValue teamRangliste = NumberCellValue.from(ranglisteNachCadrage.getSheet(), posTeamNr).setValue((double) team.getNr());
90 getSheetHelper().setNumberValueInCell(teamRangliste);
91 posTeamNr.zeilePlusEins();
92 }
93 }
94 }

100 private void cadrageErstellen() throws GenerateException {
101 for (int grpCntr = 0; grpCntr < 10; grpCntr++) {
102 TeamRangliste gruppeAusRanglist = ranglisten.gruppeAusRanglisteEinlesen(grpCntr, RANGLISTE_AUS_VORUNDE_SHEET);
103 if (gruppeAusRanglist.isEmpty()) {
104 break; // fertig keine weitere gruppen
105 }
106
107 // anzahl cadrage teams ?
108 // 2,4,8,16,32
109 int anzTeamsinGruppeAusRangliste = gruppeAusRanglist.size();
110 int anzCadrageTeams = new CadrageRechner(anzTeamsinGruppeAusRangliste).anzTeams();
111 processBoxinfo("processbox.cadrage.gruppe", grpCntr, anzCadrageTeams);
112 if (anzCadrageTeams > 0) {
113 TeamRangliste cadrageRangliste = gruppeAusRanglist.ranglisteVonLetzte(anzCadrageTeams);
114 vorrunden.vorRundenEinlesen(cadrageRangliste);
115 KoRundeTeamPaarungen teamPaarungen = new KoRundeTeamPaarungen(cadrageRangliste);
116 FormeSpielrunde spielRunde = teamPaarungen.generiereSpielrunde();
117 // Paarungen
118 getSheetHelper().setActiveSheet(getXSpreadSheet());
119 spielRundeInSheet.erstelleSpielRundeInSheet(grpCntr, getXSpreadSheet(), teamPaarungen, spielRunde);
120 }
121 }
