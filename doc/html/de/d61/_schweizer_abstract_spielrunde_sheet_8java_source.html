227 processBoxinfo("processbox.gespielte.runden.einlesen", spielrunde, bisSpielrunde);
228 var xDoc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
229
230 for (; spielrunde <= bisSpielrunde; spielrunde++) {

243 for (Team team : aktiveMeldungen.teams()) {

251 private void leseRundeEin(XSpreadsheet sheet, TeamMeldungen aktiveMeldungen, Map<Integer, int[]> statsMap,
252 Map<Integer, List<Integer>> gegnerMap) throws GenerateException {
253 RangePosition readRange = RangePosition.from(TEAM_A_SPALTE, ERSTE_DATEN_ZEILE, ERG_TEAM_B_SPALTE,
254 ERSTE_DATEN_ZEILE + 999);
255 RangeData rowsData = RangeHelper
256 .from(sheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange).getDataFromRange();
257
258 for (RowData row : rowsData) {
259 if (row.size() < 2) {
260 break;
261 }
262 int nrA = resolveTeamNr(row.get(0)); // TEAM_A_SPALTE (relativ: 0)
263 if (nrA <= 0) {
264 break; // Ende der Daten
265 }
266 Team teamA = aktiveMeldungen.getTeam(nrA);
267 if (teamA == null) {
268 continue; // Team inaktiv, überspringen
269 }
270
271 int nrB = resolveTeamNr(row.get(1)); // TEAM_B_SPALTE (relativ: 1)
272 if (nrB <= 0) {
273 // Freilos für Team A
274 teamA.setHatteFreilos(true);
275 statsMap.computeIfAbsent(nrA, k -> new int[3])[0]++;
276 continue;
277 }
278 Team teamB = aktiveMeldungen.getTeam(nrB);
279 if (teamB == null) {
280 continue;
281 }
282
283 teamA.addGegner(teamB); // registriert gegenseitig
284
285 gegnerMap.computeIfAbsent(nrA, k -> new ArrayList<>()).add(nrB);
286 gegnerMap.computeIfAbsent(nrB, k -> new ArrayList<>()).add(nrA);
287
288 int ergA = (row.size() > 2) ? row.get(2).getIntVal(0) : 0; // ERG_TEAM_A_SPALTE (relativ: 2)
289 int ergB = (row.size() > 3) ? row.get(3).getIntVal(0) : 0; // ERG_TEAM_B_SPALTE (relativ: 3)
290
291 // Punkte+ für beide Teams
292 statsMap.computeIfAbsent(nrA, k -> new int[3])[2] += ergA;
293 statsMap.computeIfAbsent(nrB, k -> new int[3])[2] += ergB;
294
295 if (ergA > ergB) {
296 statsMap.computeIfAbsent(nrA, k -> new int[3])[0]++;
297 statsMap.computeIfAbsent(nrA, k -> new int[3])[1] += ergA - ergB;
298 statsMap.computeIfAbsent(nrB, k -> new int[3])[1] -= ergA - ergB;
299 } else if (ergB > ergA) {
300 statsMap.computeIfAbsent(nrB, k -> new int[3])[0]++;
301 statsMap.computeIfAbsent(nrB, k -> new int[3])[1] += ergB - ergA;
302 statsMap.computeIfAbsent(nrA, k -> new int[3])[1] -= ergB - ergA;
303 }
304 }
