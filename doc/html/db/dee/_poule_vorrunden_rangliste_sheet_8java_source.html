202 private List<Map<Integer, int[]>> leseGruppenErgebnisse(XSpreadsheet vorrundeSheet)
203 throws GenerateException {
204
205 var readRange = RangePosition.from(
206 AbstractPouleVorrundeSheet.SPALTE_POULE_NR,
207 AbstractPouleVorrundeSheet.ERSTE_DATEN_ZEILE,
208 AbstractPouleVorrundeSheet.SPALTE_ERG_B,
209 AbstractPouleVorrundeSheet.ERSTE_DATEN_ZEILE + 9999);
210
211 RangeData rowsData = RangeHelper
212 .from(vorrundeSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(), readRange)
213 .getDataFromRange();
214
215 List<Map<Integer, int[]>> gruppen = new ArrayList<>();
216 Map<Integer, int[]> aktuelleGruppe = null;
217
218 for (RowData row : rowsData) {
219 if (row.size() < 7) {
220 break;
221 }
222
223 // Spalten-Offsets: SPALTE_POULE_NR=0, TEAM_A_NR=1, TEAM_A_NAME=2, TEAM_B_NR=3, TEAM_B_NAME=4, ERG_A=5, ERG_B=6
224 // Neue Gruppe: SPALTE_POULE_NR ist nur in der ersten Zeile des Merge-Blocks nicht leer.
225 String pouleNrStr = row.get(0).getStringVal();
226 if (pouleNrStr != null && !pouleNrStr.isEmpty()) {
227 aktuelleGruppe = new HashMap<>();
228 gruppen.add(aktuelleGruppe);
229 }
230
231 if (aktuelleGruppe == null) {
232 continue;
233 }
234
235 int teamANr = row.get(1).getIntVal(0);
236 int teamBNr = row.get(3).getIntVal(0);
237
238 int ergA = row.get(5).getIntVal(0);
239 int ergB = row.get(6).getIntVal(0);
240
241 if (teamANr > 0 && teamBNr == 0) {
242 // Freilos für Team A → Sieg ohne Punkte
243 aktuelleGruppe.computeIfAbsent(teamANr, k -> new int[4])[0]++;
244 continue;
245 }
246
247 if (teamANr <= 0) {
248 continue;
249 }
250
251 // Beiden Teams initialisieren wenn nicht vorhanden
252 aktuelleGruppe.computeIfAbsent(teamANr, k -> new int[4]);
253 aktuelleGruppe.computeIfAbsent(teamBNr, k -> new int[4]);
254
255 if (ergA > 0 || ergB > 0) {
256 // [0]=siege, [1]=niederlagen, [2]=punkte+, [3]=punkte-
257 aktuelleGruppe.get(teamANr)[2] += ergA;
258 aktuelleGruppe.get(teamANr)[3] += ergB;
259 aktuelleGruppe.get(teamBNr)[2] += ergB;
260 aktuelleGruppe.get(teamBNr)[3] += ergA;
261
262 if (ergA > ergB) {
263 aktuelleGruppe.get(teamANr)[0]++;
264 aktuelleGruppe.get(teamBNr)[1]++;
265 } else if (ergB > ergA) {
266 aktuelleGruppe.get(teamBNr)[0]++;
267 aktuelleGruppe.get(teamANr)[1]++;
268 }
269 }
270 }
271
272 return gruppen;
273 }

279 private List<List<PouleTeamErgebnis>> sortiereGruppen(List<Map<Integer, int[]>> gruppenRoh) {
280 var rechner = new PouleRanglisteRechner();
281 var sortiertGruppen = new ArrayList<List<PouleTeamErgebnis>>();
282
283 for (var gruppeRoh : gruppenRoh) {
284 var ergebnisse = new ArrayList<PouleTeamErgebnis>();
285 for (var entry : gruppeRoh.entrySet()) {
286 int teamNr = entry.getKey();
287 int[] stats = entry.getValue();
288 int siege = stats[0];
289 int niederlagen = stats[1];
290 int pktPlus = stats[2];
291 int pktMinus = stats[3];
292 int diff = pktPlus - pktMinus;
293 ergebnisse.add(new PouleTeamErgebnis(teamNr, siege, niederlagen, diff, pktPlus,
294 List.of()));
295 }
296 sortiertGruppen.add(rechner.sortiere(ergebnisse));
297 }
298
299 return sortiertGruppen;
300 }

308 private int schreibeDatenZeilen(XSpreadsheet xSheet, List<List<PouleTeamErgebnis>> sortiertGruppen)
309 throws GenerateException {
310
311 var block1 = new RangeData(); // Platz, Gruppe, Nr
312 var block2 = new RangeData(); // Siege, Ndlg, PktPlus, PktMinus, Diff, Turnier
313 var aTurnierZeilen = new ArrayList<Integer>();
314 var bTurnierZeilen = new ArrayList<Integer>();
315 int aktuelleZeile = HEADER_ZEILEN;
316
317 for (int gruppenIndex = 0; gruppenIndex < sortiertGruppen.size(); gruppenIndex++) {
318 SheetRunner.testDoCancelTask();
319 var gruppe = sortiertGruppen.get(gruppenIndex);
320 int gruppenNr = gruppenIndex + 1;
321
322 for (int platzIndex = 0; platzIndex < gruppe.size(); platzIndex++) {
323 var erg = gruppe.get(platzIndex);
324 int platz = platzIndex + 1;
325 boolean istATurnier = platz <= 2;
326
327 var row1 = block1.addNewRow();
328 row1.newInt(platz);
329 row1.newInt(gruppenNr);
330 row1.newInt(erg.teamNr());
331
332 var row2 = block2.addNewRow();
333 row2.newInt(erg.siege());
334 row2.newInt(erg.niederlagen());
335 row2.newInt(erg.erzieltePunkte());
336 row2.newInt(erg.erzieltePunkte() - erg.punkteDifferenz());
337 row2.newInt(erg.punkteDifferenz());
338 row2.newString(istATurnier ? "A" : "B");
339
340 if (istATurnier) {
341 aTurnierZeilen.add(aktuelleZeile);
342 } else {
343 bTurnierZeilen.add(aktuelleZeile);
344 }
345 aktuelleZeile++;
346 }
347 }
348
349 if (block1.isEmpty()) {
350 return HEADER_ZEILEN;
351 }
352
353 var doc = getWorkingSpreadsheet().getWorkingSpreadsheetDocument();
354 RangeHelper.from(xSheet, doc, block1.getRangePosition(Position.from(SPALTE_PLATZ, HEADER_ZEILEN)))
355 .setDataInRange(block1);
356 RangeHelper.from(xSheet, doc, block2.getRangePosition(Position.from(SPALTE_SIEGE, HEADER_ZEILEN)))
357 .setDataInRange(block2);
358
359 // VLOOKUP-Formel für Name-Spalte (referenziert Nr-Zelle in gleicher Zeile)
360 for (int z = HEADER_ZEILEN; z < aktuelleZeile; z++) {
361 String vlookup = "VLOOKUP(" + Position.from(SPALTE_NR, z).getAddress()
362 + ";$'" + SheetNamen.meldeliste() + "'.$A$4:$B$999;2;0)";
363 getSheetHelper().setFormulaInCell(
364 StringCellValue.from(xSheet, Position.from(SPALTE_NAME, z), vlookup));
365 }
366
367 // Formatierung: Rahmen + Zentrierung für alle Datenzellen
368 int letzteZeile = aktuelleZeile - 1;
369 getSheetHelper().setPropertiesInRange(xSheet,
370 RangePosition.from(SPALTE_PLATZ, HEADER_ZEILEN, LETZTE_SPALTE, letzteZeile),
371 CellProperties.from().margin(MeldeListeKonstanten.CELL_MARGIN).setBorder(BorderFactory.from().allThin().toBorder())
372 .setHoriJustify(CellHoriJustify.CENTER));
373
374 // Vertikale Trennlinien aus dem Header in den Datenbereich übernehmen
375 getSheetHelper().setPropertiesInRange(xSheet,
376 RangePosition.from(SPALTE_SIEGE, HEADER_ZEILEN, SPALTE_SIEGE, letzteZeile),
377 CellProperties.from().setBorder(BorderFactory.from().allThin().doubleLn().forLeft().toBorder()));
378 getSheetHelper().setPropertiesInRange(xSheet,
379 RangePosition.from(SPALTE_PKT_PLUS, HEADER_ZEILEN, SPALTE_PKT_PLUS, letzteZeile),
380 CellProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forLeft().toBorder()));
381 getSheetHelper().setPropertiesInRange(xSheet,
382 RangePosition.from(SPALTE_DIFF, HEADER_ZEILEN, SPALTE_DIFF, letzteZeile),
383 CellProperties.from().setBorder(BorderFactory.from().allThin().boldLn().forRight().toBorder()));
384
385 // Name-Spalte: linksbündig
386 getSheetHelper().setPropertiesInRange(xSheet,
387 RangePosition.from(SPALTE_NAME, HEADER_ZEILEN, SPALTE_NAME, letzteZeile),
388 CellProperties.from().setHoriJustify(CellHoriJustify.LEFT));
389
390 // Turnier-Spalte: fett
391 getSheetHelper().setPropertiesInRange(xSheet,
392 RangePosition.from(SPALTE_TURNIER, HEADER_ZEILEN, SPALTE_TURNIER, letzteZeile),
393 CellProperties.from().setCharWeight(FontWeight.BOLD));
394
395 // Nr-Spalte: vertikal zentriert (horizontal bereits via Bulk gesetzt)
396 getSheetHelper().setPropertiesInRange(xSheet,
397 RangePosition.from(SPALTE_NR, HEADER_ZEILEN, SPALTE_NR, letzteZeile),
398 CellProperties.from().centerVertJustify());
399
400 // Platz-Spalte: fett + dicke rechte Linie
401 getSheetHelper().setPropertiesInRange(xSheet,
402 RangePosition.from(SPALTE_PLATZ, HEADER_ZEILEN, SPALTE_PLATZ, letzteZeile),
403 CellProperties.from()
404 .setCharWeight(FontWeight.BOLD)
405 .setBorder(BorderFactory.from().allThin().boldLn().forRight().toBorder()));
406
407 // Hintergrundfarbe: A-Turnier (grün)
408 for (int zeile : aTurnierZeilen) {
409 getSheetHelper().setPropertiesInRange(xSheet,
410 RangePosition.from(SPALTE_PLATZ, zeile, LETZTE_SPALTE, zeile),
411 CellProperties.from().setCellBackColor(FARBE_A_TURNIER));
412 }
413 // Hintergrundfarbe: B-Turnier (rot)
414 for (int zeile : bTurnierZeilen) {
415 getSheetHelper().setPropertiesInRange(xSheet,
416 RangePosition.from(SPALTE_PLATZ, zeile, LETZTE_SPALTE, zeile),
417 CellProperties.from().setCellBackColor(FARBE_B_TURNIER));
418 }
419
420 return aktuelleZeile;
421 }
