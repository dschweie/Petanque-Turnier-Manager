93 private void testNamenEinfuegen() throws GenerateException {
94 var meldeliste = new MaastrichterMeldeListeSheetUpdate(getWorkingSpreadsheet());
95 int ersteDatenZeile = meldeliste.getErsteDatenZiele();
96 int anzSpielerProTeam = konfigSheet.getMeldeListeFormation().getAnzSpieler();
97 boolean teamnameAktiv = konfigSheet.isMeldeListeTeamnameAnzeigen();
98 boolean vereinsnameAktiv = konfigSheet.isMeldeListeVereinsnameAnzeigen();
99 var spieler = testnamenLoader.listeMitSpielerTestNamen(anzTeams * anzSpielerProTeam);
100
101 var data = new RangeData();
102 for (int team = 0; team < anzTeams; team++) {
103 testDoCancelTask();
104 var zeile = data.addNewRow();
105 zeile.newInt(team + 1); // TeamNr - manuell vergeben
106 if (teamnameAktiv) {
107 zeile.newString("Team " + (team + 1));
108 }
109 for (int s = 0; s < anzSpielerProTeam; s++) {
110 var stn = spieler.get(team * anzSpielerProTeam + s);
111 zeile.newString(stn.vorname());
112 zeile.newString(stn.nachname());
113 if (vereinsnameAktiv) {
114 zeile.newString("Verein " + ((team % 5) + 1));
115 }
116 }
117 zeile.newEmpty(); // Setzposition
118 zeile.newInt(MaastrichterMeldeListeSheetUpdate.AKTIV_WERT_NIMMT_TEIL);
119 }
120
121 var xSheet = getXSpreadSheet();
122 var startPos = Position.from(0, ersteDatenZeile);
123 RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
124 data.getRangePosition(startPos)).setDataInRange(data);
125
126 // Formatierung + bedingte Färbung anwenden (kein doRun, keine Nummernvergabe nötig)
127 meldeliste.upDateSheet();
128 }
