165
166 Optional<StartParameter> result = Optional.empty();
167 if (okPressed) {
168 Formation formation = readFormation(xcc);
169 boolean teamnameAnzeigen = readCheckBoxState(xcc, "cbTeamname");
170 boolean vereinsnameAnzeigen = readCheckBoxState(xcc, "cbVereinsname");

173 int gruppengroesse = parseGruppengroesse(readEditText(xcc, "editGruppengroesse"));
174 boolean mitRueckrunde = readCheckBoxState(xcc, "cbRueckrunde");
175 result = Optional.of(new StartParameter(formation, teamnameAnzeigen, vereinsnameAnzeigen, spielplanAnzeige, gruppengroesse, mitRueckrunde));
176 }
177
178 Lo.qi(XComponent.class, dialog).dispose();
179 ProcessBox.from().visibleWennAutomatisch();
180
181 return result;

188 private int parseGruppengroesse(String text) {
189 try {
190 return Math.max(0, Integer.parseInt(text.trim()));
191 } catch (NumberFormatException e) {
192 return 0;
193 }
194 }

196 private String readEditText(XControlContainer xcc, String name) {
197 XControl ctrl = xcc.getControl(name);
198 if (ctrl == null) {
199 return "";
200 }
201 XTextComponent text = Lo.qi(XTextComponent.class, ctrl);
202 return text != null ? text.getText() : "";
203 }
204
205 private Formation readFormation(XControlContainer xcc) {
206 return switch (readListBoxSelected(xcc, "lstFormation")) {
207 case 1 -> Formation.DOUBLETTE;
208 case 2 -> Formation.TRIPLETTE;
209 default -> Formation.TETE;
210 };
211 }
212
213 private short readListBoxSelected(XControlContainer xcc, String name) {
214 XControl ctrl = xcc.getControl(name);
215 if (ctrl == null) {
216 return 0;
217 }
218 XListBox lb = Lo.qi(XListBox.class, ctrl);
219 return lb != null ? lb.getSelectedItemPos() : 0;
220 }
221
222 private boolean readCheckBoxState(XControlContainer xcc, String name) {
223 XControl ctrl = xcc.getControl(name);
224 if (ctrl == null) {
225 return false;
226 }
227 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
228 return cb != null && cb.getState() == 1;
229 }

231 private void attachButtonListener(XControlContainer xcc, String name, XActionListener listener) {

233 if (ctrl != null) {
234 XButton btn = Lo.qi(XButton.class, ctrl);
235 if (btn != null) {
236 btn.addActionListener(listener);
