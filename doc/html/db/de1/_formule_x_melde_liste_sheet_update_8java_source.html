163 public boolean vollstaendigAktualisieren() throws GenerateException {
164 XSpreadsheet xSheet = getXSpreadSheet();
165 if (xSheet == null) {
166 logger.warn("Formule X Meldeliste nicht gefunden");
167 return false;
168 }
169 stringsBesinigen(xSheet);
170 teamnummernVergeben(xSheet);
171 aktivDefaultSetzen(xSheet);
172 pruefeAufDoppelteTeamNr(xSheet);
173 nachNrSortieren(xSheet);
174 upDateSheet();
175 return true;
176 }

178 private void aktivDefaultSetzen(XSpreadsheet xSheet) throws GenerateException {
179 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
180 if (letzteZeile < ERSTE_DATEN_ZEILE) {
181 return;
182 }
183 int vornameSpalte = delegate.getVornameSpalte(0);
184 int aktivSpalte = delegate.getAktivSpalte();
185
186 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
187 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
188 if (vorname == null || vorname.isBlank()) {
189 continue;
190 }
191 int aktiv = getSheetHelper().getIntFromCell(xSheet, Position.from(aktivSpalte, zeile));
192 if (aktiv <= 0) {
193 getSheetHelper().setNumberValueInCell(
194 NumberCellValue.from(xSheet, Position.from(aktivSpalte, zeile))
195 .setValue(AKTIV_WERT_NIMMT_TEIL));
196 }
197 }
198 }

200 private void teamnummernVergeben(XSpreadsheet xSheet) throws GenerateException {
201 int letzteZeile = delegate.letzteZeileMitDaten(xSheet);
202 if (letzteZeile < ERSTE_DATEN_ZEILE) {
203 return;
204 }
205
206 int nrSpalte = getNrSpalte();
207 int vornameSpalte = delegate.getVornameSpalte(0);
208
209 RangePosition sortRange = RangePosition.from(nrSpalte, ERSTE_DATEN_ZEILE, delegate.getAktivSpalte(),
210 letzteZeile);
211 SortHelper.from(this, sortRange).spalteToSort(nrSpalte).abSteigendSortieren().doSort();
212
213 int letztNr = Math.max(0,
214 getSheetHelper().getIntFromCell(xSheet, Position.from(nrSpalte, ERSTE_DATEN_ZEILE)));
215
216 for (int zeile = ERSTE_DATEN_ZEILE; zeile <= letzteZeile; zeile++) {
217 String vorname = getSheetHelper().getTextFromCell(xSheet, Position.from(vornameSpalte, zeile));
218 if (vorname == null || vorname.isBlank()) {
219 continue;
220 }
221 int nr = getSheetHelper().getIntFromCell(xSheet, Position.from(nrSpalte, zeile));
222 if (nr <= 0) {
223 letztNr++;
224 getSheetHelper().setNumberValueInCell(
225 NumberCellValue.from(xSheet, Position.from(nrSpalte, zeile)).setValue(letztNr));
226 }
227 }
228 }
