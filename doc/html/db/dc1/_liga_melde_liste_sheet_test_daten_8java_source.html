75 public void testNamenEinfuegen() throws GenerateException {
76 XSpreadsheet meldelisteSheet = meldeListe.getXSpreadSheet();
77 TurnierSheet.from(meldelisteSheet, getWorkingSpreadsheet()).setActiv();
78
79 List<String> testNamen = listeMitTestNamen();
80 Collections.shuffle(testNamen, RandomSource.asJavaRandom());
81
82 RangeData data = new RangeData();
83
84 int anzTeams = geradeAnzahlMannschaften ? 5 : 6;
85 int cntr = 0;
86 for (String name : testNamen) {
87 RowData newTeam = data.addNewRow();
88 newTeam.newEmpty();
89 newTeam.newString(name);
90 if (cntr++ > anzTeams) {
91 break;
92 }
93 }
94 Position posSpielerNr = Position.from(MeldeListeKonstanten.SPIELER_NR_SPALTE,
95 MeldeListeKonstanten.ERSTE_DATEN_ZEILE - 1);
96 RangeHelper.from(this, data.getRangePosition(posSpielerNr)).setDataInRange(data);
97 meldeListe.upDateSheet();
98 }
