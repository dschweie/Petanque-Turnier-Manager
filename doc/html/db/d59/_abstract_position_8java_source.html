161 public static int spalteStringToNumber(String name) {
162 checkNotNull(name);
163 int number = 0;
164 for (int i = 0; i < name.length(); i++) {
165 number = number * 26 + (name.charAt(i) - ('A' - 1));
166 }
167 return number - 1;

177 public static String spalteNrToString(int number) {
178 checkArgument(number > -1);
179 number++;
180 StringBuilder sb = new StringBuilder();
181 while (number-- > 0) {
182 sb.append((char) ('A' + (number % 26)));
183 number /= 26;
184 }
185 return sb.reverse().toString();
186 }
