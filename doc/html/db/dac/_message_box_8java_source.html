72 public final MessageBoxResult show() {
73
74 if (forceOk || dialogeUeberspringen) {
75 switch (type) {
76 case QUESTION_YES_NO:
77 case WARN_YES_NO:
78 case WARN_YES_NO_CANCEL:
79 return MessageBoxResult.YES;
80 case QUESTION_OK_CANCEL:
81 case WARN_OK_CANCEL:
82 case WARN_OK:
83 case ERROR_OK:
84 case INFO_OK:
85 return MessageBoxResult.OK;
86 }
87 }
88
89 // Log window ausblenden, weil wenn an der gleiche Stelle, wird dieser Dialog überdeckt.
90 // Vorigen Sichtbarkeitsstand merken — nur wenn die ProcessBox vorher offen war,
91 // wird sie nach dem Dialog wieder eingeblendet. Sonst (z.B. Spieler-DB-Dialoge,
92 // die nichts mit Tournament-Runs zu tun haben) bleibt das Log-Fenster zu.
93 ProcessBox processBox = ProcessBox.from();
94 boolean processBoxWarSichtbar = processBox.istSichtbar();
95 processBox.hide();
96
97 MessageBoxResult result = null;
98 XMessageBox xMessageBox = null;
99
100 switch (type) {
101 case QUESTION_YES_NO:
102 xMessageBox = newXMessageBox(MessageBoxType.QUERYBOX, MessageBoxButtons.BUTTONS_YES_NO);
103 break;
104 case QUESTION_OK_CANCEL:
105 xMessageBox = newXMessageBox(MessageBoxType.QUERYBOX, MessageBoxButtons.BUTTONS_OK_CANCEL);
106 break;
107 case WARN_OK:
108 xMessageBox = newXMessageBox(MessageBoxType.WARNINGBOX, MessageBoxButtons.BUTTONS_OK);
109 break;
110 case WARN_OK_CANCEL:
111 xMessageBox = newXMessageBox(MessageBoxType.WARNINGBOX, MessageBoxButtons.BUTTONS_OK_CANCEL);
112 break;
113 case WARN_YES_NO:
114 xMessageBox = newXMessageBox(MessageBoxType.WARNINGBOX, MessageBoxButtons.BUTTONS_YES_NO);
115 break;
116 case WARN_YES_NO_CANCEL:
117 xMessageBox = newXMessageBox(MessageBoxType.WARNINGBOX, MessageBoxButtons.BUTTONS_YES_NO_CANCEL);
118 break;
119 case ERROR_OK:
120 xMessageBox = newXMessageBox(MessageBoxType.ERRORBOX, MessageBoxButtons.BUTTONS_OK);
121 break;
122 case INFO_OK:
123 xMessageBox = newXMessageBox(MessageBoxType.INFOBOX, MessageBoxButtons.BUTTONS_OK);
124 break;
125
126 default:
127 break;
128 }
129
130 if (xMessageBox != null) {
131 short msgBoxresult = xMessageBox.execute();
132 result = MessageBoxResult.findResult(msgBoxresult);
133 }
134
135 if (processBoxWarSichtbar) {
136 processBox.visible();
137 }
138
139 return result;
140 }
