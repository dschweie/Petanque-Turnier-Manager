156 @Override
157 protected void beiOkGeklickt() throws Exception {
158 var dauerText = leseFeld(CTRL_DAUER);
159 var bezeichnung = leseFeld(CTRL_BEZEICHNUNG);
160 var portText = leseFeld(CTRL_PORT);
161
162 long dauerSekunden;
163 try {
164 dauerSekunden = TimerManager.parseDauer(dauerText);
165 } catch (IllegalArgumentException e) {
166 zeigeValidierungsFehler(I18n.get("timer.fehler.dauer.ungueltig"));
167 throw e;
168 }
169
170 int port;
171 try {
172 port = Integer.parseInt(portText.trim());
173 if (port < 1024 || port > 65535) {
174 throw new NumberFormatException("Port außerhalb des Bereichs");
175 }
176 } catch (NumberFormatException e) {
177 zeigeValidierungsFehler(I18n.get("timer.fehler.port.ungueltig"));
178 throw new IllegalArgumentException(I18n.get("timer.fehler.port.ungueltig"), e);
179 }
180
181 TimerEinstellungen.speichern(dauerText.trim(), port, bezeichnung, hintergrundFarbeInt);
182 TimerManager.get().starten(dauerSekunden, bezeichnung, port, hintergrundFarbeInt);
183 logger.info("Timer gestartet via Dialog: {} s, Port {}, Farbe #{}", dauerSekunden, port,
184 String.format("%06x", hintergrundFarbeInt & 0xFFFFFF));
185 }

189 private void passeZeitAn(long deltaSekunden) {
190 var dauerText = leseFeld(CTRL_DAUER);
191 long aktuelle;
192 try {
193 aktuelle = TimerManager.parseDauer(dauerText);
194 } catch (IllegalArgumentException e) {
195 aktuelle = 0;
196 }
197 long neueZeit = Math.max(60, aktuelle + deltaSekunden);
198 setzeFeld(CTRL_DAUER, TimerManager.formatiere(neueZeit));
199 }
