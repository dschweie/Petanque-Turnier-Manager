45 session.setPassword(passwort);

48 hostKeyUserInfo.setPasswort(passwort);

50 session.setUserInfo(userInfo);
51 }

54 config.put("PreferredAuthentications", "password");
55 session.setConfig(config);
56 session.connect();
57
58 kanal = (ChannelSftp) session.openChannel("sftp");
59 kanal.connect();
60
61 String remotePfad = konfiguration.remotePfad();
62 erstelleVerzeichnisWennFehlt(kanal, remotePfad);
63 kanal.cd(remotePfad);
64
65 int anzahl = 0;
66 for (Path datei : dateien) {
67 Path dateiName = datei.getFileName();
68 if (dateiName == null) {
69 logger.warn("SFTP: Datei ohne Namen übersprungen: {}", datei);
70 continue;
71 }
72 try (InputStream is = Files.newInputStream(datei)) {
73 kanal.put(is, dateiName.toString());
74 anzahl++;
75 logger.info("SFTP hochgeladen: {}/{}", remotePfad, dateiName);
76 } catch (SftpException e) {

78 }
79 }

81 } catch (JSchException e) {
82 throw new IOException(formatiereVerbindungsfehler(e.getMessage(), konfiguration), e);
83 } catch (SftpException e) {
84 throw new IOException("SFTP-Verzeichniswechsel fehlgeschlagen: " + e.getMessage(), e);
85 } finally {

104 Path knownHosts = Paths.get(System.getProperty("user.home"), ".ssh", "known_hosts");
105 if (userInfo == null && !Files.isRegularFile(knownHosts)) {

151 + "Bitte den Fingerprint beim Anbieter prüfen und den known_hosts-Eintrag erneuern: "
