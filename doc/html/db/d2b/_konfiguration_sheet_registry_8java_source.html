61 public static Optional<BaseKonfigurationSheet> fuerAktivesDokument(WorkingSpreadsheet ws) {
62 if (ws == null) {
63 return Optional.empty();
64 }
65 try {
66 TurnierSystem ts = new DocumentPropertiesHelper(ws).getTurnierSystemAusDocument();
67 if (ts == null || ts == TurnierSystem.KEIN) {
68 return Optional.empty();
69 }
70 Function<WorkingSpreadsheet, BaseKonfigurationSheet> factory = FACTORIES.get(ts);
71 if (factory == null) {
72 logger.warn("Kein KonfigurationSheet registriert für Turniersystem {}", ts);
73 return Optional.empty();
74 }
75 return Optional.of(factory.apply(ws));
76 } catch (RuntimeException e) {
77 logger.warn("KonfigurationSheet konnte nicht erzeugt werden", e);
78 return Optional.empty();
79 }
80 }
