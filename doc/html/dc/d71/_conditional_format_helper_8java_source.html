197 public ConditionalFormatHelper applyAndDoReset() throws GenerateException {
198 checkNotNull(conditionOperator);
199 checkNotNull(formula1);
200 checkNotNull(styleName);
201
202 XPropertySet xPropSet = RangeHelper.from(getISheet(), rangePos).getPropertySet();
203 com.sun.star.sheet.XSheetConditionalEntries xEntries;
204 try {
205 xEntries = Lo.qi(com.sun.star.sheet.XSheetConditionalEntries.class,
206 xPropSet.getPropertyValue("ConditionalFormat"));
207 if (doClear) {
208 xEntries.clear(); // clears all condition entries.
209 }
210
211 // create a condition and apply it to the range
212 // https://www.openoffice.org/api/docs/common/ref/com/sun/star/sheet/XSheetConditionalEntries.html
213 PropertyValue[] aCondition;
214 if (formula2 != null) {
215 aCondition = new PropertyValue[4];
216 } else {
217 aCondition = new PropertyValue[3];
218 }
219 int idx = 0;
220 aCondition[idx] = new com.sun.star.beans.PropertyValue();
221 aCondition[idx].Name = "Operator";
222 aCondition[idx].Value = conditionOperator;
223
224 // string Formula1 contains the value or formula for the operation.
225 idx++;
226 aCondition[idx] = new com.sun.star.beans.PropertyValue();
227 aCondition[idx].Name = "Formula1";
228 aCondition[idx].Value = formula1;
229
230 // string Formula2 contains the second value or formula for the operation (used with ConditionOperator::BETWEEN or ConditionOperator::NOT_BETWEEN operations).
231 if (formula2 != null) {
232 idx++;
233 aCondition[idx] = new com.sun.star.beans.PropertyValue();
234 aCondition[idx].Name = "Formula2";
235 aCondition[idx].Value = formula2;
236 }
237
238 idx++;
239 aCondition[idx] = new com.sun.star.beans.PropertyValue();
240 aCondition[idx].Name = "StyleName";
241 aCondition[idx].Value = styleName;
242 xEntries.addNew(aCondition);
243 // LO verwirft setPropertyValue("ConditionalFormat") lautlos bei tab-geschütztem Sheet
244 // und löscht dabei den alten Eintrag – Scope sicherstellt dass Sheet entsperrt ist.
245 BlattschutzManager.get().ensureUnprotectedInScope();
246 xPropSet.setPropertyValue("ConditionalFormat", xEntries);
247 } catch (RuntimeException e) {
248 // LO verwirft setPropertyValue("ConditionalFormat") bei tab-geschütztem Sheet
249 // teils lautlos, teils mit RuntimeException. Loggen, Verarbeitung weiterlaufen lassen.
250 logger.warn(
251 "Bedingte Formatierung konnte nicht gesetzt werden – ggf. LO-Einschränkung "
252 + "bei tab-geschütztem Sheet (sc/source/ui/unoobj). Range: {}, Style: {}, Ursache: {}",
253 rangePos, styleName, e.getMessage());
254 } catch (UnknownPropertyException | WrappedTargetException | PropertyVetoException e) {
255 logger.error(e.getMessage(), e);
256 }
257 reset();
258 return this;
259 }
