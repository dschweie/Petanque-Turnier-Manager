99 public static void zeigeToolbarInAllenFrames(XComponentContext xContext) {
100 try {
101 var xDesktop = DocumentHelper.getCurrentDesktop(xContext);
102 if (xDesktop == null) {
103 logger.debug("zeigeToolbarInAllenFrames: kein Desktop gefunden");
104 return;
105 }
106 var xFramesSupplier = Lo.qi(XFramesSupplier.class, xDesktop);
107 if (xFramesSupplier == null) {
108 logger.debug("zeigeToolbarInAllenFrames: XFramesSupplier nicht verfügbar");
109 return;
110 }
111 var xFrames = xFramesSupplier.getFrames();
112 if (xFrames == null) {
113 logger.debug("zeigeToolbarInAllenFrames: Frames-Container ist null");
114 return;
115 }
116 int anzahl = xFrames.getCount();
117 logger.debug("zeigeToolbarInAllenFrames: {} Frame(s) gefunden", anzahl);
118 for (int i = 0; i < anzahl; i++) {
119 try {
120 var xFrame = Lo.qi(XFrame.class, xFrames.getByIndex(i));
121 zeigeToolbarInFrame(xFrame);
122 } catch (Exception e) {
123 logger.error("Fehler beim Anzeigen der Symbolleiste in Frame {}", i, e);
124 }
125 }
126 } catch (Exception e) {
127 logger.error("Fehler beim Durchsuchen aller Frames", e);
128 }
129 }

133 private void zeigeSymbolleiste(Object source) {
134 try {
135 var xModel = Lo.qi(XModel.class, source);
136 if (xModel == null) {
137 return;
138 }
139 // nur Calc-Dokumente
140 if (Lo.qi(XSpreadsheetDocument.class, xModel) == null) {
141 return;
142 }
143 var xController = xModel.getCurrentController();
144 if (xController == null) {
145 return;
146 }
147 zeigeToolbarInFrame(xController.getFrame());
148 } catch (Exception e) {
149 logger.error("Fehler beim automatischen Anzeigen der Symbolleiste", e);
150 }
151 }

153 private static void zeigeToolbarInFrame(XFrame xFrame) {
154 if (xFrame == null) {
155 return;
156 }
157 long startNs = System.nanoTime();
158 try {
159 // Druckvorschau-Frames überspringen: ScPreviewController implementiert XSpreadsheetView nicht
160 if (Lo.qi(XSpreadsheetView.class, xFrame.getController()) == null) {
161 logger.debug("zeigeToolbarInFrame '{}': Druckvorschau-Frame erkannt – übersprungen", xFrame.getName());
162 return;
163 }
164 var xFrameProps = Lo.qi(XPropertySet.class, xFrame);
165 if (xFrameProps == null) {
166 return;
167 }
168 var xLayoutManager = Lo.qi(XLayoutManager.class, xFrameProps.getPropertyValue("LayoutManager"));
169 if (xLayoutManager == null) {
170 return;
171 }
172 // Addon-Toolbar (addon_* URL) via showElement + requestElement einblenden.
173 // createElement ist NICHT nötig – LO verwaltet Addon-Toolbars intern via XCU.
174 xLayoutManager.showElement(TOOLBAR_RESOURCE_URL);
175 boolean result = xLayoutManager.requestElement(TOOLBAR_RESOURCE_URL);
176 long dauerMs = (System.nanoTime() - startNs) / 1_000_000L;
177 PerfLog.log(logger, "[STARTUP-TIMING] zeigeToolbarInFrame '{}': {} ms (showElement+requestElement={} isVisible={})",
178 xFrame.getName(), dauerMs, result, xLayoutManager.isElementVisible(TOOLBAR_RESOURCE_URL));
179 } catch (Exception e) {
180 logger.error("Fehler beim Anzeigen der Symbolleiste in Frame '{}'",
181 xFrame.getName(), e);
182 }
183 }
