139 @Override
140 protected void doRun() throws GenerateException {
141 setSpielTag(getKonfigurationSheet().getAktiveSpieltag());
142 if (naechsteSpielrundeEinfuegen()) {
143 new SpielrundeSheet_Validator(getWorkingSpreadsheet()).validateSpieltag(getSpielTag());
144
145 SpieltagRanglisteSheet spieltagRanglisteSheet = new SpieltagRanglisteSheet(getWorkingSpreadsheet());
146 String ranglisteSheetName = spieltagRanglisteSheet.getSheetName(getSpielTag());
147 XSpreadsheet xSpieltagRanglisteSheet = getSheetHelper().findByName(ranglisteSheetName);
148 if (xSpieltagRanglisteSheet != null) {
149 spieltagRanglisteSheet.generate(getSpielTag());
150 }
151
152 getSheetHelper().setActiveSheet(getXSpreadSheet());
153 }

156 public boolean naechsteSpielrundeEinfuegen() throws GenerateException {
157 SpielRundeNr aktuelleSpielrunde = getKonfigurationSheet().getAktiveSpielRunde();
158 setSpielRundeNr(aktuelleSpielrunde);
159 getMeldeListe().upDateSheet();
160 SpielerMeldungen aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
161
162 if (aktiveMeldungen.size() == 0) {
163 SpielerMeldungen alleMeldungen = getMeldeListe().getAlleMeldungen();
164 if (alleMeldungen.size() > 0) {
165 MessageBoxResult result = MessageBox.from(getxContext(), MessageBoxTypeEnum.WARN_YES_NO)
166 .caption(I18n.get("msg.caption.keine.aktiven.meldungen"))
167 .message(I18n.get("msg.text.keine.aktiven.spieler.aktivieren", alleMeldungen.size()))
168 .show();
169 if (result == MessageBoxResult.YES) {
170 getMeldeListe().alleSpielAktivieren();
171 aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
172 }
173 }
174 }
175
176 if (!canStart(aktiveMeldungen)) {
177 return false;
178 }
179
180 int neueSpielrunde = aktuelleSpielrunde.getNr();
181 if (getSheetHelper().findByName(getSheetName(getSpielTag(), getSpielRundeNr())) != null) {
182 neueSpielrunde++;
183 }
184
185 int historieAb = getKonfigurationSheet().getSpielRundeNeuAuslosenAb();
186 int historieBis = neueSpielrunde - 1;
187 gespieltenRundenEinlesen(aktiveMeldungen, historieAb, historieBis);
188
189 if (neueSpielrunde >= 2) {
190 vorNaechsterRunde();
191 }
192
193 return neueSpielrunde(aktiveMeldungen, SpielRundeNr.from(neueSpielrunde), historieAb, historieBis);
194 }

214 protected IRanglistenAktualisierer getRanglistenAktualisierer() {
215 return () -> {
216 SpieltagRanglisteSheet probe = new SpieltagRanglisteSheet(getWorkingSpreadsheet());
217 String tagName = probe.getSheetName(getSpielTag());
218 if (getSheetHelper().findByName(tagName) != null) {
219 new SpieltagRanglisteSheetUpdate(getWorkingSpreadsheet(), getSpielTag()).doRun();
220 }
221 EndranglisteSheetUpdate endUpdate = new EndranglisteSheetUpdate(getWorkingSpreadsheet());
222 if (endUpdate.getXSpreadSheet() != null && endUpdate.getAnzahlSpieltage() >= 2) {
223 endUpdate.doRun();
224 }
225 };
