55 // Cache für Dokument-Properties im Speicher, um wiederholte UNO-Aufrufe zu vermeiden.
56 // Schlüssel: UNO-OID (via UnoRuntime.generateOid), stabile Identität unabhängig von
57 // Java-Proxy-Instanzen. ConcurrentHashMap statt Hashtable für bessere Nebenläufigkeit.
58 private static final ConcurrentHashMap<String, ConcurrentHashMap<String, String>> PROPLISTE =
59 new ConcurrentHashMap<>();
60
61 private final XSpreadsheetDocument xSpreadsheetDocument;
62 private final ConcurrentHashMap<String, String> currentPropListe;
63 private boolean firstLoad = false;

69 public DocumentPropertiesHelper(XSpreadsheetDocument xSpreadsheetDocument) {
70 this.xSpreadsheetDocument = checkNotNull(xSpreadsheetDocument);
71 var oid = dokumentOid(xSpreadsheetDocument);
72 var cachedListe = PROPLISTE.get(oid);
73 if (cachedListe != null) {
74 currentPropListe = cachedListe;
75 } else {
76 // einmal laden
77 var neueListe = new ConcurrentHashMap<String, String>();
78 XMultiPropertySet xMultiPropertySet = getXMultiPropertySet();
79 XPropertySet xPropertySet = getXPropertySet();
80 Property[] properties = xMultiPropertySet.getPropertySetInfo().getProperties();
81 for (Property userProp : properties) {
82 try {
83 Object propVal = xPropertySet.getPropertyValue(userProp.Name);
84 neueListe.put(userProp.Name, propVal.toString());
85 } catch (UnknownPropertyException | WrappedTargetException e) {
86 // Property nicht lesbar – überspringen
87 }
88 }
89 var vorhandene = PROPLISTE.putIfAbsent(oid, neueListe);
90 currentPropListe = vorhandene != null ? vorhandene : neueListe;
91 firstLoad = vorhandene == null;
92 }
93 }

106 public static void removeDocument(Object source) {
107 try {
108 if (source != null) {
109 XModel xModel = Lo.qi(XModel.class, source);
110 XSpreadsheetDocument xSpreadsheetDocument = Lo.qi(XSpreadsheetDocument.class, xModel);
111 // null dann wenn kein XSpreadsheetDocument
112 if (xSpreadsheetDocument != null) {
113 var oid = dokumentOid(xSpreadsheetDocument);
114 var removed = PROPLISTE.remove(oid);
115 if (removed == null) {
116 logger.warn("removeDocument: Kein Cache-Eintrag für OID={}, Class={}",
117 oid, source.getClass().getName());
118 }
119 }
120 }
121 } catch (Exception e) {
122 logger.error(e.getMessage(), e);
123 }
124 }

132 public void setStringProperty(String propName, String val) {
133 if (val != null) {
134 String oldVal = currentPropListe.get(propName);
135 if (!Objects.equals(oldVal, val)) {
136 setStringPropertyInDocument(propName, val);
137 currentPropListe.put(propName, val);
138 if (logger.isInfoEnabled()) {
139 logger.trace("[FOKUS-TRACE] setStringProperty: name='{}' old='{}' new='{}' doc={}",
140 propName, oldVal, val,
141 de.petanqueturniermanager.comp.ProtocolHandler.beschreibeDokument(xSpreadsheetDocument));
142 }
143 PetanqueTurnierMngrSingleton.triggerTurnierEventListener(TurnierEventType.PropertiesChanged,
144 new OnProperiesChangedEvent(xSpreadsheetDocument).addChanged(propName, oldVal, val));
145 }
146 }

182 public void ohneModifiedFlag(Runnable aktion) {
183 checkNotNull(aktion);
184 XModifiable xModifiable = Lo.qi(XModifiable.class, xSpreadsheetDocument);
185 boolean warVorherModified = xModifiable != null && xModifiable.isModified();
186 aktion.run();
187 if (xModifiable != null && !warVorherModified) {
188 try {
189 xModifiable.setModified(false);
190 } catch (PropertyVetoException e) {
191 logger.warn("Modified-Flag zurücksetzen abgelehnt", e);
192 }
193 }

201 private void setStringPropertyInDocument(String propName, String val) {
202 boolean didExist = insertStringPropertyIfNotExist(propName, val); // zuerst neu einfuegen wenn nicht vorhanden
203 if (didExist) {
204 // wenn bereits vorhanden dann wert updaten!
205 XPropertySet propSet = getXPropertySet();
206 try {
207 propSet.setPropertyValue(propName, val);
208 } catch (IllegalArgumentException | UnknownPropertyException | PropertyVetoException
209 | WrappedTargetException e) {
210 logger.error(e);
211 }
212 }
213 }

222 private boolean insertStringPropertyIfNotExist(String name, String val) {
223 checkNotNull(name);
224 checkNotNull(val);
225
226 boolean didExist = false;
227 XPropertyContainer xpc = getXPropertyContainer();
228
229 try {
230 xpc.addProperty(name, (short) 0, new Any(Type.STRING, val));
231 } catch (PropertyExistException e) {
232 didExist = true;
233 } catch (IllegalTypeException | IllegalArgumentException e) {
234 logger.error(e.getMessage(), e);
235 }
236 return didExist;
237 }

259 public String getStringProperty(String propName, String defaultVal) {

261 .filter(key -> key.equalsIgnoreCase(propName)).findFirst();

272 private String getStringPropertyFromDocument(String propName, boolean ignoreNotFound, String defaultVal) {
273 XPropertySet propSet = getXPropertySet();

278 if (!ignoreNotFound) {
