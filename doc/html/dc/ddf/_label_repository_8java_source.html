155 private static boolean istUniqueVerletzung(SQLException e) {
156 String raw = e.getMessage();
157 if (raw == null) {
158 return false;
159 }
160 String msg = raw.toLowerCase(java.util.Locale.ROOT);
161 if (msg.contains("unique constraint failed") || msg.contains("sqlite_constraint_unique")) {
162 return true;
163 }
164 String state = e.getSQLState();
165 return state != null && state.startsWith("23");

169 String raw = e.getMessage();
170 if (raw == null) {
171 return false;

173 String msg = raw.toLowerCase(java.util.Locale.ROOT);
174 if (msg.contains("foreign key constraint failed") || msg.contains("sqlite_constraint_foreignkey")) {
175 return true;
176 }
177 String state = e.getSQLState();
178 return state != null && state.startsWith("23")
179 && (msg.contains("foreign") || msg.contains("integrity"));
