92 private void validateSpieler() throws GenerateException {
93
94 // Spielrunden einlesen
95 int spielrunden = getKonfigurationSheet().getAktiveSpielRunde().getNr();
96
97 HashSet<Integer> alleSpielrNrausSpielrunden = new HashSet<>();
98
99 for (int splRundeCntr = 1; splRundeCntr <= spielrunden; splRundeCntr++) {
100 SpielRundeNr spielRundeNr = SpielRundeNr.from(splRundeCntr);
101 String spielrundeName = spielrundeSheetValidator.getSheetName(getSpieltagNr(), spielRundeNr);
102 spielrundeSheetValidator.setSpielRundeNr(spielRundeNr);
103
104 // Spieler einlesen
105 XSpreadsheet spielrundeSheet = getSheetHelper().findByName(spielrundeName);
106
107 if (spielrundeSheet == null) {
108 throw new GenerateException(I18n.get("error.spielrunde.fehlt", spielrundeName));
109 }
110
111 // Spieler Nummer Block rechts neben die ergebnisse spalten einlesen
112 RangePosition rangePosSpielrNr = RangePosition.from(SpielrundeSheetKonstanten.ERSTE_SPIELERNR_SPALTE, SpielrundeSheetKonstanten.ERSTE_DATEN_ZEILE,
113 SpielrundeSheetKonstanten.LETZTE_SPALTE, 99);
114 RangeData dataFromRange = RangeHelper.from(spielrundeSheetValidator, rangePosSpielrNr).getDataFromRange();
115
116 // flatten to list
117 List<Integer> nr = dataFromRange.stream().flatMap(rowData -> rowData.stream()).map(celldata -> celldata.getIntVal(0)).filter(num -> num > 0)
118 .collect(Collectors.toList());
119 alleSpielrNrausSpielrunden.addAll(nr);
120 }
121 List<Integer> spielerNrListAusRangliste = getSpielerSpalte().getSpielerNrList();
122
123 // prüfen ob die Anzahl spieler stimmt
124 if (spielerNrListAusRangliste.size() != alleSpielrNrausSpielrunden.size()) {
125 throw new GenerateException(I18n.get("error.spieltagrangliste.anzahl.spieler"));
126 }
127
128 boolean allMatch = spielerNrListAusRangliste.stream().allMatch(nr -> alleSpielrNrausSpielrunden.contains(nr));
129
130 if (!allMatch) {
131 throw new GenerateException(I18n.get("error.spieltagrangliste.spieler.nr"));
132 }
133 }
