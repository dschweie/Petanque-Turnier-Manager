50 root.add("spielerLabels", baueJunctionArray(data));

53 Gson gson = new GsonBuilder().setPrettyPrinting().disableHtmlEscaping().create();
54 Path datei = request.target();

56 Path eltern = datei.toAbsolutePath().getParent();
57 if (eltern != null) {
58 Files.createDirectories(eltern);

61 gson.toJson(root, w);
