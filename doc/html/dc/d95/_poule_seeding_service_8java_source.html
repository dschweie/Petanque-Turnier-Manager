50 var sortiertTeams = meldungen.teams().stream()

55 int anzGruppen = gruppenGroessen.size();
56 List<List<Team>> pouleTeams = new ArrayList<>(anzGruppen);
57 for (int i = 0; i < anzGruppen; i++) {
58 pouleTeams.add(new ArrayList<>());
59 }
60
61 for (int i = 0; i < sortiertTeams.size(); i++) {
62 int runde = i / anzGruppen;
63 int posInRunde = i % anzGruppen;
64 int pouleIdx = (runde % 2 == 0) ? posInRunde : (anzGruppen - 1 - posInRunde);
65 pouleTeams.get(pouleIdx).add(sortiertTeams.get(i));
66 }
67
68 List<Poule> poules = new ArrayList<>(anzGruppen);
69 for (int i = 0; i < anzGruppen; i++) {
70 poules.add(new Poule(i + 1, List.copyOf(pouleTeams.get(i))));
71 }
