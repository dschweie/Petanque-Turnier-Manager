124 private void beimBearbeiten() {
125 Optional<LabelDatensatz> sel = ausgewaehlt();
126 if (sel.isEmpty()) {
127 return;
128 }
129 Integer nr = sel.get().nr();
130 if (nr == null) {
131 return;
132 }
133 Optional<String> eingabe = neuerNameDialog(sel.get().name());
134 if (eingabe.isEmpty()) {
135 return;
136 }
137 try {
138 labelRepo.update(nr, eingabe.get());
139 ladeListe();
140 } catch (DuplikatException e) {
141 zeigeFehler(I18n.get("spielerdb.fehler.label_name_doppelt", eingabe.get()));
142 } catch (SpielerDbException e) {
143 logger.error("Label bearbeiten fehlgeschlagen", e);
144 zeigeFehler(e.getMessage());
145 }
146 }

148 private void beimLoeschen() {
149 Optional<LabelDatensatz> sel = ausgewaehlt();
150 if (sel.isEmpty()) {
151 return;
152 }
153 Integer nr = sel.get().nr();
154 if (nr == null) {
155 return;
156 }
157 try {
158 int anz = labelRepo.countSpieler(nr);
159 if (anz > 0) {
160 zeigeFehler(I18n.get("spielerdb.fehler.label_in_benutzung", sel.get().name(), anz));
161 return;
162 }
163 MessageBoxResult res = MessageBox.from(xContext, MessageBoxTypeEnum.QUESTION_YES_NO)
164 .caption(I18n.get("spielerdb.frage.titel"))
165 .message(I18n.get("spielerdb.frage.label_loeschen", sel.get().name()))
166 .show();
167 if (res != MessageBoxResult.YES) {
168 return;
169 }
170 labelRepo.delete(nr);
171 ladeListe();
172 } catch (InBenutzungException e) {
173 zeigeFehler(I18n.get("spielerdb.fehler.label_in_benutzung", sel.get().name(), "?"));
174 } catch (SpielerDbException e) {
175 logger.error("Label löschen fehlgeschlagen", e);
176 zeigeFehler(e.getMessage());
177 }
178 }

180 private Optional<LabelDatensatz> ausgewaehlt() {

187 return Optional.empty();

189 return Optional.of(aktuelleListe.get(idx));
190 }
