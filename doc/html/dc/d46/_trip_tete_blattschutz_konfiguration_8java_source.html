72 private void sammleMeldelisteSchutzInfo(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
73 // Editierbare Spalten: Teamname bis Aktiv (max. Breite: 1 Teamname + 3×3 Spieler + 1 Aktiv = Spalte 11)
74 // Spalte 0 = Nr (gesperrt), Spalten 1..MAX = editierbar (ohne Teamname/Vereinsname-Konfigurationsabhängigkeit
75 // statische Obergrenze verwenden, damit alle Konfigurationsvarianten abgedeckt sind)
76 int ersteEditierbareSpalte = SPIELER_NR_SPALTE + 1;
77 int letzteEditierbareSpalte = 11; // max: Teamname(1) + 3 Spieler à 3 Spalten + Aktiv = Spalte 11
78 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_TRIPTETE_MELDELISTE).ifPresent(sheet ->
79 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, List.of(
80 RangePosition.from(ersteEditierbareSpalte, 3 /* ERSTE_DATEN_ZEILE_OVERRIDE */,
81 letzteEditierbareSpalte, MeldungenSpalte.MAX_ANZ_MELDUNGEN)))));
82 }
83
84 private void sammleSpielplanSchutzInfo(XSpreadsheetDocument xDoc, List<SheetSchutzInfo> infos) {
85 SheetMetadataHelper.findeSheet(xDoc, SheetMetadataHelper.SCHLUESSEL_TRIPTETE_SPIELPLAN).ifPresent(sheet -> {
86 int ersteDatenZeile = TripTeteSpielPlanSheet.ERSTE_DATEN_ZEILE;
87 int letzteZeile = ermittleLetzteSpielplanZeile(sheet);
88 var bereiche = List.of(
89 RangePosition.from(TripTeteSpielPlanSheet.BAHN_TRI_SPALTE, ersteDatenZeile,
90 TripTeteSpielPlanSheet.BAHN_TETE_SPALTE, letzteZeile),
91 RangePosition.from(TripTeteSpielPlanSheet.TRI_A_SPALTE, ersteDatenZeile,
92 TripTeteSpielPlanSheet.TETE_B_SPALTE, letzteZeile));
93 infos.add(SheetSchutzInfo.mitEditierbarenBereichen(sheet, bereiche));
94 });
95 }
96
97 private int ermittleLetzteSpielplanZeile(XSpreadsheet sheet) {
98 int letzteZeile = TripTeteSpielPlanSheet.ERSTE_DATEN_ZEILE;
99 try {
100 for (int zeile = TripTeteSpielPlanSheet.ERSTE_DATEN_ZEILE;
101 zeile <= MeldungenSpalte.MAX_ANZ_MELDUNGEN; zeile++) {
102 XCell xCell = sheet.getCellByPosition(TripTeteSpielPlanSheet.SPIEL_NR_SPALTE, zeile);
103 if (CellContentType.EMPTY.equals(xCell.getType())) {
104 break;
105 }
106 letzteZeile = zeile;
107 }
108 } catch (Exception e) {
109 logger.warn("Letzte Spielplan-Zeile konnte nicht ermittelt werden: {}", e.getMessage(), e);
110 }
111 return letzteZeile;
112 }
