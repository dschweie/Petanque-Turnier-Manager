242 private Formation readFormation(XControlContainer xcc) {
243 return switch (readListBoxSelected(xcc, "lstFormation")) {
244 case 1 -> Formation.DOUBLETTE;
245 case 2 -> Formation.TRIPLETTE;

247 };

250 private static short formationIndex(Formation formation) {
251 return switch (formation) {
252 case DOUBLETTE -> 1;
253 case TRIPLETTE -> 2;
254 default -> 0;
255 };
256 }
257
258 private static short spielbahnIndex(SpielrundeSpielbahn spielbahn) {
259 return switch (spielbahn) {
260 case L -> 1;
261 case N -> 2;
262 case R -> 3;
263 default -> 0;
264 };
265 }

267 private short readListBoxSelected(XControlContainer xcc, String name) {
268 XControl ctrl = xcc.getControl(name);
269 if (ctrl == null) {
270 return 0;
271 }
272 XListBox lb = Lo.qi(XListBox.class, ctrl);
273 return lb != null ? lb.getSelectedItemPos() : 0;
274 }

276 private boolean readCheckBoxState(XControlContainer xcc, String name) {
277 XControl ctrl = xcc.getControl(name);
278 if (ctrl == null) {
279 return false;
280 }
281 XCheckBox cb = Lo.qi(XCheckBox.class, ctrl);
282 return cb != null && cb.getState() == 1;
283 }
