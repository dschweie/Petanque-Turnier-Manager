87 private void teamNamenEinfuegen() throws GenerateException {
88 var konfiguration = delegate.getKonfigurationSheet();
89 int anzSpieler = konfiguration.getMeldeListeFormation().getAnzSpieler();
90 boolean teamnameAktiv = konfiguration.isMeldeListeTeamnameAnzeigen();
91 boolean vereinsnameAktiv = konfiguration.isMeldeListeVereinsnameAnzeigen();
92 var spieler = testnamenLoader.listeMitSpielerTestNamen(anzTeams * anzSpieler);
93
94 var data = new RangeData();
95 for (int i = 0; i < anzTeams; i++) {
96 testDoCancelTask();
97 var zeile = data.addNewRow();
98 if (teamnameAktiv) {
99 zeile.newString("Team " + (i + 1));
100 }
101 for (int s = 0; s < anzSpieler; s++) {
102 var stn = spieler.get(i * anzSpieler + s);
103 zeile.newString(stn.vorname());
104 zeile.newString(stn.nachname());
105 if (vereinsnameAktiv) {
106 zeile.newString("Verein " + ((i % 5) + 1));
107 }
108 }
109 zeile.newInt(i + 1); // Ranglistespalte (Setzreihenfolge)
110 zeile.newInt(KoListeDelegate.AKTIV_WERT_NIMMT_TEIL);
111 }
112
113 var xSheet = meldeListeNew.getXSpreadSheet();
114 var startPos = Position.from(1, KoListeDelegate.ERSTE_DATEN_ZEILE);
115 RangeHelper.from(xSheet, getWorkingSpreadsheet().getWorkingSpreadsheetDocument(),
116 data.getRangePosition(startPos)).setDataInRange(data);
117 }
