68 StringUtils.defaultIfBlank(StringUtils.strip(konfiguration.getKopfZeileMitte()),
69 TurnierSystem.KASKADE.getBezeichnung()),
70 StringUtils.strip(konfiguration.getTurnierlogoUrl()), sections)
71 .addTo(exportierteDateien);
72
