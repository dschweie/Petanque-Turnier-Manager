84 protected void befuelleTeilnehmerDaten() throws GenerateException {
85 processBoxinfo("processbox.teilnehmer.meldungen.einlesen");
86 TeamMeldungen aktiveMeldungen = meldeliste.getAktiveMeldungen();
87
88 boolean teamnameAktiv = konfigurationSheet.isMeldeListeTeamnameAnzeigen();
89
90 List<TeilnehmerEintrag> eintraege = new ArrayList<>(aktiveMeldungen.size());
91 if (aktiveMeldungen.size() > 0) {
92 TeilnehmerNamen namen = TeilnehmerNamenLeser.from(meldeliste, MELDELISTE_ERSTE_DATEN_ZEILE,
93 konfigurationSheet.getMeldeListeFormation(), teamnameAktiv,
94 konfigurationSheet.isMeldeListeVereinsnameAnzeigen()).lesen();
95 Map<Integer, String> spielerNamen = namen.spielerNamen();
96 Map<Integer, String> teamnamen = namen.teamnamen();
97 Map<Integer, String> sortNamen = namen.sortNamen();
98
99 for (Team team : aktiveMeldungen.getTeamList()) {
100 int nr = team.getNr();
101 eintraege.add(new TeilnehmerEintrag(nr,
102 teamnamen.getOrDefault(nr, ""),
103 spielerNamen.getOrDefault(nr, ""),
104 sortNamen.getOrDefault(nr, "")));
105 }
106 eintraege.sort(konfigurationSheet.getTeilnehmerListeSortModus().comparator());
107 }
108
109 processBoxinfo("processbox.teilnehmer.meldungen.einfuegen", eintraege.size());
110
111 TeilnehmerSheetBuilder builder = TeilnehmerSheetBuilder.from(this)
112 .daten(eintraege)
113 .teamnameAktiv(teamnameAktiv)
114 .maxProBlock(konfigurationSheet.getMaxAnzTeilnehmerInSpalte())
115 .headerFarbe(konfigurationSheet.getMeldeListeHeaderFarbe())
116 .zebraFarben(konfigurationSheet.getMeldeListeHintergrundFarbeGerade(),
117 konfigurationSheet.getMeldeListeHintergrundFarbeUnGerade())
118 .aufbauen();
119
120 int letzteSpalte = builder.getLetzteDatenSpalte();
121 int footerZeile = builder.getLetzteDatenZeile() + 1;
122 StringCellValue footer = StringCellValue.from(getXSpreadSheet(), Position.from(0, footerZeile))
123 .setValue(I18n.get("teilnehmer.footer.anzahl", aktiveMeldungen.size()))
124 .setEndPosMergeSpalte(letzteSpalte).setCharWeight(FontWeight.BOLD).setCharHeight(12)
125 .setShrinkToFit(true);
126 getSheetHelper().setStringValueInCell(footer);
127 builder.freezeUndPrintbereich(footerZeile);
128 }
