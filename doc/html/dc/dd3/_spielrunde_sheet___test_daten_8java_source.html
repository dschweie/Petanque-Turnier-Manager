152 public void generate() throws GenerateException {
153 if (spielTag == null) {
154 setSpielTag(SpielTagNr.from(1));
155 }
156 anmeldungenSheet.setSpielTag(getSpielTag());
157 teilnehmerSheet.setSpielTagNr(getSpielTag());
158 spieltagRanglisteSheet.setSpieltagNr(getSpielTag());
159 naechsteSpielrundeSheet.setSpielTag(getSpielTag());
160
161 if (getSpielTag().getNr() == 1) {
162 meldeListeTestDatenGenerator.testNamenEinfuegen();
163 }
164
165 meldeListeTestDatenGenerator.initialAktuellenSpielTagMitAktivenMeldungenFuellen(getSpielTag());
166
167 anmeldungenSheet.generate();
168 teilnehmerSheet.generate();
169
170 int maxspielrundeNr = 4;
171 genRunden(maxspielrundeNr, true);
172
173 SheetRunner.testDoCancelTask();
174 spieltagRanglisteSheet.generate(getSpielTag());
175 new SpieltagRangliste_Validator(getWorkingSpreadsheet()).doValidate(getSpielTag());
176
177 getKonfigurationSheet().setAktiveSpieltag(getSpielTag());
178 getKonfigurationSheet().setAktiveSpielRunde(SpielRundeNr.from(maxspielrundeNr));
179
180 spieltagRanglisteSheet.isErrorInSheet();
181 }

183 public void genRunden(int maxspielrundeNr, boolean shuffleAktivInaktiv) throws GenerateException {
184 naechsteSpielrundeSheet.setSpielTag(getSpielTag());
185
186 for (int spielrundeNr = 1; spielrundeNr <= maxspielrundeNr; spielrundeNr++) {
187 SheetRunner.testDoCancelTask();
188 setSpielRundeNr(SpielRundeNr.from(spielrundeNr));
189
190 if (shuffleAktivInaktiv && spielrundeNr > 1) {
191 meldeListeTestDatenGenerator.spielerAufAktivInaktivMischen(getSpielTag());
192 }
193
194 SpielerMeldungen aktiveMeldungen = getMeldeListe().getAktiveMeldungen();
195 int historieAb = getKonfigurationSheet().getSpielRundeNeuAuslosenAb();
196 int historieBis = spielrundeNr - 1;
197 naechsteSpielrundeSheet.gespieltenRundenEinlesen(aktiveMeldungen, historieAb, historieBis);
198 if (!neueSpielrunde(aktiveMeldungen, SpielRundeNr.from(spielrundeNr), true, historieAb, historieBis)) {
199 return;
200 }
201
202 XSpreadsheet xsheet = getXSpreadSheet();
203 Position letztePos = letztePositionRechtsUnten();
204
205 if (letztePos != null && xsheet != null) {
206 for (int zeileCntr = ERSTE_DATEN_ZEILE; zeileCntr <= letztePos.getZeile(); zeileCntr++) {
207 SheetRunner.testDoCancelTask();
208
209 Position pos = Position.from(ERSTE_SPALTE_ERGEBNISSE, zeileCntr);
210
211 int welchenTeamHatGewonnen = RandomSource.nextInt(0, 2);
212 int gewinnerPunkte = RandomSource.nextInt(9, 14);
213 int verliererPunkte = RandomSource.nextInt(0, gewinnerPunkte);
214 int valA = (welchenTeamHatGewonnen == 0 ? verliererPunkte : gewinnerPunkte);
215 int valB = (welchenTeamHatGewonnen == 0 ? gewinnerPunkte : verliererPunkte);
216
217 NumberCellValue numberCellValue = NumberCellValue.from(xsheet, pos, valA);
218 getSheetHelper().setNumberValueInCell(numberCellValue);
219 getSheetHelper().setNumberValueInCell(numberCellValue.spaltePlusEins().setValue((double) valB));
220 }
221 }
222 }
223 }
