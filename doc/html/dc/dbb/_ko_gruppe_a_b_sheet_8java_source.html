71 private void koRundeErstellen() throws GenerateException {
72 int anzGruppen = 0;
73 int maxPaarungen = 0;
74 for (int grpCntr = 0; grpCntr < 10; grpCntr++) {
75 TeamRangliste gruppeAusRanglist = ranglisten.gruppeAusRanglisteEinlesen(grpCntr, CadrageSheet.RANGLISTE_NACH_CADRAGE_SHEET);
76 if (gruppeAusRanglist.isEmpty()) {
77 break;
78 }
79 vorrunden.vorRundenEinlesen(gruppeAusRanglist);
80 KoRundeTeamPaarungen teamPaarungen = new KoRundeTeamPaarungen(gruppeAusRanglist);
81 FormeSpielrunde spielRunde = teamPaarungen.generiereSpielrunde();
82 getSheetHelper().setActiveSheet(getXSpreadSheet());
83 spielRundeInSheet.erstelleSpielRundeInSheet(grpCntr, getXSpreadSheet(), teamPaarungen, spielRunde);
84 anzGruppen = grpCntr + 1;
85 maxPaarungen = Math.max(maxPaarungen, spielRunde.getTeamPaarungen().size());
86 }
87 if (anzGruppen > 0) {
88 int letzteSpalte = anzGruppen * 2 - 1;
89 int letzteZeile = SpielRundeInSheet.ERSTEZEILE + maxPaarungen - 1;
90 getSheetHelper().setOptimaleBreiteUndHoeheAlles(getXSpreadSheet(), 0, letzteZeile, 0, letzteSpalte);
91 }
92 }
