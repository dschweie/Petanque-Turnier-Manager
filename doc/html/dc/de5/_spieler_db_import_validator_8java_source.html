62 private static List<ValVerein> validiereVereine(List<RohVerein> roh, List<String> fehler,
63 List<ImportWarnung> warnungen) {
64 LinkedHashMap<String, ValVerein> dedup = new LinkedHashMap<>();
65 for (int i = 0; i < roh.size(); i++) {
66 RohVerein v = roh.get(i);
67 String name = v.name() == null ? "" : v.name().strip();
68 if (name.isEmpty()) {
69 fehler.add("Verein-Eintrag " + (i + 1) + ": Name fehlt");
70 continue;
71 }
72 String key = SpielerMatchKeyNormalizer.vereinSchluessel(name);
73 ValVerein neu = new ValVerein(v.nr(), name);
74 if (dedup.put(key, neu) != null) {
75 warnungen.add(new ImportWarnung(
76 "Verein '" + name + "' kommt mehrfach in der Datei vor — letzter Eintrag gewinnt"));
77 }
78 }
79 return new ArrayList<>(dedup.values());

99 private static void ergaenzeImpliziteVereine(List<RohSpieler> spieler,
100 List<ValVerein> vereine, Map<String, Integer> nameZuAltNr) {
101 int sentinel = -1;
102 // Schon vergebene negative AltNrs (z.B. aus expliziten RohVereinen mit nr=null
103 // wären in vereine schon mit altNr=null vermerkt — diese betreffen den Sentinel
104 // nicht). Wir starten frisch bei -1, da Sentinel-NRs nur dann benutzt werden,
105 // wenn der explizite RohVerein keine altNr hatte; das stört das Mapping nicht.
106 Set<Integer> bereitsBenutzteSentinels = new HashSet<>();
107 for (RohSpieler s : spieler) {
108 if (s.vereinNr() != null) {
109 continue;
110 }
111 String vereinName = s.vereinName();
112 if (vereinName == null) {
113 continue;
114 }
115 String getrimmt = vereinName.strip();
116 if (getrimmt.isEmpty()) {
117 continue;
118 }
119 String key = SpielerMatchKeyNormalizer.vereinSchluessel(getrimmt);
120 if (nameZuAltNr.containsKey(key)) {
121 continue;
122 }
123 while (bereitsBenutzteSentinels.contains(sentinel)) {
124 sentinel--;
125 }
126 int altNr = sentinel--;
127 bereitsBenutzteSentinels.add(altNr);
128 vereine.add(new ValVerein(altNr, getrimmt));
129 nameZuAltNr.put(key, altNr);
130 }
131 }

153 private static List<ValSpieler> validiereSpieler(List<RohSpieler> roh,
154 Map<String, Integer> vereinNameZuAltNr, List<String> fehler,
155 List<ImportWarnung> warnungen) {
156 LinkedHashMap<String, ValSpieler> dedup = new LinkedHashMap<>();
157 for (int i = 0; i < roh.size(); i++) {
158 RohSpieler s = roh.get(i);
159 String vorname = s.vorname() == null ? "" : s.vorname().strip();
160 String nachname = s.nachname() == null ? "" : s.nachname().strip();
161 String herkunft = herkunftsLabel(s, i);
162 if (vorname.isEmpty() || nachname.isEmpty()) {
163 fehler.add(herkunft + ": Vor-/Nachname fehlt");
164 continue;
165 }
166 Integer effektiveVereinNr = effektiveVereinNr(s, vereinNameZuAltNr);
167 String key = SpielerMatchKeyNormalizer.spielerSchluesselMitVereinNr(
168 vorname, nachname, effektiveVereinNr);
169 ValSpieler neu = new ValSpieler(s.nr(), vorname, nachname,
170 effektiveVereinNr, s.vereinName(), s.lizenznr(), s.quellZeile());
171 if (dedup.put(key, neu) != null) {
172 String praefix = s.quellZeile() != null ? "Zeile " + s.quellZeile() + ": " : "";
173 warnungen.add(new ImportWarnung(praefix
174 + "Spieler '" + vorname + " " + nachname + "' kommt mehrfach in der Datei vor"
175 + " — letzter Eintrag gewinnt"));
176 }
177 }
178 return new ArrayList<>(dedup.values());
179 }

182 private static Integer effektiveVereinNr(RohSpieler s, Map<String, Integer> nameZuAltNr) {
183 if (s.vereinNr() != null) {
184 return s.vereinNr();
185 }
186 String name = s.vereinName();
187 if (name == null) {
188 return null;

191 if (getrimmt.isEmpty()) {
192 return null;
193 }
194 return nameZuAltNr.get(SpielerMatchKeyNormalizer.vereinSchluessel(getrimmt));

203 private static List<ValSpielerLabel> validiereJunction(ImportRohdaten roh,
204 List<ImportWarnung> warnungen) {
205 Set<Integer> bekannteSpielerNrs = new HashSet<>();
206 for (RohSpieler s : roh.spieler()) {
207 if (s.nr() != null) {
208 bekannteSpielerNrs.add(s.nr());
209 }
210 }
211 Set<Integer> bekannteLabelNrs = new HashSet<>();
212 for (RohLabel l : roh.labels()) {
213 if (l.nr() != null) {
214 bekannteLabelNrs.add(l.nr());
215 }
216 }
217 List<ValSpielerLabel> erg = new ArrayList<>(roh.spielerLabels().size());
218 for (RohSpielerLabel j : roh.spielerLabels()) {
219 if (!bekannteSpielerNrs.contains(j.spielerNr())) {
220 warnungen.add(new ImportWarnung("Junction-Eintrag verweist auf unbekannte spielerNr="
221 + j.spielerNr() + " — übersprungen"));
222 continue;
223 }
224 if (!bekannteLabelNrs.contains(j.labelNr())) {
225 warnungen.add(new ImportWarnung("Junction-Eintrag verweist auf unbekannte labelNr="
226 + j.labelNr() + " — übersprungen"));
227 continue;
228 }
229 erg.add(new ValSpielerLabel(j.spielerNr(), j.labelNr()));
230 }
231 return erg;
232 }
